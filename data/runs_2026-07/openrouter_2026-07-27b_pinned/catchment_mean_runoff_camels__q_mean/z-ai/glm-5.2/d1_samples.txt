import numpy as np

def predict(X):
    # Primary inputs
    P = X[:, 0]      # p_mean
    PET = X[:, 1]    # pet_mean
    
    # Aridity index
    phi = PET / P
    
    # Original Budyko (1974) curve: 
    # E/P = [ phi * tanh(1/phi) * (1 - exp(-phi)) ]^(1/2)
    E_over_P = np.sqrt(phi * np.tanh(1.0 / phi) * (1.0 - np.exp(-phi)))
    
    # Ensure physical bounds for the evapotranspiration ratio
    E_over_P = np.clip(E_over_P, 0.0, 1.0)
    
    # Long-term mean daily runoff Q = P - E
    Q = P * (1.0 - E_over_P)
    
    # Enforce physical lower bound (Q >= 0)
    Q = np.clip(Q, 0.0, None)
    
    return Q

def predict(X):

    PET = X[:, 1]
    # Aridity index φ = PET / P (avoid division by zero)
    phi = np.divide(PET, P, out=np.zeros_like(PET, dtype=float), where=P > 0)
    # Budyko (1974) closed-form curve: E/P = [φ * tanh(1/φ) * (1 - e^{-φ})]^{1/2}
    with np.errstate(divide='ignore', invalid='ignore'):
        e_over_p = np.sqrt(phi * np.tanh(1.0 / phi) * (1.0 - np.exp(-phi)))
    # Clean up any non-finite values at φ=0 or φ=∞ (not expected with P,PET>0)
    e_over_p = np.where(np.isfinite(e_over_p), e_over_p, 0.0)
    E = P * e_over_p
    Q = P - E
    # Enforce physical bounds: 0 ≤ Q ≤ P
    Q = np.clip(Q, 0.0, P)
    return Q

import numpy as np

def predict(X):
    P = X[:, 0]          # long-term mean daily precipitation [mm/day]
    Ep = X[:, 1]         # long-term mean daily potential evapotranspiration [mm/day]
    phi = Ep / P         # aridity index (PET/P)
    # Budyko (1974) closed-form curve: geometric mean of Schreiber (1904) and Ol'dekop (1911)
    # E_a/P = sqrt( (1 - exp(-phi)) * (phi * tanh(1/phi)) )
    Ea_over_P = np.sqrt((1.0 - np.exp(-phi)) * (phi * np.tanh(1.0 / phi)))
    Q = P * (1.0 - Ea_over_P)
    # Enforce physical bounds [0, P]
    Q = np.clip(Q, 0.0, P)
    return Q

    P = X[:, 0]

    # Compute aridity index (phi = PET / P), safely handling P = 0
    phi = np.divide(PET, P, out=np.full_like(PET, 1e9), where=P!=0)
    
    # Turc-Pike equation (a standard canonical form of the Budyko curve with parameter n=2)
    # Evaporative index: E/P = phi / sqrt(1 + phi^2)
    E_P = phi / np.sqrt(1.0 + phi**2)
    
    # Long-term mean daily runoff Q = P - E
    Q = P * (1.0 - E_P)
    
    # Enforce physical bounds: 0 <= Q <= P
    Q = np.clip(Q, 0, P)
    
    return Q

import numpy as np

def predict(X):
    X = np.asarray(X, dtype=float)
    P = X[:, 0]   # p_mean [mm/day]
    Ep = X[:, 1]  # pet_mean [mm/day]

    # Budyko (1974) long-term curve:
    # E/P = [ phi * tanh(1/phi) * (1 - exp(-phi)) ]^0.5,  phi = PET/P
    # Q = P - E
    phi = np.divide(Ep, P, out=np.full_like(P, np.inf), where=P > 0.0)

    with np.errstate(divide="ignore", invalid="ignore"):
        et_over_p = np.sqrt(phi * np.tanh(1.0 / phi) * (1.0 - np.exp(-phi)))

    # Enforce physical limits: 0 <= Q <= P
    et_over_p = np.where(np.isfinite(et_over_p), et_over_p, 1.0)
    Q = P * (1.0 - et_over_p)
    return np.clip(Q, 0.0, P)
