import numpy as np

def predict(X):
    T = X[:, 0]
    
    # Standard Vogel-Fulcher-Tammann (VFT) equation for silicate melt viscosity:
    # log10(η) = A + B / (T - C)
    # where A, B, and C are composition-dependent parameters.
    
    # Base parameters for a generic silicate melt
    A_intercept = -4.31
    B_intercept = 5000.0
    C_intercept = 500.0
    
    # Linear coefficients for the 12 oxide components:
    # [sio2, tio2, al2o3, feo, fe2o3, mno, mgo, cao, na2o, k2o, p2o5, h2o]
    A_coef = np.zeros(12)
    B_coef = np.array([10.0, 20.0, 15.0, -5.0, -5.0, -5.0, -10.0, -15.0, -20.0, -15.0, 5.0, -30.0])
    C_coef = np.array([2.0, 4.0, 3.0, -1.0, -1.0, -1.0, -2.0, -3.0, -4.0, -3.0, 1.0, -6.0])
    
    # Calculate composition-dependent VFT parameters
    A = A_intercept + np.dot(X[:, 1:], A_coef)
    B = B_intercept + np.dot(X[:, 1:], B_coef)
    C = C_intercept + np.dot(X[:, 1:], C_coef)
    
    # Ensure T - C is strictly positive to avoid division by zero
    T_diff = T - C
    T_diff = np.where(T_diff <= 1.0, 1.0, T_diff)
    
    # Calculate log10 viscosity
    log_viscosity = A + B / T_diff
    
    return log_viscosity

import numpy as np

def predict(X):
    # Extract inputs
    T = X[:, 0]
    sio2 = X[:, 1]
    tio2 = X[:, 2]
    al2o3 = X[:, 3]
    feo = X[:, 4]
    fe2o3 = X[:, 5]
    mno = X[:, 6]
    mgo = X[:, 7]
    cao = X[:, 8]
    na2o = X[:, 9]
    k2o = X[:, 10]
    p2o5 = X[:, 11]
    h2o = X[:, 12]
    
    # Giordano, Russell, and Dingwell (2008) VFT model parameterization
    # log10(η) = A + B / (T - C)
    # where A, B, and C depend on structural components:
    # SM = network formers (SiO2 + Al2O3)
    # FM = field strength modifiers (FeO + Fe2O3 + MnO + MgO + CaO)
    # NKM = alkali modifiers (Na2O + K2O)
    
    SM = sio2 + al2o3
    FM = feo + fe2o3 + mno + mgo + cao
    NKM = na2o + k2o
    
    A = -4.31 * np.ones_like(T)
    B = 5809.19 + 445.92 * SM + 12.32 * FM - 601.32 * NKM - 810.93 * h2o
    C = 623.13 - 26.24 * SM - 31.14 * FM + 68.17 * NKM + 118.41 * h2o
    
    # Avoid division by zero or negative denominator
    denom = T - C
    denom = np.where(denom <= 1e-6, 1e-6, denom)
    
    log_eta = A + B / denom
    return log_eta
