import numpy as np

def predict(X):
    Q2 = X[:, 0]
    mp = 0.938272  # proton mass [GeV]
    tau = Q2 / (4.0 * mp**2)

    # Kelly (2004) parametrization of the proton electric Sachs form factor
    a1 = -0.24
    b1 = 10.98
    b2 = 12.82
    b3 = 21.97
    GE = (1.0 + a1 * tau) / (1.0 + b1 * tau + b2 * tau**2 + b3 * tau**3)

    # Standard dipole form factor
    GD = 1.0 / (1.0 + Q2 / 0.71)**2

    return GE / GD
