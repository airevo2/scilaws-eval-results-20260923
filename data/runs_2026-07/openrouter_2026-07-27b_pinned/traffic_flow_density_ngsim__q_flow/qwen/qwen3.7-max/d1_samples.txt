import numpy as np

def predict(X):
    k = X[:, 0]
    v_f = 90.0  # free-flow speed in km/h
    k_j = 150.0 # jam density in veh/km/lane
    
    # Greenshields' parabolic fundamental diagram
    q = v_f * k * (1.0 - k / k_j)
    
    return np.maximum(q, 0.0)
