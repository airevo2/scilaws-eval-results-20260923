import numpy as np

def predict(X):
    # Greenshields fundamental diagram: q = v_f * k * (1 - k/k_jam)
    k = X[:, 0]
    v_f = 100.0      # free-flow speed [km/h]
    k_jam = 250.0    # jam density [veh/km/lane]
    q = v_f * k * (1.0 - k / k_jam)
    return q
