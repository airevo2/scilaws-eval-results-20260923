import numpy as np

def predict(X):
    # Extract inputs
    T = X[:, 0]          # Temperature [K]
    rho_liq = X[:, 1]    # Liquid density [kg/m^3]
    T_c = X[:, 2]        # Critical temperature [K]
    M_mol = X[:, 3]      # Molar mass [kg/mol]
    
    # Eötvös constant (J / (K * mol^(2/3)))
    k = 2.12e-7
    
    # Molar volume [m^3/mol]
    V_m = M_mol / rho_liq
    
    # Eötvös rule: sigma * V_m^(2/3) = k * (T_c - T)
    sigma = k * (T_c - T) / np.power(V_m, 2.0 / 3.0)
    
    return sigma

    Predict liquid surface tension sigma [N/m] along the saturation curve

    rho_liq = X[:, 1]
    T_c = X[:, 2]
    M_mol = X[:, 3]

    V_m = M_mol / rho_liq
    sigma = k * (T_c - T) / (V_m ** (2.0 / 3.0))
    return sigma
