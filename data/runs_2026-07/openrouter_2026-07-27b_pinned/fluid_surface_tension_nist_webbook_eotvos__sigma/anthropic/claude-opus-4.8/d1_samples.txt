import numpy as np

def predict(X):
    T = X[:, 0]
    rho = X[:, 1]
    Tc = X[:, 2]
    M = X[:, 3]

    # Eötvös rule: sigma * V_m^(2/3) = k * (T_c - T)
    # V_m = M / rho  [m^3/mol]
    # k ~ 2.1e-7 J/(K mol^(2/3))
    k = 2.1e-7

    V_m = M / rho
    sigma = k * (Tc - T) / np.power(V_m, 2.0 / 3.0)
    return sigma

import numpy as np

def predict(X):
    T      = X[:, 0]          # K
    rho    = X[:, 1]          # kg/m^3
    Tc     = X[:, 2]          # K
    M      = X[:, 3]          # kg/mol

    # Molar volume V_m [m^3/mol]
    Vm = M / rho

    # Eötvös rule: sigma * Vm^(2/3) = k * (T_c - T)
    # with the commonly cited Eötvös constant k ~ 2.1e-7 J/(K mol^(2/3))
    k = 2.1e-7

    sigma = k * (Tc - T) / np.power(Vm, 2.0/3.0)

    return sigma
