import numpy as np

def predict(X):
    """
    Predicts liquid surface tension along the saturation curve using the Eötvös rule:
    sigma * V_m^(2/3) = k_E * (T_c - T)
    where V_m = M_mol / rho_liq is the molar volume.
    
    Parameters
    ----------
    X : np.ndarray, shape (n_samples, 4)
        X[:, 0] : T_K [K]       - Temperature
        X[:, 1] : rho_liq [kg/m³] - Liquid density
        X[:, 2] : T_c [K]        - Critical temperature
        X[:, 3] : M_mol [kg/mol] - Molar mass
    
    Returns
    -------
    sigma : np.ndarray, shape (n_samples,)
        Surface tension [N/m]
    """
    T = X[:, 0]
    rho_liq = X[:, 1]
    T_c = X[:, 2]
    M_mol = X[:, 3]
    
    # Eötvös constant (approximate universal value in SI units: J/(K mol^(2/3)))
    k_E = 2.1e-7
    
    # Molar volume [m³/mol]
    V_m = M_mol / rho_liq
    
    # Eötvös rule: sigma = k_E * (T_c - T) / V_m^(2/3)
    sigma = k_E * (T_c - T) / (V_m ** (2.0 / 3.0))
    
    return sigma
