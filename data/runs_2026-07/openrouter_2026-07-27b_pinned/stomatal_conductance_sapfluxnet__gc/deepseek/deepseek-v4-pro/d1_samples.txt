import numpy as np

def predict(X):
    """
    Predict daily daytime canopy conductance proxy Gc (cm3 cm-2 h-1 kPa-1)
    from vapour pressure deficit D (kPa) using the log-linear decline model:
        Gc = Gc_ref - m * ln(D)
    
    Parameters
    ----------
    X : 2-D numpy array
        X[:, 0] = vpd (D) [kPa], daily daytime mean VPD >= 0.3 kPa
    
    Returns
    -------
    Gc : 1-D numpy array of canopy conductance proxy values.
    
    Notes
    -----
    Coefficients are based on the widely cited Oren et al. (1999) synthesis,
    with Gc_ref = 1.3 cm3 cm-2 h-1 kPa-1 (conductance at D = 1 kPa) and
    m = 0.78 (approximating the typical ratio m ≈ 0.6 * Gc_ref).
    """
    D = X[:, 0]          # vapour pressure deficit (kPa)
    Gc_ref = 1.3         # reference conductance at D = 1 kPa
    m = 0.78             # slope of log-linear decline
    return Gc_ref - m * np.log(D)

import numpy as np

def predict(X):
    """
    Predict daily daytime canopy conductance proxy (Gc) from vapour pressure deficit (VPD).

    Parameters
    ----------
    X : np.ndarray, shape (n_samples, 1)
        X[:, 0] = vpd (D) [kPa], daily daytime mean VPD (>= 0.3 kPa).

    Returns
    -------
    Gc : np.ndarray, shape (n_samples,)
        Predicted Gc [cm3 cm-2 h-1 kPa-1] using the log-linear model:
        Gc = Gc_ref - m * ln(D)
    """
    # Literature-based parameters for canopy conductance proxy
    # Gc_ref: reference conductance at D = 1 kPa  [cm3 cm-2 h-1 kPa-1]
    # m: sensitivity of conductance to ln(D)  [same units]
    Gc_ref = 2.5
    m = 0.8

    D = X[:, 0]          # vapour pressure deficit (already >= 0.3 kPa)
    Gc = Gc_ref - m * np.log(D)
    return Gc
