import numpy as np

def predict(X):
    """
    Predicts daily-mean streamflow discharge Q(t) during a recession event 
    using the standard Maillet exponential decay model (baseflow recession).
    
    Q_t = Q_0 * exp(-k * t)
    
    where k is a typical baseflow recession constant.
    """
    t = X[:, 0]
    Q_0 = X[:, 1]
    
    # Typical recession constant (k) in 1/days. 
    # A common representative value for daily streamflow recession is ~0.05.
    k = 0.05 
    
    Q_t = Q_0 * np.exp(-k * t)
    
    return Q_t
