import numpy as np

def predict(X):
    # Classic Maillet exponential recession: Q(t) = Q_0 * exp(-alpha * t)
    t = X[:, 0]
    Q0 = X[:, 1]
    alpha = 0.05  # recession constant [1/day], typical value
    return Q0 * np.exp(-alpha * t)
