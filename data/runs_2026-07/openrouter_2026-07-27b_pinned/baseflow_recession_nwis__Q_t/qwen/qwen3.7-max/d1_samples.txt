import numpy as np

def predict(X):
    t = X[:, 0]
    Q_0 = X[:, 1]
    
    # Standard exponential recession model (Maillet, 1905)
    # Q(t) = Q_0 * exp(-alpha * t)
    alpha = 0.05  # Typical baseflow recession constant [1/day]
    
    Q_t = Q_0 * np.exp(-alpha * t)
    return Q_t
