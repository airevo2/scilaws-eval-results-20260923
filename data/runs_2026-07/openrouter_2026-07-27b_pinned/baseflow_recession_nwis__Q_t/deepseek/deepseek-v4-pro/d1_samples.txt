import numpy as np

def predict(X):
    """
    Predict daily-mean streamflow discharge Q(t) during a recession event.
    
    Uses the standard exponential recession equation:
    Q(t) = Q_0 * exp(-alpha * t)
    
    where alpha is the recession constant (day^-1).
    A typical value for alpha is around 0.1 day^-1, which represents
    a moderately fast recession. This can vary based on catchment characteristics.
    """
    t = X[:, 0]
    Q_0 = X[:, 1]
    
    # Recession constant (day^-1) - typical value
    alpha = 0.1
    
    Q_t = Q_0 * np.exp(-alpha * t)
    
    return Q_t

import numpy as np

def predict(X):
    """
    Predict daily-mean streamflow discharge Q(t) during a recession event.
    
    Uses the standard exponential decay model (Maillet's law):
        Q(t) = Q_0 * exp(-alpha * t)
    
    Parameters
    ----------
    X : ndarray of shape (n_samples, 2)
        X[:,0] : t   (days since start of recession, integer)
        X[:,1] : Q_0 (discharge at t=0, ft3/s)
    
    Returns
    -------
    Qt : ndarray of shape (n_samples,)
        Predicted discharge at time t, ft3/s.
    """
    t = X[:, 0]
    Q0 = X[:, 1]
    # Typical baseflow recession coefficient (1/day); in practice this would be
    # calibrated for a given watershed. A commonly cited value is around 0.05.
    ALPHA = 0.05  
    Qt = Q0 * np.exp(-ALPHA * t)
    return Qt
