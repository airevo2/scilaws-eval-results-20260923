import numpy as np

def predict(X):
    # Holling Type II functional response (disc equation)
    # Na = a*T*N / (1 + a*T*h*N)
    N = X[:, 0]
    a = 1.0      # attack rate
    T = 1.0      # total time
    h = 0.01     # handling time
    Na = (a * T * N) / (1.0 + a * T * h * N)
    return Na
