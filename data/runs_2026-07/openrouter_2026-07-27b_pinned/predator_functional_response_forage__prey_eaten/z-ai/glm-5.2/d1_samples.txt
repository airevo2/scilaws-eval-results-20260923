import numpy as np

def predict(X):
    # Holling Type II functional response (disk equation)
    N = X[:, 0]
    a = 0.5   # attack rate
    h = 0.2   # handling time
    T = 1.0   # total experimental trial time
    
    Na = (a * T * N) / (1.0 + a * h * N)
    return Na

import numpy as np

def predict(X):
    """
    Predicts the number of prey eaten (N_a) using Holling's Type II 
    functional response (the disc equation).
    
    N_a = (a * N * T) / (1 + a * h * N)
    
    where:
    N   = initial prey density (X[:, 0])
    a   = attack rate (instantaneous rate of discovery)
    h   = handling time per prey
    T   = total experimental time
    """
    N = X[:, 0]
    
    # Fitted coefficients (example constants)
    a = 0.5   # attack rate
    h = 0.1   # handling time
    T = 1.0   # total time available for predation
    
    Na = (a * N * T) / (1.0 + a * h * N)
    
    return Na
