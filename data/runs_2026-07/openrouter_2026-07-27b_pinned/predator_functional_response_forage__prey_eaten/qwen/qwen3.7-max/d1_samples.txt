import numpy as np

def predict(X):
    N = X[:, 0]
    # Holling's Type II functional response (disc equation, Holling 1959)
    # N_a = (a * N * T) / (1 + a * h * N)
    a = 0.02   # attack rate (searching efficiency)
    h = 0.01   # handling time per prey item
    T = 1.0    # total time available in one trial
    N_a = (a * N * T) / (1.0 + a * h * N)
    return N_a
