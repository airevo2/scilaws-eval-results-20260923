import numpy as np

def predict(X):
    """
    Predict prey eaten (Na) from prey density (N) using the Holling Type II
    functional response (disc equation).

    Parameters
    ----------
    X : np.ndarray, shape (n_samples, 1)
        X[:, 0] = prey density (N)

    Returns
    -------
    Na : np.ndarray, shape (n_samples,)
        Number of prey eaten during the trial.
    """
    # Coefficients (attack rate and handling time)
    a = 0.1   # attack rate
    Th = 0.05 # handling time (in units of trial duration)

    N = X[:, 0]
    Na = (a * N) / (1.0 + a * Th * N)
    return Na

import numpy as np

def predict(X):
    """
    Predicts the number of prey eaten (N_a) using Holling's Type II functional response
    (disc equation) with total trial time T = 1.
    
    Parameters:
    X : 2-D numpy array, X[:, 0] = prey density (N)
    
    Returns:
    1-D array of predicted prey eaten.
    """
    N = X[:, 0]
    # Attack rate (per prey per unit time) and handling time (time units per prey)
    ATTACK_RATE = 1.0      # a
    HANDLING_TIME = 0.1    # h
    # Holling's disc equation: N_a = (a * N) / (1 + a * h * N)
    return (ATTACK_RATE * N) / (1.0 + ATTACK_RATE * HANDLING_TIME * N)
