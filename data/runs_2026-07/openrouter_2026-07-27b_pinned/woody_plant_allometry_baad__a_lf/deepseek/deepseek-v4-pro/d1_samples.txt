import numpy as np

def predict(X):
    """
    Predict total whole-plant leaf area A_L [m2] from sapwood area at breast height
    A_S [m2] and total plant height H [m] using the pipe model relationship.
    
    A_L = k * A_S^b * H^c
    or the linear pipe model form:
    A_L = k * A_S  (when b=1, c=0)
    
    Here we use the commonly cited allometric form with height correction:
    A_L = alpha * A_S * H^beta
    """
    a_s = X[:, 0]  # sapwood area at breast height
    h_t = X[:, 1]  # total height
    
    # Pipe model allometric coefficients (representative values from literature)
    alpha = 0.5   # scaling constant
    beta = 0.5    # height exponent (often near 0.5 when A_L ~ A_S * sqrt(H))
    
    a_lf = alpha * a_s * (h_t ** beta)
    
    return a_lf

import numpy as np

def predict(X):
    """
    Predict total whole-plant leaf area (A_L) from sapwood cross-sectional area
    at breast height (A_S) and total plant height (H).

    The most commonly cited closed-form relationship is the pipe model
    (Shinozaki et al. 1964): A_L = k * A_S, where k is the leaf area to
    sapwood area ratio. Typical values of k for trees range from 1000 to 6000
    (dimensionless). This function uses a representative constant of 4000.
    Height (X[:,1]) is not required in the canonical pipe model but is
    accepted as an input to match the given interface.
    """
    A_S = X[:, 0]  # sapwood area at breast height [m2]
    k = 4000.0     # leaf area : sapwood area ratio [m2 m-2]
    A_L = k * A_S
    return A_L
