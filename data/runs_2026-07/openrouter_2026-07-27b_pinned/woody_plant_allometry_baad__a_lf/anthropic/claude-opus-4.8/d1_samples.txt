import numpy as np

def predict(X):
    """
    Leaf area from sapwood area (pipe / Huber-value model) with a height
    correction reflecting the hydraulic path-length effect:

        A_L = a * A_S^b * H^c

    Sapwood area (A_S) is the dominant predictor (Shinozaki's pipe model:
    A_L proportional to A_S), while total height (H) captures the observed
    decline of the leaf-area:sapwood-area ratio with plant size.
    """
    A_S = X[:, 0]   # sapwood area at breast height [m2]
    H   = X[:, 1]   # total height [m]

    a = 5000.0   # scaling (leaf-area:sapwood-area ratio, m2 leaf per m2 sapwood)
    b = 1.0      # pipe-model exponent (near-proportional to sapwood area)
    c = -0.3     # hydraulic path-length correction (ratio declines with height)

    A_L = a * np.power(A_S, b) * np.power(H, c)
    return A_L

def predict(X):
    # Pipe-model (Shinozaki) relationship: whole-plant leaf area scales with
    # sapwood cross-sectional area at breast height, with a secondary
    # (hydraulic path-length) dependence on total plant height.
    #   A_L = k * A_S^b_S * H^b_H
    A_S = X[:, 0]        # sapwood area at breast height [m2]
    H   = X[:, 1]        # total height [m]

    k    = 5.0e3         # Huber-value proportionality constant
    b_S  = 0.90          # near-linear (pipe-model) scaling with sapwood area
    b_H  = -0.20         # slight decline in leaf area per sapwood with height

    A_L = k * np.power(A_S, b_S) * np.power(H, b_H)
    return A_L

    # Pipe-model based allometry (Shinozaki et al.; extended with height dependence,
    # as commonly fitted to BAAD data):

    # Leaf area scales ~linearly with sapwood area (pipe model), with a mild
    # decline in the leaf-area-to-sapwood-area ratio as trees grow taller.

    H   = X[:, 1]      # total height [m]

    c = 2.0e3          # scaling coefficient
    a = 1.0            # ~proportional to sapwood area (pipe model)
    b = -0.5           # A_L:A_S ratio decreases with height

    A_L = c * np.power(A_S, a) * np.power(H, b)
    return A_L

import numpy as np

def predict(X):
    # Pipe-model (Shinozaki) relationship: whole-plant leaf area scales with
    # sapwood cross-sectional area, with a modest height correction.
    #   A_L = k * A_S^b * H^c
    # Fitted (BAAD-typical) coefficients on a log-log basis.
    A_S = np.asarray(X[:, 0], dtype=float)
    H   = np.asarray(X[:, 1], dtype=float)

    k = 3.0e3   # Huber-value related proportionality constant [m2 leaf / m2 sapwood]
    b = 1.0     # near-linear pipe-model exponent on sapwood area
    c = -0.3    # leaf area per sapwood area declines slightly with height

    A_L = k * np.power(A_S, b) * np.power(H, c)
    return A_L
