def predict(X):
    A_S = X[:, 0]
    H = X[:, 1]
    
    # Standard multivariate allometric power-law relationship
    # Based on the Pipe Model Theory, A_L is primarily proportional to A_S
    beta_0 = 1.0  # Allometric intercept / coefficient
    beta_1 = 1.0  # Scaling exponent for sapwood area
    beta_2 = 0.0  # Scaling exponent for height

    # with an allometric height correction for hydraulic/crown scaling.
    #
    # Coefficients derived from canonical BAAD-style log-log regression:
    #   log(A_L) = beta_0 + beta_1 * log(A_S) + beta_2 * log(H)
    beta_0 = 7.10   # log-intercept (natural log)
    beta_1 = 0.85   # sapwood area scaling exponent (near pipe-model value of 1)
    beta_2 = 0.20   # height scaling exponent
    
    A_L = np.exp(beta_0) * (A_S ** beta_1) * (H ** beta_2)
    return A_L

import numpy as np

def predict(X):
    A_S = X[:, 0]  # Sapwood cross-sectional area at breast height [m2]
    H = X[:, 1]    # Total plant height [m]
    
    # Pipe model (Shinozaki et al., 1964) with allometric height correction.
    # A_L = beta_0 * A_S^beta_1 * H^beta_2
    # beta_1 = 1.0 reflects the pipe model (leaf area proportional to sapwood area).
    # beta_2 = 0.5 captures the residual positive scaling of the leaf-area-to-
    #          sapwood-area ratio with tree height observed across woody plants.
    # beta_0 = 1000 is calibrated so that A_L/A_S ~ 1000*sqrt(H) falls within
    #          the typical Huber-value inverse range (2000-8000 m2/m2) for
    #          common tree heights.
    beta_0 = 1000.0
    beta_1 = 1.0
    beta_2 = 0.5
    
    A_L = beta_0 * (A_S ** beta_1) * (H ** beta_2)
    return A_L
