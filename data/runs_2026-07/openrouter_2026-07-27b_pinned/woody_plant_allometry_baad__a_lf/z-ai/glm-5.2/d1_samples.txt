import numpy as np

def predict(X):
    """
    Predicts total whole-plant leaf area (A_L) based on the pipe model theory 
    with hydraulic limitation (where the leaf area to sapwood area ratio 
    decreases with plant height).
    
    Canonical form:
    A_L = beta_0 * (A_S ^ beta_1) * (H ^ beta_2)
    
    Inputs:
    X[:, 0] = A_S (sapwood area at breast height) [m2]
    X[:, 1] = H (total plant height) [m]
    """
    A_S = X[:, 0]
    H = X[:, 1]
    
    # Coefficients based on typical allometric scaling for trees
    # A_L / A_S is typically on the order of 2000-4000 m2/m2
    beta_0 = 10000.0
    beta_1 = 1.0
    beta_2 = -0.5
    
    # Ensure no division by zero or negative powers on zero
    A_S = np.where(A_S <= 0, 1e-9, A_S)
    H = np.where(H <= 0, 1e-9, H)
    
    A_L = beta_0 * np.power(A_S, beta_1) * np.power(H, beta_2)
    
    return A_L
