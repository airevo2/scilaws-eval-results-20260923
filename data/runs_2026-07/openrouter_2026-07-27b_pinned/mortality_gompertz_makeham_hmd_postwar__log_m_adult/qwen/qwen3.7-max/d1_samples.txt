import numpy as np

def predict(X):
    age = X[:, 0]
    year = X[:, 1]
    
    # Gompertz law of mortality with a linear period (time) trend
    # ln(m_{x,t}) = a + b * x + c * (t - t_ref)
    
    a = -9.0       # Extrapolated baseline log mortality at age 0 in reference year
    b = 0.085      # Gompertz slope (rate of aging, typically ~0.08-0.09 for humans)
    c = -0.015     # Annual mortality improvement rate (period effect)
    t_ref = 2000.0 # Reference calendar year
    
    ln_m = a + b * age + c * (year - t_ref)
    
    return ln_m
