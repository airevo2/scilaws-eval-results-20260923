import numpy as np

def predict(X):
    # Shinozaki–Kira reciprocal yield–density law (law of constant final yield):
    #   1/w = a + b*N   =>   w = 1 / (a + b*N)
    # where w = mean dry mass per plant (g), N = plants per pot.
    N = np.asarray(X[:, 0], dtype=float)

    a = 0.05   # intercept: 1 / (mass of an isolated plant at N -> 1)
    b = 0.02   # slope: reflects competition intensity per added plant

    w = 1.0 / (a + b * N)
    return w
