import numpy as np

def predict(X):
    """
    Predicts mean oven-dry mass per surviving plant (w) based on plant density (N).
    Uses the canonical self-thinning rule (Yoda's -3/2 power law).
    """
    N = X[:, 0]
    # Parameter 'a' absorbs the constant scalar converting plants/pot to plants/m^2
    a = 1.0  # fitted coefficient (can be adjusted)
    
    w = a * np.power(N, -1.5)
    return w
