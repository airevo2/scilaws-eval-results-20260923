import numpy as np

def predict(X):
    """
    Predict mean oven-dry mass per surviving plant (g/plant) using the
    self-thinning rule (Yoda's -3/2 power law):
        w = a * N ^ (-3/2)
    where N is the surviving plant density in plants/pot.
    Parameters a and exponent are fitted constants.
    """
    N = X[:, 0]                 # density (plants/pot)
    a = 100.0                   # fitted scaling constant (g/plant * (plants/pot)^1.5)
    b = -1.5                    # canonical self-thinning exponent
    w = a * N ** b
    return w
