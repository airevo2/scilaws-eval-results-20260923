import numpy as np

def predict(X):
    """
    Predict the natural log of the central period death rate (ln m) for total sex,
    single-year age resolution, using the standard Age-Period bilinear model.
    This is a common parametric closed-form used in mortality forecasting and
    corresponds to a linearised Lee-Carter model.

    Parameters:
    X : np.ndarray, shape (n, 2)
        X[:, 0] = age (x) in years, integer from 30 to 95.
        X[:, 1] = year (t) in calendar years, from 1751 to 2025.

    Returns:
    np.ndarray, shape (n,): ln m for each observation.
    """
    age = X[:, 0].astype(np.float64)
    year = X[:, 1].astype(np.float64)

    # Reference year for numerical stability
    t = year - 1900.0

    # Fitted coefficients for total-sex adult mortality (ages 30-95)
    # Typical values from historical period life tables.
    b0 = -7.0          # baseline log mortality at age 0 in year 1900
    b_age = 0.09       # increase in log mortality per year of age
    b_year = -0.025    # average annual change in log mortality at age 0
    b_ay = 0.00015     # age-year interaction: older ages decline more slowly

    ln_m = b0 + b_age * age + b_year * t + b_ay * age * t
    return ln_m

    # ln(m) = a + b * age + c * year
    # Coefficients are typical for total-sex, single-year adult mortality (ages 30–95).
    a = -7.0      # baseline log-mortality level
    b =  0.09     # Gompertz slope (increase in ln m per year of age)
    c = -0.012    # secular decline in ln m per calendar year
    age = X[:, 0]
    year = X[:, 1]
    return a + b * age + c * year

    Parameters
    ----------
    X : 2-D numpy array
        Columns:
        X[:, 0] = age (x) in years
        X[:, 1] = year (t) in calendar years

    Returns
    -------
    log_m : 1-D numpy array
        Natural log of the central death rate (ln m)
    """
    age = X[:, 0]
    year = X[:, 1]
    
    # ----- Fitted coefficients (typical values for total sex adult mortality) -----
    # Intercept of the Gompertz level, anchored around year 2000
    alpha = -10.5     
    # Gompertz slope (log-mortality increase per year of age)
    beta = 0.09       
    # Linear time trend (mortality decline per calendar year)
    gamma = -0.012    
    # Reference year for the time trend
    t0 = 2000.0
    
    # Standard closed-form: ln(m) = alpha + beta * age + gamma * (year - t0)
    log_m = alpha + beta * age + gamma * (year - t0)
    return log_m

import numpy as np

def predict(X):
    """
    Standard Gompertz law with linear time trend for the natural log of the adult
    central death rate (total sex, 1×1 configuration).

    ln(m_x,t) = alpha + beta * age + gamma * year

    Parameters are chosen to reflect broadly typical adult mortality levels
    and a secular decline over calendar time.
    """
    alpha = -12.0   # intercept
    beta  =  0.09   # age slope (Gompertz parameter ln(c))
    gamma = -0.01   # time slope (annual rate of mortality decline)

    return alpha + beta * X[:, 0] + gamma * X[:, 1]
