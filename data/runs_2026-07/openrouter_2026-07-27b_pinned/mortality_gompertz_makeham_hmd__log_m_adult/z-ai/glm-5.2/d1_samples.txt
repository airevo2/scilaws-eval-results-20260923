import numpy as np

def predict(X):
    """
    Predicts the natural log of the central period death rate (ln m) 
    for adult ages using the Gompertz law of mortality with a linear time trend.
    
    The standard Gompertz model posits that mortality increases exponentially 
    with age. To account for period improvements over time, the baseline 
    mortality parameter is modeled as a linear function of the calendar year.
    
    Model: ln(m(x, t)) = a(t) + b * x
    where: a(t) = a0 + a1 * (t - t0)
    
    Inputs:
    X[:, 0] = age (x) in years
    X[:, 1] = year (t) in years
    
    Returns:
    1-D numpy array of predicted ln(m)
    """
    age = X[:, 0]
    year = X[:, 1]
    
    # Representative fitted constants for human adult mortality
    # b is the rate of aging (Gompertz slope), typically ~0.08 to 0.11
    b = 0.095
    
    # Baseline mortality intercept parameters
    t0 = 1900.0  # reference year
    a0 = -8.5    # baseline intercept at t0
    a1 = -0.012  # annual rate of mortality decline
    
    # Compute ln(m)
    log_m = (a0 + a1 * (year - t0)) + b * age
    
    return log_m
