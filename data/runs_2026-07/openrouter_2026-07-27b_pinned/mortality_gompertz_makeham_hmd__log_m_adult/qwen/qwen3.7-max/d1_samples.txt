    # ln m(x, t) = alpha + beta * x + gamma * t
    alpha = 12.5     # Intercept (calibrated for absolute year)
    beta = 0.085     # Gompertz aging rate (log-linear slope with age)
    gamma = -0.01    # Period mortality improvement rate (time trend)
    
    ln_m = alpha + beta * age + gamma * year
    return ln_m
