import numpy as np

def predict(X):
    """
    Predict ln M(x,t) using the Lee-Carter mortality model,
    with approximate parameters for total sex US population.
    Reference: Lee, R.D. and Carter, L.R. (1992) "Modeling and Forecasting U.S. Mortality".
    """
    age = X[:, 0].astype(int)
    year = X[:, 1]

    # Knot ages for interpolation
    knot_ages = np.array([0, 1, 5, 10, 15, 20, 25, 35, 45, 55, 65, 75, 85, 95])

    # Target log mortality for year 2000 (k_t = -7) at knot ages
    target_log_m_2000 = np.log(np.array([
        0.007,    # age 0
        0.0003,   # age 1
        0.00015,  # age 5
        0.00015,  # age 10
        0.0008,   # age 15
        0.001,    # age 20
        0.0012,   # age 25
        0.002,    # age 35
        0.005,    # age 45
        0.012,    # age 55
        0.03,     # age 65
        0.08,     # age 75
        0.18,     # age 85
        0.3       # age 95
    ]))

    # b_x values at knot ages (age-specific sensitivity to k_t)
    b_knots = np.array([
        0.020,   # age 0
        0.018,   # age 1
        0.015,   # age 5
        0.015,   # age 10
        0.020,   # age 15
        0.018,   # age 20
        0.018,   # age 25
        0.015,   # age 35
        0.012,   # age 45
        0.010,   # age 55
        0.008,   # age 65
        0.007,   # age 75
        0.006,   # age 85
        0.005    # age 95
    ])

    # Compute a_x at knots: a_x = target_log_m_2000 + 7 * b_x
    # (since k_t for year 2000 is set to -7)
    a_knots = target_log_m_2000 + 7.0 * b_knots

    # Interpolate a_x and b_x to all integer ages 0..95
    all_ages = np.arange(96)
    a_x = np.interp(all_ages, knot_ages, a_knots)
    b_x = np.interp(all_ages, knot_ages, b_knots)

    # Time-varying mortality index (linear decline with drift -0.35 per year)
    k_t = -0.35 * (year - 1980.0)

    # Lee-Carter model: ln m(x,t) = a_x + b_x * k_t
    log_m = a_x[age] + b_x[age] * k_t

    return log_m

import numpy as np

def predict(X):
    # Extract age and year
    x = X[:, 0]
    t = X[:, 1]
    
    # Reference year for time trend
    t0 = 2000.0
    
    # Gompertz-Makeham baseline with age and time components
    # ln M(x,t) = a + b*x + c*exp(d*x) + e*(t - t0) + f*(t - t0)^2 + g*x*(t - t0)
    # This is the standard Lee-Carter type extension of the Gompertz model
    # for central death rates over full age range
    
    a = -8.5       # intercept / Makeham constant
    b = 0.085      # linear age slope (senescence)
    c = 1.2e-5     # Gompertz exponential coefficient
    d = 0.095      # Gompertz exponential rate
    e = -0.012     # linear time trend (mortality improvement)
    f = -0.00008   # quadratic time trend (acceleration of improvement)
    g = -0.0006    # age-time interaction (differential improvement by age)
    
    # Compute log mortality
    log_m_full = a + b * x + c * np.exp(d * x) + e * (t - t0) + f * (t - t0)**2 + g * x * (t - t0)
    
    return log_m_full

import numpy as np

def predict(X):
    age = X[:, 0]
    year = X[:, 1]
    
    # Reference year for baseline parameters
    t0 = 1900.0
    
    # Siler model parameters with time-varying levels
    # Infant/child mortality component
    a1 = 0.015 * np.exp(-0.01 * (year - t0))
    b1 = 1.5
    
    # Background mortality (age-independent)
    a2 = 0.0005 * np.exp(-0.02 * (year - t0))
    
    # Senescent mortality component
    a3 = 0.00015 * np.exp(-0.01 * (year - t0))
    b3 = 0.085
    
    # Central death rate from the Siler competing-hazards model
    m = a1 * np.exp(-b1 * age) + a2 + a3 * np.exp(b3 * age)
    
    # Natural log of the central death rate
    log_m = np.log(m)
    return log_m
