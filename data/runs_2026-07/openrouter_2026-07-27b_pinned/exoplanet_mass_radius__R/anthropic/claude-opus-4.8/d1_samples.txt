import numpy as np

def predict(X):
    # Chen & Kipping (2017) "Forecaster" mass-radius relation (in Earth units)
    # log10(R) = C + S * log10(M), piecewise in mass
    M = np.asarray(X[:, 0], dtype=float)

    # Transition masses (M_earth)
    T1 = 2.04      # Terran -> Neptunian
    T2 = 131.58    # Neptunian -> Jovian (0.414 M_Jup)
    T3 = 26600.0   # Jovian -> Stellar

    # Slopes and intercepts (log10)
    def branch(logM, C, S):
        return C + S * logM

    logM = np.log10(M)
    logR = np.empty_like(logM)

    terran    = M < T1
    neptunian = (M >= T1) & (M < T2)
    jovian    = (M >= T2) & (M < T3)
    stellar   = M >= T3

    logR[terran]    = branch(logM[terran],    0.00346,  0.2790)
    logR[neptunian] = branch(logM[neptunian], -0.0925,  0.5890)
    logR[jovian]    = branch(logM[jovian],     1.2500, -0.0440)
    logR[stellar]   = branch(logM[stellar],   -2.8500,  0.8810)

    return np.power(10.0, logR)
