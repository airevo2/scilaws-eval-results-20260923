import numpy as np

def predict(X):
    """
    Predicts Log Reduction Value (LRV) using the Chick-Watson kinetic model.
    Assuming the coefficient of dilution (n) = 1, the model simplifies to:
    LRV = k * C * t = k * CT
    where k is the inactivation rate constant.
    """
    CT = X[:, 0]
    k = 0.01  # Example fitted inactivation rate constant
    LRV = k * CT
    LRV = np.maximum(0.0, LRV)  # Ensure LRV is always non-negative
    return LRV

import numpy as np

def predict(X):
    """
    Predicts Log10 microbial reduction (LRV) based on the Chick-Watson model.
    
    The Chick-Watson kinetic model is given by:
    log10(N0/N) = k * C^n * t
    
    where N0 is the initial microbial count, N is the surviving count, 
    C is the disinfectant concentration, t is the contact time, 
    k is the inactivation rate constant, and n is the dilution coefficient.
    
    Given only CT (C * t) as input, we assume the commonly cited case where n = 1,
    reducing the equation to:
    LRV = k * CT
    """
    # Extract CT (Disinfection dose)
    CT = X[:, 0]
    
    # Inactivation rate constant (k) - typical generic fitted coefficient
    k = 0.05 
    
    # Chick-Watson model with n=1
    LRV = k * CT
    
    return LRV
