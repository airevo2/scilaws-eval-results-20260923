def predict(X):
    """
    Predict log reduction value (LRV) from CT using the Chick-Watson model.
    LRV = k * CT, where k = 1/CT_1log.
    CT_1log is the CT required for 1-log inactivation (a typical value for Giardia
    cysts with free chlorine at 5°C, pH 7 is ~50 mg·min/L).
    """

    return X[:, 0] / CT_1log

import numpy as np

def predict(X):
    """
    Predicts log reduction value (LRV) from CT using the Chick-Watson model.

    The standard first-order Chick-Watson disinfection model:
        log10(N0/N) = k * C * t = k * CT
    where:
        N0 = initial microbial count
        N  = surviving microbial count
        C  = disinfectant residual concentration (mg/L)
        t  = contact time (min)
        CT = C * t (mg/L * min)
        k  = disinfection rate constant (L/mg/min)

    Parameters
    ----------
    X : np.ndarray, shape (n_samples, 1)
        X[:, 0] = CT values in mg/L*min.

    Returns
    -------
    lrv : np.ndarray, shape (n_samples,)
        Predicted log reduction values.
    """
    # Disinfection rate constant (organism- and disinfectant-specific).
    # A placeholder value of 1.0 is used here; in practice this would be
    # calibrated to experimental data.
    k = 1.0  # L/(mg*min)

    CT = X[:, 0]
    lrv = k * CT
    return lrv
