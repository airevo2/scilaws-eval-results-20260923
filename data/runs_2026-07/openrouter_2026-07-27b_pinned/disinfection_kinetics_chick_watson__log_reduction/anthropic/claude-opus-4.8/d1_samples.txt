import numpy as np

def predict(X):
    # Chick-Watson model: LRV = k * CT (base-10 log inactivation)
    # log10(N0/N) = (k_CW / ln(10)) * C^n * t, with n=1 => proportional to CT
    # Use a representative disinfection rate constant (L/mg/min)
    CT = np.asarray(X[:, 0], dtype=float)
    k = 0.10  # inactivation rate coefficient [L/(mg*min)] in log10 units
    lrv = k * CT
    return np.maximum(lrv, 0.0)
