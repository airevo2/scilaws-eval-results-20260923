import numpy as np

def predict(X):
    """
    Chick-Watson model for microbial inactivation.
    
    LRV = k_CW * CT
    
    where k_CW is the base-10 Chick-Watson inactivation rate constant
    and CT is the disinfection dose (concentration × contact time).
    Assumes coefficient of dilution n = 1 (standard simplification).
    """
    CT = X[:, 0]
    
    # Chick-Watson base-10 rate constant [L/(mg·min)]
    # Representative generic value; organism/disinfectant specific in practice
    k_CW = 1.0
    
    LRV = k_CW * CT
    
    return np.maximum(LRV, 0.0)
