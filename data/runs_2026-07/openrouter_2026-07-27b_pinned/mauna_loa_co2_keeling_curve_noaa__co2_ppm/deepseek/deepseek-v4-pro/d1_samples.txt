import numpy as np

def predict(X):
    """
    Predict monthly mean CO2 mole fraction at Mauna Loa (ppm, WMO X2019 scale)
    using the standard Keeling curve model: a polynomial secular trend plus
    a four-harmonic seasonal cycle.

    The model uses only the decimal year (X[:,0]); all other inputs are distractors.

    The fitted coefficients below are rounded approximations derived from
    the long-term Mauna Loa record (1958–2024) and are typical of the
    closed-form parameterisations used in the literature.
    """
    t = X[:, 0]                     # decimal year
    t0 = 1958.0                     # reference year

    # Polynomial trend coefficients (quadratic + cubic)
    a0 = 315.42                     # mean CO2 at t0 (ppm)
    a1 = 0.684                      # linear trend (ppm/yr)
    a2 = 0.0087                     # quadratic term (ppm/yr²)
    a3 = 0.00005                    # cubic term (ppm/yr³) – small but often included

    trend = a0 + a1*(t - t0) + a2*(t - t0)**2 + a3*(t - t0)**3

    # Four-harmonic seasonal cycle coefficients (amplitude in ppm)
    # b_k = sine coefficients, c_k = cosine coefficients for harmonic k
    b = np.array([2.90, 0.50, 0.15, 0.05])
    c = np.array([0.80, 0.10, 0.05, 0.02])

    w = 2.0 * np.pi
    seasonal = (
        b[0] * np.sin(w * t)     + c[0] * np.cos(w * t) +
        b[1] * np.sin(2*w * t)   + c[1] * np.cos(2*w * t) +
        b[2] * np.sin(3*w * t)   + c[2] * np.cos(3*w * t) +
        b[3] * np.sin(4*w * t)   + c[3] * np.cos(4*w * t)
    )

    return trend + seasonal

import numpy as np

def predict(X):
    """
    Standard closed-form relationship for monthly mean CO2 at Mauna Loa,
    consisting of a quadratic long-term trend plus annual and semi-annual
    harmonic cycles.  Uses only year_decimal (X[:,0]) as predictor.
    """
    t = X[:, 0]   # decimal year

    # Quadratic trend anchored at 1958.0
    t0 = 1958.0
    a0 = 315.2      # ppm at t0
    a1 = 0.685      # linear term [ppm/yr]
    a2 = 0.0125     # quadratic term [ppm/yr^2]
    trend = a0 + a1 * (t - t0) + a2 * (t - t0)**2

    # Annual harmonic (amplitude ~2.9 ppm, peak in mid-May)
    amp_ann = 2.9
    annual = amp_ann * np.sin(2 * np.pi * (t - 0.125))

    # Semi-annual harmonic (amplitude ~0.45 ppm)
    amp_semi = 0.45
    semi_annual = amp_semi * np.sin(4 * np.pi * (t - 0.05))

    return trend + annual + semi_annual

    Coefficients are approximate constants for the WMO X2019 scale over 1958–2026.
    Only uses year_decimal (X[:,0]); other columns are ignored.

    # Reference year for the quadratic polynomial
    t_ref = 1958.0
    dt = t - t_ref

    # --- Quadratic trend coefficients ---
    C0 = 315.5                      # ppm at 1958
    C1 = 0.69                       # ppm / year linear term
    C2 = 0.0145                     # ppm / year^2 acceleration

    # --- Seasonal harmonics ---

    A1_sin =  1.98                  # coefficient for sin(2πt)

import numpy as np

def predict(X):
    """
    Standard closed-form relationship for Mauna Loa CO2 using the NOAA baseline
    functional form: a quadratic trend plus a seasonal cycle of four harmonics.
    The single physical predictor is year_decimal t = X[:,0].
    Coefficients are approximate values that yield realistic CO2 concentrations.
    """
    t = X[:, 0]                     # decimal year
    t0 = 2000.0                     # reference year for numerical stability
    dt = t - t0

    # Quadratic trend coefficients (roughly match Mauna Loa record)
    a0 = 369.3                      # [ppm] baseline at t0 = 2000.0
    a1 = 2.0                        # [ppm/yr] linear growth rate at t0
    a2 = 0.0125                     # [ppm/yr^2] acceleration of growth rate

    # Seasonal harmonic coefficients (annual, semi-annual, tri-annual, quad-annual)
    # Amplitudes chosen to ~ 6 ppm peak-to-peak seasonal cycle
    b1, c1 = 2.9, -0.2             # 1 cycle/yr
    b2, c2 = 0.8, -0.7             # 2 cycles/yr
    b3, c3 = 0.3, 0.3              # 3 cycles/yr
    b4, c4 = 0.1, 0.2              # 4 cycles/yr

    # Trend component
    trend = a0 + a1 * dt + a2 * dt**2

    # Seasonal component
    seasonal = (b1 * np.sin(2.0 * np.pi * t) + c1 * np.cos(2.0 * np.pi * t) +
                b2 * np.sin(4.0 * np.pi * t) + c2 * np.cos(4.0 * np.pi * t) +
                b3 * np.sin(6.0 * np.pi * t) + c3 * np.cos(6.0 * np.pi * t) +
                b4 * np.sin(8.0 * np.pi * t) + c4 * np.cos(8.0 * np.pi * t))

    co2_ppm = trend + seasonal
    return co2_ppm
