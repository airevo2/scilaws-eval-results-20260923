import numpy as np

def predict(X):
    t = X[:, 0]
    # Reference epoch
    t0 = 1958.0
    dt = t - t0
    # Quadratic secular trend (Keeling curve)
    c0 = 315.0
    a = 0.80      # linear ppm/yr
    b = 0.0125    # quadratic ppm/yr^2
    trend = c0 + a * dt + b * dt**2
    # Seasonal cycle (annual + semiannual harmonics)
    seasonal = (2.8 * np.cos(2 * np.pi * (t - 1958.0)) +
                1.2 * np.sin(2 * np.pi * (t - 1958.0)) +
                0.6 * np.cos(4 * np.pi * (t - 1958.0)) +
                0.4 * np.sin(4 * np.pi * (t - 1958.0)))
    return trend + seasonal
