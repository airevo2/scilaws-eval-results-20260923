import numpy as np

def predict(X):
    # X[:, 0] is the decimal year (t)
    t = X[:, 0]
    
    # Time elapsed since the start of the Mauna Loa record (1958)
    dt = t - 1958.0
    
    # Long-term trend: Quadratic fit to the annual mean CO2 growth
    # ~315 ppm in 1958, accelerating to ~420 ppm by the 2020s
    trend = 315.0 + 0.85 * dt + 0.0115 * dt**2
    
    # Seasonal cycle: ~3 ppm amplitude (6 ppm peak-to-peak)
    # Maximum CO2 typically occurs in May (fractional year ~0.37)
    # sin(2*pi*(t - phase)) peaks at t - phase = 0.25 => phase = 0.37 - 0.25 = 0.12
    seasonal = 3.0 * np.sin(2 * np.pi * (t - 0.12))
    
    # Total predicted CO2 mole fraction
    c = trend + seasonal
    
    return c

import numpy as np

def predict(X):
    t = X[:, 0]
    tau = t - 1958.0
    
    # Long-term trend (quadratic approximation of the Keeling curve)
    trend = 315.0 + 0.75 * tau + 0.013 * tau**2

    amplitude = 3.0 + 0.05 * tau
    seasonal = amplitude * np.sin(2 * np.pi * (t - 0.12))
    
    return trend + seasonal
