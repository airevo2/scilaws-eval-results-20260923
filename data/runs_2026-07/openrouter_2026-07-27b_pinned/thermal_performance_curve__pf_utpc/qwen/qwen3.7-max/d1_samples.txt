import numpy as np

def predict(X):
    """
    Predicts biological performance (P) as a function of temperature (T) 
    using the Sharpe-Schoolfield equation (with high-temperature inactivation),
    the canonical mechanistic model in thermal physiology for rate-like traits.
    """
    T_C = X[:, 0]
    T_K = T_C + 273.15
    
    # Boltzmann constant (eV / K)
    k = 8.62e-5 
    
    # Typical canonical parameters for biological thermal performance curves
    P_ref = 1.0       # Scale parameter (absorbs trait-specific units)
    T_ref = 293.15    # Reference temperature (20 °C in Kelvin)
    E = 0.60          # Average activation energy (eV)
    E_h = 4.00        # High-temperature deactivation energy (eV)
    T_h = 303.15      # Transition temperature for high-T inactivation (30 °C in Kelvin)
    
    # Arrhenius term (exponential increase driven by kinetic energy)
    arrhenius = P_ref * np.exp(-E / k * (1.0 / T_K - 1.0 / T_ref))
    
    # Inactivation term (sigmoidal decrease due to protein denaturation at high T)
    inactivation = 1.0 + np.exp(-E_h / k * (1.0 / T_K - 1.0 / T_h))
    
    # Combined performance
    P = arrhenius / inactivation
    
    # Biological rates cannot be negative
    P = np.maximum(P, 0.0)
    
    return P

    Predicts biological performance (P) as a function of temperature

    thermal physiology for rate-like traits incorporating both 
    Arrhenius activation and high-temperature enzyme deactivation.

    # Boltzmann constant (eV/K)
    k = 8.62e-5
    
    # Default fitted coefficients / named constants
    R_tref = 1.0    # Scale parameter (absorbs trait-specific units)
    E = 0.65        # Activation energy (eV)

    T_ref = 293.15  # Reference temperature (20°C in Kelvin)
    T_h = 303.15    # Temperature at which 50% deactivation occurs (30°C in Kelvin)

    numerator = R_tref * np.exp((E / k) * (1.0 / T_ref - 1.0 / T_K))
    denominator = 1.0 + np.exp((E_h / k) * (1.0 / T_h - 1.0 / T_K))
    
    P = numerator / denominator
