import numpy as np

def predict(X):
    # Sharpe-Schoolfield thermal performance model (high-temperature deactivation)
    T = np.asarray(X[:, 0], dtype=float) + 273.15  # deg C -> Kelvin

    k = 8.617333262e-5   # Boltzmann constant [eV/K]

    # Fitted / typical coefficients
    B0   = 1.0    # rate scale (unit-absorbing constant per cluster)
    E    = 0.65   # activation energy [eV]
    Eh   = 2.0    # high-temperature deactivation energy [eV]
    Th   = 305.0  # temperature of half deactivation [K]
    Tref = 283.15 # reference temperature [K]

    # Arrhenius rise
    rise = np.exp((E / k) * (1.0 / Tref - 1.0 / T))
    # High-temperature inactivation
    decay = 1.0 + np.exp((Eh / k) * (1.0 / Th - 1.0 / T))

    P = B0 * rise / decay
    return P

import numpy as np

def predict(X):
    # Sharpe-Schoolfield thermal performance model (Boltzmann-Arrhenius with high-T inactivation)
    T = X[:, 0] + 273.15          # convert Celsius to Kelvin
    k = 8.617333262145e-5         # Boltzmann constant [eV/K]

    # Fitted / typical constants
    P0   = 1.0        # scale parameter (per-cluster unit absorbs actual scale)
    E    = 0.65       # activation energy [eV]
    Ed   = 3.0        # deactivation energy [eV]
    Tref = 293.15     # reference temperature [K] (20 C)
    Td   = 305.15     # temperature of half high-T inactivation [K] (~32 C)

    rise = np.exp(-E / k * (1.0 / T - 1.0 / Tref))
    fall = 1.0 + np.exp(Ed / k * (1.0 / Td - 1.0 / T))

    return P0 * rise / fall
