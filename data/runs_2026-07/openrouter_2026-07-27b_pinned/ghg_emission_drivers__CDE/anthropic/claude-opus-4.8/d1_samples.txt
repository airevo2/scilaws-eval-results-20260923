def predict(X):
    # STIRPAT / Kaya-identity based multiplicative model:
    #   I = a * P^b * A^c * (AP)^d * (DP)^e * (UR)^f * exp(g*AT)
    # Emissions scale ~ population * affluence (GDP per capita),
    # modulated by demographic, density, urbanization and climate factors.
    A  = X[:, 0]   # GDP per capita (USD)
    P  = X[:, 1]   # total population (persons)
    AP = X[:, 2]   # active-population share (%)
    DP = X[:, 3]   # population density (per km2)
    UR = X[:, 4]   # urbanization (%)
    AT = X[:, 5]   # annual mean temperature (deg C)

    # log-linear STIRPAT coefficients (commonly cited elasticities)
    ln_a = -9.0     # scale constant
    b_P  = 1.00     # population elasticity ~ 1 (Kaya)
    c_A  = 0.90     # affluence elasticity (near 1, slightly < 1)
    d_AP = 0.50     # working-age share effect
    e_DP = -0.05    # density (compactness reduces per-capita energy use)
    f_UR = 0.30     # urbanization effect
    g_AT = -0.02    # temperature (warmer -> less heating demand)

    lnI = (ln_a
           + b_P  * np.log(P)
           + c_A  * np.log(A)
           + d_AP * np.log(AP)
           + e_DP * np.log(DP)
           + f_UR * np.log(UR)
           + g_AT * AT)

    return np.exp(lnI)

import numpy as np

def predict(X):
    # STIRPAT model (Stochastic Impacts by Regression on Population,
    # Affluence, and Technology): I = a * P^b * A^c * ... 
    # Estimated in log-linear form:
    #   ln(I) = ln(a) + b*ln(P) + c*ln(A) + d*ln(AP) + e*ln(DP)
    #           + f*ln(UR) + g*ln(AT_shift)
    #
    # I  = CO2 emissions [kg CO2]
    # P  = total population
    # A  = GDP per capita (affluence)
    # AP = active/working-age population share
    # DP = population density
    # UR = urbanization rate
    # AT = annual mean temperature

    A  = X[:, 0]
    P  = X[:, 1]
    AP = X[:, 2]
    DP = X[:, 3]
    UR = X[:, 4]
    AT = X[:, 5]

    # Canonical STIRPAT elasticities (commonly cited magnitudes)
    ln_a = -1.5     # scaling constant
    b    = 1.00     # population elasticity (~unit elasticity)
    c    = 0.95     # affluence elasticity
    d    = 0.50     # working-age share
    e    = 0.05     # population density
    f    = 0.30     # urbanization
    g    = -0.10    # temperature (heating/cooling demand)

    # shift temperature to keep it positive for the log transform
    AT_shift = AT + 20.0

    ln_I = (ln_a
            + b * np.log(P)
            + c * np.log(A)
            + d * np.log(AP)
            + e * np.log(DP)
            + f * np.log(UR)
            + g * np.log(AT_shift))

    # convert from tonnes-scale fit to kg CO2
    I_CO2 = np.exp(ln_I) * 1.0e3

    return I_CO2
