import numpy as np

def predict(X):
    """
    Predict total proton-proton hadronic cross-section sigma_tot [mb] 
    at a given center-of-mass energy squared s [GeV^2].
    
    Uses the standard Donnachie-Landshoff (DL) parametrization:
    sigma_tot(s) = X * s^epsilon + Y * s^{-eta}
    with epsilon ~ 0.0808 (Pomeron intercept - 1), eta ~ 0.4525 (Reggeon intercept - 1),
    X ~ 21.70 mb, Y ~ 56.08 mb (fitted to high-energy pp data).
    Coefficients are for s in GeV^2.
    """
    s = X[:, 0]  # s in GeV^2
    
    # DL parametrization coefficients (Donnachie & Landshoff, Phys.Lett.B 296 (1992) 227)
    X_coeff = 21.70   # mb
    Y_coeff = 56.08   # mb
    epsilon = 0.0808
    eta = 0.4525
    
    sigma_tot = X_coeff * s**epsilon + Y_coeff * s**(-eta)
    
    return sigma_tot
