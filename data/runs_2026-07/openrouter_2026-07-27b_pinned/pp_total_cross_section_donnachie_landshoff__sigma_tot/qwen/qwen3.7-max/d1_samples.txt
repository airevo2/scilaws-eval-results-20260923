import numpy as np

def predict(X):
    s = X[:, 0]
    
    # Donnachie-Landshoff (1992) parameterization for proton-proton total cross-section
    # sigma_tot(s) = X_P * s^epsilon + Y_R * s^-eta
    X_P = 21.70       # Pomeron coupling (mb)
    epsilon = 0.0808  # Pomeron intercept minus 1
    Y_R = 56.08       # Reggeon coupling for pp (mb)
    eta = 0.4525      # Reggeon intercept (1 - alpha_R)
    
    sigma_tot = X_P * (s ** epsilon) + Y_R * (s ** (-eta))
    
    return sigma_tot

import numpy as np

def predict(X):
    s = X[:, 0]
    # Donnachie-Landshoff parameterization for proton-proton total cross-section
    A = 21.70       # Pomeron coupling (mb)
    B = 56.08       # Reggeon coupling (mb)
    epsilon = 0.0808 # Pomeron intercept - 1
    eta = 0.4525    # 1 - Reggeon intercept
    
    sigma_tot = A * (s ** epsilon) + B * (s ** -eta)
    return sigma_tot
