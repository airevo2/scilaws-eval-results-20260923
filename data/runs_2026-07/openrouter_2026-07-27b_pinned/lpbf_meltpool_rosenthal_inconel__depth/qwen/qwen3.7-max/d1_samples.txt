import numpy as np

def predict(X):
    """
    Predicts the melt pool depth based on the standard dimensional analysis 
    scaling law for laser melting. 
    
    The melt pool depth (d) scales with the ratio of laser power (P) to the 
    product of scan velocity (V) and beam diameter (D), which represents an 
    effective areal energy density divided by a characteristic volumetric 
    melting enthalpy.
    """
    P = X[:, 0]  # Laser power [W]
    V = X[:, 1]  # Scan velocity [mm/s]
    D = X[:, 2]  # Beam diameter [um]
    
    # Proportionality constant (named constant) that absorbs material properties 
    # (density, specific heat, latent heat, melting temperature) and unit conversions.
    # Calibrated to yield depth in [um] for typical L-PBF metals (e.g., Ti-6Al-4V, SS316L).
    C_material = 45000.0 
    
    d = C_material * (P / (V * D))
    
    return d
