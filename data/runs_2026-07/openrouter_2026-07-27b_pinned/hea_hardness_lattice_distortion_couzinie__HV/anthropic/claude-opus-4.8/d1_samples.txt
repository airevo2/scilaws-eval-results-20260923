import numpy as np

def predict(X):
    # Vickers hardness of refractory/complex-concentrated HEAs
    # Common empirical linear-correlation form:
    # HV ≈ a + b·delta + c·VEC + d·dHmix
    #
    # Physical rationale:
    #  - Larger atomic-size mismatch (delta) => stronger solid-solution
    #    strengthening / lattice distortion => higher hardness.
    #  - Lower VEC favors BCC (harder) structures; higher VEC favors FCC
    #    (softer) => negative VEC coefficient.
    #  - More negative dHmix => stronger bonding => higher hardness
    #    (positive coefficient means less-negative dHmix lowers HV).

    delta = X[:, 0]
    VEC   = X[:, 1]
    dHmix = X[:, 2]

    a =  600.0   # intercept [kgf/mm^2]
    b =   45.0   # per % delta
    c =  -80.0   # per VEC unit
    d =   -6.0   # per kJ/mol dHmix

    HV = a + b * delta + c * VEC + d * dHmix
    return HV
