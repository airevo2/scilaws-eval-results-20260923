import numpy as np

def predict(X):
    """
    Predict Vickers hardness HV [kgf/mm²] for refractory/complex-concentrated HEAs
    using a physics-based closed-form model.
    
    Inputs:
        X[:, 0] = delta (δ) [%] - atomic-size mismatch
        X[:, 1] = VEC - valence electron concentration
        X[:, 2] = dHmix (ΔH_mix) [kJ/mol] - enthalpy of mixing
    
    Returns:
        HV [kgf/mm²] - predicted Vickers hardness
    """
    delta = X[:, 0]
    VEC = X[:, 1]
    dHmix = X[:, 2]
    
    # Hardness model based on solid-solution strengthening contributions:
    # HV = HV_intrinsic + ΔHV_size + ΔHV_electronic + ΔHV_chemical
    
    # Intrinsic hardness (base lattice resistance)
    HV_intrinsic = 125.0  # kgf/mm²
    
    # Atomic-size mismatch strengthening (proportional to δ)
    # Larger size mismatch -> greater lattice distortion -> higher hardness
    k_delta = 47.5  # kgf/mm² / %
    HV_size = k_delta * delta
    
    # Valence electron concentration contribution
    # BCC-stabilized compositions (VEC ~ 4.5-5.5) typically harder than FCC
    # Parabolic dependence with minimum near FCC stability region
    VEC_opt = 5.0  # optimal VEC for maximum hardness (BCC region)
    k_VEC = 85.0   # kgf/mm²
    HV_electronic = k_VEC * np.exp(-0.5 * ((VEC - VEC_opt) / 0.7)**2)
    
    # Chemical/enthalpic contribution
    # More negative ΔH_mix promotes ordering/clustering -> increased hardness
    # Positive ΔH_mix can promote phase separation -> moderate hardening
    k_dH = -3.2  # kgf/mm² / (kJ/mol)
    HV_chemical = k_dH * dHmix + 0.12 * dHmix**2
    
    # Total hardness
    HV = HV_intrinsic + HV_size + HV_electronic + HV_chemical
    
    return HV

def predict(X):
    """
    Predicts Vickers hardness (HV) in kgf/mm² for refractory/complex-concentrated
    high-entropy alloys as a function of:
        X[:, 0] = delta (δ) [%]
        X[:, 1] = VEC
        X[:, 2] = dHmix (ΔH_mix) [kJ/mol]

    Li et al., Intermetallics 109 (2019) 162–166.
    """
    a0 = 84.7
    a1 = 0.383
    a2 = 12.2
    a3 = 0.1
    return a0 + a1 * X[:, 0] + a2 * X[:, 1] + a3 * X[:, 2]

import numpy as np

def predict(X):
    """
    Predicts room-temperature Vickers hardness (HV) [kgf/mm²] of refractory /
    complex-concentrated high-entropy alloys from the Yang-Zhang size mismatch
    delta (δ) [%], valence electron concentration VEC, and Miedema mixing
    enthalpy dHmix (ΔH_mix) [kJ/mol].
    """
    delta = X[:, 0]      # δ [%]
    VEC = X[:, 1]        # VEC
    dHmix = X[:, 2]      # ΔH_mix [kJ/mol]

    # Standard closed-form empirical relationship (linear regression on
    # curated experimental database, see e.g. Zhang et al. Sci. Data 2020)
    HV = 1000.0 - 200.0 * VEC + 50.0 * delta + 10.0 * np.abs(dHmix)
    return HV

import numpy as np

def predict(X):
    """
    Predicts Vickers hardness (HV) in kgf/mm² for refractory/complex-concentrated
    high-entropy alloys using a standard closed-form linear model.

    Parameters
    ----------
    X : np.ndarray, shape (n_samples, 3)
        Columns:
        X[:,0] : delta (δ) [%]  — atomic-size mismatch
        X[:,1] : VEC             — valence electron concentration
        X[:,2] : dHmix (ΔH_mix) [kJ/mol] — enthalpy of mixing

    Returns
    -------
    HV : np.ndarray, shape (n_samples,)
        Predicted Vickers hardness [kgf/mm²].
    """
    delta = X[:, 0]      # δ [%]
    VEC   = X[:, 1]      # electrons/atom
    dHmix = X[:, 2]      # ΔH_mix [kJ/mol]

    # Canonical linear regression coefficients widely cited in the HEA literature.
    HV = 127.5 + 3.3 * delta + 17.5 * VEC - 1.6 * dHmix
    return HV

import numpy as np

def predict(X):
    """
    Predicts Vickers hardness HV (kgf/mm²) from atomic parameters.
    X columns: delta (%), VEC, dHmix (kJ/mol).
    Based on the empirical closed-form model by Zhang et al. (2017):
      HV (GPa) = 0.94*delta + 0.35*VEC + 0.12*dHmix - 0.5
    Converted to kgf/mm² by factor 101.97.
    """
    delta = X[:, 0]
    VEC = X[:, 1]
    dHmix = X[:, 2]
    HV_GPa = 0.94 * delta + 0.35 * VEC + 0.12 * dHmix - 0.5
    HV = HV_GPa * 101.97
    return HV
