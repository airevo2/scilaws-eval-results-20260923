import numpy as np

def predict(X):
    """
    Predicts the base-10 logarithm of the 1.3 mm continuum flux density (in Jy)
    of a protoplanetary disk at an assumed distance of 140 pc.
    
    Uses the standard empirical power-law scaling between disk millimeter flux, 
    stellar mass, and age (e.g., Andrews et al. 2013; Ansdell et al. 2016; 
    Pascucci et al. 2016). The upper limit flag (X[:, 3]) is an observational 
    property and does not alter the expected underlying physical flux in the 
    closed-form mean relation.
    """
    # Extract the primary stellar mass (DM97) and age (DM97)
    log10_M_star = X[:, 0]
    log10_age = X[:, 4]
    
    # Canonical power-law fit coefficients for the disk mm flux vs stellar mass relation.
    # F_1.3mm is found to scale roughly as M_star^1.5 to M_star^1.8.
    # A weak negative correlation with age is also typically observed for disk dissipation.
    alpha = 1.8      # stellar mass exponent
    beta = -0.3      # age exponent
    gamma = -4.2     # normalization constant for log10(F / Jy) at 140 pc
    
    log10_F_1p3_Jy = alpha * log10_M_star + beta * log10_age + gamma
    
    return log10_F_1p3_Jy

import numpy as np

def predict(X):
    X = np.asarray(X, dtype=float)
    logM = X[:, 0]   # log10(M_*/M_sun), DM97
    logt = X[:, 4]   # log10(t/Myr), DM97

    # Andrews & Williams (2005) Taurus scaling:
    # F_1.3mm(140 pc) ≈ 0.1 Jy * (M_*/M_sun)^1.0 * (t/Myr)^-0.5
    logF = -1.0 + 1.0 * logM - 0.5 * logt
    return logF
