import numpy as np

def predict(X):
    """
    Predicts the absolute Gaia Wesenheit magnitude (M_W) for RR Lyrae stars 
    using a standard Period-Luminosity-Metallicity (PLZ) relation.
    """
    log_P = X[:, 0]
    feh = X[:, 1]
    
    # Canonical PLZ relation coefficients for Gaia Wesenheit magnitude 
    # (e.g., Muraveva et al. 2018; Clementini et al. 2019)
    a = -0.49
    b = -2.30
    c = -0.20
    
    M_W = a + b * log_P + c * feh
    return M_W

import numpy as np

def predict(X):
    """
    Predicts the absolute Gaia Wesenheit magnitude (M_W) for fundamental-mode RR Lyrae stars.
    
    Uses the period-metallicity relation from Ripepi et al. (2022, Gaia DR3) for the 
    Gaia Wesenheit magnitude W(G, G_BP-G_RP) = G - 1.9 * (G_BP - G_RP).
    
    Inputs:
    X[:, 0] : log_P (base-10 logarithm of the period in days)
    X[:, 1] : feh ([Fe/H] in dex)
    
    Returns:
    M_W : 1-D numpy array of absolute Wesenheit magnitudes
    """
    log_P = X[:, 0]
    feh = X[:, 1]
    
    # Coefficients from Ripepi et al. (2022, Gaia DR3)
    a = -2.629  # Period slope
    b = 0.169   # Metallicity slope
    c = -0.553  # Zero-point
    
    M_W = a * log_P + b * feh + c
    return M_W
