import numpy as np

def predict(X):
    """
    Predict absolute Gaia Wesenheit magnitude M_W for Classical Cepheids
    using the Period-Luminosity-Metallicity (PLZ) relation.
    
    M_W = alpha * (log_P - 1.0) + beta * [Fe/H] + gamma
    
    Canonical form referenced to log P_0 = 1.0 (P = 10 days).
    Coefficients representative of Gaia DR3 calibrations
    (e.g., Ripepi et al. 2022; Breuval et al. 2024).
    """
    log_P = X[:, 0]
    feh   = X[:, 1]

    # PLZ coefficients for Gaia Wesenheit: W_G = G - 1.90*(BP - RP)
    alpha = -3.31   # slope in log P
    beta  = -0.23   # metallicity dependence [mag/dex]
    gamma = -5.90   # zero-point at log P = 1.0 and [Fe/H] = 0

    M_W = alpha * (log_P - 1.0) + beta * feh + gamma
    return M_W
