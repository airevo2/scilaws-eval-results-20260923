import numpy as np

def predict(X):
    log_P = X[:, 0]
    feh = X[:, 1]
    # Gaia Wesenheit period-luminosity-metallicity relation for
    # classical Cepheids (fundamental mode), pivoted at log P = 1:
    # M_W = alpha + beta*(logP - 1) + gamma*[Fe/H]
    alpha = -5.96   # zero-point [mag]
    beta  = -3.31   # PL slope
    gamma = -0.20   # metallicity term [mag/dex]
    M_W = alpha + beta * (log_P - 1.0) + gamma * feh
    return M_W
