import numpy as np

def predict(X):
    """
    Predicts the equilibrium CO2 adsorbed amount (q) using the standard 
    Langmuir isotherm model, which is the most commonly cited canonical 
    form for single-component gas adsorption on porous materials like MOFs.
    
    Since no material-specific descriptors are provided in X, representative 
    average coefficients for CO2 adsorption on typical MOFs at 298 K are used.
    """
    P = X[:, 0]  # CO2 equilibrium partial pressure in bar
    
    # Representative Langmuir fitted coefficients for CO2 on MOFs
    q_max = 3.5  # Saturation capacity (mmol/g)
    K = 0.8      # Langmuir affinity/equilibrium constant (bar^-1)
    
    # Langmuir isotherm equation
    q = (q_max * K * P) / (1.0 + K * P)
    
    return q
