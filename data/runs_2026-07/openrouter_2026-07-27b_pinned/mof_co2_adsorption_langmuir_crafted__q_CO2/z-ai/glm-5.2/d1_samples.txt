import numpy as np

def predict(X):
    """
    Langmuir isotherm for CO2 adsorption in MOFs.
    
    q = q_max * (K * P) / (1 + K * P)
    
    where:
        q_max [mmol/g] — saturation adsorption capacity
        K    [1/bar]  — Langmuir affinity constant
        P    [bar]    — equilibrium partial pressure
    """
    P = X[:, 0]
    
    # Representative fitted coefficients for CO2/MOF (GCMC, DDEC, UFF)
    q_max = 3.0   # mmol/g
    K = 1.0       # 1/bar
    
    q = q_max * (K * P) / (1.0 + K * P)
    
    return q
