import numpy as np

def predict(X):
    # Langmuir isotherm: q = q_sat * (K*P) / (1 + K*P)
    P = X[:, 0]
    q_sat = 5.0   # mmol/g, saturation capacity
    K = 0.5       # 1/bar, Langmuir affinity constant
    q = q_sat * (K * P) / (1.0 + K * P)
    return q
