import numpy as np

def predict(X):
    D = X[:, 0]
    # Chapman-Richards height-diameter model (3-parameter Weibull-type asymptotic function)
    # Commonly used in pantropical forest ecology (e.g., Feldpausch et al., 2011)
    a = 57.78  # asymptotic maximum height
    b = 0.045  # scale parameter related to DBH
    c = 0.87   # shape parameter
    
    H = a * (1.0 - np.exp(-b * D))**c
    return H

import numpy as np

def predict(X):

    # Chapman-Richards height-diameter equation, a standard closed-form 
    # sigmoidal model widely used in forest ecology for pantropical datasets.
    a = 45.0  # asymptotic maximum height (m)
    b = 0.03  # growth rate parameter
    c = 1.2   # shape parameter
    
    H = a * (1.0 - np.exp(-b * D))**c
    return H
