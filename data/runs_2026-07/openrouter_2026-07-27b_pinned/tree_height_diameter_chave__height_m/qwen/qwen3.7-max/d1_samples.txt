import numpy as np

def predict(X):
    D = X[:, 0]
    
    # Standard allometric power-law relationship for tree height and diameter
    # H = a * D^b
    # Coefficients represent typical pantropical averages (e.g., Feldpausch et al., 2011)
    a = 1.35
    b = 0.55
    
    H = a * (D ** b)
    return H

import numpy as np

def predict(X):
    D = X[:, 0]
    lnD = np.log(D)
    # Chave et al. (2014) pantropical height-diameter allometry (baseline, E=0)
    lnH = 0.893 + 0.731 * lnD - 0.0461 * (lnD ** 2)
    H = np.exp(lnH)
    return H
