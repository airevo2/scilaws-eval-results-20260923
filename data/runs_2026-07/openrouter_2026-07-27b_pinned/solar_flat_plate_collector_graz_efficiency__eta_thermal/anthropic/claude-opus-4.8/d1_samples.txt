import numpy as np

def predict(X):
    # ISO 9806 / EN 12975 Hottel-Whillier-Bliss steady-state efficiency:
    # eta = eta0 - a1*(T_in - T_amb)/G - a2*(T_in - T_amb)^2/G
    G = X[:, 0]
    T_in = X[:, 1]
    T_amb = X[:, 2]

    eta0 = 0.75      # optical (zero-loss) efficiency
    a1 = 3.5         # W/(m^2 K) linear loss coefficient
    a2 = 0.015       # W/(m^2 K^2) quadratic loss coefficient

    dT = T_in - T_amb
    Tstar = dT / G

    eta = eta0 - a1 * Tstar - a2 * G * Tstar**2
    return eta
