    # eta = FR*(tau*alpha) - FR*UL * (T_in - T_amb) / G
    eta_0 = 0.65  # Zero-loss (optical) efficiency

    return np.clip(eta, 0.0, 1.0)

import numpy as np

def predict(X):
    G = X[:, 0]
    T_in = X[:, 1]
    T_amb = X[:, 2]
    
    # Core Hottel-Whillier-Bliss (HWB) equation parameters
    eta_0 = 0.65  # Optical (zero-loss) efficiency
    U_L = 3.5     # Overall heat loss coefficient [W/(m^2 K)]
    
    # Reduced temperature difference
    x = (T_in - T_amb) / G
    
    # Linear HWB efficiency model
    eta = eta_0 - U_L * x
    
    return np.clip(eta, 0.0, 1.0)
