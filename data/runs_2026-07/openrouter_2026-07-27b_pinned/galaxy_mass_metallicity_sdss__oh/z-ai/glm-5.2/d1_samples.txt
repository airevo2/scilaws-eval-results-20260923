import numpy as np

def predict(X):
    log_Mstar = X[:, 0]
    log_SFR = X[:, 1]
    
    # Fundamental Metallicity Relation (FMR) parameters from Mannucci et al. (2010)
    alpha = 0.32
    mu_pivot = 10.5
    oh_saturation = 9.08
    slope = 0.47
    
    # Calculate the mass-SFR parameter mu
    mu = log_Mstar - alpha * log_SFR
    
    # Initialize with the saturation metallicity for high-mu galaxies
    oh = np.full_like(mu, oh_saturation)
    
    # Apply the linear FMR for galaxies below the pivot mu
    mask = mu < mu_pivot
    oh[mask] = oh_saturation + slope * (mu[mask] - mu_pivot)
    
    return oh

import numpy as np

def predict(X):
    log_Mstar = X[:, 0]
    log_SFR = X[:, 1]
    
    # Fundamental Metallicity Relation (FMR) projection parameter (Mannucci et al. 2010)
    mu = log_Mstar - 0.32 * log_SFR
    
    # Linear relation below the saturation limit, constant above it
    oh = 8.90 + 0.47 * (mu - 10.5)
    oh = np.where(mu >= 10.5, 8.90, oh)
    
    return oh
