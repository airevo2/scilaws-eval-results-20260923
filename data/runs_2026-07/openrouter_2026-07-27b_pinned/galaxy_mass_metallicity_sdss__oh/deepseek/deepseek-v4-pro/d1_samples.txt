import numpy as np

def predict(X):
    """
    Predict 12 + log(O/H) using the Fundamental Metallicity Relation (FMR)
    from Mannucci et al. (2010, MNRAS 408, 2115).

    Inputs:
        X : 2-D numpy array of shape (n_samples, 2)
            X[:,0] = log_Mstar [log(M_sun)]
            X[:,1] = log_SFR   [log(M_sun/yr)]
    Returns:
        1-D numpy array of 12 + log(O/H) [dex]
    """
    logM = X[:, 0]
    logSFR = X[:, 1]

    m = logM - 10.0
    s = logSFR

    oh = 8.90 + 0.37*m - 0.14*s - 0.19*m**2 + 0.12*m*s - 0.054*s**2
    return oh

    Returns
    -------
    oh : np.ndarray, shape (n_samples,)
        12 + log10(O/H) in dex.
    """
    log_Mstar = X[:, 0]
    log_SFR   = X[:, 1]

    # Shifted stellar mass
    m = log_Mstar - 10.0
    s = log_SFR

    # Mannucci+2010, Equation (4) and Table 3 (SDSS calibration)
    oh = (8.90
          + 0.37  * m
          - 0.14  * s
          - 0.19  * m**2
          + 0.12  * m * s
          - 0.054 * s**2)

    return oh

    Predict gas-phase oxygen abundance 12 + log(O/H) [dex]
    using the fundamental metallicity relation (FMR) from
    Mannucci et al. (2010, MNRAS, 408, 2115).

    Input X: 2-D numpy array
        X[:, 0] : log10(stellar mass / M_sun)   [log(M_sun)]
        X[:, 1] : log10(SFR / (M_sun/yr))       [log(M_sun/yr)]
    Returns: 1-D array of 12 + log(O/H) in dex.
    """

    logSFR   = X[:, 1]

    # standardized variables from Mannucci+10
    m = logMstar - 10.0   # log M* - 10

    # polynomial coefficients (Table 3 / Eq. 2 of Mannucci+10)
    oh = (8.90
          + 0.37 * m
          - 0.14 * s
          - 0.19 * m**2
          + 0.12 * m * s
          - 0.054 * s**2)

    return oh
