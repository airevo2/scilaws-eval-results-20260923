def predict(X):

    M2 = X[:, 3]
    C1 = X[:, 4]
    C2 = X[:, 5]

    # In-plane (radial) velocity coefficients
    a = -0.219
    b =  0.444
    c = -2.324

    v_rho = (a * (M1 / M2) * (1.0 + c * C1)
             + a * (M2 / M1) * (1.0 + c * C2)
             + b)

    # Out-of-plane velocity coefficients
    d = -0.315
    e =  0.0
    f =  0.679

    v_z = (d * (M1 / M2) * (1.0 + e * C1)
           + d * (M2 / M1) * (1.0 + e * C2)
           + f)

    v_ej = np.sqrt(v_rho**2 + v_z**2)
    return v_ej
