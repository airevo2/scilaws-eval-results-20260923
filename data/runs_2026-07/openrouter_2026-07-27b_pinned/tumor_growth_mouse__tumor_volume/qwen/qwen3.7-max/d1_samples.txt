import numpy as np

def predict(X):
    t = X[:, 0]
    
    # Gompertz tumor growth model (canonical form for solid tumors)
    # V(t) = V0 * exp( (r0 / alpha) * (1 - exp(-alpha * t)) )
    V0 = 1.0       # Initial volume at t=0 (mm^3)
    r0 = 0.20      # Initial exponential growth rate (1/day)
    alpha = 0.05   # Rate of exponential decay of the growth rate (1/day)
    
    V = V0 * np.exp((r0 / alpha) * (1.0 - np.exp(-alpha * t)))
    
    return V

import numpy as np

def predict(X):
    t = X[:, 0]
    
    # Gompertz tumor growth model parameters
    V0 = 1.0        # Initial tumor volume (mm^3), approx 10^6 cells
    K = 2000.0      # Carrying capacity (mm^3), typical maximum before ulceration/necrosis
    alpha = 0.05    # Gompertz growth rate constant (1/day)
    
    # Closed-form Gompertz equation: V(t) = K * exp( -ln(K/V0) * exp(-alpha * t) )
    V = K * np.exp(-np.log(K / V0) * np.exp(-alpha * t))
    
    return V
