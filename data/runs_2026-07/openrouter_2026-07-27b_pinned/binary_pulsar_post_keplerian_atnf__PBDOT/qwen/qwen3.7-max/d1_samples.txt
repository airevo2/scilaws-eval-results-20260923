def predict(X):
    """
    Predicts the orbital period derivative (Pbdot) due to gravitational wave emission 
    using the Peters (1964) formula. 
    
    Since component masses are not provided in the inputs, canonical neutron star 
    masses of 1.4 M_sun are assumed (typical for DNS systems like the Hulse-Taylor binary).
    """
    Pb_days = X[:, 0]
    e = X[:, 1]
    
    # Canonical double neutron star masses (in Solar masses)
    m1 = 1.4
    m2 = 1.4
    M_chirp = (m1 * m2)**0.6 / (m1 + m2)**0.2
    
    # Constants
    # T_sun = G * M_sun / c^3 in seconds
    T_sun = 4.925490947e-6 
    
    # Orbital period in seconds
    Pb_sec = Pb_days * 86400.0
    
    # Peters (1964) formula for gravitational wave driven orbital decay
    coeff = - (192.0 * np.pi / 5.0)
    gw_term = (2.0 * np.pi * T_sun * M_chirp / Pb_sec)**(5.0 / 3.0)
    ecc_term = (1.0 + (73.0 / 24.0) * e**2 + (37.0 / 96.0) * e**4) / (1.0 - e**2)**(7.0 / 2.0)
    
    Pbdot = coeff * gw_term * ecc_term
    
    return Pbdot

import numpy as np

def predict(X):
    # Physical constants
    G = 6.67430e-11       # Gravitational constant [m^3 kg^-1 s^-2]
    c = 299792458.0       # Speed of light [m/s]
    M_sun = 1.98840e30    # Solar mass [kg]
    
    # Typical chirp mass for a Double Neutron Star (DNS) system
    # Assuming canonical component masses m1 = m2 = 1.4 M_sun
    # M_c = (m1 * m2)**(3/5) / (m1 + m2)**(1/5) ~ 1.218 M_sun
    M_c = 1.218 * M_sun   
    
    # Extract inputs
    Pb_days = X[:, 0]
    e = X[:, 1]
    
    # Convert orbital period to seconds
    Pb_sec = Pb_days * 86400.0
    
    # Peters (1964) quadrupole formula for intrinsic Pbdot due to GW emission
    term1 = 192.0 * np.pi / 5.0
    term2 = (2.0 * np.pi * G * M_c / (c**3 * Pb_sec)) ** (5.0 / 3.0)
    
    # Eccentricity enhancement factor
    f_e = (1.0 + (73.0 / 24.0) * e**2 + (37.0 / 96.0) * e**4) / (1.0 - e**2)**3.5
    
    # Calculate dot_P_b (dimensionless rate, s/s)
    dot_Pb = -term1 * term2 * f_e
    
    return dot_Pb
