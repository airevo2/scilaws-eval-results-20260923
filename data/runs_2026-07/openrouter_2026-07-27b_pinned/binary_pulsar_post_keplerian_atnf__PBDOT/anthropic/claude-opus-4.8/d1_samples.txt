import numpy as np

def predict(X):
    # GR quadrupole orbital decay (Peters 1964)
    # PBDOT = -(192*pi/5) * (2*pi*G/c^3)^(5/3) * (Pb/2pi)^(-5/3)
    #          * f(e) * m1*m2 / (m1+m2)^(1/3)
    # Masses unavailable -> assume canonical double-neutron-star system (1.4 Msun each)

    Pb = X[:, 0] * 86400.0        # orbital period [s]
    e  = X[:, 1]                  # eccentricity

    # physical constants (SI)
    G    = 6.67430e-11            # m^3 kg^-1 s^-2
    c    = 2.99792458e8           # m/s
    Msun = 1.98892e30             # kg

    m2 = 1.4 * Msun

    # eccentricity enhancement factor
    fe = (1.0 + (73.0/24.0)*e**2 + (37.0/96.0)*e**4) / (1.0 - e**2)**3.5

    prefactor = -(192.0 * np.pi / 5.0) * (2.0 * np.pi * G / c**3)**(5.0/3.0)

    dot_Pb = prefactor * (Pb / (2.0 * np.pi))**(-5.0/3.0) * fe \
             * (m1 * m2) / (m1 + m2)**(1.0/3.0)

    return dot_Pb

import numpy as np

def predict(X):
    # Peters (1964) / GR quadrupole orbital decay for a binary pulsar.
    # Pbdot = -(192*pi/5) * (2*pi*G/(Pb*c^3))^(5/3)
    #          * (m1*m2)/(m1+m2)^(1/3) * f(e)
    # with f(e) = (1 + 73/24 e^2 + 37/96 e^4) / (1-e^2)^(7/2)
    # Only Pb and e are given, so assume canonical double-NS masses m1=m2=1.4 Msun.

    Pb = np.asarray(X[:, 0], dtype=float) * 86400.0   # days -> seconds
    e  = np.asarray(X[:, 1], dtype=float)

    G      = 6.67430e-11          # m^3 kg^-1 s^-2
    c      = 2.99792458e8         # m/s
    Msun   = 1.98892e30           # kg

    m1 = 1.4 * Msun
    m2 = 1.4 * Msun
    M  = m1 + m2

    # eccentricity enhancement function
    fe = (1.0 + (73.0/24.0)*e**2 + (37.0/96.0)*e**4) / np.power(1.0 - e**2, 3.5)

    prefac = -(192.0 * np.pi / 5.0)
    orbital = np.power(2.0 * np.pi * G / (Pb * c**3), 5.0/3.0)
    mass_term = (m1 * m2) / np.power(M, 1.0/3.0)

    Pbdot = prefac * orbital * mass_term * fe
    return Pbdot

import numpy as np

def predict(X):
    # Gravitational-wave orbital decay (Peters 1964 / GR quadrupole formula)
    # Pb_dot = -(192*pi/5) * (2*pi*G/(c^3 * Pb))^(5/3)
    #           * (m1*m2)/(m1+m2)^(1/3) * f(e)
    # with f(e) = (1 + 73/24 e^2 + 37/96 e^4) / (1-e^2)^(7/2)
    # Masses assumed to be canonical double-neutron-star values (1.4 Msun each).

    G = 6.67430e-11        # m^3 kg^-1 s^-2
    c = 2.99792458e8       # m/s
    Msun = 1.98892e30      # kg
    day = 86400.0          # s

    Pb = X[:, 0] * day     # orbital period in seconds
    e = X[:, 1]

    m1 = 1.4 * Msun
    m2 = 1.4 * Msun
    M = m1 + m2

    fe = (1.0 + (73.0/24.0)*e**2 + (37.0/96.0)*e**4) / (1.0 - e**2)**3.5

    prefac = -(192.0 * np.pi / 5.0)
    Pbdot = prefac * (2.0*np.pi*G/(c**3 * Pb))**(5.0/3.0) \
            * (m1*m2) / M**(1.0/3.0) * fe

    return Pbdot
