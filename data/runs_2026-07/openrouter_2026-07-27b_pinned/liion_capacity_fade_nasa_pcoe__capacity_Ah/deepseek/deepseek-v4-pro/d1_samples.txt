import numpy as np

def predict(X):
    """
    Predict discharge capacity Q [Ah] as a function of cycle index k.
    Uses the standard empirical capacity fade model:
    Q(k) = Q0 - A * k^z
    where Q0 is initial capacity, A is fade rate coefficient,
    and z is the power-law exponent (typically ~0.5 for SEI growth).
    """
    Q0 = 2.0      # initial capacity [Ah]
    A = 0.05      # capacity fade coefficient [Ah/(cycle^z)]
    z = 0.5       # power-law exponent for fade

    k = X[:, 0]   # cycle index
    Q = Q0 - A * k**z
    return Q

import numpy as np

def predict(X):
    """
    Standard closed-form capacity fade model based on diffusion-limited SEI growth.
    Capacity loss is proportional to the square root of the cycle index.

    Input:  X : 2-D numpy array, X[:, 0] = cycle_index (k), starting at 1
    Output: Q : 1-D numpy array of discharge capacity per cycle [Ah]
    """
    k = X[:, 0]
    # Typical parameters for a lithium-ion cell (e.g., 2 Ah nominal)
    Q0 = 2.0    # initial capacity, Ah
    alpha = 0.1 # fade coefficient, Ah / sqrt(cycle)
    Q = Q0 - alpha * np.sqrt(k)
    return Q
