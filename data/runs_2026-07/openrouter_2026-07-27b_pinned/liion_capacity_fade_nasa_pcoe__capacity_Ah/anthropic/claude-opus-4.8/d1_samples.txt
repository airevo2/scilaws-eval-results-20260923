import numpy as np

def predict(X):
    """
    Discharge capacity vs. cycle number.

    Canonical empirical capacity-fade model (exponential/linear degradation):
        Q(k) = Q0 * exp(-alpha * k)

    where:
        Q0    = nominal (beginning-of-life) discharge capacity [Ah]
        alpha = per-cycle fade rate constant
        k     = cycle index

    This is the most commonly cited closed-form relationship for
    calendar/cycle capacity fade in Li-ion cells.
    """
    k = np.asarray(X[:, 0], dtype=float)

    # Fitted / nominal constants
    Q0 = 1.10       # nominal discharge capacity at cycle 1 [Ah]
    alpha = 1.5e-3  # per-cycle exponential fade rate [1/cycle]

    Q = Q0 * np.exp(-alpha * k)
    return Q

import numpy as np

def predict(X):
    """
    Discharge capacity fade with cycle number.

    Canonical empirical capacity-fade model (exponential decay from an
    initial capacity toward the observed loss with cycling):

        Q(k) = Q0 * exp(-alpha * (k - 1))

    where:
        Q0    = initial (first-cycle) discharge capacity [Ah]
        alpha = per-cycle fade rate constant [1/cycle]
        k     = cycle index (starts at 1)

    Typical Li-ion coin/pouch-cell values are used for the fitted constants.
    """
    k = np.asarray(X[:, 0], dtype=float)

    # Fitted / representative constants
    Q0 = 1.10        # initial discharge capacity [Ah]
    alpha = 1.0e-3   # fade rate per cycle [1/cycle]

    Q = Q0 * np.exp(-alpha * (k - 1.0))
    return Q

import numpy as np

def predict(X):
    # Empirical linear capacity-fade model:
    # Q(k) = Q0 - beta * k
    # where Q0 is the initial (nominal) discharge capacity and
    # beta is the per-cycle capacity loss rate.
    k = np.asarray(X[:, 0], dtype=float)

    Q0 = 1.10      # Ah, initial discharge capacity
    beta = 0.0015  # Ah lost per cycle

    Q = Q0 - beta * k
    return Q
