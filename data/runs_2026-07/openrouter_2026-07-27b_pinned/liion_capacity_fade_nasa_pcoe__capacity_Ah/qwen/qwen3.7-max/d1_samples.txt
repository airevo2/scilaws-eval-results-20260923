import numpy as np

def predict(X):
    k = X[:, 0]
    # Square-root cycle fade model (diffusion-limited SEI growth)
    # Q(k) = Q_0 - beta * sqrt(k)
    Q_0 = 2.0       # initial discharge capacity [Ah]
    beta = 0.046    # fade coefficient [Ah / cycle^0.5]
    Q = Q_0 - beta * np.sqrt(k)
    return Q

import numpy as np

def predict(X):
    """
    Predict discharge capacity Q (Ah) as a function of cycle index k
    using the canonical power-law capacity-fade model:
        Q(k) = Q_0 - alpha * k^beta
    where Q_0 is the nominal (beginning-of-life) capacity,
    alpha is the degradation rate coefficient, and
    beta is the power-law exponent (SEI-growth dominated fade ~ 0.5–0.8).
    """
    k = X[:, 0].astype(float)

    # Named constants (representative Li-ion 18650 cell)
    Q_0   = 2.0      # nominal capacity [Ah]
    alpha = 0.005    # fade rate coefficient [Ah / cycle^beta]
    beta  = 0.75     # power-law exponent

    Q = Q_0 - alpha * np.power(k, beta)
    return Q
