import numpy as np

def predict(X):
    """
    Predicts the discharge capacity Q (Ah) as a function of cycle index k.
    Uses the canonical power-law capacity fade model:
    Q(k) = Q_0 - A * k^z
    where Q_0 is the initial capacity, A is the fade rate, 
    and z is the empirical exponent (typically ~0.5 for diffusion-limited 
    processes like SEI layer growth).
    """
    k = X[:, 0]
    
    # Fitted/typical coefficients for a standard Li-ion cell
    Q_0 = 1.1   # Initial capacity [Ah]
    A = 0.015   # Fade rate coefficient [Ah / cycles^z]
    z = 0.5     # Power-law exponent (sqrt dependence)
    
    Q = Q_0 - A * np.power(k, z)
    return Q

def predict(X):
    """
    Predicts discharge capacity Q (Ah) based on cycle index k.
    Uses the standard empirical power-law capacity fade model:
    Q(k) = Q_BOL - A * k^z
    where Q_BOL is the beginning-of-life capacity, A is the fade coefficient, 
    and z is the empirical exponent (often ~0.5 for diffusion-limited degradation).
    """
    k = X[:, 0]
    
    # Empirical constants for a typical Li-ion cell
    Q_BOL = 1.1   # Beginning of life capacity [Ah]
    A = 0.005     # Fade rate coefficient
    z = 0.5       # Power-law exponent
    
    Q = Q_BOL - A * np.power(k, z)
    
    return Q
