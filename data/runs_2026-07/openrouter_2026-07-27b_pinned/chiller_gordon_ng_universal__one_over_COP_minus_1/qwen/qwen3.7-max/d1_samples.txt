import numpy as np

def predict(X):
    T_ei = X[:, 0]  # Evaporator (chilled-water) inlet temperature [K]
    T_ci = X[:, 1]  # Condenser (cooling-water) inlet temperature [K]
    Q_e  = X[:, 2]  # Evaporator cooling load [kW]

    # Gordon-Ng universal chiller model
    # 1/COP = a1*(T_ci/T_ei) + a2*(Q_e/T_ei) + a3*((T_ci - T_ei)/Q_e)
    # Term 1: Carnot-like thermodynamic limit (dimensionless)
    # Term 2: Heat-exchanger irreversibility losses (proportional to load)
    # Term 3: Compressor mechanical/thermal losses (inversely proportional to load)
    a1 = 0.1     # dimensionless
    a2 = 0.05    # [K/kW]
    a3 = 1.0     # [kW/K]

    inv_COP = a1 * (T_ci / T_ei) + a2 * (Q_e / T_ei) + a3 * ((T_ci - T_ei) / Q_e)

    return inv_COP - 1.0
