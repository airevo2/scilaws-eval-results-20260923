    #
    # The GN model expresses the reciprocal COP as a reversible (Carnot-like)
    # term plus internal-entropy and heat-loss corrections that scale with

    Q_e  = X[:, 2]   # evaporator cooling load [kW]

    a1 = 12.0
    a2 = 15.0
    a3 = 20.0

               + a1 * (1.0 / Q_e)
               + a2 * T_ci / (T_ei * Q_e)
               + a3 * (T_ci - T_ei) / (T_ei * Q_e))

    return inv_COP - 1.0

               + T_ci / T_ei
               + a0 / Q_e

    # COP_Carnot = T_cold / (T_hot - T_cold)
    # => 1/COP = (T_hot - T_cold) / T_cold
    # => 1/COP - 1 = (T_hot - 2*T_cold) / T_cold
    T_ei = X[:, 0]   # evaporator (cold-side) inlet temperature [K]
    T_ci = X[:, 1]   # condenser (hot-side) inlet temperature [K]

    inv_COP = (T_ci - T_ei) / T_ei
    return inv_COP - 1.0

import numpy as np

def predict(X):
    """
    Gordon-Ng universal thermodynamic chiller model.

    The reversible (Carnot) baseline for a chiller is
        1/COP = T_cond/T_evap - 1   ->   1/COP - 1 = T_ci/T_ei - 2
    with three physically-motivated loss terms:
      - internal entropy production   ~ a1 * (T_ei / Q_e)
      - heat-exchanger / leakage loss ~ a2 * (T_ci - T_ei)/(T_ei * Q_e)
      - finite-rate (load) losses      ~ a3 * (Q_e / T_ci)
    """
    T_ei = X[:, 0]   # evaporator (chilled water) inlet temp [K]
    T_ci = X[:, 1]   # condenser (cooling water) inlet temp [K]
    Q_e  = X[:, 2]   # evaporator cooling load [kW]

    # Fitted Gordon-Ng coefficients (representative values)
    a1 =  2.5e-1
    a2 =  1.2e+2
    a3 =  6.0e-4

    carnot = T_ci / T_ei - 2.0

    loss = (a1 * (T_ei / Q_e)
            + a2 * ((T_ci - T_ei) / (T_ei * Q_e))
            + a3 * (Q_e / T_ci))

    return carnot + loss
