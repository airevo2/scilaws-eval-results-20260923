def predict(X):
    """
    Standard Gordon–Ng universal thermodynamic model for a vapor-compression chiller.
    Returns 1/COP - 1 given:
        X[:, 0] = T_ei  (evaporator inlet temperature, K)
        X[:, 1] = T_ci  (condenser inlet temperature, K)
        X[:, 2] = Q_e   (evaporator cooling load, kW)
    """
    T_ei = X[:, 0]
    T_ci = X[:, 1]
    Q_e  = X[:, 2]

    # Fitted coefficients (replace with calibrated values for a specific machine)
    A = 1.0   # heat-leak / friction constant
    B = 1.0   # heat-leak / friction temperature coefficient
    C = 1.0   # evaporator heat-exchanger irreversibility coefficient
    D = 1.0   # condenser heat-exchanger irreversibility coefficient

    # Gordon–Ng form:
    # 1/COP = -1 + (T_ci / T_ei) + (1/Q_e)*(A + B*T_ci) + Q_e*(C/T_ei + D/T_ei**2)
    # Therefore:
    # 1/COP - 1 = (T_ci / T_ei) - 2 + (1/Q_e)*(A + B*T_ci) + Q_e*(C/T_ei + D/T_ei**2)
    one_over_COP_minus_1 = (
        (T_ci / T_ei) - 2.0
        + (1.0 / Q_e) * (A + B * T_ci)
        + Q_e * (C / T_ei + D / T_ei**2)
    )

    return one_over_COP_minus_1

import numpy as np

def predict(X):
    """
    Predict 1/COP - 1 using the Gordon–Ng universal thermodynamic model
    for vapor-compression chillers.

    Inputs (X is a 2-D numpy array):
        X[:, 0] = T_ei  Evaporator (chilled-water) inlet temperature [K]
        X[:, 1] = T_ci  Condenser (cooling-water) inlet temperature [K]
        X[:, 2] = Q_e   Evaporator cooling load [kW]

    Returns:
        1-D numpy array of 1/COP - 1 [dimensionless]
    """
    T_ei = X[:, 0]
    T_ci = X[:, 1]
    Q_e  = X[:, 2]

    # Gordon–Ng universal model coefficients (placeholder values)
    A = 1.0   # kW
    B = 1.0   # kW·K
    C = 1.0   # kW

    # 1/COP = -1 + (T_ci/T_ei)
    #         + (A/Q_e)*(T_ci/T_ei)
    #         + B*(T_ci/T_ei)/(Q_e*T_ei)
    #         + C/Q_e
    ratio = T_ci / T_ei
    inv_COP = (-1.0
               + ratio
               + (A / Q_e) * ratio
               + (B / (Q_e * T_ei)) * ratio
               + C / Q_e)

    # Return the requested quantity: 1/COP - 1
    return inv_COP - 1.0

import numpy as np

def predict(X):
    T_ei = X[:, 0]
    T_ci = X[:, 1]
    Q_e = X[:, 2]
    
    # Gordon-Ng universal thermodynamic model for chillers:
    # 1/COP = -1 + (T_ci / T_ei) + (R / Q_e) * (T_ci / T_ei - 1) + (S * T_ci / Q_e)
    # Therefore, 1/COP - 1 = (T_ci / T_ei) - 2 + (R / Q_e) * (T_ci / T_ei - 1) + (S * T_ci / Q_e)
    
    # Representative thermodynamic coefficients
    R = 1.0  # Coefficient representing internal irreversibilities
    S = 0.01 # Coefficient representing heat leaks
    
    T_ratio = T_ci / T_ei
    Q_inv = 1.0 / Q_e
    
    return T_ratio - 2.0 + (R * Q_inv) * (T_ratio - 1.0) + (S * T_ci * Q_inv)
