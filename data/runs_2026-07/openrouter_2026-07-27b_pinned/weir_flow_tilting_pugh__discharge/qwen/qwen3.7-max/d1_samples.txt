import numpy as np

def predict(X):
    h = X[:, 0]  # upstream head over weir crest [m]
    p = X[:, 1]  # weir height (crest above bed) [m]
    b = X[:, 2]  # weir crest length (channel width) [m]
    
    g = 9.80665  # gravitational acceleration [m/s^2]
    
    # Rehbock (1929) discharge coefficient for sharp-crested weir
    C_d = 0.611 + 0.075 * (h / p)
    
    # Standard rectangular sharp-crested weir equation
    Q = (2.0 / 3.0) * C_d * b * np.sqrt(2.0 * g) * h ** 1.5
    
    return Q
