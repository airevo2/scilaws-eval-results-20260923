import numpy as np

def predict(X):
    g = 9.81
    h = X[:, 0]
    p = X[:, 1]
    b = X[:, 2]
    # Rehbock formula for a sharp-crested rectangular weir
    Ce = 0.602 + 0.083 * (h / p)
    he = h + 0.0012  # effective head correction [m]
    Q = (2.0 / 3.0) * Ce * np.sqrt(2.0 * g) * b * he ** 1.5
    return Q

def predict(X):

    p = X[:, 1]
    b = X[:, 2]
    # Rehbock formula for a sharp-crested rectangular weir

    Q = (2.0 / 3.0) * Cd * np.sqrt(2.0 * g) * b * he**1.5
    return Q
