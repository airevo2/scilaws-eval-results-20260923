import numpy as np

def predict(X):
    h = X[:, 0]
    p = X[:, 1]
    b = X[:, 2]
    g = 9.81
    
    # Rehbock formula for the discharge coefficient (Cd)
    Cd = 0.611 + 0.075 * (h / p)
    
    # Standard sharp-crested weir discharge equation
    Q = (2.0 / 3.0) * Cd * np.sqrt(2.0 * g) * b * (h ** 1.5)
    
    return Q
