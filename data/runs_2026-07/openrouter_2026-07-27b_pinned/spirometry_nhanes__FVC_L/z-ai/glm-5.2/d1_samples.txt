import numpy as np

def predict(X):
    """
    Predicts Forced Vital Capacity (FVC) in liters.
    Uses the NHANES III reference equation (Hankinson et al., 1999) 
    for Males, age < 20 years.
    """
    age = X[:, 0]
    height_cm = X[:, 1]
    
    # NHANES III closed-form relationship for FVC (Male, < 20 yrs)
    fvc = -0.2511 + 0.00533 * age + 0.000156 * np.power(height_cm, 2)
    
    return fvc
