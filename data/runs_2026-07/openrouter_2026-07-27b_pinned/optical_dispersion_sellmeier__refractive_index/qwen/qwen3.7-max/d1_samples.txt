    below correspond to Fused Silica (SiO2) as a canonical default for the

import numpy as np

def predict(X):
    lam = X[:, 0]
    lam2 = lam ** 2
    # Modified Sellmeier equation for LiNbO3 (ordinary), Bond 1965
    # n^2 = A + B / (lambda^2 - C^2) - D * lambda^2
    A = 4.9130
    B = 0.1173
    C = 0.212
    D = 0.0278
    n2 = A + B / (lam2 - C**2) - D * lam2
    return np.sqrt(n2)
