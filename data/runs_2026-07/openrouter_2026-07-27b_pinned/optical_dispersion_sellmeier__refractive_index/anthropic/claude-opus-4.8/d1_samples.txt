import numpy as np

def predict(X):
    lam = X[:, 0]  # wavelength in micrometers
    l2 = lam**2
    # Three-term Sellmeier equation (canonical form)
    # (coefficients for a single-crystal, ordinary ray, room temperature)
    B1, C1 = 1.4313493, 0.0726631**2
    B2, C2 = 0.65054713, 0.1193242**2
    B3, C3 = 5.3414021, 18.028251**2
    n2 = 1.0 + B1*l2/(l2 - C1) + B2*l2/(l2 - C2) + B3*l2/(l2 - C3)
    return np.sqrt(n2)

import numpy as np

def predict(X):
    """
    Real part of the refractive index vs vacuum wavelength (micrometers)
    for single-crystal diamond (min-deviation goniometer, Bond 1965 style),
    using the standard two-term Sellmeier dispersion formula.

        n^2(lambda) = 1 + B1*lambda^2/(lambda^2 - C1) + B2*lambda^2/(lambda^2 - C2)
    """
    lam = np.asarray(X[:, 0], dtype=float)
    lam2 = lam**2

    # Canonical Sellmeier coefficients for diamond
    B1, C1 = 0.3306, 0.1750**2   # C1 in um^2
    B2, C2 = 4.3356, 0.1060**2   # C2 in um^2

    n2 = 1.0 + B1 * lam2 / (lam2 - C1) + B2 * lam2 / (lam2 - C2)
    return np.sqrt(n2)

import numpy as np

def predict(X):
    lam = np.asarray(X[:, 0], dtype=float)
    l2 = lam**2
    # Sellmeier coefficients (sapphire, ordinary ray; Malitson-type form)
    n2 = (1.0
          + 1.4313493 * l2 / (l2 - 0.0726631**2)
          + 0.65054713 * l2 / (l2 - 0.1193242**2)
          + 5.3414021 * l2 / (l2 - 18.028251**2))
    return np.sqrt(n2)
