import numpy as np

def predict(X):
    """
    Predict mean deconvolved calcium fluorescence for a V1 neuron as a function of orientation.
    Uses a von Mises tuning curve, which is the standard closed-form for circular orientation data.
    Parameters are typical for a well-tuned V1 neuron.
    """
    theta_deg = X[:, 0]                     # orientation in degrees
    theta_rad = np.deg2rad(theta_deg)       # convert to radians

    # Parameters (canonical values for a typical V1 neuron)
    theta_pref_deg = 90.0                   # preferred orientation [deg]
    theta_pref_rad = np.deg2rad(theta_pref_deg)
    baseline = 0.5                          # spontaneous activity [au]
    amplitude = 1.0                         # peak evoked response above baseline [au]
    kappa = 2.0                             # concentration (~inverse variance)

    # Von Mises tuning curve on doubled angle (periodic with 180 deg)
    R = baseline + amplitude * np.exp(kappa * (np.cos(2 * (theta_rad - theta_pref_rad)) - 1))
    return R

import numpy as np

def predict(X):
    """
    Predict the mean deconvolved calcium fluorescence response R(theta) of a V1 neuron
    to a static sinusoidal grating of orientation theta (degrees).

    The standard model for orientation tuning is a von Mises function (circular Gaussian),
    which respects the 180-degree periodicity of orientation (0 deg = 180 deg).
    """
    # Extract orientation (degrees) from the input array
    theta = X[:, 0]

    # Canonical parameters of a typical orientation-tuned V1 neuron
    R0 = 0.5          # baseline response (arbitrary units)
    A = 10.0          # peak amplitude above baseline
    theta_pref = 45.0 # preferred orientation (degrees)
    k = 2.0           # concentration (inversely related to tuning width)

    # Convert to radians and compute the von Mises tuning curve
    theta_rad = np.deg2rad(theta)
    theta_pref_rad = np.deg2rad(theta_pref)

    # R(theta) = R0 + A * exp( k * (cos(2*(theta - theta_pref)) - 1) )
    R = R0 + A * np.exp(k * (np.cos(2 * (theta_rad - theta_pref_rad)) - 1.0))

    return R
