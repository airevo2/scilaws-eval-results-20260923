import numpy as np

def predict(X):
    theta = X[:, 0]
    # von Mises (circular Gaussian) orientation tuning curve
    # Factor of 2 in the angle enforces 180-degree periodicity (orientation, not direction)
    R0 = 0.3          # baseline deconvolved fluorescence [au]
    A = 2.0           # amplitude above baseline [au]
    theta_pref = 45.0 # preferred orientation [deg]
    kappa = 2.5       # concentration parameter (~inverse variance)
    R = R0 + A * np.exp(kappa * (np.cos(2 * np.radians(theta - theta_pref)) - 1))
    return R

import numpy as np

def predict(X):
    theta = X[:, 0]
    
    # Typical fitted coefficients for a V1 orientation tuning curve
    R_baseline = 0.3   # baseline calcium fluorescence
    R_amplitude = 2.0  # peak response above baseline
    theta_pref = 90.0  # preferred orientation in degrees
    sigma = 25.0       # tuning width (standard deviation) in degrees
    
    # Orientation is 180-degree periodic (0 and 180 are identical), 
    # so we use a circular Gaussian (wrapped Gaussian) with angle doubled.
    delta_rad = np.deg2rad(2 * (theta - theta_pref))
    sigma_rad = np.deg2rad(sigma)
    
    # Circular Gaussian form: R(theta) = R0 + A * exp((cos(2*(theta - theta_pref)) - 1) / (2 * sigma^2))
    # For small angles, this approximates a standard Gaussian with std dev `sigma` in degrees.
    R = R_baseline + R_amplitude * np.exp((np.cos(delta_rad) - 1) / (2 * sigma_rad**2))
    
    return R

def predict(X):

    # Typical fitted parameters for a V1 orientation tuning curve
    R0 = 0.5         # Baseline calcium fluorescence (au)
    A = 2.0          # Amplitude above baseline
    theta_pref = 90.0 # Preferred orientation (deg)
    sigma = 20.0     # Standard deviation of the tuning curve (deg)
    
    # Compute the shortest circular distance between theta and theta_pref
    # Since static grating orientation is periodic with 180 degrees, max distance is 90
    dtheta = np.mod(theta - theta_pref + 90.0, 180.0) - 90.0
    
    # Circular Gaussian tuning curve
    R = R0 + A * np.exp(-0.5 * (dtheta / sigma)**2)
    
    return R

    sigma = 15.0     # Tuning width / standard deviation (deg)
    
    # Calculate circular distance on the [0, 180) domain
    diff = np.abs(theta - theta_pref)
    circular_diff = np.minimum(diff, 180.0 - diff)
    
    # Circular Gaussian tuning curve
    R = R_0 + A * np.exp(-(circular_diff**2) / (2 * sigma**2))
    
    return R

import numpy as np

def predict(X):
    theta = X[:, 0]
    theta_rad = np.deg2rad(theta)
    
    # Standard circular Gaussian (wrapped) parameters for a V1 orientation tuning curve
    R0 = 0.2       # baseline response [au]
    A = 1.5        # amplitude above baseline [au]
    theta_pref = np.deg2rad(90.0)  # preferred orientation
    sigma = np.deg2rad(25.0)       # tuning width
    
    # Circular Gaussian form: R = R0 + A * exp(-(1 - cos(2 * (theta - theta_pref))) / sigma^2)
    delta = theta_rad - theta_pref
    R = R0 + A * np.exp(-(1.0 - np.cos(2.0 * delta)) / sigma**2)
    
    return R
