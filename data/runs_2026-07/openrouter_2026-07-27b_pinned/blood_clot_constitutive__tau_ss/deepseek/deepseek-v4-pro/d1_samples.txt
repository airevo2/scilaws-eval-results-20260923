import numpy as np

def predict(X):
    """
    Predicts simple-shear Cauchy stress τ_ss [kPa] from:
    - gamma (engineering shear strain)
    - conc_mM (fibrinogen concentration, mM)
    - time_min (clot maturation time, min)

    Standard closed-form: linear elastic (neo-Hookean) response
    τ = G * gamma, with shear modulus G depending linearly on
    fibrinogen concentration and maturation time.
    Constants are plausible values for whole-blood clots.
    """
    gamma = X[:, 0]
    conc_mM = X[:, 1]
    time_min = X[:, 2]

    # Shear modulus coefficients (kPa)
    G0 = 0.5       # base modulus
    alpha = 0.02   # kPa per mM fibrinogen
    beta = 0.003   # kPa per minute maturation time

    G = G0 + alpha * conc_mM + beta * time_min
    tau_ss = G * gamma
    return tau_ss
