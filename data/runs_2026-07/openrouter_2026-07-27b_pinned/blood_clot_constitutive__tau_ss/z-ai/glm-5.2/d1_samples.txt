import numpy as np

def predict(X):
    """
    Predicts the simple-shear Cauchy stress (tau_ss) on a whole-blood thrombus 
    specimen using the standard homogeneous simple-shear closed-form approximation.
    
    The baseline constitutive relationship assumes a linear elastic / incompressible 
    Neo-Hookean response under simple shear, where the shear stress is directly 
    proportional to the engineering shear strain gamma:
        tau_ss = G * gamma
    
    Here, G represents the global shear modulus of the material. The reference 
    baseline treats G as a single fitted constant across all specimens (ignoring 
    conc_mM and time_min), though a richer formula could replace G with a function 
    of the fibrinogen concentration and maturation time.
    """
    gamma = X[:, 0]
    G = 1.0  # kPa, representative fitted global shear modulus for the baseline
    
    return G * gamma
