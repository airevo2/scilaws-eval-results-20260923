import numpy as np

def predict(X):
    gamma = X[:, 0]
    c_fib = X[:, 1]
    t_mat = X[:, 2]

    # ---------------------------------------------------------------
    # Standard closed-form for homogeneous simple shear of an
    # incompressible neo-Hookean (equivalently Mooney-Rivlin)
    # hyperelastic solid:
    #
    #     tau_ss = G * gamma
    #
    # For simple shear (F_12 = gamma, all other F_ij = delta_ij):
    #   B = F F^T  =>  sigma_12 = mu * B_12 = mu * gamma
    # where mu = G is the ground-state shear modulus.
    # ---------------------------------------------------------------

    # Reference shear modulus at reference conditions (kPa)
    G_ref = 1.0

    # Reference concentration and maturation time (midpoints of design)
    c_ref = 20.0   # mM
    t_ref = 90.0   # min

    # ---- Modulus scaling from semiflexible-network / clot physics ----
    # G ~ c^alpha : crosslinked fibrin network scaling
    #   (MacKintosh / Storm theory and experiments give alpha ~ 2-2.5
    #    for tightly cross-linked semiflexible networks)
    alpha = 2.0

    # G ~ t^beta : FXIIIa-mediated cross-linking kinetics
    #   (alpha-chain cross-linking continues on the time-scale of hours;
    #    a weak power law captures the gradual stiffening)
    beta = 0.5

    # Specimen-specific shear modulus
    G = G_ref * (c_fib / c_ref) ** alpha * (t_mat / t_ref) ** beta

    # Simple-shear Cauchy stress (neo-Hookean closed form)
    tau_ss = G * gamma

    return tau_ss
