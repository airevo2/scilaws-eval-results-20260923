import numpy as np

def predict(X):
    lambda11 = X[:, 0]
    lambda22 = X[:, 1]
    
    # Fung-type anisotropic hyperelastic model parameters (representative constants)
    c = 0.05   # MPa
    a1 = 15.0
    a2 = 25.0
    a3 = 10.0
    
    # Green-Lagrange strains
    E11 = 0.5 * (lambda11**2 - 1.0)
    E22 = 0.5 * (lambda22**2 - 1.0)
    
    # Exponential argument
    Q = a1 * E11**2 + a2 * E22**2 + 2.0 * a3 * E11 * E22
    
    # Second Piola-Kirchhoff stress in the cranial-caudal direction
    S22 = c * np.exp(Q) * (a2 * E22 + a3 * E11)
    
    # True Cauchy stress (assuming incompressibility, J=1)
    sigma22 = (lambda22**2) * S22
    
    return sigma22
