    # W = (C / 2) * (exp(Q) - 1)
    # Q = c1 * E11^2 + c2 * E22^2 + 2 * c12 * E11 * E22
    # E_ii = 0.5 * (lambda_ii^2 - 1)

    """

import numpy as np

def predict(X):
    """
    Predicts the Cauchy stress sigma_22 [MPa] for a planar mouse-skin specimen 
    under biaxial loading using a standard Fung-type exponential hyperelastic model.
    
    Inputs:
    X[:, 0] = lambda_11 (lateral stretch)
    X[:, 1] = lambda_22 (cranial-caudal stretch)
    """
    lambda_11 = X[:, 0]
    lambda_22 = X[:, 1]
    
    # Representative material constants for skin (can be fitted to specific data)
    c = 0.01   # MPa
    b1 = 10.0  # lateral stiffness parameter
    b2 = 10.0  # cranial-caudal stiffness parameter
    b3 = 2.0   # coupling parameter
    
    # Green-Lagrange strains
    E11 = 0.5 * (lambda_11**2 - 1.0)
    E22 = 0.5 * (lambda_22**2 - 1.0)
    
    # Exponential strain energy term Q
    Q = b1 * E11**2 + b2 * E22**2 + 2.0 * b3 * E11 * E22
    
    # Second Piola-Kirchhoff stress S22
    S22 = c * (b2 * E22 + b3 * E11) * np.exp(Q)
    
    # Cauchy stress sigma_22 (assuming incompressibility J=1 and plane stress sigma_33=0)
    sigma_22 = lambda_22**2 * S22
    
    return sigma_22
