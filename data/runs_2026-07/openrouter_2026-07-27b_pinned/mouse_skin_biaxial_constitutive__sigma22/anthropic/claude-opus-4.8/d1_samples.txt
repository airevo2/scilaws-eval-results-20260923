import numpy as np

def predict(X):
    # Incompressible, isotropic Fung-type exponential (Demiray/Delfino) model
    # commonly used for soft biological membranes such as skin.
    #
    # Strain energy:  W = (c/(2b)) * (exp(b*(I1 - 3)) - 1)
    # Incompressibility (planar biaxial):  lambda33 = 1/(lambda11 * lambda22)
    # Cauchy stress (sigma33 = 0 plane-stress condition):
    #   sigma22 = c * (lambda22^2 - lambda33^2) * exp(b*(I1 - 3))

    lam11 = np.asarray(X[:, 0], dtype=float)
    lam22 = np.asarray(X[:, 1], dtype=float)

    lam33 = 1.0 / (lam11 * lam22)

    I1 = lam11**2 + lam22**2 + lam33**2

    # Fitted material constants (representative for mouse skin, MPa)
    c = 0.30   # shear-like stiffness parameter [MPa]
    b = 2.5    # dimensionless exponential stiffening parameter

    sigma22 = c * (lam22**2 - lam33**2) * np.exp(b * (I1 - 3.0))

    return sigma22

import numpy as np

def predict(X):
    # Fung-type exponential (orthotropic) strain-energy model for planar skin
    #   W = (c/2) * ( exp(Q) - 1 )
    #   Q = a1*E11^2 + a2*E22^2 + 2*a3*E11*E22
    # 2nd Piola-Kirchhoff:  S22 = c*exp(Q)*(a2*E22 + a3*E11)
    # Cauchy (planar, incompressible): sigma22 = lambda22^2 * S22
    lam11 = X[:, 0]
    lam22 = X[:, 1]

    # Green-Lagrange strains
    E11 = 0.5 * (lam11**2 - 1.0)
    E22 = 0.5 * (lam22**2 - 1.0)

    # Fitted material constants (representative mouse-skin values)
    c  = 0.30      # MPa, stiffness scale
    a1 = 2.0       # lateral fiber term
    a2 = 3.5       # cranial-caudal fiber term
    a3 = 0.8       # coupling term

    Q = a1 * E11**2 + a2 * E22**2 + 2.0 * a3 * E11 * E22
    S22 = c * np.exp(Q) * (a2 * E22 + a3 * E11)

    sigma22 = lam22**2 * S22
    return sigma22

import numpy as np

def predict(X):
    # Fung-type exponential (orthotropic) strain-energy model for planar
    # biaxial soft tissue, one of the most commonly cited skin models.
    #
    #   W = (c/2) * [ exp(Q) - 1 ],
    #   Q = a11*E11^2 + a22*E22^2 + 2*a12*E11*E22
    #
    # with Green-Lagrange strains  E_ii = (lambda_ii^2 - 1)/2.
    # Second Piola-Kirchhoff:  S22 = dW/dE22 = c*(a22*E22 + a12*E11)*exp(Q)
    # Cauchy stress (fiber/CC direction): sigma22 = lambda22^2 * S22
    #
    # Fitted material constants (representative for mouse skin, MPa):
    c   = 0.030     # MPa, stress-like scaling
    a11 = 6.0       # dimensionless
    a22 = 9.0       # dimensionless (stiffer cranial-caudal direction)
    a12 = 2.0       # dimensionless coupling

    lam11 = np.asarray(X[:, 0], dtype=float)
    lam22 = np.asarray(X[:, 1], dtype=float)

    E11 = 0.5 * (lam11**2 - 1.0)
    E22 = 0.5 * (lam22**2 - 1.0)

    Q = a11 * E11**2 + a22 * E22**2 + 2.0 * a12 * E11 * E22
    S22 = c * (a22 * E22 + a12 * E11) * np.exp(Q)
    sigma22 = lam22**2 * S22

    return sigma22

import numpy as np

def predict(X):
    # Fung-type anisotropic exponential model for planar biaxial skin.
    # Green-Lagrange strains from principal stretches
    lam11 = X[:, 0]
    lam22 = X[:, 1]
    E11 = 0.5 * (lam11**2 - 1.0)
    E22 = 0.5 * (lam22**2 - 1.0)

    # Fitted material constants (Fung strain-energy: W = c/2 (exp(Q)-1))
    c  = 0.05     # MPa, stiffness scale
    a1 = 8.0      # lateral   (1-1)
    a2 = 10.0     # cranial-caudal (2-2)
    a3 = 1.5      # coupling

    Q = a1 * E11**2 + a2 * E22**2 + 2.0 * a3 * E11 * E22

    # Cauchy stress in the 2-direction (in-plane Fung form)
    sigma22 = c * lam22**2 * (a2 * E22 + a3 * E11) * np.exp(Q)
    return sigma22
