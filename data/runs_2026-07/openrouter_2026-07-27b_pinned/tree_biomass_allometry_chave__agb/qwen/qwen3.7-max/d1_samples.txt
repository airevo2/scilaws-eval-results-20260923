import numpy as np

def predict(X):
    rho = X[:, 0]  # wood density [g/cm^3]
    D   = X[:, 1]  # diameter at breast height [cm]
    H   = X[:, 2]  # total tree height [m]

    # Chave et al. (2005) pantropical allometric model
    # AGB = 0.0673 * (rho * D^2 * H)^0.976
    AGB = 0.0673 * (rho * D**2 * H) ** 0.976

    return AGB

import numpy as np

def predict(X):
    rho = X[:, 0]
    D = X[:, 1]
    H = X[:, 2]
    
    # Chave et al. (2005) pantropical allometric equation for moist forests
    agb_kg = 0.0509 * rho * (D ** 2) * H
    
    return agb_kg
