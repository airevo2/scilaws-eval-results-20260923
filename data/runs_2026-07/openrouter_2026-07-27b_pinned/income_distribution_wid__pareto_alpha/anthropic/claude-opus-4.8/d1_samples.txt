import numpy as np

def predict(X):
    # Top income shares follow a Pareto law:
    #   S(p) = p^(1 - 1/alpha),   so  ln S(p) = (1 - 1/alpha) * ln p
    # with S(1) = 1 (the top 100% holds all income => intercept 0).
    #
    # alpha is the Pareto index; the inverted-Pareto coefficient is
    # b = alpha/(alpha-1).  Empirically alpha ~ 1.6-1.8 for pre-tax
    # national income, and inequality has risen slightly over time
    # (alpha falling), which we capture with a small time trend.

    ln_p = X[:, 0]
    year = X[:, 2]

    # Baseline Pareto index ~1.7 in 2000, slowly declining (rising inequality)
    alpha = 1.70 - 0.004 * (year - 2000.0)
    alpha = np.clip(alpha, 1.2, 2.5)

    exponent = 1.0 - 1.0 / alpha   # (alpha-1)/alpha, ~0.41

    ln_S = exponent * ln_p
    return ln_S

import numpy as np

def predict(X):
    # Pareto / power-law form for top income shares:
    #   S(p) proportional to p^(1 - 1/alpha)
    #   => ln S(p) = c + b * ln p,   with b = 1 - 1/alpha  (0<b<1)
    #
    # Empirically (WID), top-10% ~ 0.50, top-1% ~ 0.20, giving b ~ 0.4
    # (Pareto index alpha ~ 1.67, inverted-Pareto beta ~ 2.5).
    # A small positive time trend captures the rise in top shares.

    ln_p = X[:, 0].astype(float)   # ln p
    year = X[:, 2].astype(float)

    b   = 0.40      # power-law (Pareto) slope, 1 - 1/alpha
    c0  = 0.22      # intercept at reference year
    trend = 0.003   # per-year drift in log top share
    t0  = 2000.0

    log_top_share = c0 + b * ln_p + trend * (year - t0)

    return log_top_share

import numpy as np

def predict(X):
    # Pareto law for top income shares:
    #   S(p) = C * p^((alpha-1)/alpha)   =>   ln S(p) = ln C + ((alpha-1)/alpha) * ln p
    # The exponent b = (alpha-1)/alpha lies in (0,1); smaller b => more concentration.
    ln_p   = X[:, 0].astype(float)          # ln p
    year   = X[:, 2].astype(float)
    region = X[:, 3]

    # Region-specific baseline Pareto exponent (concentration level)
    region_exp = {
        'Europe':   0.62,
        'Asia':     0.55,
        'Africa':   0.50,
        'Americas': 0.52,
        'Oceania':  0.60,
    }
    # Region-specific intercept ln C
    region_int = {
        'Europe':   -0.02,
        'Asia':      0.03,
        'Africa':    0.05,
        'Americas':  0.04,
        'Oceania':   0.00,
    }

    reg = np.array([str(r) for r in region])
    b   = np.array([region_exp.get(r, 0.57) for r in reg])
    c   = np.array([region_int.get(r, 0.00) for r in reg])

    # Inequality has risen over recent decades: exponent falls slightly with time
    # (lower exponent -> higher top share).
    b_eff = b - 0.0015 * (year - 1980.0)
    b_eff = np.clip(b_eff, 0.30, 0.95)

    ln_S = c + b_eff * ln_p
    return ln_S

import numpy as np

def predict(X):
    ln_p   = X[:, 0].astype(float)   # ln(p), p <= 1
    year   = X[:, 2].astype(float)
    region = X[:, 3]

    # --- Pareto model for top shares ---------------------------------
    # For a Pareto upper tail with coefficient alpha, the share of income
    # held by the top fraction p is  S(p) = p^(1 - 1/alpha),
    # hence   ln S(p) = (1 - 1/alpha) * ln p.
    # The inverted-Pareto (Gabaix/Piketty) coefficient b = 1/(1-1/alpha)
    # controls tail fatness; alpha ~ 1.5 -> slope ~ 0.33.
    # -----------------------------------------------------------------

    # Baseline Pareto exponent, drifting down over time (rising inequality)
    alpha = 1.55 - 0.004 * (year - 2000.0)

    # Regional adjustment to the tail exponent (higher alpha = less unequal)
    reg = np.asarray(region).astype(str)
    reg_adj = np.zeros_like(alpha)
    reg_adj[reg == 'Europe']   =  0.15
    reg_adj[reg == 'Oceania']  =  0.10
    reg_adj[reg == 'Asia']     = -0.05
    reg_adj[reg == 'Americas'] = -0.10
    reg_adj[reg == 'Africa']   = -0.20
    alpha = alpha + reg_adj

    # keep alpha in a sensible (>1) range
    alpha = np.clip(alpha, 1.05, 3.0)

    slope = 1.0 - 1.0 / alpha          # (1 - 1/alpha), between 0 and 1

    # Small positive intercept: at p=1 the true share is 1 (ln=0);
    # a slight curvature offset captures departure from a pure power law.
    intercept = 0.05

    log_top_share = intercept + slope * ln_p
    return log_top_share

import numpy as np

def predict(X):
    # Pareto-Lorenz relationship for top income shares:
    #   S(p) = C * p^(1 - 1/alpha)   =>   ln S(p) = (1 - 1/alpha) * ln p + ln C
    # where alpha is the Pareto exponent (inverted Pareto beta = alpha/(alpha-1)).
    # Empirically for WID top shares, slope (1-1/alpha) ~ 0.43 (alpha ~ 1.75, beta ~ 2.35).

    ln_p = X[:, 0].astype(float)      # ln(p)
    year = X[:, 2].astype(float)
    region = X[:, 3]

    # Pareto power-law slope and intercept
    slope = 0.43                      # = 1 - 1/alpha
    intercept = 0.06                  # = ln C (baseline concentration)

    # Mild upward time trend in concentration since ~1980
    trend = 0.0025 * (year - 1980.0)

    # Region shifts (inequality level differences); Europe as reference
    region = np.asarray(region, dtype=object)
    region_effect = np.zeros_like(ln_p)
    offsets = {
        "Europe":   0.00,
        "Asia":     0.10,
        "Africa":   0.20,
        "Americas": 0.15,
        "Oceania":  0.05,
    }
    for r, off in offsets.items():
        region_effect[region == r] = off

    return slope * ln_p + intercept + trend + region_effect
