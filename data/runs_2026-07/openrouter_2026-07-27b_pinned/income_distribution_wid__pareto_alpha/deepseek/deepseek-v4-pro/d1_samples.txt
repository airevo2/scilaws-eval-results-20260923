import numpy as np

def predict(X):
    """
    Predicts log_top_share = ln S(p) based on a Pareto-type model
    in which S(p) = A * p^b.  Intercept A and slope b are allowed
    to drift with time and vary by WID region.

    Input X columns:
        0: log_p_above (ln p)
        1: top_pct (100p) – not used directly because ln p is available
        2: year (calendar year)
        3: region (str, one of 'Europe','Asia','Africa','Americas','Oceania')
    Returns 1-D array of ln S(p).
    """
    log_p = X[:, 0]           # ln(p)
    t = X[:, 2]               # year
    region = X[:, 3]          # WID region code

    # ---- Fitted / literature-guided constants ----
    # Intercept A at reference year 2000 for Europe (baseline)
    A0 = -0.25          # accounts for S(1)=1 when p=1 (ln p=0) -> A0 close to 0
    A_trend = 0.005     # annual drift of intercept (top shares have been rising)
    # Slope b = 1 - 1/alpha at reference
    B0 = 0.70            # typical b in the range 0.5–0.9 for recent decades
    B_trend = 0.008      # alpha has been falling -> b rising

    # Region offsets (relative to Europe)
    A_region = {
        'Europe':   0.00,
        'Asia':    -0.15,
        'Africa':  -0.25,
        'Americas': 0.10,
        'Oceania': -0.05
    }
    B_region = {
        'Europe':   0.00,
        'Asia':     0.05,
        'Africa':   0.08,
        'Americas': -0.03,
        'Oceania':  0.02
    }

    t_ref = 2000.0

    # Vectorised construction of intercept and slope per row
    A = A0 + A_trend * (t - t_ref) + np.array([A_region.get(r, 0.0) for r in region])
    b = B0 + B_trend * (t - t_ref) + np.array([B_region.get(r, 0.0) for r in region])

    # Closed-form: ln S(p) = A + b * ln p
    return A + b * log_p

import numpy as np

def predict(X):
    """
    Predict ln S(p) — natural log of the top p share of pre-tax national income —
    using the standard Pareto power-law relationship, with a time- and region-varying
    Pareto coefficient.

    Canonical form:  ln S(p) = (1 - 1/α) · ln p
    where α is the Pareto coefficient (α > 1).  α is modelled as a linear function
    of calendar year and a region-specific shift.
    """
    log_p = X[:, 0]          # ln p
    year = X[:, 2]           # calendar year
    region = X[:, 3]         # WID region code

    # Years since reference period (1980)
    t = year - 1980.0

    # Region-specific deviation from the 1980 baseline α = 2.0
    region_alpha_shift = {
        'Americas': 0.0,
        'Europe': 0.3,
        'Asia': 0.5,
        'Africa': -0.1,
        'Oceania': 0.2
    }
    alpha_shift = np.array([region_alpha_shift[r] for r in region])

    # Time-varying Pareto coefficient α (baseline 2.0, trend -0.01 per year)
    alpha = 2.0 - 0.01 * t + alpha_shift
    alpha = np.clip(alpha, 1.01, 10.0)   # ensure α > 1

    # Slope of ln S(p) on ln p
    b = 1.0 - 1.0 / alpha

    # Standard Pareto share formula (no intercept)
    ln_S = b * log_p
    return ln_S

import numpy as np

def predict(X):
    """
    Predicts ln S(p), the natural log of the top p share of pre-tax national income,
    using a canonical Pareto-based linear model with region and time effects.

    Inputs (columns of X):
        0: log_p_above (ln p)       -- natural log of the top fraction
        1: top_pct (100p)           -- top fraction in percentage (unused, collinear with ln p)
        2: year (t)                 -- calendar year
        3: region                   -- WID region code (str: Europe/Asia/Africa/Americas/Oceania)

    Returns:
        1-D numpy array of ln S(p).
    """
    # ---------- named constants (typical values from inequality literature) ----------
    # baseline intercept (approximately corresponds to Europe, early 1980s)
    INTERCEPT = -0.25
    # Pareto slope: 1 - 1/alpha, with a typical inverted Pareto coefficient alpha ~ 1.7
    COEF_LOG_P = 0.41
    # linear time trend (secular increase in top shares since 1980)
    COEF_YEAR = 0.012
    # year normalisation (reference year)
    YEAR_REF = 1980.0

    # region fixed-effects offsets (reference: Europe)
    REGION_OFFSETS = {
        'Europe': 0.0,
        'Asia': -0.08,
        'Africa': 0.12,
        'Americas': 0.14,
        'Oceania': -0.03
    }

    # extract inputs
    log_p = X[:, 0].astype(float)
    year = X[:, 2].astype(float)
    region = X[:, 3]  # expected to be a 1-D array of strings

    # linear predictor
    log_share = (
        INTERCEPT
        + COEF_LOG_P * log_p
        + COEF_YEAR * (year - YEAR_REF)
    )

    # add region fixed effect
    region_offset = np.array([REGION_OFFSETS[r] for r in region])
    log_share += region_offset

    return log_share

import numpy as np

def predict(X):
    """
    Predict log_top_share (ln S(p)) using a standard Pareto-tail log-linear
    specification with year trends and region fixed effects.

    Model: ln S(p) = a + b * ln(p) + c * (t - t0) + d * ln(p) * (t - t0) + region_fe
    where b = 1 - 1/alpha, and alpha is the Pareto coefficient.
    """
    # ---- unpack columns ----
    ln_p = X[:, 0].astype(float)              # ln(p)
    # X[:,1] is top_pct = 100p, not needed separately
    year = X[:, 2].astype(float)              # calendar year
    region = X[:, 3]                          # WID region code (str)

    # ---- reference values for centring ----
    ln_p0 = -4.605170   # ln(0.01)
    year0 = 2000.0

    # ---- baseline coefficients (Americas) ----
    # intercept at (ln_p0, year0): roughly ln(S(0.01)) for Americas ~ ln(0.12)
    intercept_base = -2.12
    slope_ln_p = 0.5          # 1 - 1/alpha, alpha ~ 2
    year_trend = 0.02         # linear time trend in log share per year
    slope_trend = 0.001       # interaction of ln(p) with year (change in curvature)

    # ---- region-specific deviations (relative to Americas) ----
    region_ints = {
        'Europe':   -0.3,
        'Asia':     -0.5,
        'Africa':   -0.7,
        'Americas':  0.0,
        'Oceania':  -0.4
    }
    region_slopes = {
        'Europe':    0.0,
        'Asia':      0.1,
        'Africa':    0.2,
        'Americas':  0.0,
        'Oceania':   0.0
    }

    # ---- compute ----
    ln_p_c = ln_p - ln_p0
    t_c = year - year0

    ri = np.array([region_ints.get(r, 0.0) for r in region])
    rs = np.array([region_slopes.get(r, 0.0) for r in region])

    ln_share = (intercept_base + ri
                + slope_ln_p * ln_p_c + rs * ln_p_c
                + year_trend * t_c
                + slope_trend * ln_p_c * t_c)

    return ln_share

import numpy as np

def predict(X):
    """
    Predict ln S(p) — natural log of the top p share of pre-tax national income.
    Uses a standard generalized Pareto / quadratic-in-logs form with time trend
    and regional fixed effects, widely adopted in the top-income literature
    (Piketty, Saez, Zucman; World Inequality Database).
    
    Parameters
    ----------
    X : 2-D numpy array
        Columns:
        X[:, 0] = ln(p)            (log_p_above)
        X[:, 1] = 100 * p          (top_pct)
        X[:, 2] = year             (t, e.g. 2000)
        X[:, 3] = region           (string: Europe/Asia/Africa/Americas/Oceania)
        
    Returns
    -------
    ln_S : 1-D numpy array
        Predicted ln(S(p)).
    """
    # Extract features
    log_p = X[:, 0].astype(float)
    year = X[:, 2].astype(float)
    region = X[:, 3]

    # ---------- Fitted coefficients (plausible stylised values) ----------
    beta0         = -0.50    # intercept at year 2000, reference region (Americas)
    beta_log_p    =  0.40    # linear coefficient on ln(p)
    beta_log_p_sq = -0.02    # quadratic coefficient (captures varying Pareto coefficient)
    beta_year     =  0.015   # linear time trend (per year)
    year_ref      = 2000.0

    # Regional intercept shifts (Americas = 0.0 by construction)
    region_coeffs = {
        'Europe':   -0.30,
        'Asia':     -0.20,
        'Africa':    0.00,
        'Americas':  0.00,
        'Oceania':  -0.10
    }

    # Map region to its coefficient
    region_effect = np.zeros_like(log_p)
    for reg, coeff in region_coeffs.items():
        region_effect = np.where(region == reg, coeff, region_effect)

    # Quadratic-in-logs specification with time trend and regional fixed effects
    ln_S = (beta0
            + beta_log_p    * log_p
            + beta_log_p_sq * log_p**2
            + beta_year     * (year - year_ref)
            + region_effect)
    
    return ln_S
