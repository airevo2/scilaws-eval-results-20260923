import numpy as np

def predict(X):
    """
    Predict log10(L/Lsun) from log10(M/Msun) using the canonical
    main-sequence mass-luminosity relation L ∝ M^3.5.
    Metallicity (column 1) is a distractor and is ignored.
    """

    log_L = 3.5 * log_M
