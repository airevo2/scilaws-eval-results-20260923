import numpy as np

def predict(X):
    # Standard mass-luminosity relation: L/L_sun = (M/M_sun)^alpha
    # Canonical exponent alpha ~ 3.5 for main-sequence stars
    log_M = X[:, 0]
    alpha = 3.5
    log_L = alpha * log_M
    return log_L
