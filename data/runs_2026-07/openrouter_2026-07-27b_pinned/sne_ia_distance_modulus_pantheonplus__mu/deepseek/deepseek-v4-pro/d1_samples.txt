import numpy as np

def predict(X):
    """
    Predict the standardised Type Ia supernova distance modulus mu (mag)
    as a function of CMB-frame redshift z_hd.

    Parameters
    ----------
    X : np.ndarray
        A 2-D array with shape (N, 1) or (N,). X[:, 0] contains the
        redshift values z_hd.

    Returns
    -------
    mu : np.ndarray
        A 1-D array of the corresponding distance modulus values.
        mu = 5 * log10(d_L / 10 pc), with d_L in parsecs.
    """
    z = np.asarray(X[:, 0]).ravel()
    
    # Cosmological parameters (flat ΛCDM, typical values from SNe Ia analyses)
    c_km_s = 299792.458      # speed of light in km/s
    H0 = 70.0                # Hubble constant in km/s/Mpc
    Omega_m = 0.3            # matter density parameter
    Omega_L = 0.7            # dark energy density parameter
    
    # Hubble distance in Mpc
    dh = c_km_s / H0  # ~ 4282.75 Mpc
    
    mu = np.empty_like(z)
    for i, zi in enumerate(z):
        if zi <= 0:
            mu[i] = -np.inf
            continue
        
        # Numerical integration of 1/E(z) from 0 to zi
        z_prime = np.linspace(0, zi, 2000)  # dense grid for accuracy
        E = np.sqrt(Omega_m * (1 + z_prime)**3 + Omega_L)
        integral = np.trapz(1.0 / E, z_prime)
        
        # Luminosity distance in Mpc
        d_L = (1 + zi) * dh * integral
        
        # Distance modulus: d_L in Mpc -> pc: d_L_pc = d_L * 1e6
        # mu = 5 * log10(d_L * 1e6 / 10) = 5 * log10(d_L) + 25
        mu[i] = 5.0 * np.log10(d_L) + 25.0
    
    return mu

import numpy as np

def predict(X):
    """
    Compute the standardised Type Ia supernova distance modulus mu (magnitudes)
    as a function of CMB-frame redshift z_hd, assuming a flat Lambda-CDM cosmology.

    mu = 5 * log10( d_L(z) / 10 pc ) with d_L in Mpc: mu = 5 * log10(d_L) + 25.

    Parameters
    ----------
    X : np.ndarray, shape (N, 1)
        X[:, 0] = z_hd, the Hubble-diagram CMB-frame redshift.

    Returns
    -------
    mu : np.ndarray, shape (N,)
        Distance modulus in magnitudes.
    """
    z = X[:, 0].astype(np.float64)

    # Cosmological parameters (flat Lambda-CDM)
    H0 = 70.0          # km/s/Mpc
    Omega_m = 0.3
    Omega_L = 0.7
    c = 299792.458     # km/s

    # Hubble distance in Mpc
    DH = c / H0        # Mpc

    # E(z) = sqrt(Omega_m * (1+z)^3 + Omega_Lambda)
    def E(z_arr):
        return np.sqrt(Omega_m * (1.0 + z_arr)**3 + Omega_L)

    # Numerical integration: 1000 steps per redshift
    N_steps = 1000
    mu = np.empty_like(z)

    for i, zi in enumerate(z):
        if zi <= 0.0:
            mu[i] = -np.inf   # mathematically undefined at z=0 (or set to 0 for very small)
            continue
        z_grid = np.linspace(0.0, zi, N_steps)
        integrand = 1.0 / E(z_grid)
        integral = np.trapz(integrand, z_grid)
        dL = (1.0 + zi) * DH * integral          # Mpc
        mu[i] = 5.0 * np.log10(dL) + 25.0        # distance modulus

    return mu

import numpy as np

def predict(X):
    """
    Predict the standardised Type Ia supernova distance modulus mu (mag)
    as a function of CMB-frame redshift z (X[:, 0]) using the standard
    flat Lambda-CDM cosmological model.

    Parameters
    ----------
    X : np.ndarray, shape (N,)
        Redshift values z = X[:, 0].

    Returns
    -------
    mu : np.ndarray, shape (N,)
        Distance modulus values mu = 5 * log10(d_L / 10 pc).
    """
    # Cosmological constants (flat Lambda-CDM)
    c = 299792.458  # speed of light, km/s
    H0 = 70.0       # Hubble constant, km/s/Mpc
    Omega_m = 0.3
    Omega_L = 1.0 - Omega_m

    # Redshift array
    z = X[:, 0].astype(np.float64)

    # Hubble parameter E(z) = H(z)/H0
    def E(z):
        return np.sqrt(Omega_m * (1.0 + z)**3 + Omega_L)

    # Integration grid from 0 to max(z) (fine enough for typical cosmology)
    z_max = np.max(z)
    if z_max < 1e-4:
        z_max = 0.1
    n_grid = 2000
    z_grid = np.linspace(0.0, z_max, n_grid)
    E_grid = E(z_grid)

    # Cumulative trapezoidal integration of 1/E(z)
    dz = np.diff(z_grid)
    integrand = 1.0 / E_grid
    integral = np.zeros(n_grid)
    for i in range(1, n_grid):
        integral[i] = integral[i-1] + 0.5 * (integrand[i-1] + integrand[i]) * dz[i-1]

    # Interpolate the integral at input redshifts
    I_z = np.interp(z, z_grid, integral)

    # Luminosity distance in Mpc: d_L = (1+z) * (c/H0) * I(z)
    d_L_Mpc = (1.0 + z) * (c / H0) * I_z

    # Distance modulus: mu = 5*log10(d_L / 10 pc)
    # 1 Mpc = 10^6 pc, so d_L in pc = d_L_Mpc * 1e6
    # mu = 5*log10(d_L_Mpc * 1e6) - 5 = 5*log10(d_L_Mpc) + 25
    mu = 5.0 * np.log10(d_L_Mpc) + 25.0

    return mu

import numpy as np
from scipy.integrate import quad

def predict(X):
    """
    Standard distance modulus for SN Ia in a flat Lambda-CDM universe.
    mu = 5 * log10(d_L / 10 pc) with d_L in parsecs.
    Input: X[:,0] = z_hd (CMB-frame redshift)
    Returns: mu [mag]
    """
    # Cosmological parameters (Planck 2018: TT,TE,EE+lowE+lensing)
    H0 = 67.4          # km/s/Mpc
    Omega_m = 0.315
    Omega_L = 0.685
    c = 299792.458     # speed of light in km/s

    z = X[:, 0].astype(float)

    def E(z):
        return np.sqrt(Omega_m * (1 + z)**3 + Omega_L)

    # Compute luminosity distance in Mpc for each redshift
    d_L_Mpc = np.empty_like(z)
    for i, zi in enumerate(z):
        if zi == 0:
            d_L_Mpc[i] = 0.0
        else:
            integral, _ = quad(lambda z_: 1.0 / E(z_), 0.0, zi)
            d_L_Mpc[i] = (1.0 + zi) * (c / H0) * integral

    # Distance modulus: mu = 5*log10(d_L/pc) - 5 = 5*log10(d_L/Mpc) + 25
    mu = 5.0 * np.log10(d_L_Mpc) + 25.0
    return mu

import numpy as np

def predict(X):
    """
    Predict the standardised distance modulus μ for Type Ia supernovae
    using a flat ΛCDM cosmology with Ω_m = 0.3, Ω_Λ = 0.7, H₀ = 70 km/s/Mpc.
    The luminosity distance is computed via numerical integration, and
    μ = 5 log₁₀(d_L / 10 pc) = 5 log₁₀(d_L/Mpc) + 25.
    """
    z = X[:, 0]                     # redshift

    # Canonical flat ΛCDM parameters
    Om = 0.3
    OL = 0.7
    H0 = 70.0                        # km/s/Mpc
    c = 299792.458                   # km/s
    DH = c / H0                      # Hubble distance in Mpc

    # Integration grid over [0, zmax]
    zmax = np.max(z)
    nz = 10000
    zgrid = np.linspace(0, zmax, nz)
    dz = zgrid[1] - zgrid[0]
    Ez = np.sqrt(Om * (1 + zgrid)**3 + OL)

    # Cumulative trapezoidal integral of 1/E(z) from 0
    cum_int = np.zeros_like(zgrid)
    cum_int[1:] = np.cumsum(0.5 * (Ez[1:] + Ez[:-1]) * dz)

    # Interpolate to get the integral for each z
    I_z = np.interp(z, zgrid, cum_int)

    # Luminosity distance in Mpc
    dL = (1 + z) * DH * I_z

    # Distance modulus
    mu = 5.0 * np.log10(dL) + 25.0
    return mu
