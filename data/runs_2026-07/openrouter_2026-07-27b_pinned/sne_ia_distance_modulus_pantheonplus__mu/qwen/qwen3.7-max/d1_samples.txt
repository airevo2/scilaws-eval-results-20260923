import numpy as np

def predict(X):
    z = X[:, 0]
    
    # Standard fiducial flat Lambda-CDM cosmology
    c = 299792.458  # Speed of light in km/s
    H0 = 70.0       # Hubble constant in km/s/Mpc
    Om = 0.3        # Matter density parameter
    Ol = 0.7        # Dark energy density parameter
    
    dH = c / H0     # Hubble distance in Mpc
    
    # Determine upper limit for integration grid
    z_max = np.max(z) if z.size > 0 else 2.5
    z_max = max(z_max, 2.5)
    
    # High-resolution grid for numerical integration
    N = 5000
    z_grid = np.linspace(0, z_max, N)
    E_grid = np.sqrt(Om * (1 + z_grid)**3 + Ol)
    inv_E = 1.0 / E_grid
    
    # Cumulative trapezoidal integration
    dz = z_grid[1] - z_grid[0]
    cum_int = np.zeros(N)
    cum_int[1:] = np.cumsum((inv_E[:-1] + inv_E[1:]) / 2.0) * dz
    
    # Interpolate the integral to the exact redshift values
    int_z = np.interp(z, z_grid, cum_int)
    
    # Luminosity distance in Mpc
    dL = (1 + z) * dH * int_z
    
    # Distance modulus: mu = 5 * log10(d_L / 10 pc)
    # Since d_L is in Mpc, 10 pc = 10^-5 Mpc
    mu = 5.0 * np.log10(dL) + 25.0
    
    return mu

import numpy as np

def predict(X):
    z = X[:, 0]

    # Flat LambdaCDM parameters (commonly cited canonical values)
    H0 = 70.0             # Hubble constant [km/s/Mpc]
    c_light = 299792.458  # Speed of light [km/s]
    Omega_m = 0.3         # Matter density parameter
    Omega_L = 0.7         # Dark energy density parameter (flat: Omega_L = 1 - Omega_m)

    # Compute the comoving line-of-sight integral: int_0^z dz' / E(z')
    # E(z) = sqrt(Omega_m * (1+z)^3 + Omega_L)
    n_steps = 2000
    t = np.linspace(0.0, 1.0, n_steps + 1)          # integration parameter [0, 1]
    z_grid = z[:, np.newaxis] * t[np.newaxis, :]    # shape (N, n_steps+1)

    E_z = np.sqrt(Omega_m * (1.0 + z_grid)**3 + Omega_L)
    integrand = 1.0 / E_z

    # Trapezoidal rule integration
    dz_step = z / n_steps
    integral = np.sum((integrand[:, :-1] + integrand[:, 1:]) / 2.0, axis=1) * dz_step

    # Luminosity distance in Mpc:  d_L = (c / H0) * (1 + z) * integral
    d_L = (c_light / H0) * (1.0 + z) * integral

    # Distance modulus:  mu = 5 * log10(d_L / 10 pc)
    # Since d_L is in Mpc and 1 Mpc = 1e6 pc:
    #   d_L / 10 pc = d_L * 1e5
    #   mu = 5 * log10(d_L) + 5 * log10(1e5) = 5 * log10(d_L) + 25
    mu = 5.0 * np.log10(d_L) + 25.0

    return mu

import numpy as np

def predict(X):
    z = X[:, 0]
    
    # Fiducial flat LambdaCDM cosmological parameters
    c = 299792.458  # Speed of light in km/s
    H0 = 70.0       # Hubble constant in km/s/Mpc
    Om = 0.3        # Matter density parameter
    Ol = 0.7        # Dark energy density parameter
    
    # Handle empty input gracefully
    if z.size == 0:
        return np.array([])
        
    z_max = np.max(z)
    N = 10000
    zp = np.linspace(0, z_max, N)
    dz = z_max / (N - 1)
    
    # Hubble parameter E(z) = H(z)/H0
    Ez = np.sqrt(Om * (1 + zp)**3 + Ol)
    inv_Ez = 1.0 / Ez
    
    # Cumulative trapezoidal integration for the comoving distance integral
    integral_grid = np.zeros(N)
    integral_grid[1:] = np.cumsum((inv_Ez[:-1] + inv_Ez[1:]) / 2.0) * dz
    
    # Interpolate the integral to the exact requested redshifts
    integral_z = np.interp(z, zp, integral_grid)
    
    # Luminosity distance in Mpc
    dL = (1 + z) * (c / H0) * integral_z
    
    # Standardised distance modulus
    mu = 5.0 * np.log10(dL) + 25.0
    
    return mu

import numpy as np

def predict(X):
    """
    Standardised Type Ia SN distance modulus mu(z) for a flat LCDM cosmology.
    
    mu = 5 * log10(d_L [Mpc]) + 25
    d_L(z) = (c / H0) * (1 + z) * integral_0^z dz' / E(z')
    E(z) = sqrt(Omega_m * (1+z)^3 + Omega_Lambda)
    """
    z = X[:, 0]

    # Fiducial flat-LCDM cosmological parameters
    c_light = 299792.458   # speed of light [km/s]
    H0 = 70.0              # Hubble constant [km/s/Mpc]
    Omega_m = 0.3          # matter density parameter
    Omega_Lambda = 0.7     # dark energy density parameter (flat: 1 - Omega_m)

    # Build a fine integration grid covering all requested redshifts
    N = 2000
    z_max = np.max(z)
    z_grid = np.linspace(0.0, max(z_max * 1.01, 0.01), N)

    # Dimensionless Hubble parameter E(z) and integrand 1/E(z)
    E_z = np.sqrt(Omega_m * (1.0 + z_grid) ** 3 + Omega_Lambda)
    integrand = 1.0 / E_z

    # Cumulative trapezoidal integration of int_0^z dz'/E(z')
    dz_grid = np.diff(z_grid)
    cum_int = np.zeros(N)
    cum_int[1:] = np.cumsum(0.5 * (integrand[:-1] + integrand[1:]) * dz_grid)

    # Interpolate the cumulative integral to each target redshift
    int_z = np.interp(z, z_grid, cum_int)

    # Luminosity distance in Mpc
    d_L = (c_light / H0) * (1.0 + z) * int_z

    # Distance modulus: mu = 5 log10(d_L / 10 pc) = 5 log10(d_L [Mpc]) + 25
    mu = 5.0 * np.log10(d_L) + 25.0

    return mu

import numpy as np

def predict(X):
    z = X[:, 0]

    # Flat ΛCDM cosmological parameters
    c_light = 299792.458   # speed of light [km/s]
    H0 = 70.0              # Hubble constant [km/s/Mpc]
    Om = 0.3               # matter density parameter
    Ode = 0.7              # dark energy density parameter

    # Luminosity distance:
    #   d_L = (c/H0) * (1+z) * ∫₀ᶻ dz' / E(z')
    #   E(z') = √[Ω_m(1+z')³ + Ω_Λ]
    #
    # Distance modulus:
    #   μ = 5·log₁₀(d_L [Mpc]) + 25

    N = 10000
    z_max = np.max(z)
    z_grid = np.linspace(0.0, z_max, N + 1)
    E_inv = 1.0 / np.sqrt(Om * (1.0 + z_grid) ** 3 + Ode)

    # Cumulative trapezoidal integration of 1/E(z')
    dz = z_grid[1] - z_grid[0]
    cum_int = np.zeros(N + 1)
    cum_int[1:] = np.cumsum(0.5 * (E_inv[:-1] + E_inv[1:]) * dz)

    # Interpolate the integral to each target redshift
    int_val = np.interp(z, z_grid, cum_int)

    # Luminosity distance in Mpc and distance modulus
    d_L = (c_light / H0) * (1.0 + z) * int_val
    mu = 5.0 * np.log10(d_L) + 25.0

    return mu
