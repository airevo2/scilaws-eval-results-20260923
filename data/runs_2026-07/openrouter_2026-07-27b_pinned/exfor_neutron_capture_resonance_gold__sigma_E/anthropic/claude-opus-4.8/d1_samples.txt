import numpy as np

def predict(X):
    E = np.asarray(X[:, 0], dtype=float)
    E = np.clip(E, 1e-6, None)

    # --- 1/v thermal capture term ---
    sigma_th = 98.65      # barn at thermal energy
    E_th     = 0.0253     # eV
    sigma_1v = sigma_th * np.sqrt(E_th / E)

    # --- Single-level Breit-Wigner resonance at 4.9 eV ---
    E0       = 4.906      # eV, resonance energy
    Gamma    = 0.140      # eV, total width
    sigma_pk = 33000.0    # barn, peak capture cross section
    half     = 0.5 * Gamma
    sigma_res = sigma_pk * (half * half) / ((E - E0) ** 2 + half * half) \
                * np.sqrt(E0 / E)

    sigma = sigma_1v + sigma_res
    return np.log10(sigma)
