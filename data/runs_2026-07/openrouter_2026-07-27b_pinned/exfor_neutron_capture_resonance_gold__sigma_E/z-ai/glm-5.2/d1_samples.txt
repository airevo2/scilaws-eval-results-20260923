import numpy as np

def predict(X):
    """
    Predicts log10 of the Au-197(n,gamma) inclusive capture cross section in barns.
    Model: 1/v thermal background + single Breit-Wigner resonance at 4.906 eV.
    """
    E = X[:, 0]
    # Avoid division by zero for sub-thermal energies
    E = np.where(E < 1e-10, 1e-10, E)
    
    # 1/v thermal background
    # Thermal cross section is ~98.65 b at 0.0253 eV
    sigma_thermal = 98.65 * np.sqrt(0.0253 / E)
    
    # Breit-Wigner resonance for the 4.9 eV peak
    E_r = 4.906          # Resonance energy in eV
    Gamma = 0.1236       # Total width in eV
    Gamma2_4 = (Gamma**2) / 4.0
    sigma_0 = 31500.0    # Peak cross section in barns at E_r
    
    # The neutron width scales as sqrt(E), giving the 1/sqrt(E) factor in the BW formula
    sigma_res = sigma_0 * np.sqrt(E_r / E) * (Gamma2_4 / ((E - E_r)**2 + Gamma2_4))
    
    sigma = sigma_thermal + sigma_res
    return np.log10(sigma)
