    b2 = 1.0
    b3 = 1.0
    b4 = 1.0
    b5 = 1.0
    b6 = 0.0
    
    # Closed-form multiplicative relationship
    # Temperature (AT) is kept in levels (exponential term) to accommodate negative values
    I = a * (P ** b1) * (A ** b2) * (AP ** b3) * (DP ** b4) * (UR ** b5) * np.exp(b6 * AT)
    
    return I
