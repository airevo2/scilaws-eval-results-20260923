import numpy as np

def predict(X):
    sigma0 = X[:, 0]
    # Kormendy & Ho (2013) M_BH-sigma relation for classical bulges and ellipticals
    # log10(M_BH/M_sun) = 8.49 + 4.38 * log10(sigma0 / 200 km/s)
    log_sigma_ratio = np.log10(sigma0 / 200.0)
    log_M_BH = 8.49 + 4.38 * log_sigma_ratio
    return log_M_BH

import numpy as np

def predict(X):
    """
    Predicts log10(M_BH/M_sun) using the Kormendy & Ho (2013) M_BH-sigma relation.
    
    The relation is given by:
    log10(M_BH/M_sun) = alpha + beta * log10(sigma_0 / 200 km/s)
    
    where alpha = 8.49 and beta = 4.38 are the canonical fitted coefficients 
    for the M_BH-sigma relation.
    
    Parameters:
    X : np.ndarray
        A 2-D array of shape (n_samples, 4). Only the first column (sigma_0) 
        is used for this canonical relation.
        
    Returns:
    np.ndarray
        A 1-D array of predicted log10(M_BH/M_sun) values.
    """
    sigma0 = X[:, 0]
    
    # Kormendy & Ho (2013) canonical coefficients
    alpha = 8.49
    beta = 4.38
    
    log_M_BH = alpha + beta * np.log10(sigma0 / 200.0)
    
    return log_M_BH
