import numpy as np

def predict(X):
    sigma0 = X[:, 0]
    log_M_sph = X[:, 1]
    B_T = X[:, 2]
    P = X[:, 3]
    
    # Canonical M-sigma relation (e.g., McConnell & Ma 2013)
    # log10(M_BH/M_sun) = alpha + beta * log10(sigma0 / 200 km/s)
    alpha = 8.32
    beta = 5.64
    
    log_M_BH = alpha + beta * np.log10(sigma0 / 200.0)
    
    # The other variables (spheroid mass, B/T, pseudobulge indicator) are 
    # secondary or used for sample selection (e.g., excluding pseudobulges 
    # in some canonical fits like Kormendy & Ho 2013). We include them with 
    # zero weight to acknowledge the input signature.
    log_M_BH = log_M_BH + 0.0 * log_M_sph + 0.0 * B_T + 0.0 * P
    
    return log_M_BH
