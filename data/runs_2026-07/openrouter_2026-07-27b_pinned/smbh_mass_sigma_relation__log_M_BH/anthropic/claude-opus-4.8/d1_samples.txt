import numpy as np

def predict(X):
    # M-sigma relation (Kormendy & Ho 2013 canonical form)
    # log10(M_BH/M_sun) = alpha + beta * log10(sigma/200)
    sigma0 = X[:, 0]
    alpha = 8.49
    beta = 4.38
    return alpha + beta * np.log10(sigma0 / 200.0)
