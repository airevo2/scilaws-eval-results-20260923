import numpy as np

def predict(X):
    """
    Predict average race velocity (v̄) [m/s] from event distance and sex.

    Uses the classic Riegel endurance formula (T = k * d^1.06), rearranged to
    velocity form:  v = c * d^{-0.06}, with sex-specific scaling constants.

    Parameters
    ----------
    X : 2-D numpy array
        X[:, 0] = distance_m (d) [m], typical range [60, 100000]
        X[:, 1] = sex_M (s_M), 1 for men, 0 for women

    Returns
    -------
    v : 1-D numpy array of average velocities [m/s]
    """
    d = X[:, 0]
    sex_M = X[:, 1]

    # Canonical constants based on the Riegel power law (exponent = 1.06)
    MEN_FACTOR   = 13.0    # scaling factor such that v ≈ 10.4 m/s at d=100 m
    WOMEN_FACTOR = 11.5    # women's factor, reflecting ~12% slower world records
    EXPONENT     = -0.06   # from v ∝ d^{-0.06}

    factor = np.where(sex_M == 1, MEN_FACTOR, WOMEN_FACTOR)
    v = factor * (d ** EXPONENT)
    return v

    d = X[:, 0]  # distance in meters
    s_M = X[:, 1]  # sex indicator: 1 for men, 0 for women
    
    # Coefficients for Riegel's endurance equation: T = a * d^b
    # where T is time in seconds, d is distance in meters
    # The standard form uses fatigue factor b ≈ 1.06 for world-class performances
    # Baseline scaling factor a is calibrated so that T_men = a_M * (d/1000)^b gives minutes
    # Then converted to m/s: v = d / T
    
    # Fatigue exponent (Riegel's constant across sexes for elite performances)
    b = 1.06
    
    # Scaling factor: derived from world-record marathon performance (~2:01:39 for men)
    # a_M calibrates time for men: T_M = a_M * d^b (T in seconds)
    a_M = 0.0135  # calibrated to match world record data
    a_F = 0.0148  # slightly higher for women (slower times)
    
    # Compute time in seconds using Riegel formula: T = a * d^b
    T_M = a_M * (d ** b)
    T_F = a_F * (d ** b)
    
    # Average velocity: v = d / T
    v_M = d / T_M
    v_F = d / T_F
    
    # Combine using sex indicator
    v_avg = s_M * v_M + (1 - s_M) * v_F
    
    return v_avg

import numpy as np

def predict(X):
    """
    Predict average race velocity v̄ [m/s] for world-record performances.

    The standard closed-form relationship is a power law:
        v̄ = A * d^B
    where d is race distance [m], and A, B are sex-dependent constants.
    For men:   A_m ≈ 16.4, B ≈ -0.1
    For women: A_w ≈ 15.0, B ≈ -0.1
    """
    d = X[:, 0]
    s_M = X[:, 1]

    # Coefficients (fitted to approximate world-record trends)
    A_m = 16.4   # men's scaling constant
    A_w = 15.0   # women's scaling constant
    B = -0.1     # universal exponent for running (speed ∝ distance^-0.1)

    # Compute velocity using sex-specific coefficient
    A = np.where(s_M == 1, A_m, A_w)
    v_bar = A * (d ** B)
    return v_bar
