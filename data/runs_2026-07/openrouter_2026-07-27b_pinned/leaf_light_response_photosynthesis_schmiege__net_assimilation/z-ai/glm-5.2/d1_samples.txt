import numpy as np

def predict(X):
    """
    Predicts net CO2 assimilation rate (A_n) using the non-rectangular hyperbola 
    (Thornley, 1976) light response curve.
    """
    # Input: Incident photosynthetically active radiation (Qin)
    Q = X[:, 0]
    
    # Typical fitted parameters for the non-rectangular hyperbola
    alpha = 0.05  # Apparent quantum yield (mol CO2 / mol photons)
    A_max = 8.0   # Light-saturated gross photosynthesis (umol CO2 m-2 s-1)
    theta = 0.7   # Curvature parameter (dimensionless, 0 < theta <= 1)
    R_d = 0.8     # Dark respiration rate (umol CO2 m-2 s-1)
    
    # Non-rectangular hyperbola equation for gross photosynthesis
    term1 = alpha * Q + A_max
    # Use np.maximum to prevent negative values under the square root due to floating point precision
    term2 = np.sqrt(np.maximum(0.0, term1**2 - 4.0 * theta * alpha * Q * A_max))
    
    A_gross = (term1 - term2) / (2.0 * theta)
    
    # Net CO2 assimilation rate
    A_n = A_gross - R_d
    
    return A_n

import numpy as np

def predict(X):
    """
    Predicts net CO2 assimilation rate (A_n) using the non-rectangular hyperbola (NRH) 
    light response model (Thornley, 1976).
    """
    Q = X[:, 0]
    
    # Typical ecophysiological parameters
    alpha = 0.05      # Initial quantum yield (mol CO2 / mol photons)
    Amax = 12.0       # Light-saturated gross assimilation rate (umol CO2 m-2 s-1)
    theta = 0.7       # Curvature parameter (dimensionless, 0 to 1)
    Rd = 0.8          # Dark respiration rate (umol CO2 m-2 s-1)
    
    # Non-rectangular hyperbola equation
    term1 = alpha * Q + Amax
    term2 = term1**2 - 4 * theta * alpha * Q * Amax
    
    # Protect against negative values due to floating point precision
    term2 = np.maximum(term2, 0)
    
    A_n = (1.0 / (2.0 * theta)) * (term1 - np.sqrt(term2)) - Rd
    
    return A_n

    sqrt_term = np.sqrt(discriminant)

    return An
