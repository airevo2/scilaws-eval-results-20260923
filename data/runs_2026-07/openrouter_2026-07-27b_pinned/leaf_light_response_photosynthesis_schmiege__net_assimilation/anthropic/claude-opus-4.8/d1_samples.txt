import numpy as np

def predict(X):
    Q = X[:, 0]
    # Non-rectangular hyperbola light-response model (Prioul & Chartier / Farquhar)
    A_max = 10.0    # light-saturated gross assimilation asymptote [umol m-2 s-1]
    phi = 0.05      # apparent quantum yield [mol CO2 / mol photons]
    theta = 0.7     # convexity (curvature) [dimensionless]
    Rd = 0.8        # dark respiration [umol m-2 s-1]

    # Solve non-rectangular hyperbola for gross photosynthesis
    a = theta
    b = -(phi * Q + A_max)
    c = phi * Q * A_max
    Ag = (-b - np.sqrt(np.maximum(b**2 - 4 * a * c, 0.0))) / (2 * a)

    An = Ag - Rd
    return An
