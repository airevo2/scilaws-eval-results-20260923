import numpy as np

def predict(X):
    phi = np.clip(X[:, 0], 1e-6, 1.0 - 1e-6)
    # Kozeny-Carman equation: k = C * phi^3 / (1 - phi)^2
    # In log10 form: log10(k) = log10(C) + 3*log10(phi) - 2*log10(1 - phi)
    # C ~ 3162 mD (log10(C) = 3.5) is a representative constant for
    # cemented sandstones with k in millidarcies.
    log_k = 3.5 + 3.0 * np.log10(phi) - 2.0 * np.log10(1.0 - phi)
    return log_k

import numpy as np

def predict(X):
    phi = np.clip(X[:, 0], 1e-5, 1.0 - 1e-5)
    
    # Kozeny-Carman equation: k = C * (phi^3 / (1 - phi)^2)
    # For typical sandstones with k in millidarcies (mD), log10(C) is approximately 4.0
    C_log10 = 4.0 
    
    log_k = 3.0 * np.log10(phi) - 2.0 * np.log10(1.0 - phi) + C_log10
    
    return log_k
