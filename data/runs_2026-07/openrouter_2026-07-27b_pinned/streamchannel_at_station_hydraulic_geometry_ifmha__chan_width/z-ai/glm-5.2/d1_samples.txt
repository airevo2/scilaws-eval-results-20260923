import numpy as np

def predict(X):
    """
    Predicts channel width (w) in feet from discharge (Q) in cfs using the 
    standard downstream hydraulic geometry power-law relationship (Leopold and Maddock, 1953):
    w = a * Q^b
    """
    Q = X[:, 0]
    a = 8.0  # typical empirical coefficient for US streams
    b = 0.26 # typical empirical exponent for channel width
    
    # Ensure Q is treated as a numpy array and handle potential zero discharges
    Q = np.array(Q, dtype=float)
    w = a * np.power(Q, b)
    
    return w
