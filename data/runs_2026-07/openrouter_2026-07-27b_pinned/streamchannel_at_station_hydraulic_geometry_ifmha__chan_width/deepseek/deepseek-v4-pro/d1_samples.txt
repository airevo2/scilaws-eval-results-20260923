import numpy as np

def predict(X):
    """
    Predict channel water-surface width (w, ft) from discharge (Q, cfs)
    using the classic at-a-station hydraulic geometry power-law relationship:
    w = a * Q^b
    Coefficients are typical literature values for at-a-station width.
    """
    # At-a-station hydraulic geometry coefficients
    a = 10.0    # coefficient (ft per cfs^b)
    b = 0.12    # exponent (dimensionless)

    Q = X[:, 0]                  # discharge in cfs
    w = a * np.power(Q, b)       # width in ft
    return w

import numpy as np

def predict(X):
    """
    Predict water-surface width (w, ft) from discharge (Q, cfs) using the
    classic at-a-station hydraulic geometry power law: w = a * Q^b.

    Parameters
    ----------
    X : 2-D numpy array
        X[:, 0] = channel discharge Q [cfs]

    Returns
    -------
    w : 1-D numpy array
        Predicted water-surface width [ft]
    """
    Q = X[:, 0]                     # discharge in cfs
    # Coefficients typical for natural river cross-sections (Leopold & Maddock, 1953)
    a = 16.6                        # ft / (cfs^b)
    b = 0.26                        # dimensionless exponent
    w = a * np.power(Q, b)
    return w
