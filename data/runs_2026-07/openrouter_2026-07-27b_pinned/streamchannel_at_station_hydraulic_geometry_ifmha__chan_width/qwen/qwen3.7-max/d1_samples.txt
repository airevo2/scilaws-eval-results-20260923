import numpy as np

def predict(X):
    """
    At-a-station hydraulic geometry: width-discharge relationship
    following Leopold & Maddock (1953).
    
    w = a * Q^b
    
    Typical at-a-station exponents (Leopold & Maddock, 1953):
        b ≈ 0.26  (width exponent)
        a ≈ 4.8   (coefficient for Q in cfs, w in ft)
    """
    Q = X[:, 0]
    a = 4.8
    b = 0.26
    w = a * Q ** b
    return w
