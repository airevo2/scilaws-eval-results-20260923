import numpy as np

def predict(X):
    # Hydraulic geometry (Leopold & Maddock, 1953): w = a * Q^b
    # Commonly cited at-a-station/downstream values: a ~ 1.5, b ~ 0.5 (width in ft, Q in cfs)
    Q = np.asarray(X[:, 0], dtype=float)
    a = 1.5
    b = 0.5
    return a * np.power(Q, b)
