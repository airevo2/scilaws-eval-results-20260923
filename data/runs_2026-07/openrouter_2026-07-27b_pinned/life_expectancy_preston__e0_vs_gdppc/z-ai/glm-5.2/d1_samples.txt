import numpy as np

def predict(X):
    """
    Predicts sex-averaged life expectancy at birth (e0) based on Preston (1975).
    The standard Preston curve uses a linear relationship between life expectancy 
    and the natural logarithm of GDP per capita.
    """
    Y = X[:, 0]
    
    # Canonical form from Preston (1975) for the 1960s cross-section:
    # e0 = a + b * ln(gdppc)
    # Using fitted coefficients representative of the 1960s data (in 1963 USD)
    a = -10.0
    b = 10.5
    
    e0 = a + b * np.log(Y)
    
    return e0
