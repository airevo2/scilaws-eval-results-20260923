    # Preston (1975) semi-logarithmic curve for the 1960s cross-section

import numpy as np

def predict(X):
    """
    Predicts sex-averaged life expectancy at birth (e0) based on GDP per capita 
    using the canonical Preston Curve (1975) semi-logarithmic relationship.
    
    Parameters:
    X : numpy.ndarray
        2-D array where X[:, 0] is GDP per capita in constant 1963 USD.
        
    Returns:
    numpy.ndarray
        1-D array of predicted life expectancy at birth (years).
    """
    # Canonical semi-log functional form: e0 = a + b * ln(Y)
    # Coefficients are representative of Preston's 1960s cross-section estimates
    a = 15.0
    b = 8.0
    
    gdppc = X[:, 0]
    e0 = a + b * np.log(gdppc)
    
    return e0
