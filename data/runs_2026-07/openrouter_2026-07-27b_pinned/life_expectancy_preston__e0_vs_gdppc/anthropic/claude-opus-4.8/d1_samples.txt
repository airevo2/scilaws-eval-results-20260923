import numpy as np

def predict(X):
    # Preston (1975) curve: life expectancy as a logarithmic function of income per head.
    # Most commonly cited functional form: e0 = a + b * ln(gdppc)
    Y = np.asarray(X[:, 0], dtype=float)
    a = -13.0
    b = 12.0
    return a + b * np.log(Y)

import numpy as np

def predict(X):
    # Preston curve: sex-averaged life expectancy at birth as a
    # logarithmic function of national income per head (constant 1963 USD).
    Y = np.asarray(X[:, 0], dtype=float)
    a = -0.5     # intercept
    b = 9.0      # slope on ln(income)
    e0 = a + b * np.log(Y)
    return e0
