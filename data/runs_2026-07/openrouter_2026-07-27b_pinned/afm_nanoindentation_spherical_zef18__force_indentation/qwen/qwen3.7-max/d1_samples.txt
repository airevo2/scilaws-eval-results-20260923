    R = X[:, 1]      # spherical probe tip radius (m)
    
    # Hertz contact model for a rigid spherical indenter on an elastic half-space:
    # F = (4/3) * E_eff * sqrt(R) * delta^(3/2)
    # where E_eff = E / (1 - nu^2) is the reduced elastic modulus.
    
    E_eff = 1000.0  # Effective elastic modulus (Pa); placeholder for soft biological sample
    
    F = (4.0 / 3.0) * E_eff * np.sqrt(R) * np.power(np.maximum(delta, 0.0), 1.5)
    return F
