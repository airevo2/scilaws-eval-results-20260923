import numpy as np

def predict(X):
    # Hertz model for a spherical (colloidal) AFM probe indenting an elastic half-space:
    #   F = (4/3) * E/(1 - nu^2) * sqrt(R) * delta^(3/2)
    delta = np.asarray(X[:, 0], dtype=float)
    R     = np.asarray(X[:, 1], dtype=float)

    E  = 1.0e3   # Young's modulus [Pa] (soft biological sample, e.g. cells/tissue)
    nu = 0.5     # Poisson ratio (incompressible, standard assumption)

    delta = np.clip(delta, 0.0, None)  # only contact (positive indentation) gives force
    F = (4.0 / 3.0) * (E / (1.0 - nu**2)) * np.sqrt(R) * delta**1.5
    return F
