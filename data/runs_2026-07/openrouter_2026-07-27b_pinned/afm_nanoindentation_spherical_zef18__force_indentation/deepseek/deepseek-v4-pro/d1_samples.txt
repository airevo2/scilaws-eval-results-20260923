def predict(X):
    """
    Predict indentation force F (N) for a spherical AFM tip using the Hertz model.
    
    Parameters
    ----------
    X : np.ndarray, shape (n, 2)
        X[:, 0] : indentation_m (delta) in m
        X[:, 1] : R_m (R) in m (spherical tip radius)

    Returns
    -------
    F : np.ndarray, shape (n,)
        Indentation force in N (positive during approach contact).
    """
    delta = X[:, 0]
    R = X[:, 1]

    # Effective Young's modulus (E*) placeholder.
    # In practice this would be fitted to data: E* = E_sample / (1 - nu_sample^2)
    # assuming a rigid tip.  Units: Pa.
    E_star = 1.0

    # Force is zero when not in contact (delta <= 0)
    delta_pos = np.maximum(delta, 0.0)

    # Hertz model for a sphere on an elastic half-space:
    # F = (4/3) * E* * sqrt(R) * delta^(3/2)
    F = (4.0 / 3.0) * E_star * np.sqrt(R) * delta_pos ** 1.5

    return F
