import numpy as np

def predict(X):
    """
    Predicts Annual national primary energy consumption (I_E) in TJ 
    using an extended STIRPAT (Stochastic Impacts by Regression on Population, 
    Affluence, and Technology) / Kaya identity framework.
    
    Inputs:
    X[:, 0] = A (GDP per capita)
    X[:, 1] = P (Population)
    X[:, 2] = f_active (Working-age fraction)
    X[:, 3] = rho (Population density)
    X[:, 4] = f_urban (Urbanization rate)
    X[:, 5] = T (Mean annual temperature)
    """
    A = X[:, 0]
    P = X[:, 1]
    f_active = X[:, 2]
    rho = X[:, 3]
    f_urban = X[:, 4]
    T = X[:, 5]
    
    # Named constants for the multiplicative STIRPAT-style elasticities
    # Base scaling factor (Energy intensity coefficient)
    alpha = 1.0 
    
    # Elasticities / coefficients
    beta_P = 1.0        # Population elasticity
    beta_A = 1.0        # Affluence (GDP per capita) elasticity
    beta_fa = 0.0       # Working-age fraction elasticity
    beta_rho = 0.0      # Population density elasticity
    beta_fu = 0.0       # Urbanization rate elasticity
    beta_T = 0.0        # Temperature semi-elasticity
    
    # Closed-form multiplicative relationship
    # Temperature is modeled with an exponential term to handle negative values safely
    I_E = alpha * (P ** beta_P) * (A ** beta_A) * \
          (f_active ** beta_fa) * (rho ** beta_rho) * \
          (f_urban ** beta_fu) * np.exp(beta_T * T)
          
    return I_E

import numpy as np

def predict(X):
    """
    Predicts Annual national primary energy consumption (I_E) in TJ 
    using the STIRPAT (Stochastic Impacts by Regression on Population, 
    Affluence, and Technology) framework, standard in energy economics.
    """
    A = X[:, 0]          # GDP per capita (Affluence)
    P = X[:, 1]          # Total population
    f_active = X[:, 2]   # Working-age fraction
    rho = X[:, 3]        # Population density
    f_urban = X[:, 4]    # Urbanization rate
    T = X[:, 5]          # Mean annual temperature

    # Base intercept calibrated for TJ units 
    alpha = -11.5 
    
    # Standard elasticities derived from energy STIRPAT literature
    b_P = 1.05           # Population elasticity
    b_A = 0.95           # Affluence elasticity
    b_active = 0.30      # Demographic dividend (working-age) elasticity
    b_rho = -0.05        # Density elasticity (negative due to agglomeration/transport efficiencies)
    b_urban = 0.20       # Urbanization elasticity
    
    # Temperature effect: proxy for heating/cooling degree days 
    # Deviation from a comfortable baseline (~15°C)
    T_base = 15.0
    b_T = 0.015          
    
    # Safeguard against non-positive values for logarithmic terms
    P_safe = np.maximum(P, 1.0)
    A_safe = np.maximum(A, 1.0)
    f_active_safe = np.maximum(f_active, 1.0)
    rho_safe = np.maximum(rho, 0.1)
    f_urban_safe = np.maximum(f_urban, 1.0)
    
    # Log-linear STIRPAT formulation
    ln_I = (alpha 
            + b_P * np.log(P_safe) 
            + b_A * np.log(A_safe) 
            + b_active * np.log(f_active_safe) 
            + b_rho * np.log(rho_safe) 
            + b_urban * np.log(f_urban_safe) 
            + b_T * np.abs(T - T_base))
            
    I_E = np.exp(ln_I)
    
    return I_E

import numpy as np

def predict(X):
    """
    Predict annual national primary energy consumption I_E [TJ]
    using the STIRPAT (Stochastic Impacts by Regression on Population,
    Affluence, and Technology) model — the standard closed-form
    relationship in energy-environmental economics.

    I_E = alpha * P^b1 * A^b2 * f_active^b3 * rho^b4 * f_urban^b5 * exp(b6 * T)

    Equivalently in log-linear form:
    ln(I_E) = b0 + b1*ln(P) + b2*ln(A) + b3*ln(f_active)
              + b4*ln(rho) + b5*ln(f_urban) + b6*T
    """
    A        = X[:, 0]                # GDP per capita [constant 2015 USD]
    P        = X[:, 1]                # Total population [persons]
    f_active = X[:, 2] / 100.0        # Working-age fraction (0–1)
    rho      = X[:, 3]                # Population density [persons / km²]
    f_urban  = X[:, 4] / 100.0        # Urban fraction (0–1)
    T        = X[:, 5]                # Mean annual temperature [°C]

    # STIRPAT elasticities / coefficients
    b0 = -15.53    # intercept  (ln-scale, TJ)
    b1 =  1.00     # population elasticity
    b2 =  0.75     # affluence (GDP per capita) elasticity
    b3 =  0.30     # working-age share elasticity
    b4 = -0.15     # population-density elasticity
    b5 =  0.20     # urbanisation elasticity
    b6 = -0.01     # temperature coefficient [per °C]

    ln_IE = (b0
             + b1 * np.log(P)
             + b2 * np.log(A)
             + b3 * np.log(f_active)
             + b4 * np.log(rho)
             + b5 * np.log(f_urban)
             + b6 * T)

    return np.exp(ln_IE)
