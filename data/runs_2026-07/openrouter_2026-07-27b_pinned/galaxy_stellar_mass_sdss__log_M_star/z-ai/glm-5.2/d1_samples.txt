    return log_M_star

import numpy as np

def predict(X):
    """
    Predict log10(M_star/M_sun) using the standard Bell et al. (2003)
    mass-to-light ratio vs. color relation for a Kroupa IMF.
    
    log10(M/L_r) = -0.306 + 1.097 * (g - r)
    log10(L_r/L_sun) = 0.4 * (M_r_sun - M_r)
    
    where M_r = r - DM (K-corrections neglected in this closed-form proxy)
    and M_r_sun = 4.64 mag (AB).
    """
    # Extract input columns
    g = X[:, 1]
    r = X[:, 2]
    DM = X[:, 6]
    
    # Constants
    M_r_sun = 4.64          # solar absolute r-band magnitude (AB)
    a = -0.306              # Bell et al. (2003) intercept
    b = 1.097               # Bell et al. (2003) slope for (g - r)
    
    # Absolute r-band magnitude (ignoring K-corrections)
    M_r = r - DM
    
    # Log stellar mass
    log_L_r = 0.4 * (M_r_sun - M_r)
    log_ML_r = a + b * (g - r)
    log_M_star = log_L_r + log_ML_r
    
    return log_M_star
