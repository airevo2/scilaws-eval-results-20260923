    canonical baseline (climatological mean) for the ISN v2.0 dataset is returned.
    """
    # Long-term mean of the SILSO ISN v2.0 monthly sunspot number (1749-present)
    R_I_mean = 50.0 
    
    return np.full(X.shape[0], R_I_mean)

import numpy as np

def predict(X):
    t = X[:, 0]
    # Canonical sinusoidal approximation for the ~11-year Schwabe solar cycle
    # Since SSN is a chaotic, quasi-periodic time series, a simple harmonic 
    # model is the standard closed-form approximation when only time is given.
    period = 11.0
    mean_val = 50.0
    amp = 50.0
    phase = 2 * np.pi * (t - 1749.0) / period
    return np.maximum(0, mean_val + amp * np.sin(phase))

import numpy as np

def predict(X):
    t = X[:, 0]
    # Approximate 11-year Schwabe cycle (sinusoidal climatological model)
    # Long-term mean ~ 50, Amplitude ~ 50, Period ~ 11.0 years
    period = 11.0
    phase = 1750.0
    ssn = 50.0 + 50.0 * np.sin(2.0 * np.pi * (t - phase) / period)
    return np.maximum(ssn, 0.0)
