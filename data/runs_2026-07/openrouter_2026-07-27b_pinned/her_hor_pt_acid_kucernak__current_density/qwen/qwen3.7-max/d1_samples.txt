    alpha_c = 0.5       # Cathodic transfer coefficient

    T_ref = 298.15      # Reference temperature [K]
    Ea = 25000.0        # Apparent activation energy [J mol-1]
    m = 0.5             # Reaction order in P_H2 (Tafel quasi-equilibrated)
    n = 0.5             # Reaction order in c_acid (kinetic H+ dependence)

import numpy as np

def predict(X):
    eta = X[:, 0]       # Overpotential vs RHE [V]
    T = X[:, 1]         # Temperature [K]
    P_H2 = X[:, 2]      # H2 partial pressure [bar]
    c_acid = X[:, 3]    # HClO4 concentration [mol dm-3]

    # Physical constants
    F = 96485.3329      # Faraday constant [C mol-1]
    R = 8.314462618     # Gas constant [J mol-1 K-1]

    # Butler-Volmer kinetic parameters for HOR/HER on polycrystalline Pt
    j0_ref = 0.8e-3     # Reference exchange current density [A cm-2]
                        # (at T_ref, P_H2 = 1 bar, c_acid = 1 M)
    T_ref = 298.15      # Reference temperature [K]
    c_ref = 1.0         # Reference acid concentration [mol dm-3]
    alpha = 1.0         # Effective (apparent) transfer coefficient
                        # For the overall 2e- reaction: alpha = n * alpha_sym = 2 * 0.5
    gamma = 0.5         # Reaction order w.r.t. P_H2 (Tafel dissociation)
    delta = 0.5         # Apparent reaction order w.r.t. c_acid
    E_a = 13000.0       # Apparent activation energy [J mol-1]

    # Exchange current density with pressure, concentration, and temperature dependence
    j0 = (j0_ref
          * (P_H2 / 1.0) ** gamma
          * (c_acid / c_ref) ** delta
          * np.exp(-(E_a / R) * (1.0 / T - 1.0 / T_ref)))

    # Symmetric Butler-Volmer equation: j = j0 * [exp(aFη/RT) - exp(-aFη/RT)]
    j = 2.0 * j0 * np.sinh(alpha * F * eta / (R * T))

    return j
