import numpy as np

def predict(X):
    # van Genuchten (1980) water retention curve
    h = np.asarray(X[:, 0], dtype=float)
    h = np.abs(h)

    # Typical loam-like parameters (van Genuchten, 1980)
    theta_r = 0.078     # residual water content [m3/m3]
    theta_s = 0.43      # saturated water content [m3/m3]
    alpha   = 3.6       # [1/m]
    n       = 1.56      # shape parameter [-]
    m       = 1.0 - 1.0 / n

    Se = (1.0 + (alpha * h) ** n) ** (-m)
    theta = theta_r + (theta_s - theta_r) * Se

    return theta

import numpy as np

def predict(X):
    # Van Genuchten (1980) water retention model, most commonly cited form
    h = np.asarray(X[:, 0], dtype=float)

    # Typical loam-like parameters (fitted constants)
    theta_r = 0.05     # residual water content [m^3/m^3]
    theta_s = 0.45     # saturated water content [m^3/m^3]
    alpha   = 0.02     # inverse air-entry [1/m]
    n       = 1.5      # shape parameter
    m       = 1.0 - 1.0 / n

    h = np.maximum(h, 0.0)
    Se = (1.0 + (alpha * h) ** n) ** (-m)
    theta = theta_r + (theta_s - theta_r) * Se

    return theta
