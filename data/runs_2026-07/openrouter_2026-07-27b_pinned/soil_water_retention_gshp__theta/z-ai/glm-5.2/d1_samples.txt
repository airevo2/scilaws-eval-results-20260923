import numpy as np

def predict(X):
    h = X[:, 0]
    # Typical van Genuchten (1980) parameters for a medium-textured soil
    theta_r = 0.045      # residual water content [m^3/m^3]
    theta_s = 0.43       # saturated water content [m^3/m^3]
    alpha   = 0.015      # inverse air-entry pressure [1/m]
    n       = 2.0        # pore-size distribution parameter [-]
    m       = 1.0 - 1.0 / n   # common restriction m = 1 - 1/n
    theta = theta_r + (theta_s - theta_r) / (1.0 + (alpha * h) ** n) ** m
    return theta

import numpy as np

def predict(X):
    """
    Standard van Genuchten (1980) closed-form soil water retention curve.
    
    theta(h) = theta_r + (theta_s - theta_r) * [1 + (alpha * h)^n]^(-m)
    
    where m = 1 - 1/n, and h is positive matric suction in metres.
    
    Parameters use canonical literature values (e.g., VG 1980, Table / 
    typical sandy loam): theta_r = 0.067, theta_s = 0.45, 
    alpha = 2.0 m^-1 (≈0.02 cm^-1), n = 1.41.
    """
    h = X[:, 0].astype(float)
    
    # --- van Genuchten parameters ---
    theta_r = 0.067      # residual water content [m^3/m^3]
    theta_s = 0.45       # saturated water content [m^3/m^3]
    alpha = 2.0          # inverse of air-entry pressure [1/m]
    n = 1.41             # pore-size distribution parameter [-]
    m = 1.0 - 1.0 / n    # VG constraint m = 1 - 1/n
    
    # Effective saturation term
    Se = 1.0 / (1.0 + (alpha * h) ** n) ** m
    
    theta = theta_r + (theta_s - theta_r) * Se
    return theta
