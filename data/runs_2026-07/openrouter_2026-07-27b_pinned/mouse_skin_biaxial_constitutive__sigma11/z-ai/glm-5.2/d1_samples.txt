import numpy as np

def predict(X):
    """
    Predicts the Cauchy stress sigma_11 [MPa] for a planar mouse-skin specimen 
    under biaxial loading using a standard 2D Fung-type hyperelastic model.
    
    The strain energy density function is:
    W = 0.5 * c * (exp(Q) - 1)
    where Q = A1 * E11^2 + A2 * E22^2 + 2 * A3 * E11 * E22
    
    The Cauchy stress is derived assuming incompressibility and plane stress.
    """
    lambda11 = X[:, 0]
    lambda22 = X[:, 1]
    
    # Green-Lagrange strains
    E11 = 0.5 * (lambda11**2 - 1.0)
    E22 = 0.5 * (lambda22**2 - 1.0)
    
    # Representative material parameters for mouse skin
    c = 0.5     # MPa
    A1 = 12.0
    A2 = 15.0
    A3 = 4.0
    
    # Exponential argument Q
    Q = A1 * E11**2 + A2 * E22**2 + 2.0 * A3 * E11 * E22
    
    # Second Piola-Kirchhoff stress S11 = dW/dE11
    S11 = c * (A1 * E11 + A3 * E22) * np.exp(Q)
    
    # Convert to Cauchy stress (assuming 3D incompressibility, plane stress sigma_33 = 0)
    # For a 2D reduced strain energy function, this reduces to sigma_11 = lambda_11^2 * S_11
    sigma11 = lambda11**2 * S11
    
    return sigma11

    b12 = 2.0    # Coupling parameter
