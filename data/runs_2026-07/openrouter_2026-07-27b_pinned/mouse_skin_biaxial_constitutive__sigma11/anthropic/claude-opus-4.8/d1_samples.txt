import numpy as np

def predict(X):
    # Incompressible neo-Hookean model for planar biaxial loading of skin.
    # Incompressibility: lambda33 = 1 / (lambda11 * lambda22)
    # Cauchy stress (deviatoric, with plane-stress condition sigma33 = 0):
    #   sigma11 = mu * (lambda11^2 - lambda33^2)
    lam11 = X[:, 0]
    lam22 = X[:, 1]
    lam33 = 1.0 / (lam11 * lam22)

    mu = 0.20  # shear modulus [MPa], representative of mouse skin

    sigma11 = mu * (lam11**2 - lam33**2)
    return sigma11

import numpy as np

def predict(X):
    """
    Fung-type (exponential) anisotropic hyperelastic model for planar biaxial skin.
    sigma11 (Cauchy stress, MPa) derived from strain-energy
        W = (c/2) * (exp(Q) - 1),
        Q = a1*E11^2 + a2*E22^2 + 2*a12*E11*E22,
    with Green-Lagrange strains E_ii = (lambda_ii^2 - 1)/2 and an
    incompressible thin-membrane (sigma33 = 0) assumption, giving
        sigma11 = c * exp(Q) * (2*a1*E11 + 2*a12*E22) * lambda11^2.
    """
    lam11 = np.asarray(X[:, 0], dtype=float)
    lam22 = np.asarray(X[:, 1], dtype=float)

    # Green-Lagrange strains
    E11 = 0.5 * (lam11**2 - 1.0)
    E22 = 0.5 * (lam22**2 - 1.0)

    # Fitted material constants (typical mouse-skin order of magnitude)
    c   = 0.05   # MPa  (stiffness scale)
    a1  = 8.0    # dimensionless, lateral direction
    a2  = 8.0    # dimensionless, cranial-caudal direction
    a12 = 2.0    # dimensionless, cross-coupling

    Q = a1 * E11**2 + a2 * E22**2 + 2.0 * a12 * E11 * E22

    sigma11 = c * np.exp(Q) * (2.0 * a1 * E11 + 2.0 * a12 * E22) * lam11**2
    return sigma11

import numpy as np

def predict(X):
    """
    Sigma_11 (Cauchy stress, MPa) for an incompressible, isotropic
    exponential (Demiray/Delfino-type) hyperelastic skin model:

        W = (mu / (2*b)) * ( exp(b*(I1 - 3)) - 1 )

    With incompressibility (lambda33 = 1/(lambda11*lambda22)) and the
    plane-stress condition sigma33 = 0, the closed-form lateral stress is

        sigma11 = mu * exp(b*(I1 - 3)) * (lambda11^2 - lambda33^2)
    """
    lam11 = X[:, 0]
    lam22 = X[:, 1]
    lam33 = 1.0 / (lam11 * lam22)          # incompressibility

    I1 = lam11**2 + lam22**2 + lam33**2

    # Fitted material constants (typical soft-tissue/skin values)
    mu = 0.20   # MPa  (initial shear-like stiffness)
    b  = 2.50   # dimensionless exponential stiffening

    sigma11 = mu * np.exp(b * (I1 - 3.0)) * (lam11**2 - lam33**2)
    return sigma11
