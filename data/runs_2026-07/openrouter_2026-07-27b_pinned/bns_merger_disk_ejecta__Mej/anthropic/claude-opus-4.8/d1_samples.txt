import numpy as np

def predict(X):
    # Dietrich & Ujevic (2017), CQG 34, 105014 - dynamical ejecta mass fit
    M1  = X[:, 0]
    M2  = X[:, 1]
    Mb1 = X[:, 2]
    Mb2 = X[:, 3]
    C1  = X[:, 4]
    C2  = X[:, 5]

    a = -1.35695
    b =  6.11252
    c = -49.43355
    d =  16.1144
    n = -2.5484

    # term for star 1 (uses companion mass ratio Mb2/Mb1)
    term1 = (a * (Mb2 / Mb1)**(1.0/3.0) * (1.0 - 2.0*C1) / C1
             + b * (Mb2 / Mb1)**n
             + c * (1.0 - M1 / Mb1)) * Mb1

    # term for star 2 (1 <-> 2 swap)
    term2 = (a * (Mb1 / Mb2)**(1.0/3.0) * (1.0 - 2.0*C2) / C2
             + b * (Mb1 / Mb2)**n
             + c * (1.0 - M2 / Mb2)) * Mb2

    Mej_1e3 = term1 + term2 + d  # in units of 1e-3 M_sun

    Mej = Mej_1e3 * 1.0e-3
    Mej = np.clip(Mej, 0.0, None)
    return Mej
