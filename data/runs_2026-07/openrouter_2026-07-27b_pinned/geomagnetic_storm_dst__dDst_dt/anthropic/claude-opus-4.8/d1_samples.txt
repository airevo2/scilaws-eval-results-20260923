import numpy as np

def predict(X):
    # Burton et al. (1975) ring-current equation:
    #   dDst*/dt = Q(t) - Dst*/tau
    #   Dst* = Dst - b*sqrt(P_dyn) + c   (pressure-corrected Dst)
    #   Q = a*(Ey - Ec) for Ey > Ec, else 0
    Dst   = X[:, 0]
    Ey    = X[:, 1]      # dawn-dusk solar wind electric field [mV/m]
    P_dyn = X[:, 2]

    # Canonical Burton constants
    a   = -4.4    # nT/hr per (mV/m)
    Ec  = 0.5     # mV/m  (injection threshold)
    b   = 0.20    # nT / sqrt(nPa)
    c   = 20.0    # nT
    tau = 7.7     # hr    (ring-current decay time)

    # Injection term Q (only for Ey above threshold)
    Q = np.where(Ey > Ec, a * (Ey - Ec), 0.0)

    # Pressure-corrected Dst
    Dst_star = Dst - b * np.sqrt(np.clip(P_dyn, 0.0, None)) + c

    # Tendency
    dDst_dt = Q - Dst_star / tau
    return dDst_dt

    c   = 20.0    # nT  (quiet-time offset)

    Q = np.where(Ey > Ec, a * (Ey - Ec), 0.0)

    Dst_star = Dst - b * np.sqrt(np.maximum(P_dyn, 0.0)) + c

    c   = 20.0      # nT               (quiet-time offset)

    Ey    = X[:, 1]
    P_dyn = X[:, 2]

    Dst_star = Dst - b * np.sqrt(np.clip(P_dyn, 0.0, None)) + c

    Q = np.where(Ey > Ec, -a * (Ey - Ec), 0.0)

    dDst_dt = Q - Dst_star / tau
    return dDst_dt

    b = 7.26   # nT / nPa^0.5

    # Injection term Q (nT/hr), active only for Ey above threshold Ec
    Ec = 0.49  # mV/m
    Q = np.where(Ey > Ec, -4.4 * (Ey - Ec), 0.0)

    # Decay time constant (hr)
    tau = 2.4 * np.exp(9.74 / (4.69 + np.clip(Ey, 0.0, None)))

    dDst_dt = Q - Dst_star / tau
    return dDst_dt

import numpy as np

def predict(X):
    # Burton / O'Brien-McPherron ring-current equation:
    #   dDst*/dt = Q(Ey) - Dst*/tau
    # with the pressure-corrected Dst*:  Dst* = Dst - b*sqrt(Pdyn) + c
    Dst   = X[:, 0]
    Ey    = X[:, 1]
    P_dyn = X[:, 2]

    # Pressure correction constants (O'Brien & McPherron 2000)
    b = 7.26   # nT / sqrt(nPa)
    c = 11.0   # nT

    # Injection term Q(t): active only for Ey above threshold Ec
    a  = -4.4    # nT/hr per mV/m
    Ec = 0.49    # mV/m
    Q = np.where(Ey > Ec, a * (Ey - Ec), 0.0)

    # Ey-dependent decay time (hr)
    tau = 2.40 * np.exp(9.74 / (4.69 + Ey))

    # Pressure-corrected Dst
    Dst_star = Dst - b * np.sqrt(np.clip(P_dyn, 0, None)) + c

    dDst_dt = Q - Dst_star / tau
    return dDst_dt
