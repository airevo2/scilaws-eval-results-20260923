import numpy as np

def predict(X):
    m_i = X[:, 0]
    m_j = X[:, 1]
    d_ij = X[:, 2]
    # s_ij = X[:, 3] (Intervening opportunities, typically used in competing destinations or radiation models, 
    # but omitted here in favor of the most commonly cited standard closed-form Gravity Model)
    
    # Standard Gravity Model: T_ij = G * (m_i * m_j) / d_ij^gamma
    # Using canonical literal values G=1, gamma=2 (Newtonian analogy)
    T_ij = (m_i * m_j) / (d_ij**2)
    
    return np.log10(T_ij + 1)

def predict(X):
    m_i = X[:, 0]
    m_j = X[:, 1]
    s_ij = X[:, 3]

    # Radiation model (Simini et al., Nature 2012)
    # Transition probability: p_ij = m_i * m_j / ((m_i + s_ij)(m_i + s_ij + m_j))
    # Expected flow: T_ij = m_i * p_ij
    T_ij = (m_i**2 * m_j) / ((m_i + s_ij) * (m_i + s_ij + m_j))

    return np.log10(T_ij + 1.0)

import numpy as np

def predict(X):
    m_i = X[:, 0]
    m_j = X[:, 1]
    d_ij = X[:, 2]
    s_ij = X[:, 3]

    # Standard log-linear gravity model with intervening opportunities
    # T_ij = K * m_i^alpha * m_j^beta * d_ij^(-gamma) * (s_ij + 1)^(-delta)
    alpha = 1.0       # origin mass elasticity (production)
    beta  = 1.0       # destination mass elasticity (attraction)
    gamma = 2.0       # distance-decay exponent (inverse-square law)
    delta = 0.5       # intervening-opportunities elasticity (competition)
    log10_K = -2.0    # intercept / calibration constant (log10 scale)

    log10_Tij_plus1 = (log10_K
                       + alpha * np.log10(m_i)
                       + beta  * np.log10(m_j)
                       - gamma * np.log10(d_ij)
                       - delta * np.log10(s_ij + 1))

    return log10_Tij_plus1
