def predict(X):
    """
    Predict national annual N2O emissions (kg N2O) excluding LULUCF using a
    standard STIRPAT / Environmental Kuznets Curve (EKC) log-linear specification.
    
    Model:
        ln(I_N2O) = b0 + b1*ln(P) + b2*ln(A) + b3*[ln(A)]^2
                    + b4*AP + b5*ln(DP) + b6*UR + b7*AT + b8*AT^2
        I_N2O = exp(ln(I_N2O))
    
    Inputs (columns of X):
        0: A  - GDP per capita (constant 2015 USD)
        1: P  - Total population (persons)
        2: AP - Active population share (%)
        3: DP - Population density (persons per km2)
        4: UR - Urbanization rate (%)
        5: AT - Annual mean temperature (°C)
    """

    P  = X[:, 1]
    AP = X[:, 2]
    DP = X[:, 3]
    UR = X[:, 4]
    AT = X[:, 5]

    # Canonical fitted coefficients (illustrative literature-consistent values)
    b0 = -8.0      # intercept
    b1 = 1.0       # population elasticity
    b2 = 0.6       # affluence elasticity
    b3 = -0.04     # EKC quadratic term (turning point at high income)
    b4 = 0.02      # active population share
    b5 = 0.1       # population density elasticity
    b6 = 0.015     # urbanization rate
    b7 = 0.05      # linear temperature effect

    ln_P  = np.log(P)
    ln_DP = np.log(DP)

            + b1 * ln_P
            + b2 * ln_A
            + b3 * ln_A**2
            + b4 * AP
            + b5 * ln_DP
            + b6 * UR
            + b7 * AT
            + b8 * AT**2)

    I_N2O = np.exp(ln_I)
    return I_N2O

    b5 = 0.50   # urbanization rate

import numpy as np

def predict(X):
    """
    Predict national annual N2O emissions (kg N2O per country per year)
    using a multiplicative STIRPAT-type closed-form model.
    
    I_N2O = alpha * GDP^beta_gdp * Pop^beta_pop * AP^beta_ap * DP^beta_dp
            * UR^beta_ur * exp(beta_at * AT)
    """
    # Fitted coefficients (illustrative values consistent with STIRPAT literature)
    alpha = 1.0e-6
    beta_gdp = 0.65
    beta_pop = 1.0
    beta_ap = 0.25
    beta_dp = 0.15
    beta_ur = 0.40
    beta_at = 0.03

    X = np.asarray(X, dtype=float)
    if X.ndim == 1:
        X = X.reshape(1, -1)

    gdp = X[:, 0]
    pop = X[:, 1]
    ap = X[:, 2]
    dp = X[:, 3]
    ur = X[:, 4]
    at = X[:, 5]

    # Evaluate in log-space for numerical stability
    log_I = (np.log(alpha)
             + beta_gdp * np.log(gdp)
             + beta_pop * np.log(pop)
             + beta_ap * np.log(ap)
             + beta_dp * np.log(dp)
             + beta_ur * np.log(ur)
             + beta_at * at)

    I = np.exp(log_I)
    return I
