import numpy as np

def predict(X):
    # STIRPAT / Kaya-type multiplicative model for national N2O emissions
    # I = a * P^b * A^c * exp(other elasticities)
    # Emissions scale ~linearly with population and sub-linearly with affluence.
    A  = X[:, 0]   # GDP per capita (USD)
    P  = X[:, 1]   # total population (persons)
    AP = X[:, 2]   # active population share (%)
    DP = X[:, 3]   # population density (persons/km2)
    UR = X[:, 4]   # urbanization (%)
    AT = X[:, 5]   # annual mean temperature (deg C)

    # small positive floor to avoid log(0)
    eps = 1e-9

    # Fitted coefficients (log-linear STIRPAT form)
    a  = -3.0     # intercept (ln scale)
    b  = 1.00     # population elasticity (~unit)
    c  = 0.45     # affluence elasticity (sub-linear)
    d  = -0.10    # population density (agglomeration effect)
    e  = -0.20    # urbanization (less agricultural N2O)
    f  =  0.30    # active-population share
    g  =  0.010   # temperature effect

    lnI = (a
           + b * np.log(P + eps)
           + c * np.log(A + eps)
           + d * np.log(DP + eps)
           + e * np.log(UR + eps)
           + f * np.log(AP + eps)
           + g * AT)

    return np.exp(lnI)

import numpy as np

def predict(X):
    # STIRPAT-type multiplicative model:
    # I = a * P^b * A^c * f(other factors)
    # N2O is dominated by agriculture, so it scales strongly with population
    # and moderately with affluence (GDP per capita).
    A  = X[:, 0]   # GDP per capita
    P  = X[:, 1]   # total population
    AP = X[:, 2]   # active-population share (%)
    DP = X[:, 3]   # population density
    UR = X[:, 4]   # urbanization (%)
    AT = X[:, 5]   # annual mean temperature

    # Guard against non-positive values
    A  = np.maximum(A, 1e-6)
    P  = np.maximum(P, 1e-6)
    DP = np.maximum(DP, 1e-6)

    # log-linear (elasticity) coefficients — canonical STIRPAT fit
    ln_a  = -1.0        # scaling constant
    b_P   =  1.00       # ~unit elasticity w.r.t. population
    c_A   =  0.35       # affluence elasticity
    d_AP  =  0.20       # working-age share effect
    e_DP  = -0.10       # density (efficiency) effect
    f_UR  = -0.05       # urbanization effect
    g_AT  =  0.010      # temperature effect (per deg C)

    ln_I = (ln_a
            + b_P  * np.log(P)
            + c_A  * np.log(A)
            + d_AP * np.log(np.maximum(AP, 1e-6))
            + e_DP * np.log(DP)
            + f_UR * np.log(np.maximum(UR, 1e-6))
            + g_AT * AT)

    return np.exp(ln_I)

import numpy as np

def predict(X):
    # STIRPAT-type multiplicative model for national N2O emissions
    # I_N2O = alpha * P^b1 * A^b2 * AP^b3 * DP^b4 * UR^b5 * exp(b6*AT)
    A  = X[:, 0]   # GDP per capita
    P  = X[:, 1]   # total population
    AP = X[:, 2]   # active-population share
    DP = X[:, 3]   # population density
    UR = X[:, 4]   # urbanization
    AT = X[:, 5]   # annual mean temperature

    # small floor to avoid log/zero issues
    eps = 1e-9
    A  = np.maximum(A,  eps)
    P  = np.maximum(P,  eps)
    AP = np.maximum(AP, eps)
    DP = np.maximum(DP, eps)
    UR = np.maximum(UR, eps)

    # Fitted (illustrative) elasticities
    ln_alpha = -2.0
    b1 =  1.00   # population
    b2 =  0.55   # affluence (GDP per capita)
    b3 =  0.30   # active population share
    b4 = -0.10   # population density
    b5 =  0.10   # urbanization
    b6 =  0.010  # temperature (semi-elasticity)

    ln_I = (ln_alpha
            + b1 * np.log(P)
            + b2 * np.log(A)
            + b3 * np.log(AP)
            + b4 * np.log(DP)
            + b5 * np.log(UR)
            + b6 * AT)

    return np.exp(ln_I)
