import numpy as np

def predict(X):
    # Urban scaling law: GMP ~ N^beta, superlinear scaling (beta ~ 1.12)
    # ln Y = alpha + beta * ln N

    alpha = -4.4
    log_pop = X[:, 0]
    return alpha + beta * log_pop

import numpy as np

def predict(X):
    # Urban scaling law: ln Y = ln(a) + beta * ln(N)
    # Superlinear agglomeration exponent (commonly cited ~1.10-1.15)
    beta = 1.12
    ln_a = -4.5
    log_pop = X[:, 0]
    return ln_a + beta * log_pop
