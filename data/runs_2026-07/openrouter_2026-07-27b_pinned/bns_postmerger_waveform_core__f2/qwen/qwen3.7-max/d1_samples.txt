def predict(X):
    kap2t = X[:, 2]

    # as a function of the tidal coupling constant kappa_2^T

import numpy as np

def predict(X):
    # Extract the combined tidal coupling constant kappa_2^T
    kap2t = X[:, 2]
    
    # Canonical universal relation for the dominant post-merger peak frequency f2
    # as a function of the tidal coupling constant kappa_2^T.
    # Based on the tight empirical correlation established in the literature 
    # (e.g., Bernuzzi et al. 2015, Takami et al. 2015).
    # Coefficients are derived from high-resolution NR simulation fits.
    a = 3.52
    b = -0.0012
    
    f2 = a + b * kap2t
    
    return f2

import numpy as np

def predict(X):
    # Extract the combined tidal coupling constant (kappa_2^T)
    kap2t = X[:, 2]
    
    # Universal relation for the dominant post-merger frequency f2 (in kHz)
    # as a function of kappa_2^T, canonically established by Bernuzzi et al. (2015).
    # The use of kappa_2^T absorbs the mass and mass-ratio dependencies, 
    # yielding a highly accurate, equation-of-state-independent linear fit.
    a = 4.044
    b = -0.00214
    
    f2 = a + b * kap2t
    
    return f2

import numpy as np

def predict(X):
    # f2-kappa2T universal relation (Bernuzzi et al. 2015, PRL 114, 161103)
    # Quadratic fit of the dominant post-merger GW peak frequency f2 [kHz]
    # as a function of the tidal coupling constant kappa_2^T.
    kap2t = X[:, 2]
    f2 = 4.194 - 2.556e-3 * kap2t + 5.329e-7 * kap2t**2
    return f2
