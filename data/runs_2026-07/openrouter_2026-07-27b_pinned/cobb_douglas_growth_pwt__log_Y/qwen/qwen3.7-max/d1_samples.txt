def predict(X):
    K = X[:, 0]
    L = X[:, 1]
    h = X[:, 2]
    
    # Standard Cobb-Douglas production function augmented with human capital
    # Y = A * K^alpha * (h * L)^(1 - alpha)
    # ln(Y/L) = ln(A) + alpha * ln(K/L) + (1 - alpha) * ln(h)
    
    alpha = 1.0 / 3.0  # Standard capital share of income
    ln_A = 6.1         # Empirical intercept (log TFP) to align PWT 11.0 2017 USD units
    
    log_y = ln_A + alpha * np.log(K / L) + (1.0 - alpha) * np.log(h)
    return log_y

import numpy as np

def predict(X):
    K = X[:, 0]
    L = X[:, 1]
    h = X[:, 2]
    
    # Standard Cobb-Douglas production function in per-worker terms:
    # Y/L = A * (K/L)^alpha * h^(1-alpha)
    # Using the canonical capital share alpha = 1/3 and assuming A=1 (ln(A)=0) 
    # for the theoretical closed-form relationship.
    alpha = 1.0 / 3.0
    
    ln_Y_L = alpha * np.log(K / L) + (1.0 - alpha) * np.log(h)
    
    return ln_Y_L
