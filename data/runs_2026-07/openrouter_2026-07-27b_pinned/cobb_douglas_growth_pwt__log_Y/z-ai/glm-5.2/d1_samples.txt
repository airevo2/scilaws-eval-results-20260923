import numpy as np

def predict(X):
    K = X[:, 0]
    L = X[:, 1]
    h = X[:, 2]
    
    # Standard Cobb-Douglas production function:
    # Y = A * K^alpha * (h * L)^(1 - alpha)
    # where alpha is the capital share (typically 1/3) and A is TFP (normalized to 1).
    # ln(Y/L) = alpha * ln(K) + (1 - alpha) * ln(h) + (1 - alpha) * ln(L) - ln(L)
    # ln(Y/L) = alpha * (ln(K) - ln(L)) + (1 - alpha) * ln(h)
    
    alpha = 1.0 / 3.0
    
    log_Y = alpha * (np.log(K) - np.log(L)) + (1.0 - alpha) * np.log(h)
    
    return log_Y

import numpy as np

def predict(X):
    """
    Standard augmented Solow (Cobb-Douglas) closed-form relationship.
    Y = A * K^alpha * (h * L)^(1-alpha)
    => ln(Y/L) = ln(A) + alpha * ln(K/L) + (1-alpha) * ln(h)
    
    Canonical capital share alpha = 1/3.
    TFP term ln(A) normalized to 0 (A=1) for the structural form.
    """
    K = X[:, 0]          # capital stock, millions 2017 USD
    L = X[:, 1]          # persons engaged, millions
    h = X[:, 2]          # human capital index

    alpha = 1.0 / 3.0
    log_A = 0.0          # normalized TFP

    log_K_over_L = np.log(K) - np.log(L)
    log_h = np.log(h)

    log_Y_per_L = log_A + alpha * log_K_over_L + (1.0 - alpha) * log_h
    return log_Y_per_L

        Y = A * K^alpha * (h * L)^(1-alpha)
    =>  ln(Y/L) = ln(A) + alpha * ln(K/L) + (1-alpha) * ln(h)
    
    Here we set A = 1 (constant absorbed) and alpha = 1/3, the canonical
    capital share in macro development accounting.
    """

    L = X[:, 1]   # persons engaged (millions)

    log_Y = alpha * (np.log(K) - np.log(L)) + (1.0 - alpha) * np.log(h)
    return log_Y
