import numpy as np

def predict(X):
    """
    Predict log(Y/L) using the standard Cobb-Douglas production function
    in intensive form with human capital: Y/L = (K/L)^alpha * h^(1-alpha),
    where alpha is the capital share of output.

    Inputs:
        X[:, 0] = cn  (K)   -- Capital stock [millions 2017 USD PPP]
        X[:, 1] = emp (L)   -- Employment      [millions of persons]
        X[:, 2] = hc  (h)   -- Human capital index (unitless, typically 1 to 4)

    Returns:
        log_Y -- Natural log of output-side real GDP per person engaged
                 (ln(Y/L)), in log-millions-2017-USD per log-millions-persons.
    """
    # Standard capital share in macroeconomics (often 1/3 or 0.3)
    ALPHA = 1.0 / 3.0

    K = X[:, 0]
    L = X[:, 1]
    h = X[:, 2]

    # Intensive-form production function (TFP normalised to 1)
    log_Y = ALPHA * np.log(K / L) + (1.0 - ALPHA) * np.log(h)
    return log_Y

def predict(X):
    """
    Predict log output per worker using the standard Cobb-Douglas production function
    with labor-augmenting human capital: Y = A * K^alpha * (L * h)^(1 - alpha)
    => log(Y/L) = log_A + alpha * log(K/L) + (1 - alpha) * log(h).

    Coefficients:
        alpha = 1/3  (commonly cited capital share)
        log_A = 0.0  (TFP set to 1, as TFP is not provided among inputs)
    """
    K = X[:, 0]          # cn: capital stock
    L = X[:, 1]          # emp: employment
    h = X[:, 2]          # hc: human capital index

    alpha = 1.0 / 3.0
    log_A = 0.0          # could be a named constant, e.g., TFP_log

    log_Y_per_L = log_A + alpha * np.log(K / L) + (1.0 - alpha) * np.log(h)
    return log_Y_per_L

    L = X[:, 1]   # Persons engaged (emp, millions)

    # Y = A * K^alpha * (h * L)^(1 - alpha)
    # => Y/L = A * (K/L)^alpha * h^(1 - alpha)
    # log(Y/L) = log(A) + alpha * log(K/L) + (1 - alpha) * log(h)
    # Using standard capital share alpha = 1/3 and normalising TFP (A = 1)
    alpha = 1.0 / 3.0
    log_YL = alpha * np.log(K / L) + (1.0 - alpha) * np.log(h)
    return log_YL
