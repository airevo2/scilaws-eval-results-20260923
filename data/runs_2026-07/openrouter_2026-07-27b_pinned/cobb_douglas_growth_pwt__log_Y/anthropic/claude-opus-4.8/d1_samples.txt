import numpy as np

def predict(X):
    # Cobb-Douglas per-worker form: Y = A * K^alpha * (h*L)^(1-alpha)
    # => ln(Y/L) = ln A + alpha*ln(K/L) + (1-alpha)*ln(h)
    K = X[:, 0]
    L = X[:, 1]
    h = X[:, 2]

    alpha = 1.0 / 3.0        # capital share
    lnA = -1.5               # TFP / units constant

    ln_Y_over_L = lnA + alpha * np.log(K / L) + (1.0 - alpha) * np.log(h)
    return ln_Y_over_L
