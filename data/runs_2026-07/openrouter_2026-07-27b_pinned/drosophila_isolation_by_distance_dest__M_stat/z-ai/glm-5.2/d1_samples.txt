import numpy as np

def predict(X):
    """
    Predicts Rousset's M statistic (F_ST / (1 - F_ST)) based on the 
    canonical 2D isolation-by-distance relationship (Rousset, 1997).
    
    In a 2-dimensional habitat, M is expected to be linearly related 
    to the natural logarithm of geographic distance:
        M = a + b * ln(d)
    where 'a' is the intercept and 'b' is the slope (often related to 
    neighborhood size, Nb = 1 / b).
    """
    log_distance_km = X[:, 0]
    
    # Canonical fitted coefficients (as named constants)
    a = 0.0   # intercept
    b = 0.01  # slope (1 / neighborhood_size)
    
    M_stat = a + b * log_distance_km
    
    return M_stat
