import numpy as np

def predict(X):
    # Rousset (1997) isolation-by-distance:
    # For a two-dimensional habitat, the linearised differentiation statistic
    # M = F_ST / (1 - F_ST) increases linearly with the logarithm of geographic distance.
    #   M_stat = a + b * ln(d)
    # where the slope b = 1 / (4 * pi * D * sigma^2) relates to neighbourhood size.
    log_d = X[:, 0]

    a = 0.0        # intercept
    b = 0.1        # slope (~ 1 / (4 pi D sigma^2))

    return a + b * log_d
