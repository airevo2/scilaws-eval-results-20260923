def predict(X):
    """
    Predict Rousset's M-statistic (M = F_ST / (1 - F_ST)) under isolation by distance
    in a two-dimensional habitat.
    
    Standard closed-form relationship:
        M = (1 / (4 * pi * D * sigma^2)) * ln(d) + C
    where
        D      = effective population density,
        sigma^2 = mean squared parent-offspring dispersal distance,
        C      = a constant depending on mutation rate and other factors.
        
    Input X:
        X[:, 0] = log_distance_km (ln(d))
        X[:, 1] = distance_km (d)  (not used directly in this 2D IBD formula)
    """
    # Biological parameters (named constants, can be replaced with fitted values)
    D = 1.0          # effective population density (individuals per km^2)
    sigma2 = 1.0     # mean squared dispersal distance (km^2)
    C = 0.0          # intercept constant

    # Slope of the isolation-by-distance relationship in 2D
    slope = 1.0 / (4.0 * np.pi * D * sigma2)

    # Natural log of distance (provided as X[:, 0])
    ln_d = X[:, 0]

    # Predicted M-statistic
    M = slope * ln_d + C
    return M

def predict(X):
    """
    Predicts Rousset's M-statistic (M = F_ST / (1 - F_ST)) for pairs of populations
    under isolation by distance in a two-dimensional continuous habitat.

        M = (1 / (4 * pi * D * sigma^2)) * ln(d) + constant
    where:
        D       = effective population density (individuals per km^2)
        sigma^2 = mean square parent-offspring distance (km^2)
        d       = great-circle distance between populations (km)

    Input:
        X : 2-D numpy array

            X[:, 1] = distance_km = d   (not used in this canonical form)

    """

    D = 1.0          # effective population density (ind/km^2)
    sigma2 = 1.0     # mean square parent-offspring distance (km^2)
    const = 0.0      # additive constant (often negligible or absorbed in fitting)

    slope = 1.0 / (4.0 * np.pi * D * sigma2)
    M = slope * X[:, 0] + const

import numpy as np

def predict(X):
    """
    Predict Rousset's linearised differentiation statistic M = F_ST / (1 - F_ST)
    under isolation by distance in a two-dimensional continuum.
    
    The standard closed-form relationship is:
        M = a + b * ln(distance)
    where b = 1 / (4 * pi * D * sigma^2),
    D = population density, sigma^2 = dispersal variance,
    and a is a constant depending on mutation rate and habitat shape.
    
    Here we use default theoretical coefficients (D=1, sigma^2=1, a=0).
    """
    # Named constants (fitted or theoretical coefficients)
    D = 1.0          # population density
    sigma2 = 1.0     # dispersal variance
    a = 0.0          # intercept
    b = 1.0 / (4.0 * np.pi * D * sigma2)  # slope

    # X[:, 0] is log_distance_km = ln(d)
    return a + b * X[:, 0]
