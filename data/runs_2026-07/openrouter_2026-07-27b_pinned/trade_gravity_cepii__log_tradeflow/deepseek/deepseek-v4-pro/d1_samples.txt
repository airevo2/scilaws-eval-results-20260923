      X[:, 1] = log_gdp_d      (ln Y_n)

      X[:, 3] = contig         (c_{ij})
      X[:, 4] = comlang_off    (l_{ij})
      X[:, 5] = col_dep_ever   (k_{ij})
      X[:, 6] = fta_wto        (r_{ij})

    Returns:
      1-D array of ln X_{ij} (log thousands current USD)

    log_gdp_o = X[:, 0]
    log_gdp_d = X[:, 1]
    log_dist  = X[:, 2]
    contig   = X[:, 3]
    comlang  = X[:, 4]
    colony   = X[:, 5]
    fta      = X[:, 6]

    # Canonical gravity coefficients (unit income elasticities,
    # distance elasticity = -1, plus typical border/historical/trade agreement effects)
    CONST       = -25.0   # absorbs world GDP and scale (thousands USD)
    BETA_GDP    =  1.0
    BETA_DIST   = -1.0
    BETA_CONTIG =  0.5
    BETA_COMLANG=  0.3
    BETA_COLONY =  0.8

    const = -20.0          # intercept, capturing scale and multilateral resistance

    beta_log_dist = -1.14  # distance elasticity
    beta_contig = 0.53     # shared border
    beta_comlang_off = 0.29  # common official language
    beta_col_dep_ever = 0.65  # colonial/dependency link ever

                 + beta_log_gdp_o * X[:, 0]
