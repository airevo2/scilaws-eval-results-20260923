def predict(X):
    # Standard gravity equation (log-linear form)
    log_gdp_o = X[:, 0]
    log_gdp_d = X[:, 1]
    log_dist  = X[:, 2]
    contig    = X[:, 3]
    comlang   = X[:, 4]
    col       = X[:, 5]
    fta       = X[:, 6]

    # Typical fitted coefficients from standard gravity estimations
    ln_X = (-30.0
            + 1.0  * log_gdp_o
            + 0.9  * log_gdp_d
            - 1.0  * log_dist
            + 0.6  * contig
            + 0.4  * comlang
            + 0.8  * col
            + 0.4  * fta)
    return ln_X
