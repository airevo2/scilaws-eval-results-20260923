import numpy as np

def predict(X):
    """
    Predicts the log of bilateral trade flows using the standard log-linear 
    Gravity Model of International Trade. 
    
    Coefficients are based on typical empirical estimates from the gravity 
    literature (e.g., Disdier & Head 2008 meta-analysis, Head & Mayer 2014), 
    adjusted for the specified units (Trade in thousands USD, GDP in USD).
    """
    # Intercept (ln G - ln Y_world + ln 1000 + average multilateral resistance effects)
    BETA_0 = -26.50
    
    # Elasticities and semi-elasticities
    BETA_GDP_O = 0.92    # Origin GDP elasticity (typically ~0.9 - 1.0)
    BETA_GDP_D = 0.94    # Destination GDP elasticity (typically ~0.9 - 1.0)
    BETA_DIST = -0.78    # Distance elasticity (typically ~ -0.7 to -0.9)
    BETA_CONTIG = 0.45   # Contiguity premium
    BETA_LANG = 0.35     # Common official language premium
    BETA_COLONY = 0.28   # Historical colonial tie premium
    BETA_FTA = 0.32      # Regional Trade Agreement / WTO premium
    
    # Extract features
    log_gdp_o = X[:, 0]
    log_gdp_d = X[:, 1]
    log_dist = X[:, 2]
    contig = X[:, 3]
    comlang_off = X[:, 4]
    col_dep_ever = X[:, 5]
    fta_wto = X[:, 6]
    
    # Compute log trade flow
    log_tradeflow = (
        BETA_0 + 
        BETA_GDP_O * log_gdp_o + 
        BETA_GDP_D * log_gdp_d + 
        BETA_DIST * log_dist + 
        BETA_CONTIG * contig + 
        BETA_LANG * comlang_off + 
        BETA_COLONY * col_dep_ever + 
        BETA_FTA * fta_wto
    )
    
    return log_tradeflow
