import numpy as np

def predict(X):
    # Kleiber's law: B (kcal/day) = 70 * M(kg)^0.75
    # Convert to Watts: 1 kcal/day = 4184 J / 86400 s = 0.048426 W
    # Convert mass: M(kg) = M(g) / 1000
    # B(W) = 70 * (4184/86400) * (M_g/1000)^0.75
    # log10(B_W) = log10(70 * 4184 / 86400) + 0.75 * (log10(M_g) - 3)
    #            = log10(3.3898) - 2.25 + 0.75 * log10(M_g)
    #            = -1.7198 + 0.75 * log10(M_g)
    log10_M = np.log10(X[:, 0])
    log10_B = -1.7198 + 0.75 * log10_M
    return log10_B
