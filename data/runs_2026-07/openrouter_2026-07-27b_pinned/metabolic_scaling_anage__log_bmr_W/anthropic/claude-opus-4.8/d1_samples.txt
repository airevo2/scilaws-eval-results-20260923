import numpy as np

def predict(X):
    M_g = X[:, 0]
    # Kleiber's law: BMR = 70 kcal/day * M_kg^0.75
    # 70 kcal/day = 70 * 4184 J / 86400 s = 3.39 W
    coef_W = 3.39      # normalization constant in watts (at 1 kg)
    exponent = 0.75    # Kleiber scaling exponent
    M_kg = M_g / 1000.0
    B = coef_W * np.power(M_kg, exponent)
    return np.log10(B)

import numpy as np

def predict(X):

    # Kleiber's law: BMR ~ 70 kcal/day * M_kg^0.75
    # 70 kcal/day = 3.39 W  =>  B(W) = 3.39 * (M/1000)^0.75
    log_M_kg = np.log10(M / 1000.0)
    return np.log10(3.39) + 0.75 * log_M_kg
