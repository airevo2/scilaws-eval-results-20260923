import numpy as np

def predict(X):
    """
    Predicts log10 of basal metabolic rate (B) in watts given body mass in grams.
    Uses the standard allometric scaling: B = a * M^b, with b = 0.75 (Kleiber's law).
    The coefficient is derived from Kleiber's original equation: B (kcal/day) = 70 * M_kg^0.75,
    converted to watts and mass in grams.
    """
    M_g = X[:, 0]  # body mass in grams

    # Constants for unit conversion
    kleiber_kcal_per_day_kg = 70.0      # kcal/day for a 1 kg animal^0.75
    kcal_to_J = 4184.0                  # J per kcal
    sec_per_day = 86400.0               # s per day
    g_per_kg = 1000.0                   # g per kg

    # B (W) = 70 * (M_g/1000)^0.75 * 4184/86400
    # log10(B) = log10(70 * 4184/86400 / 1000^0.75) + 0.75 * log10(M_g)
    intercept = np.log10(kleiber_kcal_per_day_kg * kcal_to_J / sec_per_day / g_per_kg**0.75)
    slope = 0.75

    return slope * np.log10(M_g) + intercept

import numpy as np

def predict(X):
    """
    Predict log10(BMR) in Watts from body mass in grams.
    Uses Kleiber's law: BMR (W) = a * (mass_kg)^0.75.
    Commonly cited a = 3.4 W/kg^0.75 for eutherian mammals.
    """
    mass_g = X[:, 0]
    # Convert mass to kg: mass_kg = mass_g / 1000.
    # log10(B) = log10(3.4) + 0.75 * log10(mass_kg)
    #          = log10(3.4) + 0.75 * (log10(mass_g) - 3)
    #          = 0.75 * log10(mass_g) + log10(3.4) - 2.25
    logB = 0.75 * np.log10(mass_g) + np.log10(3.4) - 2.25
    return logB

import numpy as np

def predict(X):
    """
    Predict log10 of basal metabolic rate (BMR) in watts given body mass in grams.
    Uses Kleiber's law: BMR (kcal/day) = 70 * (body_mass_kg)^0.75
    Converted to watts and log10 form.
    
    Parameters:
    X : np.ndarray
        2D array where X[:, 0] is body_mass_g (grams)
    
    Returns:
    np.ndarray
        1D array of log10(BMR) in watts
    """
    body_mass_g = X[:, 0]
    # Kleiber's law constants
    kcal_per_day_to_watts = 4184 / 86400       # 1 kcal/day in watts
    a_kcal_per_day = 70.0                       # coefficient in kcal/day for kg^0.75
    a_watts = a_kcal_per_day * kcal_per_day_to_watts  # coefficient in W for kg^0.75
    # BMR (W) = a_watts * (mass_kg)^0.75
    # mass_kg = body_mass_g / 1000
    # log10(BMR) = log10(a_watts) + 0.75 * (log10(body_mass_g) - 3)
    log10_bmr_W = np.log10(a_watts) + 0.75 * (np.log10(body_mass_g) - 3.0)
    return log10_bmr_W
