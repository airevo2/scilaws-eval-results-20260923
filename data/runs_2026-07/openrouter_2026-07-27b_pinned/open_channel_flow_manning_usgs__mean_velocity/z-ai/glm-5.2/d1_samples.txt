import numpy as np

def predict(X):
    R = X[:, 0]
    S = X[:, 1]
    n = 0.035  # typical Manning's roughness coefficient for natural streams
    V = (1.0 / n) * np.power(R, 2.0 / 3.0) * np.sqrt(S)
    return V
