import numpy as np

def predict(X):
    """
    Predict cross-section mean flow velocity (V, m/s) using Manning's equation.
    
    Inputs:
        X[:, 0] = R_m (hydraulic depth, m)
        X[:, 1] = SLOPE (friction slope, dimensionless)
    
    Returns:
        1-D numpy array of V (m/s)
    
    For natural streams, a typical Manning's n is assumed.
    """
    # Extract inputs
    R = X[:, 0]      # hydraulic depth [m]
    S_f = X[:, 1]    # friction slope [-]

    # Manning's roughness coefficient (typical for natural channels)
    MANNING_N = 0.03  # s/m^(1/3)

    # Manning equation in SI units: V = (1/n) * R^(2/3) * S_f^(1/2)
    V = (1.0 / MANNING_N) * (R ** (2.0 / 3.0)) * np.sqrt(S_f)

    return V

import numpy as np

def predict(X):
    """
    Predict mean flow velocity V (m/s) from hydraulic depth R (m) and slope S (-)
    using the Manning equation (the standard closed-form relationship for open-channel flow).

    Manning's equation (SI units):
        V = (1/n) * R^(2/3) * S^(1/2)
    where n is Manning's roughness coefficient.
    A typical value for natural streams (n = 0.03) is used as the default.
    """
    # Named constant: Manning's n for natural channels
    MANNING_N = 0.03

    R = X[:, 0]   # hydraulic depth (m)
    S = X[:, 1]   # slope (dimensionless)

    # Manning formula
    V = (1.0 / MANNING_N) * (R ** (2.0 / 3.0)) * np.sqrt(S)

    return V
