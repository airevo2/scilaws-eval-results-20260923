import numpy as np

def predict(X):
    # Sørensen similarity decays exponentially with environmental distance
    # (distance-decay of similarity; Nekola & White 1999)
    d = X[:, 0]
    S0 = 0.5      # initial similarity at zero environmental distance
    k = 0.3       # decay rate
    return S0 * np.exp(-k * d)

import numpy as np

def predict(X):
    # Distance-decay of similarity: exponential decay model (most commonly cited)
    # S(d) = S0 * exp(-alpha * d)
    d = X[:, 0]
    S0 = 0.5      # initial similarity at zero environmental distance
    alpha = 0.3   # decay rate
    S = S0 * np.exp(-alpha * d)
    return np.clip(S, 0.0, 1.0)
