import numpy as np

def predict(X):
    """
    Predicts Sørensen similarity (S) from environmental distance (d).
    Uses the widely cited exponential decay model: S = S_max * exp(-alpha * d)
    with S_max = 1.0 (theoretical similarity at d=0) and alpha a decay rate.
    """
    d = X[:, 0]  # environmental distance (z-scored Euclidean)
    S_max = 1.0
    alpha = 0.5   # typical decay constant for standardised environment
    S = S_max * np.exp(-alpha * d)
    return S
