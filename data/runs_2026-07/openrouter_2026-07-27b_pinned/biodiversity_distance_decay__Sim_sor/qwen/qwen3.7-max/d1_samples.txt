import numpy as np

def predict(X):
    """
    Distance-decay of Sørensen similarity as a negative exponential
    of environmental distance (the canonical Nekola & White 1999 form).
    
    S = S0 * exp(-beta * d)
    """
    S0 = 1.0       # intercept: maximum similarity at zero environmental distance
    beta = 0.30    # exponential decay rate per unit z-scored Euclidean distance
    
    d = X[:, 0]
    S = S0 * np.exp(-beta * d)
    return np.clip(S, 0.0, 1.0)

import numpy as np

def predict(X):
    env_dist = X[:, 0]
    
    # Distance-decay of similarity (exponential decay model)
    # S0: similarity at zero distance (intercept)
    # beta: decay rate with respect to standardized environmental distance
    S0 = 1.0
    beta = 0.25 
    
    Sim_sor = S0 * np.exp(-beta * env_dist)
    
    # Ensure bounds [0, 1]
    return np.clip(Sim_sor, 0.0, 1.0)
