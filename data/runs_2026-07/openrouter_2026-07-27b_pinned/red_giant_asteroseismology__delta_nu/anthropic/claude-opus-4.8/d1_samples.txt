import numpy as np

def predict(X):
    # Canonical scaling relation: Delta_nu ~ alpha * nu_max^beta
    # (Stello et al. 2009 / Huber et al. 2011 style)
    # Delta_nu_sun ~ 135.1 uHz, nu_max_sun ~ 3090 uHz
    nu_max = X[:, 0]
    alpha = 0.267
    beta = 0.760
    delta_nu = alpha * np.power(nu_max, beta)
    return delta_nu
