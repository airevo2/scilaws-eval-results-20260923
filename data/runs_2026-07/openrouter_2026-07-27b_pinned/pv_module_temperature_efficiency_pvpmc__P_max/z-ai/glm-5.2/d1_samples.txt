import numpy as np

def predict(X):
    """
    Predicts per-module DC power at the maximum-power point (P_mp) [W].
    Uses the PVWatts v5 model with the Sandia air mass modifier.
    Wind speed (WS) is omitted because back-of-cell module temperature (T_c) 
    is directly provided.
    """
    # Unpack inputs
    G = X[:, 0]      # Plane-of-array irradiance [W/m^2]
    T_c = X[:, 1]    # Back-of-cell module temperature [deg C]
    # WS = X[:, 2]  # Wind speed [m/s] (unused, T_c is directly measured)
    AM = X[:, 3]     # Absolute air mass [dimensionless]
    
    # Constants and coefficients
    P_mp0 = 250.0    # Typical rated module power at STC [W]
    G0 = 1000.0      # Reference irradiance [W/m^2]
    T0 = 25.0        # Reference temperature [deg C]
    gamma = -0.0045  # Maximum power temperature coefficient [1/deg C]
    
    # PVWatts v5 Air Mass modifier polynomial coefficients
    c0 = 0.918093
    c1 = 0.086257
    c2 = -0.024599
    c3 = 0.001635
    c4 = -0.000034
    
    # Calculate air mass modifier
    f_AM = c0 + c1*AM + c2*AM**2 + c3*AM**3 + c4*AM**4
    
    # PVWatts v5 model for P_mp
    P_mp = P_mp0 * (G / G0) * (1 + gamma * (T_c - T0)) * f_AM
    
    # Ensure non-negative power output
    P_mp = np.maximum(P_mp, 0.0)
    
    return P_mp

import numpy as np

def predict(X):
    """
    Predict per-module DC power at the maximum-power point (P_mp) using the
    PVUSA (Photovoltaics for Utility Scale Applications) empirical model.
    
    The standard closed-form relationship is:
        P_mp = G * (C1 + C2*G + C3*T_c + C4*WS*ln(G) + C5*AM*(ln(G))^2)
    
    Inputs:
        X[:, 0] = G      [W/m²]   Plane-of-array irradiance
        X[:, 1] = T_c    [°C]     Module temperature
        X[:, 2] = WS     [m/s]    Wind speed
        X[:, 3] = AM     [—]      Absolute air mass
    
    Returns:
        P_mp [W] as a 1-D numpy array.
    """
    G = X[:, 0]
    T_c = X[:, 1]
    WS = X[:, 2]
    AM = X[:, 3]

    # Empirical PVUSA coefficients (typical values for a c-Si module).
    # In practice these are fitted to field measurements.
    C1 = 0.275        # intercept (effective efficiency at STC)
    C2 = -1.0e-6      # irradiance non-linearity
    C3 = -1.0e-3      # temperature coefficient (per °C)
    C4 = -1.0e-3      # wind-speed cooling coefficient
    C5 = 1.0e-4       # air-mass spectral coefficient

    lnG = np.log(G)
    P_mp = G * (C1 + C2*G + C3*T_c + C4*WS*lnG + C5*AM*(lnG**2))

    return P_mp

import numpy as np

def predict(X):
    G = X[:, 0]
    T_c = X[:, 1]
    # WS = X[:, 2]  # Wind speed is not used here since T_c is measured directly
    AM = X[:, 3]
    
    G0 = 1000.0
    T0 = 25.0
    
    # Standard reference parameters for a typical c-Si module
    P_mp0 = 250.0       # Reference maximum power [W]
    alpha_Pmp = -0.004  # Temperature coefficient for P_mp [1/deg C]
    C1 = -0.04          # SAPM irradiance coefficient 1
    C2 = 0.005          # SAPM irradiance coefficient 2
    
    # Sandia Array Performance Model (SAPM) air mass modifier coefficients
    a0 = 0.918093
    a1 = 0.086257
    a2 = -0.024459
    a3 = 0.002816
    a4 = -0.000126
    
    # Calculate air mass modifier
    F1_AM = a0 + a1*AM + a2*AM**2 + a3*AM**3 + a4*AM**4
    F1_AM = np.clip(F1_AM, 0.0, None)
    
    # Effective irradiance (assuming incidence angle modifier = 1 as AOI is not provided)
    Ee = (G / G0) * F1_AM
    Ee = np.clip(Ee, 1e-6, None)  # Prevent log(0)
    
    # SAPM closed-form equation for P_mp
    P_mp = P_mp0 * Ee * (1 + C1 * np.log(Ee) + C2 * (np.log(Ee))**2) * (1 + alpha_Pmp * (T_c - T0))
    
    return P_mp
