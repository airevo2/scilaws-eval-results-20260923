import numpy as np

def predict(X):
    # Standard empirical scaling for bulk modulus (Cohen-type):
    # bulk modulus scales inversely with atomic volume (bond length d ~ V_a^(1/3),
    # and K ~ d^-3.5 ~ V_a^(-1.167)).
    V_a = X[:, 0]          # atomic volume [Angstrom^3]
    C = 3300.0             # proportionality constant [GPa * A^3.5]
    p = 1.167              # ~ 3.5/3 exponent from Cohen's relation
    K_VRH = C * np.power(V_a, -p)
    return K_VRH

import numpy as np

def predict(X):
    """
    Closed-form estimate of the Voigt-Reuss-Hill bulk modulus.

    Physical basis:
    The bulk modulus of a solid scales inversely with the atomic
    volume (K ~ d(E)/dV per atom); the most commonly cited empirical
    scaling for a broad set of crystalline solids is a power law

        K_VRH = C * V_atomic^(-n)

    with n close to unity.  This captures the dominant trend that
    denser-packed (small atomic volume) materials are much stiffer.
    """
    V_a = np.asarray(X[:, 0], dtype=float)   # atomic volume [A^3]

    # Fitted constants (power-law of atomic volume)
    C = 2000.0     # [GPa * A^3^n]
    n = 1.0        # exponent

    K_VRH = C * np.power(V_a, -n)

    return K_VRH
