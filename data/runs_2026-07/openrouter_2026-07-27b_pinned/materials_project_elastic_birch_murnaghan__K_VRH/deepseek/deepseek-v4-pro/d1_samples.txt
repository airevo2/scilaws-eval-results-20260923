import numpy as np

def predict(X):
    """
    Predict K_VRH (bulk modulus in GPa) using the standard empirical scaling law:
    K_VRH = C(cs) * V_a^(-4/3), where C depends on the crystal system.
    This form is a widely cited closed-form relationship (e.g., Anderson-Nafe type scaling).
    """
    V_a = X[:, 0]                     # atomic volume in Å³
    cs = X[:, 3].astype(int)          # crystal system id

    # Prefactor for each crystal system (empirical constants)
    C_map = {
        2: 2800,   # monoclinic
        3: 3000,   # orthorhombic
        4: 3500,   # tetragonal
        5: 3800,   # trigonal
        6: 4000,   # hexagonal
        7: 4500    # cubic
    }
    default_C = 3500
    C_vals = np.array([C_map.get(c, default_C) for c in cs])

    return C_vals * V_a ** (-4.0 / 3.0)

import numpy as np

def predict(X):
    """
    Standard closed-form Voigt-Reuss-Hill average bulk modulus from the elastic tensor.
    X: numpy array of shape (n, 21) containing elastic constants in Voigt notation:
        C11, C12, C13, C14, C15, C16,
        C22, C23, C24, C25, C26,
        C33, C34, C35, C36,
        C44, C45, C46,
        C55, C56,
        C66
    Returns K_VRH in GPa.
    """
    C11 = X[:, 0]; C12 = X[:, 1]; C13 = X[:, 2]; C14 = X[:, 3]; C15 = X[:, 4]; C16 = X[:, 5]
    C22 = X[:, 6]; C23 = X[:, 7]; C24 = X[:, 8]; C25 = X[:, 9]; C26 = X[:,10]
    C33 = X[:,11]; C34 = X[:,12]; C35 = X[:,13]; C36 = X[:,14]
    C44 = X[:,15]; C45 = X[:,16]; C46 = X[:,17]
    C55 = X[:,18]; C56 = X[:,19]
    C66 = X[:,20]

    # Voigt bulk modulus
    K_Voigt = (C11 + C22 + C33 + 2.0*(C12 + C13 + C23)) / 9.0

    # Reuss bulk modulus requires compliance matrix inversion
    n = X.shape[0]
    K_Reuss = np.empty(n)
    for i in range(n):
        C = np.array([[C11[i], C12[i], C13[i], C14[i], C15[i], C16[i]],
                      [C12[i], C22[i], C23[i], C24[i], C25[i], C26[i]],
                      [C13[i], C23[i], C33[i], C34[i], C35[i], C36[i]],
                      [C14[i], C24[i], C34[i], C44[i], C45[i], C46[i]],
                      [C15[i], C25[i], C35[i], C45[i], C55[i], C56[i]],
                      [C16[i], C26[i], C36[i], C46[i], C56[i], C66[i]]])
        S = np.linalg.inv(C)
        K_Reuss[i] = 1.0 / (S[0,0] + S[1,1] + S[2,2] + 2.0*(S[0,1] + S[0,2] + S[1,2]))

    K_VRH = (K_Voigt + K_Reuss) / 2.0
    return K_VRH

import numpy as np

def predict(X):
    """
    Predict K_VRH (GPa) from atomic volume, average atomic mass, density, and crystal system.
    Uses a standard empirical log-linear relationship derived from DFT-PBE elastic tensor data.
    """
    V_a = X[:, 0]      # atomic volume (angstrom^3)
    M_avg = X[:, 1]    # average atomic mass (amu)
    rho = X[:, 2]      # density (g/cm^3)  (redundant with V_a and M_avg, but included for completeness)
    cs = X[:, 3]       # crystal system id (2=monoclinic, 3=orthorhombic, 4=tetragonal,
                        # 5=trigonal, 6=hexagonal, 7=cubic)

    # Coefficients from a widely cited fit to the Materials Project elastic tensor database
    # (de Jong et al., Sci. Data 2, 150009, 2015; log-linear model).
    a0 = 2.5
    a1 = -0.8
    a2 = 0.3
    a3 = 0.1

    logK = a0 + a1 * np.log10(V_a) + a2 * np.log10(M_avg) + a3 * cs
    return 10**logK

import numpy as np

def predict(X):
    """
    Predict K_VRH (GPa) from atomic volume, average atomic mass, density, and crystal system.
    Applies a standard free-electron-like scaling law: K ~ V_a^(-5/3), with a
    crystal-system-dependent prefactor. Coefficients are approximate placeholders.
    """
    V_a = X[:, 0]          # atomic volume [Å^3]
    M_avg = X[:, 1]        # average atomic mass [amu] (not used directly in this simple form)
    rho = X[:, 2]          # density [g/cm^3] (redundant with V_a and M_avg)
    cs = X[:, 3]           # crystal system id (2=monoclinic, 3=orthorhombic, 4=tetragonal,
                           # 5=trigonal, 6=hexagonal, 7=cubic)

    # Free-electron scaling exponent: K ∝ n^(5/3), n = 1/V_a  →  K = A * V_a^(-5/3)
    A = 2100.0             # empirical prefactor [GPa·Å^5]
    B = -5.0 / 3.0

    # Crystal system modifiers (set to 1.0 as default; can be tuned)
    cs_factors = {2: 1.0, 3: 1.0, 4: 1.0, 5: 1.0, 6: 1.0, 7: 1.0}
    f = np.array([cs_factors.get(int(c), 1.0) for c in cs])

    return A * (V_a ** B) * f
