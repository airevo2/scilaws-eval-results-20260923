import numpy as np

def predict(X):
    phi = X[:, 0]  # aridity index PET/P
    # Budyko (Choudhury-Yang / Turc-Pike style) Fu equation
    w = 2.6  # commonly cited parameter for Fu's equation
    et_over_p = 1.0 + phi - (1.0 + phi**w)**(1.0 / w)
    return np.clip(et_over_p, 0.0, 1.0)

def predict(X):
    # Budyko framework - Choudhury-Yang single-parameter form
    # E/P = phi / (1 + phi^n)^(1/n), with n a fitted shape parameter
    phi = X[:, 0]
    n = 2.0  # commonly cited representative value
    et_over_p = phi / np.power(1.0 + np.power(phi, n), 1.0 / n)
