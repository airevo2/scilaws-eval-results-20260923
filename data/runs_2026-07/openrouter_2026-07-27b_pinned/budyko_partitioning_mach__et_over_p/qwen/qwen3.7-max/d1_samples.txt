import numpy as np

def predict(X):
    phi = X[:, 0]  # aridity index PET/P
    # Budyko (1974) equation — the canonical non-parametric Budyko curve
    # E/P = sqrt( phi * tanh(1/phi) * (1 - exp(-phi)) )
    et_over_p = np.sqrt(phi * np.tanh(1.0 / phi) * (1.0 - np.exp(-phi)))
    return et_over_p

import numpy as np

def predict(X):
    phi = X[:, 0]       # PET/P aridity index
    P   = X[:, 1]       # mean annual precipitation [mm/yr]
    Ep  = X[:, 2]       # mean annual potential ET [mm/yr]
    T   = X[:, 3]       # mean annual temperature [°C]
    A   = X[:, 4]       # drainage area [km²]
    S   = X[:, 5]       # basin slope [%]
    Z   = X[:, 6]       # mean elevation [m]
    c   = X[:, 7]       # clay fraction [%]
    BFI = X[:, 8]       # base-flow index

    # --- Fu (1981) Budyko equation ---
    # E/P = 1 + phi - (1 + phi^omega)^(1/omega)
    #
    # omega controls the partitioning curve and is parameterised
    # from catchment attributes following the common finding that
    # omega increases with soil water-holding capacity and
    # groundwater support, and decreases with steep slopes and
    # cold/high-elevation environments.

    omega = (2.0
             + 0.30 * (c   / 100.0)
             + 0.50 * BFI
             - 0.30 * (S   / 100.0)
             - 0.10 * (Z   / 1000.0)
             + 0.20 * (T   / 20.0)
             + 0.05 * np.log10(np.maximum(A, 1.0)))

    et_over_p = 1.0 + phi - (1.0 + np.power(phi, omega)) ** (1.0 / omega)

    return np.clip(et_over_p, 0.0, 1.0)

import numpy as np

def predict(X):
    phi = X[:, 0]
    with np.errstate(divide='ignore', invalid='ignore'):
        et_over_p = np.sqrt(phi * np.tanh(phi) * (1.0 - np.exp(-1.0 / phi)))
    et_over_p = np.where(phi == 0, 0.0, et_over_p)
    return et_over_p

import numpy as np

def predict(X):
    phi = X[:, 0]
    
    # Avoid division by zero in the tanh term
    phi_safe = np.where(phi == 0, 1e-12, phi)
    
    # Budyko (1974) canonical equation for the evaporative fraction
    e_over_p = np.sqrt(phi_safe * np.tanh(1.0 / phi_safe) * (1.0 - np.exp(-phi_safe)))
    
    # Handle the exact zero case and enforce physical bounds
    e_over_p = np.where(phi <= 0, 0.0, e_over_p)
    e_over_p = np.clip(e_over_p, 0.0, 1.0)
    
    return e_over_p
