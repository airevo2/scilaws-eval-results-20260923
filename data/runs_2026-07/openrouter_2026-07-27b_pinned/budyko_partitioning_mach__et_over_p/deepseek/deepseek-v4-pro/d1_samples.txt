import numpy as np

def predict(X):
    """
    Predict long-term mean annual E/P ratio using the standard Budyko (1974) curve.
    
    X[:, 0] : phi, climate aridity index (PET / P)
    Other columns are ignored in this closed-form relationship.
    """
    phi = X[:, 0]
    # Budyko (1974) formula: E/P = sqrt( phi * tanh(1/phi) * (1 - exp(-phi)) )
    # This is the geometric mean of the Schreiber (1904) and Ol'dekop (1911) curves.
    ep = phi * np.tanh(1.0 / phi) * (1.0 - np.exp(-phi))
    return np.sqrt(ep)

import numpy as np

def predict(X):
    """
    Predict long-term mean annual evaporative fraction E/P using the classic Budyko (1974) curve.
    
    The Budyko equation relates E/P to the aridity index φ = Ep/P:
        E/P = sqrt( φ * tanh(1/φ) * (1 - exp(-φ)) )
    
    Parameters
    ----------
    X : 2-D numpy array
        Feature matrix. Only column 0 (pet_over_p, aridity index) is used.
    
    Returns
    -------
    et_over_p : 1-D numpy array
        Predicted E/P values, bounded between 0 and 1.
    """
    phi = X[:, 0]  # aridity index φ = Ep/P
    # Avoid division by zero (φ > 0 in the given typical range)
    et_over_p = np.sqrt(phi * np.tanh(1.0 / phi) * (1.0 - np.exp(-phi)))
    return et_over_p
