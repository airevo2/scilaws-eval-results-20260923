    """
    Compute Planck spectral radiance L(λ, T) for the solar disk at 1 AU.
    
    Parameters
    ----------
    X : np.ndarray, shape (n, 1)
        X[:, 0] = wavelength_nm (λ) in nm.
    
    Returns
    -------
    L : np.ndarray, shape (n,)
        Spectral radiance in W m⁻² nm⁻¹ sr⁻¹.
    
    Notes
    -----
    The formula uses the solar effective temperature T = 5772 K.
    The Planck spectral radiance in wavelength units is

    h = 6.62607015e-34      # Planck constant, J s

import numpy as np

def predict(X):
    """
    Compute Planck spectral radiance L(λ, T) [W m⁻² nm⁻¹ sr⁻¹]
    for a solar effective temperature T = 5772 K.

    Input:
        X : 2-D numpy array, X[:, 0] = wavelength in nm
    Output:
        L : 1-D numpy array of spectral radiance
    """
    # Fundamental constants (SI)
    h = 6.62607015e-34      # Planck constant, J·s
    c = 2.99792458e8        # speed of light, m/s
    k_B = 1.380649e-23      # Boltzmann constant, J/K
    T = 5772                # solar effective temperature, K

    # Wavelength conversion: nm -> m
    wl_nm = X[:, 0]               # nm
    wl_m = wl_nm * 1.0e-9         # m

    # Planck law: spectral radiance per unit wavelength (m)
    B_m = ( (2.0 * h * c**2) / (wl_m**5) ) / ( np.exp( h * c / (wl_m * k_B * T) ) - 1.0 )
    # B_m is in W m⁻² sr⁻¹ m⁻¹. Convert to per nm: 1 m⁻¹ = 1e-9 nm⁻¹
    L = B_m * 1.0e-9              # W m⁻² sr⁻¹ nm⁻¹

    return L

import numpy as np

def predict(X):
    # Physical constants (SI)
    h = 6.62607015e-34      # Planck constant, J s
    c = 299792458.0        # speed of light, m/s
    kB = 1.380649e-23      # Boltzmann constant, J/K

    # Solar effective temperature (K) — standard value consistent with SORCE SIM
    T_sun = 5778.0

    # Extract wavelength in nm from the input array
    lambda_nm = X[:, 0]

    # Convert to metres
    lambda_m = lambda_nm * 1e-9

    # Planck spectral radiance per unit wavelength in metres [W m^-2 m^-1 sr^-1]
    L_per_m = (2.0 * h * c**2 / lambda_m**5) / (np.exp(h * c / (lambda_m * kB * T_sun)) - 1.0)

    # Convert to per nanometre: 1 m = 1e9 nm, so multiply by 1e-9 m/nm
    L_per_nm = L_per_m * 1e-9

    return L_per_nm
