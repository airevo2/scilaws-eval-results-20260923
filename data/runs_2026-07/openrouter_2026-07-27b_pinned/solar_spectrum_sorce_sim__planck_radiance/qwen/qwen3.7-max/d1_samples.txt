    lam_m = lam_nm * 1e-9
    
    # Planck spectral radiance in W m^-2 m^-1 sr^-1
    B_lam = (2.0 * h * c**2 / lam_m**5) / (np.exp(h * c / (lam_m * k_B * T)) - 1.0)
    
    # Convert to W m^-2 nm^-1 sr^-1
    L = B_lam * 1e-9
    
    return L

import numpy as np

def predict(X):
    lam = X[:, 0]
    
    # Physical constants (SI)
    h = 6.62607015e-34      # Planck constant [J s]
    c = 299792458.0         # Speed of light [m/s]
    k_B = 1.380649e-23      # Boltzmann constant [J/K]
    
    # Standard solar effective temperature (IAU 2015 nominal)
    T = 5772.0              # [K]
    
    # First radiation constant for spectral radiance, scaled for nm
    # 2 * h * c^2 converted from W m^-2 m^-1 sr^-1 to W m^-2 nm^-1 sr^-1
    c1 = 2.0 * h * c**2 * 1e36
    
    # Second radiation constant, scaled for wavelength in nm
    c2 = h * c / (k_B * 1e-9)
    
    # Planck spectral radiance L(lambda, T)
    L = c1 / (lam**5 * (np.exp(c2 / (lam * T)) - 1.0))
    
    return L
