import numpy as np

def predict(X):
    """
    Predict summertime thermocline depth z_t (m) from lake surface area and maximum depth.

    Parameters
    ----------
    X : np.ndarray, shape (n, 2)
        X[:, 0] = A_s_km2 : lake surface area (km^2)
        X[:, 1] = Z_max_m : lake maximum depth (m)

    Returns
    -------
    z_t_m : np.ndarray, shape (n,)
        Predicted thermocline depth (m)
    """
    A_s = X[:, 0]
    Z_max = X[:, 1]
    # Classic empirical rule: z_t ≈ 2 * sqrt(area in km^2), capped by maximum depth.
    z_t = np.minimum(2.0 * np.sqrt(A_s), Z_max)
    return z_t

import numpy as np

def predict(X):
    """
    Predict summertime thermocline depth (z_t) in meters.

    Uses the standard empirical relationship from Fee et al. (1996),
    where thermocline depth (defined as the depth of maximum temperature
    gradient) scales with the square root of lake area:
        z_t = 2.0 * sqrt(A_ha)
    with A in hectares. The input area is in km^2 (1 km^2 = 100 ha).
    The prediction is capped by the lake maximum depth (Z_max) as a
    physical upper bound.
    """
    A_km2 = X[:, 0]
    Z_max = X[:, 1]
    
    # Convert area from km^2 to hectares
    A_ha = 100.0 * A_km2
    
    # Fee et al. (1996) thermocline depth
    z_t_unbounded = 2.0 * np.sqrt(A_ha)
    
    # Apply maximum depth as upper bound
    z_t = np.minimum(z_t_unbounded, Z_max)
    
    return z_t

    and maximum depth (Z_max_m) using the widely cited empirical relationship
    from Gorham & Boyce (1989) Limnology & Oceanography 34: 301-311.

    z_t = 0.3388 * A_s^{0.24} * Z_max^{0.55}
    where A_s is in km², Z_max in m, and z_t in m.
    """
    A_s = X[:, 0]
    Z_max = X[:, 1]
    z_t = 0.3388 * (A_s ** 0.24) * (Z_max ** 0.55)
    return z_t

import numpy as np

def predict(X):
    """
    Predict summertime thermocline depth z_t (m) from lake area and maximum depth.

    Parameters
    ----------
    X : np.ndarray, shape (n, 2)
        X[:,0] = A_s (km^2) : lake surface area
        X[:,1] = Z_max (m)  : lake maximum depth

    Returns
    -------
    z_t : np.ndarray, shape (n,)
        Predicted thermocline depth (m), capped by Z_max.
    """
    A_s = X[:, 0]
    Z_max = X[:, 1]

    # Fee et al. (1996) empirical relationship:
    # log10(z_t) = 0.336 + 0.283 * log10(A_s)
    # => z_t = 10^0.336 * A_s^0.283
    # 10^0.336 ≈ 2.168
    z_t = 2.168 * (A_s ** 0.283)

    # Physical upper bound: thermocline cannot be deeper than the maximum depth
    z_t = np.minimum(z_t, Z_max)

    return z_t

import numpy as np

def predict(X):
    """
    Predict summertime thermocline depth (z_t) [m] from lake surface area
    A_s [km^2] and maximum depth Z_max [m] using the widely cited empirical
    relationship of Gorham and Boyce (1989):
        log10(z_t) = 0.54 + 0.24 * log10(A_s) + 0.32 * log10(Z_max)
    """
    A_s = X[:, 0]   # lake surface area, km^2
    Z_max = X[:, 1] # lake maximum depth, m

    intercept = 0.54
    coeff_A = 0.24
    coeff_Z = 0.32

    log10_z_t = intercept + coeff_A * np.log10(A_s) + coeff_Z * np.log10(Z_max)
    z_t = np.power(10.0, log10_z_t)
    return z_t
