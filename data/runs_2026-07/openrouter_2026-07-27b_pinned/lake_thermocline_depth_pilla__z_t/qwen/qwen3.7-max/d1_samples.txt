import numpy as np

def predict(X):
    A_s = X[:, 0]
    Z_max = X[:, 1]
    
    # Empirical power-law relationship for summertime thermocline depth.
    # Thermocline depth scales primarily with wind fetch (proportional to A_s^0.25)
    # and is secondarily influenced by maximum depth.
    z_t = 4.0 * (A_s ** 0.25) * (Z_max ** 0.10)
    
    # Physical constraint: thermocline cannot be deeper than the lake's maximum depth
    z_t = np.minimum(z_t, Z_max)
    
    # The source dataset aggregates on a 0.5 m grid due to interpolation spacing
    z_t = np.round(z_t * 2.0) / 2.0
    
    return z_t

    z_t = np.minimum(z_t, Z_max)
