import numpy as np

def predict(X):
    A = np.asarray(X[:, 0], dtype=float)   # lake area [km^2]
    Zmax = np.asarray(X[:, 1], dtype=float)  # max depth [m]

    # Canonical wind-fetch / area based thermocline-depth relation
    # (power law in lake surface area, e.g. Gorham & Boyce type scaling)
    z_t = 3.7 * np.power(np.maximum(A, 1e-6), 0.185)

    # physical constraint: thermocline cannot exceed lake bottom
    z_t = np.minimum(z_t, Zmax)

    return z_t

    z_t = np.minimum(z_t, 0.9 * Z_max)
