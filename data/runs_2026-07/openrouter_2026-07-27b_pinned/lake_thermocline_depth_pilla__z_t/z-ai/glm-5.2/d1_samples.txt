import numpy as np

def predict(X):
    A_s = X[:, 0]  # Lake surface area in km^2
    Z_max = X[:, 1]  # Lake maximum depth in m
    
    # Gorham & Boyce (1989) canonical empirical scaling for thermocline depth
    # z_t ∝ A_s^0.5
    z_t = 2.82 * np.sqrt(A_s)
    
    # Z_max provides a physical upper bound on thermocline depth
    z_t = np.minimum(z_t, Z_max)
    
    return z_t

import numpy as np

def predict(X):
    A_s = X[:, 0]
    Z_max = X[:, 1]
    
    # Canonical multiple regression power-law form for thermocline depth 
    # (e.g., Gorham & Boyce, 1989) using lake surface area and maximum depth.
    # log10(z_t) = c0 + c1*log10(A_s) + c2*log10(Z_max)
    c0 = -0.21
    c1 = 0.42
    c2 = 0.32
    
    z_t = 10**(c0 + c1 * np.log10(A_s) + c2 * np.log10(Z_max))
    
    # Enforce the physical upper bound provided by maximum depth
    z_t = np.minimum(z_t, Z_max)
    
    return z_t
