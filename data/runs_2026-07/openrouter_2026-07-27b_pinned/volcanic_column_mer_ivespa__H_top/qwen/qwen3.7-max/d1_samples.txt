import numpy as np

def predict(X):
    """
    Predict time-averaged eruption-column top height (km above vent)
    from mass eruption rate using the canonical buoyant-plume 1/4-power scaling.
    
    H_top = k * MER^(1/4)
    
    Based on Morton, Taylor & Turner (1956) plume theory applied to
    volcanic columns (Sparks 1986; Wilson et al. 1978; Mastin et al. 2009).
    """
    MER = X[:, 0]  # Mass eruption rate [kg/s]
    
    # Canonical buoyant-plume scaling coefficient
    # k ~ 0.3 km / (kg/s)^0.25 for standard tropical/mid-latitude atmosphere
    k = 0.3
    n = 0.25  # 1/4-power from buoyant plume theory
    
    H_top = k * MER ** n
    return H_top

import numpy as np

def predict(X):
    """
    Predicts time-averaged eruption-column top height (H_top) in km above vent level
    from the mass eruption rate (MER) in kg/s using the standard buoyant-plume 
    scaling law (H ~ MER^0.25).
    
    Parameters:
    X : 2-D numpy array
        X[:, 0] = MER_kg_per_s (Mass Eruption Rate in kg/s)
        
    Returns:
    H_top : 1-D numpy array
        Plume top height in km above vent level.
    """
    MER = X[:, 0]
    
    # Buoyant plume scaling exponent is theoretically 1/4 (0.25).
    # The empirical coefficient ~0.284 aligns the theoretical scaling 
    # with the observed global bounds (0.58 km to 36 km).
    coeff = 0.284
    exponent = 0.25
    
    H_top = coeff * (MER ** exponent)
    
    return H_top
