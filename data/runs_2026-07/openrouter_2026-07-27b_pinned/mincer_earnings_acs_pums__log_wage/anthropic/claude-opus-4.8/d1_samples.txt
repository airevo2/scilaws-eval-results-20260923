import numpy as np

def predict(X):
    # Mincer earnings equation (standard closed-form)
    # ln(Y) = beta0 + beta1 * schooling + beta2 * experience + beta3 * experience^2
    s = X[:, 0]
    x = X[:, 1]
    
    beta0 = 6.0     # intercept (log baseline wage)
    beta1 = 0.10    # return to a year of schooling (~10%)
    beta2 = 0.05    # return to experience (linear term)
    beta3 = -0.0008 # concavity of experience profile
    
    log_wage = beta0 + beta1 * s + beta2 * x + beta3 * x**2
    return log_wage
