import numpy as np

def predict(X):
    """
    Predict log wage (ln Y) using the standard Mincer earnings equation.
    
    Parameters
    ----------
    X : np.ndarray, shape (n, 2)
        X[:, 0] : years of completed schooling (s)
        X[:, 1] : potential years of work experience (x)
    
    Returns
    -------
    ln_wage : np.ndarray, shape (n,)
        Natural logarithm of annual wage and salary income.
    """
    s = X[:, 0]   # schooling
    x = X[:, 1]   # experience

    # Canonical Mincer coefficients (typical values from labor economics)
    beta_0 = 0.5       # intercept
    beta_1 = 0.10      # return to schooling
    beta_2 = 0.05      # return to experience (linear)
    beta_3 = -0.001    # return to experience (quadratic)

    ln_wage = beta_0 + beta_1 * s + beta_2 * x + beta_3 * (x ** 2)
    return ln_wage
