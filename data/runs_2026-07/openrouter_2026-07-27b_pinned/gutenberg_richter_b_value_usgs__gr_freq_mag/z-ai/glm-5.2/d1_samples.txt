import numpy as np

def predict(X):
    """
    Calculates the cumulative annual earthquake rate (log10 N) for events 
    with moment magnitude at or above the threshold M, using the 
    Gutenberg-Richter law: log10(N) = a - b*M.
    
    Canonical global average coefficients are used:
    a = 8.0 (annual rate of M >= 0 events)
    b = 1.0 (slope of the magnitude-frequency distribution)
    """
    M = X[:, 0]
    a = 8.0
    b = 1.0
    return a - b * M
