import numpy as np

def predict(X):
    """
    Predict OD600 for E. coli K-12 BW25113 using the modified Gompertz model.

    Parameters
    ----------
    X : np.ndarray, shape (n, 1)
        X[:, 0] = time_h (hours since inoculation)

    Returns
    -------
    od600 : np.ndarray, shape (n,)
        Predicted optical density at 600 nm (dimensionless, background-subtracted)
    """
    t = X[:, 0]

    # Typical growth parameters for E. coli K-12 BW25113 in a 96-well plate at 37°C
    A = 0.85         # maximum OD600 (asymptote)
    mu_max = 0.75    # maximum specific growth rate (1/h)
    lag = 1.8        # lag phase duration (h)

    # Modified Gompertz equation (Zwietering et al., Appl. Environ. Microbiol. 1990)
    # y = A * exp{-exp[ (mu_max * e / A) * (lag - t) + 1 ]}
    inner = (mu_max * np.e / A) * (lag - t) + 1.0
    # Prevent overflow in exp(inner) for extreme t values
    inner = np.clip(inner, -100, 100)
    od600 = A * np.exp(-np.exp(inner))
    return od600

import numpy as np

def predict(X):
    # Extract time in hours
    t = X[:, 0]
    
    # Parameters for E. coli K-12 BW25113 in a microplate well
    # Lower asymptote (background-subtracted)
    A = 0.0       # OD
    # Asymptotic increase (max OD - min OD)
    C = 0.8       # OD
    # Maximum specific growth rate
    mu_max = 0.8  # 1/h
    # Lag time (Zwietering definition)
    lam = 2.0     # h
    
    # Four-parameter logistic model (Zwietering et al., 1990)
    exponent = (4.0 * mu_max / C) * (lam - t) + 2.0
    od600 = A + C / (1.0 + np.exp(exponent))
    
    return od600

    lag = 1.0                 # lag phase duration (h)
    
    # closed-form logistic with lag
    od = A / (1.0 + np.exp((4.0 * mu_max / A) * (lag - t) + 2.0))
