import numpy as np

def predict(X):
    t = X[:, 0]
    
    # Standard logistic growth model for bacterial biomass (OD600)
    # Parameters are representative for E. coli K-12 BW25113 in typical 96-well microplate conditions
    OD_max = 1.5   # Maximum OD600 (carrying capacity / stationary phase)
    OD_0 = 0.02    # Initial OD600 (inoculation density)
    mu_max = 1.2   # Maximum specific growth rate (h^-1)
    
    # Logistic equation
    od600 = OD_max / (1.0 + ((OD_max - OD_0) / OD_0) * np.exp(-mu_max * t))
    
    return od600

import numpy as np

def predict(X):
    """
    Predict OD600(t) for E. coli K-12 BW25113 using the modified Gompertz model
    (Zwietering et al., 1990, Appl. Environ. Microbiol. 56:1875-1881),
    the standard closed-form growth model in predictive microbiology.
    """
    t = X[:, 0]  # time in hours

    # Model parameters (typical for BW25113 in 96-well plate, 37 °C)
    OD_0   = 0.02   # initial (inoculum) OD600
    OD_max = 2.5    # carrying capacity / stationary-phase OD600
    mu_max = 0.8    # maximum specific growth rate  [h^-1]
    t_lag  = 3.0    # lag-phase duration             [h]

    # Asymptotic log-ratio
    A = np.log(OD_max / OD_0)

    # Modified Gompertz equation:
    #   ln(OD(t)/OD_0) = A * exp(-exp((mu_max * e / A) * (t_lag - t) + 1))
    OD = OD_0 * np.exp(
        A * np.exp(-np.exp((mu_max * np.e / A) * (t_lag - t) + 1.0))
    )

    return OD
