import numpy as np

def predict(X):
    lam = X[:, 0]  # wavelength in micrometres
    T = X[:, 1]    # temperature in degC

    # Cauchy dispersion for ethylene glycol at reference temperature (~20 C)
    # n(lambda) = A + B/lambda^2 + C/lambda^4
    A = 1.4306
    B = 0.00460
    C = 0.00020

    n_ref = A + B / lam**2 + C / lam**4

    # Thermo-optic correction: dn/dT ~ -2.6e-4 per degC, referenced to 20 C
    dn_dT = -2.6e-4
    T_ref = 20.0

    n = n_ref + dn_dT * (T - T_ref)
    return n
