    """

    year = X[:, 1]
    
    # Coefficients representing the Gompertz-derived sex differential in mortality

    c2 = -0.002   # Time trend in the intercept
    c3 = 0.00005  # Time trend in the age slope
    
    ln_alpha = c0 + c1 * age + c2 * year + c3 * age * year
    return ln_alpha

def predict(X):
    """
    Glei-Horiuchi (2007) Gompertz-derived model for the natural log of the
    male/female central death rate ratio (ln α = ln(m_male / m_female)).

    Under the Gompertz law m(x) = A * exp(B * x), the log sex ratio is
    linear in age:
        ln α = ln(A_m/A_f) + (B_m - B_f) * x.
    Allowing the log-ratio intercept and the Gompertz slope difference to
    vary linearly with calendar year yields the closed-form bilinear model:
        ln α = a0 + a1*(t - t0) + [b0 + b1*(t - t0)] * x.
    The constants below are illustrative typical values for Western
    populations; replace with fitted coefficients as needed.
    """
    age = X[:, 0]
    year = X[:, 1]

    # Reference year (centering improves interpretability)
    t0 = 1900.0

    # Illustrative coefficients
    a0 = 0.42       # log-ratio intercept at t0
    a1 = 0.0088     # annual change in the intercept
    b0 = -0.004     # Gompertz slope difference (B_m - B_f) at t0
    b1 = -0.00006   # annual change in the slope difference

    dt = year - t0
    return a0 + a1 * dt + (b0 + b1 * dt) * age

import numpy as np

def predict(X):
    """
    Predicts the natural log of the ratio of male to female central death rates (ln α).
    
    Under the Gompertz mortality model, m(x) = A * exp(B * x). 
    The ratio of male to female mortality is:
    α(x) = (A_m / A_f) * exp((B_m - B_f) * x)
    
    Taking the natural log yields:
    ln α(x) = ln(A_m / A_f) + (B_m - B_f) * x
    
    Following Glei and Horiuchi (2007), this relationship is linear in age (x). 
    To account for temporal changes in the Gompertz parameters over calendar years (t), 
    the intercept and slope are modeled as linear functions of time.
    """
    age = X[:, 0]
    year = X[:, 1]
    
    # Coefficients for the time-varying intercept a(t) = a0 + a1 * t
    a0 = 50.0
    a1 = -0.025
    
    # Coefficients for the time-varying slope b(t) = b0 + b1 * t
    b0 = 0.05
    b1 = -0.00003
    
    # Time-varying parameters
    a_t = a0 + a1 * year
    b_t = b0 + b1 * year
    
    # ln(alpha) = a(t) + b(t) * x
    ln_alpha = a_t + b_t * age
    
    return ln_alpha
