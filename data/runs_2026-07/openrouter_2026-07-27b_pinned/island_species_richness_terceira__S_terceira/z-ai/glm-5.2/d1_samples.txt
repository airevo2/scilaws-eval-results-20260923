    """

    b3 = -0.0002  # Slight negative effect of excessive precipitation
    b4 = 0.15     # Positive effect of temperature
    b5 = 0.02     # Positive effect of slope (microhabitat heterogeneity)

    log_S = b0 + b1 * d + b2 * h + b3 * p + b4 * t + b5 * sl
    
    # Return expected species richness

    where:
      D  = anthropogenic habitat-modification index
      H  = altitude (m)
      P  = mean annual precipitation (mm/yr)
      T  = mean annual temperature (degC)
      Sl = terrain slope (degrees)

    b4 = 0.08     # temperature effect

    eta = (b0
           + b1 * X[:, 0]
           + b2 * X[:, 1]

import numpy as np

def predict(X):
    """
    Predicts arthropod species richness (S) using a standard log-linear 
    (Poisson regression) closed-form relationship, which is the canonical 
    model for count data like species richness.
    
    Includes quadratic terms for disturbance (d) and altitude (h) to account 
    for the Intermediate Disturbance Hypothesis and the mid-domain effect 
    (mid-elevation peak in species richness), respectively.
    """
    d = X[:, 0]
    h = X[:, 1]
    p = X[:, 2]
    t = X[:, 3]
    sl = X[:, 4]
    
    # Named constants for the fitted coefficients
    beta_0 = 1.5       # Intercept
    beta_d1 = 0.02     # Linear effect of disturbance
    beta_d2 = -0.0002  # Quadratic effect of disturbance (IDH)
    beta_h1 = 0.002    # Linear effect of altitude
    beta_h2 = -0.0000015 # Quadratic effect of altitude (mid-domain effect)
    beta_p = 0.0001    # Effect of precipitation
    beta_t = 0.05      # Effect of temperature
    beta_sl = 0.01     # Effect of slope
    
    # Log-link function to ensure strictly positive richness
    log_S = (beta_0 + 
             beta_d1 * d + beta_d2 * d**2 + 
             beta_h1 * h + beta_h2 * h**2 + 
             beta_p * p + 
             beta_t * t + 
             beta_sl * sl)
             
    S = np.exp(log_S)
    
    return S
