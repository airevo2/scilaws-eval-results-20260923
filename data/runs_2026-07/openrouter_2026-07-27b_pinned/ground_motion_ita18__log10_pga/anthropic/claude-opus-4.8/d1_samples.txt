import numpy as np

def predict(X):
    """
    log10(PGA) [log10(cm/s^2)], RotD50 horizontal PGA.
    Canonical form: Bindi et al. (2011) ITA10 GMPE for Italy.

        log10(Y) = e1 + FD(R,M) + FM(M) + FS + FSOF

    with
        FD = [c1 + c2*(M-Mref)] * log10( sqrt(Rjb^2 + h^2) )
             - c3 * sqrt(Rjb^2 + h^2)
        FM = b1*(M-Mh) + b2*(M-Mh)^2            for M <= Mh
           = b3*(M-Mh)                          for M >  Mh   (b3 = 0)
        FS = site term (EC8 classes)
        FSOF = style-of-faulting term
    """
    X = np.asarray(X, dtype=float)

    Mw  = X[:, 0]
    R   = X[:, 1]
    # X[:, 2] hypocentral depth not used explicitly (pseudo-depth h fitted)
    C_A = X[:, 3]
    C_B = X[:, 4]
    C_C = X[:, 5]
    C_D = X[:, 6]
    C_E = X[:, 7]
    E_N = X[:, 8]
    E_R = X[:, 9]
    E_SS = X[:, 10]

    # --- Coefficients for PGA (cm/s^2), Bindi et al. (2011) ---
    e1  = 3.6720
    c1  = -1.9400
    c2  = 0.4130
    c3  = 0.000134
    h   = 10.322
    b1  = -0.2620
    b2  = -0.0707
    b3  = 0.0

    # Site terms (EC8)
    sA, sB, sC, sD, sE = 0.0, 0.1620, 0.2400, 0.1050, 0.5700
    # Style-of-faulting terms
    fN, fR, fSS = -0.0503, 0.1050, 0.0

    Mref = 5.0
    Mh   = 6.75

    Rh = np.sqrt(R**2 + h**2)

    # Distance function
    FD = (c1 + c2 * (Mw - Mref)) * np.log10(Rh) - c3 * Rh

    # Magnitude function
    FM = np.where(Mw <= Mh,
                  b1 * (Mw - Mh) + b2 * (Mw - Mh)**2,
                  b3 * (Mw - Mh))

    # Site function
    FS = sA * C_A + sB * C_B + sC * C_C + sD * C_D + sE * C_E

    # Style-of-faulting function
    FSOF = fN * E_N + fR * E_R + fSS * E_SS

    return e1 + FD + FM + FS + FSOF
