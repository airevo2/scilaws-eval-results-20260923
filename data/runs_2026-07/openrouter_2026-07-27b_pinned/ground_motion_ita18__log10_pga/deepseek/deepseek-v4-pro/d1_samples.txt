    Mh = 5.5
    a = 3.074
    b1 = 0.511
    b2 = -0.059
    c1 = -1.051
    c2 = 0.192
    e1 = -0.0084

    # Site class coefficients (EC8 class A is reference)
    site_coeff = np.array([0.0, 0.060, 0.146, 0.237, 0.102])  # A, B, C, D, E

    # Style-of-faulting coefficients (strike-slip is reference)
    f_N = -0.051   # Normal

    R = X[:, 1]
    h = X[:, 2]

    style = X[:, 8] * f_N + X[:, 9] * f_R  # SS contributes 0

    R_hypo = np.sqrt(R**2 + h**2)

    # Predicted log10 PGA
    log10_pga = (a
                 + b1 * (M - Mh)
                 + b2 * (M - Mh)**2
                 + (c1 + c2 * (M - Mh)) * np.log10(R_hypo)
                 + e1 * h
                 + site
                 + style)

    return log10_pga

    ec8_D = X[:, 6]
    ec8_E = X[:, 7]

import numpy as np

def predict(X):
    """
    Standard closed-form relationship for log10(PGA) (RotD50, cm/s^2) commonly used in 
    shallow crustal ground-motion prediction equations (GMPEs) in active tectonic regions.
    
    Input X columns:
    0: Mw         - Moment magnitude
    1: R_km       - Joyner-Boore distance (km)
    2: depth_km   - Hypocentral depth (km)
    3: ec8_A      - EC8 site class A (rock, Vs30 > 800 m/s) – reference class
    4: ec8_B      - EC8 site class B (stiff soil, Vs30 360–800 m/s)
    5: ec8_C      - EC8 site class C (medium soil, Vs30 180–360 m/s)
    6: ec8_D      - EC8 site class D (soft soil, Vs30 < 180 m/s)
    7: ec8_E      - EC8 site class E (thin alluvium over rock)
    8: sof_NF     - Normal faulting
    9: sof_TF     - Reverse (Thrust) faulting
    10: sof_SS    - Strike-slip faulting (reference style)
    
    Functional form:
        log10(PGA) = beta0 + beta1*M + beta2*M^2 + 
                     (beta3 + beta4*M) * log10( sqrt(R^2 + h^2) ) +
                     gamma_B*C_B + gamma_C*C_C + gamma_D*C_D + gamma_E*C_E +
                     delta_N*F_N + delta_R*F_R
    where h is the hypocentral depth, acting as the near-source saturation distance.
    """
    # Extract variables
    M = X[:, 0]
    R = X[:, 1]
    h = X[:, 2]
    C_B = X[:, 4]
    C_C = X[:, 5]
    C_D = X[:, 6]
    C_E = X[:, 7]
    F_N = X[:, 8]
    F_R = X[:, 9]

    # Typical coefficients for illustration (e.g., derived from European strong-motion datasets)
    # Intercept (for Mw=0, reference rock A, strike-slip) – adjust as fitted
    beta0 =  1.500
    beta1 =  0.450
    beta2 = -0.010
    beta3 = -1.250
    beta4 =  0.120
    # Site amplification relative to rock A (ec8_A = 0)
    gamma_B = 0.20
    gamma_C = 0.32
    gamma_D = 0.45
    gamma_E = 0.55
    # Faulting style adjustments relative to strike-slip
    delta_N = -0.04   # normal faulting slightly lower motions
    delta_R =  0.05   # reverse faulting slightly higher motions

    # Near-source saturation distance (hypocentral depth used as pseudo-depth)
    R_sat = np.sqrt(R**2 + h**2)

    # Compute log10 PGA
    log10_pga = (beta0 
                 + beta1 * M 
                 + beta2 * M**2 
                 + (beta3 + beta4 * M) * np.log10(R_sat)
                 + gamma_B * C_B 
                 + gamma_C * C_C 
                 + gamma_D * C_D 
                 + gamma_E * C_E
                 + delta_N * F_N 
                 + delta_R * F_R)

    return log10_pga
