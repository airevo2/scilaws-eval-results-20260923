import numpy as np

def predict(X):
    H = X[:, 0]
    # Anchor points derived from the complete regime bounds and counts
    H_ref = 12.0625
    N_ref = 2.0
    alpha = np.log10(1097.0 / N_ref) / (18.0 - H_ref)
    
    # Standard exponential cumulative size-frequency distribution N(<H)
    N_cum = N_ref * np.power(10.0, alpha * (H - H_ref))
    return N_cum
