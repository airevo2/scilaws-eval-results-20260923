import numpy as np

def predict(X):
    H = X[:, 0]
    # Standard cumulative NEO size-frequency distribution:
    #   log10 N(<H) = alpha * H + beta
    # i.e. N(<H) = 10^(alpha*H + beta), a power law in H
    # (equivalently N(<H) ~ D^(-b) with H = -5*log10(D) + const, alpha = b/5).
    # Coefficients fitted to the complete regime H in [12, 18]:
    #   N(12) ~ 2,  N(18) ~ 1097  =>  alpha ~ 0.4565
    alpha = 0.4565
    beta = -5.177
    logN = alpha * H + beta
    N = np.power(10.0, logN)
    return N

import numpy as np

def predict(X):
    H = X[:, 0]
    # Standard cumulative exponential Size-Frequency Distribution (SFD) for NEOs
    # N(<H) = N_0 * 10^(a * (H - H_0))
    # Calibrated to the complete regime endpoints: N(18) ≈ 1097, N(12) ≈ 2
    N_0 = 1097.0
    a = 0.4567
    H_0 = 18.0
    
    return N_0 * np.power(10.0, a * (H - H_0))
