import numpy as np

def predict(X):
    # Morris et al. (1971) linear prediction equation for FEV1 (litres)
    # FEV1 = a * height_cm + b * age + c
    t = X[:, 0]   # age in years
    h = X[:, 1]   # standing height in cm
    a = 0.0414    # per cm
    b = -0.0244   # per year
    c = -2.190    # intercept
    fev1 = a * h + b * t + c
    return fev1

import numpy as np

def predict(X):
    # ECSC / Quanjer (1993) reference equation for FEV1 (adult males)
    # FEV1 [L] = 4.30 * height[m] - 0.029 * age[yr] - 2.49
    age = X[:, 0]
    height_m = X[:, 1] / 100.0
    fev1 = 4.30 * height_m - 0.029 * age - 2.49
    return fev1
