import numpy as np

def predict(X):
    """
    Geiger-Nuttall law for alpha-decay half-life.
    
    log10(T_1/2 / s) = a / sqrt(Q_alpha) + b
    
    This is the canonical empirical relationship between the alpha-decay
    partial half-life and the Q-value, originally observed by Geiger and
    Nuttall (1911) and later explained by Gamow's quantum tunneling theory.
    
    Coefficients are chosen as a universal average over heavy alpha-emitting
    nuclei (Z ~ 84-100), consistent with the Viola-Seaborg systematics
    evaluated at a representative atomic number.
    """
    Q = X[:, 0]  # Q_alpha in MeV
    a = 145.0
    b = -53.0
    return a / np.sqrt(Q) + b

import numpy as np

def predict(X):
    Q_alpha = X[:, 0]
    
    # Geiger-Nuttall law: log10(T_1/2) = A + B / sqrt(Q_alpha)
    # Using approximate global fit coefficients for alpha emitters
    A = -54.0
    B = 145.0
    
    return A + B / np.sqrt(Q_alpha)
