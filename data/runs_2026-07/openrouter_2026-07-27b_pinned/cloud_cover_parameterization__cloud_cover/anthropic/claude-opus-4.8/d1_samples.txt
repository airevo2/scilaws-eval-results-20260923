import numpy as np

def predict(X):
    q_c = X[:, 0]
    q_i = X[:, 1]
    RH = X[:, 2]

    # Sundqvist (1989) diagnostic cloud fraction scheme
    # C = 1 - sqrt( (1 - RH) / (1 - RH_crit) )   for RH > RH_crit
    RH_crit = 0.7  # critical relative humidity threshold

    RH_c = np.clip(RH, 0.0, 1.0)
    denom = np.maximum(1.0 - RH_crit, 1e-6)
    frac = np.clip((1.0 - RH_c) / denom, 0.0, 1.0)
    C = 1.0 - np.sqrt(frac)
    C = np.where(RH_c > RH_crit, C, 0.0)

    # Ensure saturated / condensate-containing grid boxes are (partly) cloudy
    has_condensate = (q_c + q_i) > 1e-8
    C = np.where(has_condensate, np.maximum(C, 1e-2), C)

    return np.clip(C, 0.0, 1.0)

import numpy as np

def predict(X):
    """
    Diagnostic sub-grid cloud fraction using the Sundqvist (1989) relative-humidity
    scheme, with saturation forced where cloud condensate already exists.

        C = 1 - sqrt( (1 - RH) / (1 - RH_crit) )     for RH_crit < RH < 1
        C = 0                                        for RH <= RH_crit
        C = 1                                        for RH >= 1  or  condensate present

    This is the most commonly cited closed-form relationship for coarse-grained
    cloud cover diagnosis.
    """
    q_c = X[:, 0]
    q_i = X[:, 1]
    RH  = X[:, 2]

    # Critical relative humidity threshold for cloud onset (typical large-scale value)
    RH_crit = 0.8

    # Clip RH into a sensible range
    RH_c = np.clip(RH, 0.0, 1.0)

    # Sundqvist relative-humidity cloud fraction
    denom = max(1.0 - RH_crit, 1e-6)
    arg = np.clip((1.0 - RH_c) / denom, 0.0, 1.0)
    C = 1.0 - np.sqrt(arg)

    # No cloud below critical humidity
    C = np.where(RH_c <= RH_crit, 0.0, C)

    # Where condensate is present, ensure fully/partly cloudy conditions
    condensate = q_c + q_i
    C = np.where(condensate > 1e-8, np.maximum(C, 1.0), C)

    # Bound to physical [0, 1] range
    C = np.clip(C, 0.0, 1.0)

    return C

    C = 1 - sqrt( (1 - RH) / (1 - RH_crit) )   for RH_crit < RH < 1

    which no sub-grid cloud forms. Grid boxes that already contain
    condensate (q_c + q_i > 0) are treated as fully/partly cloudy.
    """

    q_i = X[:, 1]
    RH  = X[:, 2]

    RH_crit = 0.8

    C = np.where(has_cond, np.maximum(C, 1.0 - np.sqrt(arg)), C)

import numpy as np

def predict(X):
    """
    Standard Sundqvist-type diagnostic cloud fraction scheme.

    Cloud cover is diagnosed from relative humidity relative to a
    critical threshold RH_crit (below which no cloud forms):

        C = 1 - sqrt( (1 - RH) / (1 - RH_crit) )   for RH_crit < RH < 1
        C = 0                                       for RH <= RH_crit
        C = 1                                       for RH >= 1

    In addition, if condensate (liquid + ice) is present, the grid box
    is treated as (at least partially) cloudy.
    """
    q_c = X[:, 0]
    q_i = X[:, 1]
    RH  = X[:, 2]

    # critical relative humidity threshold (commonly cited value ~0.6-0.8)
    RH_crit = 0.7

    # Sundqvist (1989) diagnostic cloud fraction
    denom = np.maximum(1.0 - RH_crit, 1e-6)
    arg = np.clip((1.0 - RH) / denom, 0.0, 1.0)
    C = 1.0 - np.sqrt(arg)

    # No cloud below the critical humidity
    C = np.where(RH <= RH_crit, 0.0, C)

    # Saturated / supersaturated -> fully cloudy
    C = np.where(RH >= 1.0, 1.0, C)

    # Ensure any existing condensate implies cloud presence
    q_cond = q_c + q_i
    C = np.where(q_cond > 1e-8, np.maximum(C, 1.0 - np.exp(-1.0e4 * q_cond)), C)

    # clip to physical bounds
    return np.clip(C, 0.0, 1.0)
