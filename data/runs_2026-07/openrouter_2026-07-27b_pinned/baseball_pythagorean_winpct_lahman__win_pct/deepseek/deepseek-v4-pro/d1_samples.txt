import numpy as np

def predict(X):
    """
    Predict season win percentage using the Pythagorean expectation (exponent = 2).

    Parameters
    ----------
    X : np.ndarray
        2-D array with shape (n_samples, 2).
        X[:, 0] = RS (runs scored)
        X[:, 1] = RA (runs allowed)

    Returns
    -------
    np.ndarray
        1-D array of predicted win percentages W / (W + L).
    """
    RS = X[:, 0]
    RA = X[:, 1]
    # Pythagorean win percentage with exponent 2
    win_pct = RS**2 / (RS**2 + RA**2)
    return win_pct
