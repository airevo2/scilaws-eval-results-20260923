def predict(X):
    """
    Predict back-calculated total length (mm) at age using the standard
    von Bertalanffy growth function (VBGF). This is the canonical closed-form
    relationship for fish length-at-age, often used as the underlying growth
    model even when the data are back-calculated lengths (Li_sp_m).

    Parameters
    ----------
    X : np.ndarray, shape (n_samples, 1)
        X[:, 0] = Agei (years since hatching)

    Returns
    -------
    np.ndarray, shape (n_samples,)
        Predicted Li_sp_m (mm)
    """
    # VBGF parameters (typical literature values; replace with fitted ones)
    L_inf = 600.0   # asymptotic length (mm)
    K = 0.15        # Brody growth coefficient (yr^-1)
    t0 = -0.5       # theoretical age at zero length (yr)

    age = X[:, 0]
    return L_inf * (1.0 - np.exp(-K * (age - t0)))
