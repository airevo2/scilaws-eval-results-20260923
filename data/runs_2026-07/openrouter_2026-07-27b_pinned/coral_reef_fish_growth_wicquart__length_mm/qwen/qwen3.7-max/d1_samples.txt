import numpy as np

def predict(X):
    """
    Back-calculated total length (mm) at age using the standard
    von Bertalanffy Growth Function (VBGF), the canonical closed-form
    model fit to back-calculated length-at-age data from otolith increments.
    
    L(t) = L_inf * (1 - exp(-k * (t - t0)))
    
    Parameters chosen as representative generic values for a marine fish species.
    """
    Agei = X[:, 0]
    
    # VBGF parameters (generic representative values)
    L_inf = 550.0   # asymptotic total length (mm)
    k = 0.25        # Brody growth-rate coefficient (yr^-1)
    t0 = -0.5       # theoretical age at zero length (yr)
    
    Li_sp_m = L_inf * (1.0 - np.exp(-k * (Agei - t0)))
    
    return Li_sp_m
