import numpy as np

def predict(X):
    # Zanna & Bolton (2020) equation-discovery closure for the subgrid
    # eddy momentum forcing (divergence of the deviatoric eddy stress tensor).
    #
    # Stress tensor (with D = shearing, D̃ = stretching deformation):
    #   S_x = kappa * [ d/dx( 1/2 * zeta^2 - zeta*D ) + d/dy( zeta*D̃ ) ]
    #       = kappa * ( 1/2 * d(zeta^2)/dx - d(zeta*D)/dx + d(zeta*D̃)/dy )
    #
    # kappa is a (negative) eddy amplitude with units of m^2 (backscatter).
    kappa = -4.87e8  # [m^2]

    d_zeta2_dx      = X[:, 5]
    d_zetaD_dx      = X[:, 6]
    d_zetaDtilde_dy = X[:, 7]

    Sx = kappa * (0.5 * d_zeta2_dx - d_zetaD_dx + d_zetaDtilde_dy)
    return Sx

                      [ zeta*Dtilde        ,   1/2 zeta^2 - zeta*D ] ]

    kappa = -4.87e4

    return Sx
