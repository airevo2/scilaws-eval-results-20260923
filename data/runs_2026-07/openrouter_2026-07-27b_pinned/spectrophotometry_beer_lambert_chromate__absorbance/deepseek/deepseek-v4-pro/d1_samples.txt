import numpy as np

def predict(X):
    """
    Predict absorbance A using the Beer-Lambert law: A = (epsilon / M) * c * l.

    Parameters
    ----------
    X : np.ndarray, shape (n, 2)
        X[:, 0] = concentration c in g/L
        X[:, 1] = path length l in cm

    Returns
    -------
    A : np.ndarray, shape (n,)
        Predicted absorbance (dimensionless).
    """
    c = X[:, 0]   # mass concentration, g/L
    l = X[:, 1]   # path length, cm

    # Molar absorptivity of potassium dichromate at 350 nm (L mol^-1 cm^-1)
    epsilon = 3160.0
    # Molar mass of potassium dichromate (K2Cr2O7), g/mol
    M = 294.185

    # Beer-Lambert law: A = epsilon * (c / M) * l
    A = epsilon * c / M * l
    return A

import numpy as np

def predict(X):
    """
    Predict absorbance A using the Beer-Lambert law for potassium dichromate.

    Parameters
    ----------
    X : ndarray, shape (n_samples, 2)
        X[:, 0] = mass concentration c in g/L
        X[:, 1] = path length l in cm

    Returns
    -------
    A : ndarray, shape (n_samples,)
        Predicted absorbance (blank-subtracted).
    """
    # Molar mass of potassium dichromate (K2Cr2O7) in g/mol
    M = 294.185
    # Molar absorptivity at 350 nm in L mol^{-1} cm^{-1} (commonly cited value)
    epsilon = 3160.0

    c_mass = X[:, 0]   # g/L
    l = X[:, 1]        # cm

    # Beer-Lambert: A = epsilon * (c_mass / M) * l
    A = epsilon * (c_mass / M) * l
    return A
