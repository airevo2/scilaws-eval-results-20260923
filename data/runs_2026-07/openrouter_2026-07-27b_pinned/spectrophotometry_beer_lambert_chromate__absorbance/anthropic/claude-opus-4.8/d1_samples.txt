import numpy as np

def predict(X):
    # Beer-Lambert law: A = epsilon * c * l
    # For potassium dichromate, using specific absorptivity (a) in L/(g*cm)
    # A typical value near the 350 nm / 257 nm band; here we use a
    # commonly cited specific absorptivity around 10 L/(g*cm).
    a = 10.0  # specific absorptivity [L/(g*cm)]
    c = X[:, 0]  # concentration [g/L]
    l = X[:, 1]  # path length [cm]
    A = a * c * l
    return A
