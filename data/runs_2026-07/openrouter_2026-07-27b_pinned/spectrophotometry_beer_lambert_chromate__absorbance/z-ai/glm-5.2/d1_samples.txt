import numpy as np

def predict(X):
    """
    Predicts the absorbance A of a potassium dichromate solution 
    using the Beer-Lambert Law: A = epsilon * c * l
    
    Inputs:
    X[:, 0] = concentration (c) in g/L
    X[:, 1] = path_length (l) in cm
    
    Returns:
    1-D numpy array of absorbance values
    """
    c = X[:, 0]
    l = X[:, 1]
    
    # epsilon represents the specific absorptivity (L g^-1 cm^-1).
    # The exact value is highly dependent on the measurement wavelength.
    # We use 1.0 as a canonical proportionality constant for the general form.
    epsilon = 1.0  
    
    A = epsilon * c * l
    return A

def predict(X):

    # where a is the specific absorptivity (L/(g*cm)).
