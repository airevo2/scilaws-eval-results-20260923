import numpy as np

def predict(X):
    # Michaelis-Menten kinetics: v = Vmax * S / (Km + S)
    S = X[:, 0]
    Vmax = 100.0   # nmol g^-1 h^-1 (maximum reaction rate)
    Km = 150.0     # uM (half-saturation constant)
    v = Vmax * S / (Km + S)
    return v
