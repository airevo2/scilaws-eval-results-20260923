import numpy as np

def predict(X):
    """
    Michaelis-Menten enzyme kinetics:
        v = Vmax * S / (Km + S)
    
    Parameters (representative soil enzyme values):
        Vmax = 200.0 nmol g^-1 h^-1  (maximum activity rate)
        Km   = 150.0 uM              (half-saturation constant)
    """
    S = X[:, 0]          # substrate concentration [uM]
    Vmax = 200.0         # nmol g^-1 h^-1
    Km = 150.0           # uM
    v = Vmax * S / (Km + S)
    return v
