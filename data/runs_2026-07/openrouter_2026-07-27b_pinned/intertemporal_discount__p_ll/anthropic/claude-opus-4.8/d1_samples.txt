    return p_ll

import numpy as np

def predict(X):
    # Hyperbolic discounting with logistic (softmax) choice rule
    log_ratio = X[:, 0]
    t_ss = X[:, 1]
    t_ll = X[:, 2]
    V_ss = X[:, 4]
    V_ll = X[:, 5]

    # Fitted parameters (canonical values)
    k = 0.01      # hyperbolic discount rate (1/day)
    beta = 2.0    # inverse temperature / choice sensitivity

    # Discounted (present) values
    dv_ll = V_ll / (1.0 + k * t_ll)
    dv_ss = V_ss / (1.0 + k * t_ss)

    # Normalize difference by SS value to keep it scale-free
    diff = (dv_ll - dv_ss) / (dv_ss + 1e-8)

    # Logistic choice probability of picking larger-later
    p_ll = 1.0 / (1.0 + np.exp(-beta * diff))
    return p_ll
