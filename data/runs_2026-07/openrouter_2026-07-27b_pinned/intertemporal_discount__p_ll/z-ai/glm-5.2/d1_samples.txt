import numpy as np

def predict(X):
    """
    Standard hyperbolic discounting model with logistic choice.
    P(LL) = 1 / (1 + exp(-beta * (V_LL/(1+k*t_LL) - V_SS/(1+k*t_SS))))
    """
    # Discount rate (per day) and choice sensitivity
    k = 0.01
    beta = 1.0

    V_ss = X[:, 4]
    V_ll = X[:, 5]
    t_ss = X[:, 1]
    t_ll = X[:, 2]

    # Hyperbolically discounted values
    d_ss = V_ss / (1.0 + k * t_ss)
    d_ll = V_ll / (1.0 + k * t_ll)

    # Logistic choice probability
    p_ll = 1.0 / (1.0 + np.exp(-beta * (d_ll - d_ss)))
    p_ll = np.clip(p_ll, 0.0, 1.0)

    return p_ll

import numpy as np

def predict(X):
    """
    Standard exponential discounting model with Luce choice rule for intertemporal choice.
    
    Under exponential discounting, subjective values are:
        SV_SS = V_SS * exp(-r * t_SS)
        SV_LL = V_LL * exp(-r * t_LL)
    
    With a Luce (ratio) choice rule, the probability of choosing LL is:
        P(LL) = SV_LL / (SV_LL + SV_SS)
              = 1 / (1 + exp(-(ln(V_LL/V_SS) - r * (t_LL - t_SS))))
    
    This reduces to a logistic function of the log amount ratio minus the
    discount rate times the delay difference.
    """
    log_amount_ratio = X[:, 0]   # ln(V_LL / V_SS)
    delay_diff_days = X[:, 3]    # t_LL - t_SS

    # Typical exponential discount rate (per day)
    r = 0.001

    # Log-odds of choosing the larger-later option
    logit = log_amount_ratio - r * delay_diff_days

    # Logistic transform, clipped to avoid overflow
    logit = np.clip(logit, -500, 500)
    p_ll = 1.0 / (1.0 + np.exp(-logit))

    return p_ll

import numpy as np

def predict(X):
    """
    Standard hyperbolic discounting model with a softmax (logistic) choice rule.
    
    The subjective value of an option is V / (1 + k * t).
    The probability of choosing the larger-later option is:
    
        P(LL) = 1 / (1 + exp(-beta * (log(V_LL/V_SS) - log((1+k*t_LL)/(1+k*t_SS)))))
    
    This is the log-odds form of the discounted-utility difference, which is
    scale-invariant and commonly used in intertemporal choice research.
    """
    log_amount_ratio = X[:, 0]   # ln(V_LL / V_SS)
    t_SS = X[:, 1]               # delay of smaller-sooner option (days)
    t_LL = X[:, 2]               # delay of larger-later option (days)
    
    # Typical parameter values (may be replaced with fitted constants)
    k = 0.01      # discount rate parameter (per day)
    beta = 1.0    # inverse temperature / sensitivity
    
    # Log ratio of hyperbolic discount factors
    log_discount_ratio = np.log((1.0 + k * t_LL) / (1.0 + k * t_SS))
    
    # Log-odds of choosing LL
    logit = beta * (log_amount_ratio - log_discount_ratio)
    
    # Logistic transform
    p_ll = 1.0 / (1.0 + np.exp(-logit))
    
    # Ensure output is within [0, 1]
    p_ll = np.clip(p_ll, 0.0, 1.0)
    return p_ll

import numpy as np

def predict(X):
    """
    Standard hyperbolic discounting model combined with a logistic choice function.
    The choice probability is determined by the difference in the log of discounted values,
    which is equivalent to the log ratio of discounted values.
    """
    # Canonical parameters (can be replaced with fitted values)
    k = 0.01   # Discount rate parameter
    mu = 1.0   # Sensitivity (inverse temperature) parameter
    
    log_amount_ratio = X[:, 0]
    t_ss = X[:, 1]
    t_ll = X[:, 2]
    
    # Calculate the log ratio of the discount factors
    log_discount_ratio = np.log((1.0 + k * t_ll) / (1.0 + k * t_ss))
    
    # Log ratio of the discounted values (LL over SS)
    log_discounted_value_ratio = log_amount_ratio - log_discount_ratio
    
    # Logistic choice function mapping the log ratio to probability
    p_ll = 1.0 / (1.0 + np.exp(-mu * log_discounted_value_ratio))
    
    return p_ll

import numpy as np

def predict(X):
    # Extract inputs
    ss_time_days = X[:, 1]
    ll_time_days = X[:, 2]
    ss_value = X[:, 4]
    ll_value = X[:, 5]
    
    # Standard hyperbolic discounting parameters (Mazur, 1987)
    # k: discount rate parameter
    # beta: inverse temperature / sensitivity parameter for the logistic choice rule
    k = 0.01
    beta = 0.05
    
    # Calculate subjective values using hyperbolic discounting
    sv_ss = ss_value / (1.0 + k * ss_time_days)
    sv_ll = ll_value / (1.0 + k * ll_time_days)
    
    # Calculate probability of choosing the larger-later option using a softmax/logistic function
    diff = sv_ll - sv_ss
    p_ll = 1.0 / (1.0 + np.exp(-beta * diff))
    
    # Ensure probabilities are strictly within (0, 1) bounds
    p_ll = np.clip(p_ll, 1e-10, 1 - 1e-10)
    
    return p_ll
