import numpy as np

def predict(X):
    """
    Predict main-stem river length using Hack's Law (1957).
    
    Hack's Law is the canonical geomorphological relationship between
    main-stem length and drainage area:
        L = C * A^h
    
    Original (Hack, 1957): L(mi) = 1.4 * A(mi^2)^0.6
    Metric conversion:
        C_km = 1.4 * 1.60934 / (2.58999^0.6) ≈ 1.27
    """
    A = X[:, 0]  # upland_area_skm (total upstream drainage area, km^2)
    
    # Hack's Law parameters (metric)
    C = 1.27
    h = 0.6
    
    L = C * np.power(A, h)
    return L
