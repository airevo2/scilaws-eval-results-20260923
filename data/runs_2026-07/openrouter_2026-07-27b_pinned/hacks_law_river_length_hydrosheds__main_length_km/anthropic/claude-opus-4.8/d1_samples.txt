import numpy as np

def predict(X):
    # Hack's law: L = c * A^h
    # Canonical fitted values (Hack, 1957): c ~ 1.4, h ~ 0.6 for L, A in km
    A = np.asarray(X[:, 0], dtype=float)
    A = np.clip(A, 1e-9, None)
    c = 1.4
    h = 0.6
    L = c * np.power(A, h)
    return L
