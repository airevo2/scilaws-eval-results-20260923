import numpy as np

def predict(X):
    """
    Predict main-stem river length (km) using Hack's law:
        L = c * A^h
    where A is the upstream drainage area (km^2), c ≈ 1.4, and h ≈ 0.6.
    This is the most commonly cited closed-form relationship for main-stem length
    versus drainage area (Hack, 1957).
    """
    A = X[:, 0]               # upland_area_skm (km^2)
    L = 1.4 * np.power(A, 0.6)  # main_length_km (km)
    return L

def predict(X):
    # Hack's law: L = c * A^h, the standard closed-form relationship for
    # main-stem river length as a function of upstream drainage area.
    # Hack (1957) gave c = 1.4 with L in miles and A in square miles;
    # converting to km and km^2 yields c ≈ 1.273 and h ≈ 0.6.
    A = X[:, 0]          # total upstream drainage area [km^2]
    c = 1.4 * 1.609344 / (2.589988**0.6)   # ≈ 1.273 km / (km^2)^0.6
    h = 0.6
    L = c * np.power(A, h)
    return L
