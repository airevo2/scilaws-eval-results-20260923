import numpy as np

def predict(X):
    """
    Period life expectancy at birth (total sex) as a logistic function
    of calendar year, capturing the mortality transition from pre-industrial
    levels to the modern asymptote (canonical bounded-growth form used in
    demographic projections, e.g., UN WPP methodology).

    Parameters of the logistic curve:
        e_min  = 33.0   — pre-transition floor (years)
        e_max  = 87.0   — upper asymptote (years)
        t0     = 1935.0 — inflection year of the transition
        k      = 0.027  — rate parameter (per year)
    """
    t = X[:, 0]
    e_min = 33.0
    e_max = 87.0
    t0 = 1935.0
    k = 0.027
    e0 = e_min + (e_max - e_min) / (1.0 + np.exp(-k * (t - t0)))
    return e0

import numpy as np

def predict(X):
    """
    Period life expectancy at birth (e0) as a linear function of calendar year.
    
    Based on the Oeppen & Vaupel (2002) "Broken Limits to Life Expectancy" 
    best-practice linear trend: ~2.43 years gained per decade.
    
    e0(t) = 45.0 + 0.243 * (t - 1840)
    """
    t = X[:, 0]
    # Oeppen-Vaupel (2002) best-practice line coefficients
    slope = 0.243        # years of e0 gained per calendar year (~2.43 per decade)
    intercept = 45.0     # e0 at reference year
    t_ref = 1840.0       # reference year (start of best-practice record)
    
    e0 = intercept + slope * (t - t_ref)
    return e0
