import numpy as np

def predict(X):
    """
    Predicts period life expectancy at birth (e0) based solely on calendar year.
    Uses a standard linear trend (drift) model, which is the most common 
    baseline closed-form relationship for e0 over time in demographic 
    time-series forecasting (e.g., Lee-Carter extensions).
    """
    t = X[:, 0]
    # Linear trend coefficients (approximate global HMD average drift)
    # e0(t) = a + b * t
    a = -340.0
    b = 0.22
    
    e0 = a + b * t
    return e0

    # Oeppen-Vaupel (2002) style linear trend for period life expectancy at birth.
    # Record/best-practice e0 has increased roughly linearly by ~0.23 years per
    # calendar year since the mid-19th century; this total-sex adaptation uses
    # a slope of 0.23 anchored near 43 years around 1840.
    e0 = 0.23 * (t - 1840.0) + 43.0
    return e0

import numpy as np

def predict(X):
    t = X[:, 0]
    # Standard linear trend approximation for period life expectancy at birth (e0)
    # reflecting the long-term average trajectory observed in HMD countries 
    # from the 18th century to the present.
    intercept = -266.4
    slope = 0.17216
    
    e0 = intercept + slope * t
    return e0
