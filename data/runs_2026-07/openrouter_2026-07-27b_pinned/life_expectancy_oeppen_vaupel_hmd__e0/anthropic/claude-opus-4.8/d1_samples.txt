import numpy as np

def predict(X):
    # Oeppen–Vaupel style linear rise in life expectancy:
    # best-practice/national e0 rose ~2.5 years per decade (0.25 yr/yr).
    year = X[:, 0]
    slope = 0.25          # years of e0 gained per calendar year
    e0_ref = 78.0         # reference e0 near year 2000
    year_ref = 2000.0
    return e0_ref + slope * (year - year_ref)
