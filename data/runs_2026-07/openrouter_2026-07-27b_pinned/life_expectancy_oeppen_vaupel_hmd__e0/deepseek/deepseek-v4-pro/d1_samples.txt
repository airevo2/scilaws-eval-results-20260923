import numpy as np

def predict(X):
    """
    Predict period life expectancy at birth from calendar year using a logistic
    (S-shaped) function that captures the historical demographic transition.
    The function is bounded between a pre-transition plateau and a post-transition
    plateau, with a steady inflection around the mid-20th century.
    """
    t = X[:, 0]                     # calendar year
    lower = 30.0                    # typical pre-transition e0 (years)
    upper = 85.0                    # asymptotic maximum e0 (years)
    mid_year = 1950.0               # inflection point of the logistic curve
    rate = 0.025                    # pace of the increase (1/year)

    e0 = lower + (upper - lower) / (1.0 + np.exp(-rate * (t - mid_year)))
    return e0

import numpy as np

def predict(X):
    """
    Predict period life expectancy at birth (e0) from calendar year using
    a logistic function, which is the standard closed-form model for the
    demographic transition in life expectancy over long historical periods.
    """
    t = X[:, 0]
    # Parameters chosen to reflect a typical long-run trajectory
    L_lower = 30.0      # pre-transitional life expectancy (years)
    L_upper = 85.0      # asymptotic maximum life expectancy (years)
    k = 0.025           # speed of transition (1/years)
    t_mid = 1925.0      # midpoint year of the transition
    e0 = L_lower + (L_upper - L_lower) / (1.0 + np.exp(-k * (t - t_mid)))
    return e0
