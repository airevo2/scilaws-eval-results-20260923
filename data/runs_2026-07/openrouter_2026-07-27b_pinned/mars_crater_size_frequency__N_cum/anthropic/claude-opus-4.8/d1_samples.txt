    # Standard single power-law form: N(>=D) = C * D^(-b)
    # with D in km and N in km^-2.
    D = np.asarray(X, dtype=float)[:, 0]

    # Fitted coefficients (typical Mars heavily-cratered SFD)

import numpy as np

def predict(X):
    # Cumulative crater size-frequency distribution for Mars (production/empirical form):
    #   N(>=D) = A * D^(-b)   [craters per km^2]
    # A single power law is the most commonly cited closed form for the
    # cumulative SFD over a wide diameter range. Only the lower bin edge D_lo is used.
    D = np.asarray(X, dtype=float)[:, 0]

    # Fitted coefficients (log10 intercept and power-law slope)
    log10_A = -2.35     # amplitude at D = 1 km  => A ~ 4.5e-3 km^-2
    b = 1.80            # cumulative slope

    log10_N = log10_A - b * np.log10(D)
    return np.power(10.0, log10_N)

import numpy as np

def predict(X):
    # Cumulative crater size-frequency distribution (production function)
    # standard single power-law form: N(>=D) = k * D^(-b)
    # calibrated so that N spans ~5 decades over D ~ 1..664 km on Mars.
    D = np.asarray(X, dtype=float)[:, 0]  # D_lower_km, the threshold diameter
    k = 0.1     # normalization at D = 1 km  [km^-2]
    b = 1.8     # cumulative power-law slope
    return k * np.power(D, -b)
