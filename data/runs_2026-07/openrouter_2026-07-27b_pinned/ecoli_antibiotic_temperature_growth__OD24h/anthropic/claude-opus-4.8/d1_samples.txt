import numpy as np

def predict(X):
    T = np.asarray(X, dtype=float)[:, 0]

    # Cardinal Temperature Model with Inflection (Rosso/CTMI)
    Tmin = 8.0     # minimum temperature for growth
    Topt = 39.0    # optimal temperature
    Tmax = 47.5    # maximum temperature for growth
    ODopt = 1.0    # OD600 at optimal temperature (24 h plateau)

    num = (T - Tmax) * (T - Tmin) ** 2
    den = (Topt - Tmin) * (
        (Topt - Tmin) * (T - Topt)
        - (Topt - Tmax) * (Topt + Tmin - 2.0 * T)
    )

    # avoid division issues
    gamma = np.where(den != 0.0, num / den, 0.0)
    gamma = np.clip(gamma, 0.0, 1.0)

    # Outside [Tmin, Tmax] growth is zero
    gamma = np.where((T <= Tmin) | (T >= Tmax), 0.0, gamma)

    OD = ODopt * gamma
    return np.maximum(OD, 0.0)
