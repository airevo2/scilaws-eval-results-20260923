import numpy as np

def predict(X):
    """
    Predict OD600(24h) for E. coli BW25113 in LB as a function of temperature.
    Uses the Ratkowsky full square-root model (1983):
        sqrt(mu) = b * (T - T_min) * (1 - exp(c * (T - T_max)))
    OD24h is taken proportional to growth rate mu, i.e. OD = a * [sqrt(mu)]^2.
    """
    T = X[:, 0]

    # Cardinal temperatures for E. coli (degC)
    T_min = 8.0       # minimum growth temperature
    T_max = 46.5      # maximum growth temperature

    # Shape parameter controlling sharpness of decline near T_max
    c = 0.2

    # Scaling coefficient (absorbs b^2, inoculum size, time, etc.)
    a = 0.0041        # yields OD ~2.5 at 37 degC

    term = (T - T_min) * (1.0 - np.exp(c * (T - T_max)))
    od = a * term ** 2

    # Zero outside the permissive growth range
    od = np.where((T >= T_min) & (T <= T_max), od, 0.0)

    return od

import numpy as np

def predict(X):
    """
    Predict OD600 after 24h for E. coli BW25113 as a function of temperature.
    
    Uses the extended Ratkowsky (Belehradek-Ratkowsky) square-root model,
    the standard closed-form relationship for bacterial growth across the
    full biokinetic temperature range:
    
        sqrt(r) = b * (T - T_min) * (1 - exp(c * (T - T_max)))
    
    OD600(24h) is taken proportional to the growth rate r(T).
    """
    T = X[:, 0]

    # Extended Ratkowsky model parameters for E. coli K-12 (BW25113)
    T_min = 10.0    # Theoretical minimum growth temperature [degC]
    T_max = 46.0    # Theoretical maximum growth temperature [degC]
    b = 0.05        # Rate scaling parameter [OD^0.5 / degC]
    c = 0.3         # High-temperature decline shape parameter [1/degC]

    # Square-root of OD proxy from the Ratkowsky model
    sqrt_od = b * (T - T_min) * (1.0 - np.exp(c * (T - T_max)))

    # Growth is physically impossible outside the cardinal range;
    # clip the square-root term to zero before squaring
    sqrt_od = np.maximum(sqrt_od, 0.0)

    od24h = sqrt_od ** 2
    return od24h
