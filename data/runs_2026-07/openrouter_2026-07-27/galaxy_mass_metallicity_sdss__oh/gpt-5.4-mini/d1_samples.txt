import numpy as np

def predict(X):
    """
    Gas-phase oxygen abundance, 12 + log(O/H), using the Mannucci et al. (2010)
    / "fundamental metallicity relation" polynomial form.

    Inputs
    ------
    X : array-like, shape (n_samples, 2)
        X[:, 0] = log_Mstar = log10(M_star / Msun)
        X[:, 1] = log_SFR   = log10(SFR / (Msun/yr))

    Returns
    -------
    oh : ndarray, shape (n_samples,)
        12 + log10(O/H)
    """
    X = np.asarray(X)
    m = X[:, 0]
    s = X[:, 1]
    mu = m - 0.32 * s
    return 8.90 + 0.37 * (mu - 10.0) - 0.14 * (mu - 10.0) ** 2

    s = X[:, 1]
    
    # Commonly used FMR variable
    mu = m - 0.32 * s

    oh = 8.90 + 0.37 * (mu - 10.0) - 0.14 * (mu - 10.0)**2
    
    return oh

import numpy as np

def predict(X):
    log_Mstar = X[:, 0]
    log_SFR = X[:, 1]

    # Fundamental Metallicity Relation (Mannucci et al. 2010), commonly used polynomial form
    m = log_Mstar - 10.0
    s = log_SFR

    oh = 8.90 + 0.37*m - 0.14*s - 0.19*m**2 + 0.12*m*s - 0.054*s**2
    return oh
