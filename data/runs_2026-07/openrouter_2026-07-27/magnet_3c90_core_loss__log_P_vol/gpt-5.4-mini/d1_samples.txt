    w = X[:, 3].astype(float)          # waveform id: 0 sinusoidal, 1 triangular, 2 trapezoidal

    # Common generalized Steinmetz form:
    # P_v = k * f^alpha * B^beta * C_waveform * C_temp
    #
    # Since no fitted coefficients are provided, use the standard canonical
    # closed-form exponents for core-loss modeling:
    #   alpha = 1.5, beta = 2.5
    # with waveform correction factors relative to sinusoid.
    #
    # This returns log10(P_vol in kW/m^3), so the scale factor 1e-3 converts W/m^3 -> kW/m^3.
    alpha = 1.5
    beta = 2.5

    # Standard waveform correction factors (canonical approximation):
    # sinusoidal: 1
    # triangular: ~1.11
    # trapezoidal: ~1.25
    Cw = np.ones_like(f)
    Cw = np.where(w == 1, 1.11, Cw)
    Cw = np.where(w == 2, 1.25, Cw)

    # Because the true material-dependent prefactor k is not provided, use k = 1.
    # This yields a standard closed-form log-linear model structure.

import numpy as np

def predict(X):
    """
    Estimate log10 volumetric core loss using a generalized Steinmetz-type model.

    Inputs
    ------
    X : ndarray, shape (n_samples, 4)
        Columns:
        0: f_Hz         excitation frequency [Hz]
        1: B_peak_T     peak flux density [T]
        2: T_C          temperature [degC]
        3: waveform_id  0 = sinusoidal, 1 = triangular, 2 = trapezoidal

    Returns
    -------
    log_P_vol : ndarray, shape (n_samples,)
        Base-10 logarithm of volumetric core loss [log10(kW/m^3)]
    """
    f = np.asarray(X[:, 0], dtype=float)
    B = np.asarray(X[:, 1], dtype=float)
    T = np.asarray(X[:, 2], dtype=float)
    w = np.asarray(X[:, 3], dtype=float)

    # Common canonical core-loss form: modified Steinmetz equation
    # P_v = k * f^alpha * B^beta
    # with simple waveform/temperature adjustment factors.
    #
    # Since no data are provided, use representative literature-style exponents
    # and a mild temperature correction.
    k = 2.0e-3
    alpha = 1.6
    beta = 2.4

    # waveform factor: sinusoidal baseline, triangular slightly higher loss,
    # trapezoidal higher still due to harmonics / dwell regions
    k_w = np.where(w == 0, 1.00,
          np.where(w == 1, 1.12, 1.22))

    # temperature factor: mild increase with temperature (generic approximation)
    k_T = 1.0 + 0.0025 * (T - 25.0)

    P_kW_m3 = k * k_w * k_T * (f ** alpha) * (B ** beta)

    # avoid log of zero/negative if inputs are pathological
    P_kW_m3 = np.maximum(P_kW_m3, 1e-30)

    return np.log10(P_kW_m3)
