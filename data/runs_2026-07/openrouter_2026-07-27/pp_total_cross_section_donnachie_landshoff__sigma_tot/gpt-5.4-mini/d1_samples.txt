    Standard Donnachie-Landshoff-type parametrization for pp total cross section:
        sigma_tot(s) = A * s^epsilon + B * s^-eta
    with s in GeV^2 and sigma in mb.

    Commonly used pp form (approximate canonical values):
        A = 21.70 mb
        epsilon = 0.0808
        B = 56.08 mb
        eta = 0.4525
    """
    s = np.asarray(X)[:, 0]
    A = 21.70
    epsilon = 0.0808
    B = 56.08
    eta = 0.4525
    return A * np.power(s, epsilon) + B * np.power(s, -eta)

import numpy as np

def predict(X):
    """
    Standard closed-form parameterization of the pp total cross section
    in terms of Mandelstam s.

    Here we use a commonly cited Regge-inspired form:
        sigma_tot(s) = A + B * log^2(s / s0) + C * s^(-eta)

    with representative canonical constants chosen for pp:
        A = 34.71 mb
        B = 0.2647 mb
        s0 = 1.0 GeV^2
        C = 27.55 mb
        eta = 0.4525

    Parameters
    ----------
    X : ndarray, shape (n_samples, 1)
        X[:, 0] = s in GeV^2

    Returns
    -------
    ndarray, shape (n_samples,)
        sigma_tot in mb
    """
    s = np.asarray(X)[:, 0].astype(float)

    A = 34.71
    B = 0.2647
    s0 = 1.0
    C = 27.55
    eta = 0.4525

    return A + B * np.log(s / s0)**2 + C * s**(-eta)
