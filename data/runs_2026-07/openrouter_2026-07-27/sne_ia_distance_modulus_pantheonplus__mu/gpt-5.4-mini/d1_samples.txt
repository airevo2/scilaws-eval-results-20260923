import numpy as np

def predict(X):
    z = np.asarray(X)[:, 0]
    c = 299792.458  # km/s
    H0 = 70.0       # km/s/Mpc
    Omega_m = 0.3
    Omega_L = 0.7

    # Flat LCDM luminosity distance:
    # d_L(z) = (1+z) * c/H0 * integral_0^z dz'/E(z')
    # E(z) = sqrt(Omega_m (1+z)^3 + Omega_L)
    z = np.asarray(z, dtype=float)

    # Numerical quadrature on a per-object basis
    mu = np.empty_like(z, dtype=float)
    for i, zi in enumerate(z):
        zz = np.linspace(0.0, zi, 512)
        Ez = np.sqrt(Omega_m * (1.0 + zz)**3 + Omega_L)
        integral = np.trapz(1.0 / Ez, zz)
        dL_Mpc = (1.0 + zi) * (c / H0) * integral  # Mpc
        dL_pc = dL_Mpc * 1.0e6
        mu[i] = 5.0 * np.log10(dL_pc / 10.0)

    return mu

import numpy as np

def predict(X):
    z = np.asarray(X)[:, 0]
    c = 299792.458  # km/s
    H0 = 70.0       # km/s/Mpc
    Omega_m = 0.3
    Omega_L = 1.0 - Omega_m

    # Luminosity distance in a flat LCDM cosmology:
    # d_L(z) = (1+z) * c/H0 * integral_0^z dz'/E(z')
    # with E(z) = sqrt(Omega_m*(1+z)^3 + Omega_L)
    z_grid = np.linspace(0.0, np.max(z), 4096)
    Ez = np.sqrt(Omega_m * (1.0 + z_grid)**3 + Omega_L)
    inv_E = 1.0 / Ez
    dz = z_grid[1] - z_grid[0]
    integral = np.cumsum((inv_E[:-1] + inv_E[1:]) * 0.5 * dz)
    integral = np.concatenate(([0.0], integral))
    chi = np.interp(z, z_grid, integral)

    dL_mpc = (1.0 + z) * (c / H0) * chi
    dL_pc = dL_mpc * 1.0e6

    mu = 5.0 * np.log10(dL_pc / 10.0)
    return mu

import numpy as np

def predict(X):
    z = np.asarray(X)[:, 0]
    c = 299792.458  # km/s
    H0 = 70.0       # km/s/Mpc
    Mpc_to_pc = 1.0e6

    # Standard flat ΛCDM distance modulus with canonical parameters
    Om0 = 0.3
    Ol0 = 1.0 - Om0

    # Numerical integration for comoving distance
    z = np.clip(z, 1e-8, None)
    z_grid = np.linspace(0.0, 1.0, 2048)

    def Ez(zz):
        return np.sqrt(Om0 * (1.0 + zz)**3 + Ol0)

    # Vectorized trapezoidal integration over a rescaled grid for each z
    # Integral_0^z dz'/E(z')
    zz = z[:, None] * z_grid[None, :]
    integrand = 1.0 / Ez(zz)
    integral = z * np.trapz(integrand, z_grid, axis=1)

    d_c_mpc = (c / H0) * integral
    d_l_mpc = (1.0 + z) * d_c_mpc
    d_l_pc = d_l_mpc * Mpc_to_pc

    mu = 5.0 * np.log10(d_l_pc / 10.0)
    return mu

import numpy as np

def predict(X):
    z = X[:, 0]
    # Standard flat-LCDM distance modulus relation:
    # mu = 5 log10(d_L / 10 pc), with
    # d_L(z) = (c/H0) * (1+z) * integral_0^z dz' / E(z')
    # E(z) = sqrt(Om0 (1+z)^3 + (1-Om0))
    #
    # Since no cosmological parameters are provided, use the standard
    # observationally common placeholder in closed form with the low-z
    # Hubble-law approximation:
    # mu ≈ 5 log10(c z / H0) + 25
    #
    # Here H0 is left as the conventional constant 70 km/s/Mpc.
    c = 299792.458  # km/s
    H0 = 70.0       # km/s/Mpc
    dL_Mpc = (c / H0) * z
    mu = 5.0 * np.log10(dL_Mpc) + 25.0
    return mu
