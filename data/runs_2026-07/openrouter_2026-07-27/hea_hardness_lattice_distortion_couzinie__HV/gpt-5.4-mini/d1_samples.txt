    )
    return HV

import numpy as np

def predict(X):
    X = np.asarray(X, dtype=float)
    delta = X[:, 0]
    vec = X[:, 1]
    dHmix = X[:, 2]

    # A common closed-form empirical representation for hardness in
    # refractory / complex-concentrated HEAs uses linear dependence on
    # atomic-size mismatch and valence electron concentration, with a
    # weaker correction for mixing enthalpy.
    HV = 27.0 + 5.2 * delta - 3.8 * vec - 0.12 * dHmix

    return HV
