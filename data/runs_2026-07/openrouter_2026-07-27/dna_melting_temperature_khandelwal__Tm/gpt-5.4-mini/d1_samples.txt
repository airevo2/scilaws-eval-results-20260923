import numpy as np

def predict(X):
    X = np.asarray(X)
    E = X[:, 0]
    N = X[:, 1]
    salt_M = X[:, 2]
    dna_conc_M = X[:, 3]

    # Common closed-form approximation for DNA duplex melting temperature:
    # Tm = 64.9 + 41*(#GC - 16.4)/N + 16.6*log10([Na+])
    # Here the sequence-specific "strength" parameter E is used as a surrogate
    # for GC/stacking contribution, so we map it linearly into the GC-like term.
    #
    # For self-complementary duplexes, a common concentration correction is:
    # Tm += 16.6*log10([Na+]) + 0.41*(%GC) - 500/N - 0.62*ln(Ct)
    # but the most widely cited simple form is the salt-corrected nearest-neighbor
    # style approximation below.
    #
    # Since only E, N, salt, and total DNA concentration are provided, we use the
    # standard log-salt and concentration correction with a linear stabilization
    # term from E.

    tm = (
        81.5
        + 16.6 * np.log10(salt_M)
        + 0.41 * (100.0 * (E / 10.0) - 16.4)
        - 675.0 / N
        - 0.62 * np.log(dna_conc_M)
    )

    return tm

import numpy as np

def predict(X):
    E = X[:, 0]
    N = X[:, 1]
    salt_M = X[:, 2]
    dna_conc_M = X[:, 3]

    # Common nearest-neighbor / empirical DNA duplex melting temperature form
    # with a stacking/stability term (E), length correction, salt correction,
    # and concentration correction.
    tm = (
        64.9
        + 41.0 * (E - 16.4) / N
        + 16.6 * np.log10(salt_M)
        - 500.0 / N
        - 0.62 * (dna_conc_M * 1e9) / N
    )

    return tm
