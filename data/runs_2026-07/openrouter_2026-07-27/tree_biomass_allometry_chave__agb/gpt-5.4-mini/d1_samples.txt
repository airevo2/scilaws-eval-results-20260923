import numpy as np

def predict(X):
    rho = X[:, 0]   # wood density (g/cm^3)
    D = X[:, 1]     # diameter at breast height (cm)
    H = X[:, 2]     # total tree height (m)
    
    # Chave et al. (2014) moist forest allometric equation for individual-tree AGB (kg)
    return 0.0673 * (rho * (D ** 2) * H) ** 0.976
