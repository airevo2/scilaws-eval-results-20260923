import numpy as np

def predict(X):
    # Simple canonical wind-turbine power curve approximation:
    # P = 0 for cut-in winds, P rises roughly with v^3 up to rated power,
    # then saturates at rated power.
    v = X[:, 0]

    # Typical wind turbine parameters (generic, not turbine-specific)
    cut_in = 3.0       # m/s
    rated_speed = 12.0  # m/s
    cut_out = 25.0     # m/s
    rated_power = 2000.0  # kW

    P = np.zeros_like(v, dtype=float)

    # Cubic region between cut-in and rated speed
    mask_cubic = (v >= cut_in) & (v < rated_speed)
    P[mask_cubic] = rated_power * ((v[mask_cubic]**3 - cut_in**3) /
                                    (rated_speed**3 - cut_in**3))

    # Rated region
    mask_rated = (v >= rated_speed) & (v < cut_out)
    P[mask_rated] = rated_power

    # Above cut-out, power stays at 0 for safety in the standard idealized model
    P[v >= cut_out] = 0.0

    return P
