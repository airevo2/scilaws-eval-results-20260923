import numpy as np

def predict(X):
    """
    Predict log10(Kd) for PFAS sorption to soil.

    Canonical closed-form relationship:
        log10(Kd) = log10(Koc) + log10(foc)

    where Koc is commonly estimated from hydrophobicity using a linear free-energy
    style relationship with log Kow:
        log10(Koc) = a * log10(Kow) + b

    For PFAS-specific soil partitioning, a widely used pragmatic form is:
        log10(Kd) = a * log10(Kow) + b + log10(foc)

    This implementation uses a standard approximate coefficient set.
    """
    X = np.asarray(X, dtype=float)

    log_Kow = X[:, 6]
    foc = X[:, 13]

    # Commonly used approximate Koc-logKow relationship for hydrophobic organic sorbates
    a = 0.54
    b = 1.47

    # Avoid log10(0) if foc is zero or extremely small
    foc_safe = np.clip(foc, 1e-12, None)

    log_Kd = a * log_Kow + b + np.log10(foc_safe)
    return log_Kd

import numpy as np

def predict(X):
    X = np.asarray(X)
    MW = X[:, 0]
    pKa1 = X[:, 1]
    C_num = X[:, 2]
    F_num = X[:, 3]
    F_pct = X[:, 4]
    log_S = X[:, 5]
    log_Kow = X[:, 6]
    pH = X[:, 7]
    CEC = X[:, 8]
    Sand = X[:, 9]
    Silt = X[:, 10]
    Clay = X[:, 11]
    Corg = X[:, 12]
    foc = X[:, 13]

    # Canonical soil sorption relationship for hydrophobic organic contaminants:
    # Kd = Koc * foc  => log10(Kd) = log10(Koc) + log10(foc)
    # For PFAS, a widely used closed-form approximation is a linear
    # hydrophobicity-based log Koc relationship with an ionization correction.
    # Here, the most common compact form is written using log Kow and pH/pKa1.
    #
    # log Kd = a + b*logKow + log10(foc) + c*max(0, pH - pKa1)
    # The pH term reduces sorption for anionic PFAS at environmental pH.
    # Coefficients are representative literature-style values.
    a = -0.49
    b = 0.81
    c = -0.50

    ion_term = np.maximum(0.0, pH - pKa1)
    log_Kd = a + b * log_Kow + np.log10(np.maximum(foc, 1e-30)) + c * ion_term

    return log_Kd
