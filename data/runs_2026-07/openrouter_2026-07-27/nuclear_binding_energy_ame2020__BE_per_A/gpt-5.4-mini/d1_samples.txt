import numpy as np

def predict(X):
    Z = X[:, 0].astype(float)
    N = X[:, 1].astype(float)
    A = X[:, 2].astype(float)
    I = X[:, 4].astype(float)  # (N - Z) / A
    pair = X[:, 6].astype(float)

    # Semi-empirical mass formula (Bethe-Weizsäcker), per nucleon:
    # B/A = a_v
    #       - a_s A^(-1/3)
    #       - a_c Z(Z-1) A^(-4/3)
    #       - a_a (N-Z)^2 A^(-2)
    #       + a_p A^(-3/2) * delta
    #
    # Using commonly cited canonical coefficients (MeV):
    a_v = 15.8
    a_s = 18.3
    a_c = 0.714
    a_a = 23.2
    a_p = 12.0

    # Pairing term: +1 even-even, 0 odd-A, -1 odd-odd
    pairing = np.zeros_like(A, dtype=float)
    mask = A > 0
    pairing[mask] = a_p * pair[mask] * A[mask] ** (-1.5)

    result = (
        a_v
        - a_s * A ** (-1.0 / 3.0)
        - a_c * Z * (Z - 1.0) * A ** (-4.0 / 3.0)
        - a_a * (I ** 2)
        + pairing
    )
    return result

import numpy as np

def predict(X):
    Z = X[:, 0].astype(float)
    N = X[:, 1].astype(float)
    A = X[:, 2].astype(float)
    delta = X[:, 6].astype(float)

    # Semi-empirical mass formula (Weizsäcker / Bethe–Weizsäcker) in binding-energy-per-nucleon form
    a_v = 15.8   # volume term [MeV]
    a_s = 18.3   # surface term [MeV]
    a_c = 0.714  # Coulomb term [MeV]
    a_a = 23.2   # asymmetry term [MeV]
    a_p = 12.0   # pairing term [MeV]

    # Pairing contribution with standard even-even / odd-odd / odd-A sign convention
    pairing = np.zeros_like(A)
    pairing[delta > 0] = +a_p / np.sqrt(A[delta > 0])   # even-even
    pairing[delta < 0] = -a_p / np.sqrt(A[delta < 0])   # odd-odd
    # odd-A remains 0

    BE = (
        a_v * A
        - a_s * A ** (2.0 / 3.0)
        - a_c * Z * (Z - 1.0) / A ** (1.0 / 3.0)
        - a_a * (A - 2.0 * Z) ** 2 / A
        + pairing * A
    )

    return BE / A

import numpy as np

def predict(X):
    Z = X[:, 0].astype(float)
    N = X[:, 1].astype(float)
    A = X[:, 2].astype(float)
    NZ_diff = X[:, 3].astype(float)
    I = X[:, 4].astype(float)
    coul_proxy = X[:, 5].astype(float)
    delta = X[:, 6].astype(float)

    # Semi-empirical mass formula (Weizsäcker / Bethe–Weizsäcker),
    # expressed per nucleon: B/A = a_v - a_s A^(-1/3) - a_c Z(Z-1) A^(-4/3)
    #                      - a_a (N-Z)^2 / A^2 + a_p delta / A^(3/2)
    a_v = 15.75
    a_s = 17.8
    a_c = 0.711
    a_a = 23.7
    a_p = 11.18

    A13 = np.cbrt(A)
    BA = (
        a_v
        - a_s / A13
        - a_c * coul_proxy / A
        - a_a * (NZ_diff ** 2) / (A ** 2)
        + a_p * delta / (A ** 1.5)
    )
    return BA

import numpy as np

def predict(X):
    Z = X[:, 0].astype(float)
    N = X[:, 1].astype(float)
    A = X[:, 2].astype(float)
    I = X[:, 4].astype(float)
    coul_proxy = X[:, 5].astype(float)
    pair = X[:, 6].astype(float)

    # Semi-empirical mass formula / liquid-drop binding energy per nucleon
    # B/A = a_v - a_s A^(-1/3) - a_c Z(Z-1) A^(-4/3) - a_a (N-Z)^2 / A^2 + delta/A
    # Using standard textbook coefficient values (in MeV)
    a_v = 15.8
    a_s = 18.3
    a_c = 0.714
    a_a = 23.2
    a_p = 12.0

    # Pairing term: + for even-even, - for odd-odd, 0 for odd-A
    pairing = np.zeros_like(A, dtype=float)
    mask = pair == 1
    pairing[mask] = +a_p / np.sqrt(A[mask])
    mask = pair == -1
    pairing[mask] = -a_p / np.sqrt(A[mask])

    BA = (
        a_v
        - a_s * A ** (-1.0 / 3.0)
        - a_c * coul_proxy / A
        - a_a * (N - Z) ** 2 / A**2
        + pairing / A
    )

    return BA
