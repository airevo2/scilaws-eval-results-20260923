import numpy as np

def predict(X):
    T = X[:, 0]
    # Sharpe-Schoolfield / Boltzmann-Arrhenius style thermal performance curve
    # Canonical closed form with low- and high-temperature inactivation terms.
    # Parameter values are generic placeholders in the absence of fitted coefficients.
    B0 = 1.0          # normalization / scale
    E = 0.65          # activation energy (eV)
    k = 8.617333262e-5 # Boltzmann constant (eV/K)
    Tref = 293.15     # reference temperature (K)
    Th = 318.15       # high-temperature deactivation midpoint (K)
    Eh = 2.0          # high-temperature deactivation energy (eV)
    Tl = 273.15       # low-temperature deactivation midpoint (K)
    El = 2.0          # low-temperature deactivation energy (eV)

    Tk = T + 273.15

    num = B0 * np.exp(-E / k * (1.0 / Tk - 1.0 / Tref))
    denom = 1.0 + np.exp(Eh / k * (1.0 / Th - 1.0 / Tk)) + np.exp(El / k * (1.0 / Tk - 1.0 / Tl))
    return num / denom
