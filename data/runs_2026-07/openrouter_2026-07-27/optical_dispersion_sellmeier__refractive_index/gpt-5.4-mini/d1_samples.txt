import numpy as np

def predict(X):
    """
    Real refractive index n as a function of vacuum wavelength (um).

    Since no specific material/system is provided, the standard closed-form
    dispersion relation most commonly used in optics is the Sellmeier form.
    Here a generic 3-term Sellmeier structure is returned, with coefficients
    left as constants placeholder defaults (unity-scale) because no dataset
    or material-specific coefficients were given.
    """
    lam = np.asarray(X)[:, 0]
    lam2 = lam * lam

    # Generic Sellmeier form:
    # n^2 = 1 + sum_i B_i * lam^2 / (lam^2 - C_i)
    # Material-specific coefficients must be supplied for a real prediction.
    B1, B2, B3 = 1.0, 0.0, 0.0
    C1, C2, C3 = 0.01, 0.1, 10.0

    n2 = 1.0 \
        + B1 * lam2 / (lam2 - C1) \
        + B2 * lam2 / (lam2 - C2) \
        + B3 * lam2 / (lam2 - C3)

    return np.sqrt(n2)
