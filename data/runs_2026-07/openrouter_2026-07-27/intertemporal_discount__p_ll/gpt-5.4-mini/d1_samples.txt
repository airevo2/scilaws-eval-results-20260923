import numpy as np

def predict(X):
    """
    Canonical hyperbolic-discounting choice rule with a softmax/logistic choice mapping.

    P(LL) = 1 / (1 + exp(-k * (V_LL/(1 + k_d * t_LL) - V_SS/(1 + k_d * t_SS))))

    Here, since no fitted coefficients/data are provided, this function uses a
    standard normalized closed-form choice probability based on the discounted
    value difference inferred from the inputs. The discount-rate parameters are
    set to 1.0 as generic placeholders.
    """
    X = np.asarray(X)
    log_amount_ratio = X[:, 0]
    t_ss = X[:, 1]
    t_ll = X[:, 2]
    v_ss = X[:, 4]
    v_ll = X[:, 5]

    # Generic canonical hyperbolic discounting with logistic choice rule
    k_d = 1.0
    beta = 1.0

    dv = (v_ll / (1.0 + k_d * t_ll)) - (v_ss / (1.0 + k_d * t_ss))
    p_ll = 1.0 / (1.0 + np.exp(-beta * dv))
    return p_ll
