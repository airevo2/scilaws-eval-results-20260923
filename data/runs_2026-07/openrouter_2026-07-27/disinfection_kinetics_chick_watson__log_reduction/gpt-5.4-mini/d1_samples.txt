import numpy as np

def predict(X):
    """
    Predict log10 microbial reduction (LRV) from CT using the classic
    Chick-Watson form:

        LRV = k * CT^n

    For the most commonly cited simplified closed-form relationship in
    disinfection practice, n is often taken as 1 (linear with CT), giving:

        LRV = k * CT

    Since no organism/disinfectant-specific calibration data are provided,
    this function returns the canonical uncalibrated form with k = 1.
    """
    CT = np.asarray(X)[:, 0]
    return np.maximum(CT, 0.0)

import numpy as np

def predict(X):
    CT = X[:, 0]
    # Classic Chick-Watson / log-linear disinfection model:
    # LRV = k * (CT)^n
    # In the simplest commonly cited form, n = 1, so LRV is proportional to CT.
    # Since no organism/process-specific fitted coefficients are provided, use the
    # standard normalized closed-form expression based on log10 survival reduction:
    # log10(N0/N) = CT / 2.303 for a first-order base-e rate constant of 1.
    return np.maximum(0.0, CT / 2.303)
