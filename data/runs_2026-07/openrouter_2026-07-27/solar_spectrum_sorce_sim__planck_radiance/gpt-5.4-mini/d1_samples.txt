import numpy as np

def predict(X):
    """
    Planck spectral radiance B_lambda(λ, T) in units of W m^-2 nm^-1 sr^-1.

    Uses the standard closed-form Planck law:
        B_lambda = (2 h c^2 / λ^5) / (exp(h c / (λ k_B T)) - 1)

    Since only wavelength is provided and the target is solar radiance inferred
    from SORCE SIM daily-mean SSI via L = SSI / Ω, this function returns the
    solar-disk-equivalent spectral radiance using a canonical effective solar
    temperature of 5778 K.

    Input:
        X[:, 0] = wavelength_nm

    Returns:
        1-D numpy array of spectral radiance in W m^-2 nm^-1 sr^-1
    """
    lam_nm = np.asarray(X[:, 0], dtype=float)
    lam_m = lam_nm * 1e-9

    h = 6.62607015e-34      # J s
    c = 299792458.0         # m s^-1
    kB = 1.380649e-23       # J K^-1
    T = 5778.0              # K

    exponent = (h * c) / (lam_m * kB * T)
    B_per_m = (2.0 * h * c**2) / (lam_m**5) / np.expm1(exponent)  # W m^-2 m^-1 sr^-1
    B_per_nm = B_per_m * 1e-9  # W m^-2 nm^-1 sr^-1

    return B_per_nm
