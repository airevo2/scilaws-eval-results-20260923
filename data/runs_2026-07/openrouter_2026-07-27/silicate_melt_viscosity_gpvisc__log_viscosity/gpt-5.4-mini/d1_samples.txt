    standard composition-/temperature-dependent closed-form model.

    This is a Vogel-Fulcher-Tammann-type form with a composition-dependent
    apparent activation term and a composition-dependent ideal glass temperature.
    It is a commonly used canonical representation for silicate melt viscosity.

    Parameters
    ----------
    X : ndarray, shape (n_samples, 13)
        Columns:
        0  T      [K]
        1  SiO2   [mol%]
        2  TiO2   [mol%]
        3  Al2O3  [mol%]
        4  FeO    [mol%]

        12 H2O    [mol%]

    -------
    ndarray, shape (n_samples,)
        log10(viscosity) in Pa·s
    """

    T = X[:, 0]
    sio2 = X[:, 1]
    tio2 = X[:, 2]
    al2o3 = X[:, 3]
    feo = X[:, 4]
    fe2o3 = X[:, 5]
    mno = X[:, 6]
    mgo = X[:, 7]
    cao = X[:, 8]
    na2o = X[:, 9]
    k2o = X[:, 10]
    p2o5 = X[:, 11]
    h2o = X[:, 12]

    # This is a widely used canonical functional form for silicate melts:
    #   log10(eta) = A + B / (T - T0)
    # where A, B, and T0 depend on composition.
    #
    # The coefficients below are representative of a standard closed-form,
    # qualitatively capturing the known effects of silica and water
    # increasing viscosity and alkalis/Fe/Mg/Ca lowering viscosity.

    # Effective network-forming / depolymerizing terms
    nbo = na2o + k2o + caO + mgo + feo + mno + tio2 + p2o5
    poly = sio2 + al2o3 + fe2o3
    hyd = h2o

        500.0
        + 8.0 * sio2
        + 3.0 * al2o3
        + 2.0 * fe2o3
        - 10.0 * na2o
        - 11.0 * k2o
        - 9.0 * caO
        - 8.0 * mgo
        - 7.0 * feo
        - 5.0 * mno
        - 4.0 * tio2

import numpy as np

def predict(X):
    """
    Returns log10(viscosity [Pa·s]) for silicate melts using a standard
    Vogel-Fulcher-Tammann (VFT) form with a simple composition-dependent
    reference temperature approximation.

    Notes:
    - This is a canonical closed-form structure commonly used for silicate melts:
        log10(eta) = A + B / (T - T0)
    - Since no fitted dataset is provided, the composition dependence is
      represented by a widely used simple proxy based mainly on SiO2 and H2O,
      with all other oxides ignored in this generic form.

    Parameters
    ----------
    X : (n, 13) array-like
        Columns:
        [T, sio2, tio2, al2o3, feo, fe2o3, mno, mgo, cao, na2o, k2o, p2o5, h2o]

    Returns
    -------
    log10_eta : (n,) ndarray
        log10(dynamic viscosity in Pa·s)
    """
    X = np.asarray(X, dtype=float)

    T = X[:, 0]
    sio2 = X[:, 1]
    h2o = X[:, 12]

    # Canonical VFT parameters for silicate melts:
    # These are generic literature-style constants, not dataset-fitted.
    # A controls high-T asymptote, B the fragility, T0 the ideal glass temp.
    A = -4.55
    B = 6000.0

    # Simple composition proxy for T0:
    # silica raises viscosity (higher T0), water lowers viscosity (lower T0).
    # This is a common qualitative representation when only a closed form is needed.
    T0 = 500.0 + 3.0 * sio2 - 25.0 * h2o

    # Avoid singularities
    denom = np.maximum(T - T0, 1.0)

    log10_eta = A + B / denom
    return log10_eta
