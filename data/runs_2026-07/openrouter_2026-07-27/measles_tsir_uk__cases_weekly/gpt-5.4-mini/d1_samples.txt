import numpy as np

def predict(X):
    I_prev = X[:, 0]
    s = X[:, 1]
    B = X[:, 2]
    log_pop = X[:, 3]
    z = X[:, 4]

    N = 10.0 ** log_pop

    # Standard measles incidence model:
    # I_t = beta_s * I_{t-1} * S_t / N
    # with S_t approximated by births accumulation minus cases accumulation
    # and a seasonal transmission term beta_s represented by a cosine forcing.
    #
    # Since no fitted data are provided, use a canonical closed-form seasonally forced
    # mass-action form with a latent susceptible proxy z.
    beta0 = 24.0
    amp = 0.25
    phase = 2.0 * np.pi * (s - 1.0) / 26.0
    beta_s = beta0 * (1.0 + amp * np.cos(phase))

    S_t = np.maximum(z, 0.0)

    I_t = beta_s * I_prev * S_t / np.maximum(N, 1.0)

    return I_t
