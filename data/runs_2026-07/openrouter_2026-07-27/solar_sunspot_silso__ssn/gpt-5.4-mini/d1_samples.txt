    year = np.asarray(X[:, 1], dtype=float)
    month = np.asarray(X[:, 2], dtype=float)

    #
    # Hathaway et al. (1994)-style profile:
    # R(t) = A * ((t - t0)^3 / exp(((t - t0)^2)/b^2))

    R = A * (tau ** 3) * np.exp(-(tau ** 2) / (b ** 2))

    # Ensure non-negative monthly mean sunspot number
    return np.maximum(R, 0.0)

import numpy as np

def predict(X):
    """
    Approximate monthly mean International Sunspot Number (ISN v2.0 scale).
    
    This is a standard solar-cycle style representation using a sum of
    asymmetric Gaussian-like cycles with a slowly varying cycle period.
    Since no fitted data are provided, this implements the widely used
    Hathaway-style parametric form for sunspot cycle shape.

    Inputs
    ------
    X : array-like, shape (n, 3)
        X[:, 0] = t_year : decimal year
        X[:, 1] = year    : integer year
        X[:, 2] = month   : month number (1-12)

    Returns
    -------
    R_I : ndarray, shape (n,)
        Monthly mean International Sunspot Number estimate.
    """
    X = np.asarray(X)
    t = X[:, 0].astype(float)

    # Cycle parameters (canonical solar-cycle shape model)
    # Hathaway, Wilson, Reichmann-style form:
    #   f(t) = a * ((t - t0)/b)^3 / (exp(((t - t0)/b)^2) - c)
    # Here we build a simple sum of such cycles anchored near historical minima.
    #
    # Since no training data are provided, we use a fixed sequence of cycle minima
    # and amplitudes consistent with the standard cycle-shape literature.
    cycle_starts = np.array([
        1745.0, 1755.2, 1766.4, 1775.5, 1784.7, 1798.3, 1810.6, 1823.3,
        1833.9, 1843.5, 1855.8, 1867.2, 1878.9, 1890.2, 1902.0, 1913.6,
        1923.6, 1933.8, 1944.2, 1954.3, 1964.9, 1976.5, 1986.8, 1996.4,
        2008.9, 2019.7
    ])
    amplitudes = np.array([
        60, 100, 85, 75, 90, 120, 95, 70, 85, 105, 140, 110, 160,
        120, 130, 150, 90, 70, 200, 110, 160, 110, 150, 120, 95, 130
    ], dtype=float)

    # Typical cycle rise/decay asymmetry parameters
    tau_rise = 3.5
    tau_decay = 6.2

    # Smooth solar-cycle envelope: asymmetric Gaussian-like pulse
    R = np.zeros_like(t, dtype=float)
    for t0, A in zip(cycle_starts, amplitudes):
        dt = t - t0
        # asymmetry: faster rise, slower decay
        sigma = np.where(dt < 0, tau_rise, tau_decay)
        pulse = A * np.exp(-0.5 * (dt / sigma) ** 2)
        R += pulse

    # Weak background floor to avoid exact zeros in minima
    R += 5.0

    return R

def predict(X):
    """
    Monthly mean total International Sunspot Number (ISN v2.0 scale).

    A simple standard closed-form representation used for solar-cycle
    climatology is the classical Schwabe-cycle cosine-squared envelope.
    Since no fitted coefficients/data are provided, this returns a generic
    cycle-shape estimate driven by decimal year, with an 11-year periodicity.

    Inputs
    ------
    X : ndarray, shape (n, 3)
        X[:, 0] = t_year (decimal year)
        X[:, 1] = year (integer year)   [unused]
        X[:, 2] = month (1-12)          [unused]

    Returns
    -------
    ndarray, shape (n,)
        Estimated monthly mean International Sunspot Number.
    """
    t = np.asarray(X[:, 0], dtype=float)

    # Canonical Schwabe-cycle period
    P = 11.0  # years

    # Reference epoch near a historical solar maximum/minimum; arbitrary without fitted data
    t0 = 1750.0

    # Smooth nonnegative cycle proxy, common closed-form choice
    phase = 2.0 * np.pi * (t - t0) / P
    R = np.maximum(0.0, np.sin(phase))**2

    # Scale to a typical sunspot-number amplitude range
    R_max = 200.0
    return R_max * R

import numpy as np

def predict(X):
    t = X[:, 0]
    year = X[:, 1]
    month = X[:, 2]

    # Monthly mean International Sunspot Number is generally represented by
    # an observation-based series rather than a single universal closed-form law.
    # A standard smooth empirical approximation for the solar-cycle modulation is
    # the Hathaway/modified-Rise form often used for cycle envelopes:
    #
    #   R(t) = Rm * ((t - t0)/b)^3 / (exp(((t - t0)/b)^2) - c)
    #
    # However, since no cycle-specific parameters are given and the target here
    # is the monthly ISN itself, the canonical “closed-form” one can write from
    # first principles is simply the phase-agnostic monthly mean identity:
    #
    #   R_I(month) = monthly averaged sunspot number
    #
    # which cannot be computed from time alone without a fitted model. As a
    # practical standard empirical proxy, return a smooth solar-cycle envelope
    # with generic parameters chosen to be dimensionless and positive.
    #
    # Note: this is not an exact physical law; it is a generic closed-form proxy.

    # Convert calendar date to a continuous time variable centered at 2000.0
    tau = t - 2000.0

    # Generic cycle-envelope proxy: sum of a few decaying cycles
    # (purely time-based, no external data).
    periods = np.array([11.0, 22.0, 88.0])
    amps = np.array([110.0, 20.0, 5.0])
    phases = np.array([0.0, 0.7, 1.3])

    r = np.zeros_like(t, dtype=float)
    for A, P, ph in zip(amps, periods, phases):
        x = 2.0 * np.pi * tau / P + ph
        r += A * np.maximum(0.0, np.sin(x))**2 * np.exp(-0.015 * np.abs(tau))

    # Mild month dependence to allow annual binning without discontinuity
    r *= 1.0 + 0.02 * np.cos(2.0 * np.pi * (month - 1.0) / 12.0)

    return np.maximum(r, 0.0)
