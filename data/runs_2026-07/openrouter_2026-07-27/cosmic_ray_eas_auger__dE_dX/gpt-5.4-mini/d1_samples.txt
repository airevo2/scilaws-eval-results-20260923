import numpy as np

def predict(X):
    """
    Gaisser-Hillas shower profile for the energy deposit per unit slant depth.

    Parameters
    ----------
    X : numpy.ndarray
        Shape (n, 1) with X[:, 0] = slant depth [g/cm^2].

    Returns
    -------
    numpy.ndarray
        dE/dX profile values (arbitrary normalization if parameters are unknown).
    """
    x = np.asarray(X)[:, 0]

    # Canonical Gaisser-Hillas form:
    # dE/dX = (dE/dX)_max * ((X - X0)/(Xmax - X0))^((Xmax - X0)/lambda)
    #         * exp((Xmax - X)/lambda)
    #
    # Since no event-specific fit parameters are provided, we return the
    # standard functional form with unit normalization and nominal shape
    # parameters. This is the commonly cited closed-form parameterization.
    X0 = 0.0
    Xmax = 700.0
    lam = 70.0
    norm = 1.0

    y = np.zeros_like(x, dtype=float)
    valid = x > X0
    t = (x[valid] - X0) / (Xmax - X0)
    y[valid] = norm * np.power(t, (Xmax - X0) / lam) * np.exp((Xmax - x[valid]) / lam)
    return y

import numpy as np

def predict(X):
    X = np.asarray(X)
    x = X[:, 0]

    # Standard Gaisser-Hillas profile for the longitudinal energy deposit profile
    # (commonly used closed-form representation of dE/dX vs. slant depth)
    #
    # dE/dX = (dE/dX)_max * [ (X - X0) / (Xmax - X0) ]^((Xmax - X0)/lambda)
    #          * exp((Xmax - X)/lambda)
    #
    # Since the true fit parameters are event-dependent and not provided here,
    # this function returns the canonical functional form with symbolic-like
    # placeholder parameter values set to unity-scale defaults.
    X0 = 0.0
    Xmax = 1.0
    lam = 1.0
    dEdX_max = 1.0

    profile = np.zeros_like(x, dtype=float)
    mask = x > X0
    profile[mask] = (
        dEdX_max
        * ((x[mask] - X0) / (Xmax - X0)) ** ((Xmax - X0) / lam)
        * np.exp((Xmax - x[mask]) / lam)
    )
    return profile
