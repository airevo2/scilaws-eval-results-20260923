    #

def predict(X):
    """
    Hydrogen electrocatalysis on Pt: Butler–Volmer / Tafel-form kinetics for HOR/HER
    with exchange current density treated as a function of temperature, H2 pressure,
    and acid concentration.

    Inputs:
        X[:, 0] = eta      [V vs RHE]
        X[:, 1] = T        [K]
        X[:, 2] = P_H2     [bar]
        X[:, 3] = c_acid   [mol dm^-3]

    Returns:
        j [A cm^-2] positive anodic (HOR), negative cathodic (HER)
    """
    eta = X[:, 0]
    T = X[:, 1]
    P_H2 = X[:, 2]
    c_acid = X[:, 3]

    # Physical constants
    F = 96485.33212      # C mol^-1
    R = 8.314462618      # J mol^-1 K^-1

    # Canonical reversible HOR/HER kinetics on Pt:
    # j = j0 * [exp((1-alpha)F eta/RT) - exp(-alpha F eta/RT)]
    #
    # For a symmetric barrier on Pt, alpha ≈ 0.5 is the most common choice.
    alpha = 0.5

    # Standard commonly used empirical dependence of exchange current density
    # for HOR/HER on Pt on reactant/product activities:
    # j0 ∝ (P_H2)^1/2 * c_H+ * exp(-Ea/RT)
    #
    # Since no data are provided, use a commonly cited Arrhenius-type
    # semi-empirical form with reasonable Pt/HOR-HER scale:
    # j0_ref ~ 1e-3 A cm^-2 at 298 K, 1 bar H2, 1 M acid
    # Ea ~ 20 kJ/mol (representative order of magnitude)
    j0_ref = 1e-3       # A cm^-2
    Ea = 2.0e4          # J mol^-1

    # Normalize activities to the reference state.
    # Proton activity in strong acids is often treated approximately as proportional
    # to concentration for a simple closed-form expression.
    j0 = j0_ref * np.sqrt(P_H2) * c_acid * np.exp(-Ea / R * (1.0 / T - 1.0 / 298.15))

    # Butler-Volmer current density

              np.exp(-alpha * F * eta / (R * T)))

    return j

import numpy as np

def predict(X):
    eta = X[:, 0]
    T = X[:, 1]
    P_H2 = X[:, 2]
    c_acid = X[:, 3]

    R = 8.31446261815324  # J mol^-1 K^-1
    F = 96485.33212       # C mol^-1

    # Common closed-form expression for HOR/HER on Pt:
    # Butler–Volmer with exchange current density depending on T, PH2, and acid concentration.
    # Here j is per real Pt area, positive anodic (HOR), negative cathodic (HER).
    #
    # j = j0(T, P_H2, c_acid) * [exp(alpha_a F eta / RT) - exp(-alpha_c F eta / RT)]
    #
    # A widely used symmetry choice for Pt hydrogen kinetics is alpha_a = alpha_c = 0.5.
    alpha = 0.5

    # Standard-state exchange current density scaling:
    # j0 ∝ P_H2^1/2 * a_H+   (for the hydrogen electrode reaction in acid)
    # with Arrhenius temperature dependence.
    # Since no data are provided, use the canonical closed-form dependence only.
    # j0_ref is an arbitrary prefactor; for a pure physics-based form, keep it as 1.
    j0_ref = 1.0

    # Normalized acid activity approximation in strong acid:
    # a_H+ ~ c_acid (common engineering approximation)
    j0 = j0_ref * np.sqrt(np.maximum(P_H2, 0.0)) * np.maximum(c_acid, 0.0) * np.exp(0.0 / T)

    j = j0 * (
        np.exp(alpha * F * eta / (R * T)) -
        np.exp(-alpha * F * eta / (R * T))
    )

    return j
