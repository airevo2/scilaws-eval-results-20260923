import numpy as np

def predict(X):
    M = np.asarray(X)[:, 0]

    # Commonly used rough mass-radius scaling for self-gravitating bodies:
    #   R ~ M^(1/3)
    # with normalization chosen so Earth (M=1) gives R=1 in Earth units.
    return M**(1.0 / 3.0)

import numpy as np

def predict(X):
    M = np.asarray(X)[:, 0]

    # Common closed-form approximation: radius is nearly constant for rocky/small
    # planets, grows weakly with mass for gas/ice giants, and then flattens again.
    # Since only mass is provided, use a simple broken power-law fit consistent
    # with the broad empirical mass-radius trend.

    R = np.where(
        M < 2.0,
        M**0.279,                 # low-mass rocky regime
        np.where(
            M < 132.0,
            1.08 * M**0.59,       # sub-Neptune / giant-planet transition
            17.74 * M**(-0.044)   # brown dwarf / high-mass flattening
        )
    )

    return R

import numpy as np

def predict(X):
    M = np.asarray(X)[:, 0]

    # Common closed-form mass-radius scaling for substellar objects:
    # R / R_earth ≈ 11.2 * (M / M_Jup)^(-1/8) for giant planets / brown dwarfs
    Mjup_in_Mearth = 317.82838
    return 11.2 * (M / Mjup_in_Mearth) ** (-1.0 / 8.0)

import numpy as np

def predict(X):
    M = np.asarray(X)[:, 0]
    
    # Simple empirical mass-radius relation, common single-parameter form:
    # R / R_earth = (M / M_earth)^(1/3)
    return np.power(M, 1.0 / 3.0)
