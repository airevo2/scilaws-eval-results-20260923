import numpy as np

def predict(X):
    P = X[:, 0].astype(float)  # laser power [W]
    V = X[:, 1].astype(float)  # scan velocity [mm/s]
    D = X[:, 2].astype(float)  # beam diameter [um]

    # Common conduction-mode scaling for melt pool depth:
    # depth ~ k * sqrt( (P/V) / D )
    # with k as an empirical constant depending on absorptivity, thermal properties, and units.
    k = 1.0
    d = k * np.sqrt((P / V) / D)

    return d
