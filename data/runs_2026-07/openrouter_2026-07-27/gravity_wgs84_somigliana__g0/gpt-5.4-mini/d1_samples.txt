import numpy as np

def predict(X):
    """
    WGS84 normal gravity on the ellipsoid surface (Somigliana formula).

    Parameters
    ----------
    X : np.ndarray
        X[:, 0] = geodetic latitude in degrees.

    Returns
    -------
    np.ndarray
        Normal gravity gamma in m/s^2.
    """
    phi = np.deg2rad(X[:, 0])

    # WGS84 constants
    gamma_e = 9.7803253359  # m/s^2, normal gravity at equator
    k = 0.00193185265241    # Somigliana constant
    e2 = 0.00669437999013   # first eccentricity squared

    sin_phi = np.sin(phi)
    sin2 = sin_phi * sin_phi

    gamma = gamma_e * (1.0 + k * sin2) / np.sqrt(1.0 - e2 * sin2)
    return gamma
