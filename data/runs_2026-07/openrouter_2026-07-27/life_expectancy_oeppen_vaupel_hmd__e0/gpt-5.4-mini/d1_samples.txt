import numpy as np

def predict(X):
    # Simple closed-form demographic approximation of period life expectancy at birth
    # as a monotone logistic trend over calendar time.
    t = np.asarray(X)[:, 0].astype(float)
    
    # Midpoint and slope chosen to reflect the long-run historical rise in life expectancy.
    # Asymptotes approximate plausible lower/upper bounds for HMD total-sex e0.
    L_low = 15.0
    L_high = 90.0
    t0 = 1950.0
    k = 0.03
    
    return L_low + (L_high - L_low) / (1.0 + np.exp(-k * (t - t0)))

import numpy as np

def predict(X):
    t = np.asarray(X)[:, 0]
    # Standard demographic/logistic-style closed form for life expectancy over time:
    # e0(t) = L / (1 + exp(-k * (t - t0)))
    # Chosen as a generic canonical saturation curve for period life expectancy.

    k = 0.03

    return L / (1.0 + np.exp(-k * (t - t0)))
