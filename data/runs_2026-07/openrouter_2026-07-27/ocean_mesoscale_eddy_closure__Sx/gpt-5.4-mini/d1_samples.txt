import numpy as np

def predict(X):
    u = X[:, 0]
    v = X[:, 1]
    zeta = X[:, 2]
    D = X[:, 3]
    Dtilde = X[:, 4]
    d_zeta2_dx = X[:, 5]
    d_zetaD_dx = X[:, 6]
    d_zetaDtilde_dy = X[:, 7]

    # Standard coarse-grained 2D QG SGS momentum forcing in the zonal direction
    # based on divergence of the subgrid stress tensor:
    #
    # S_x = -1/2 * [ ∂x(u'^2 - v'^2) + 2 ∂y(u'v') ]
    #
    # In the common isotropic/stress-invariant representation using velocity-gradient
    # invariants, this reduces to a canonical linear combination of enstrophy and
    # strain-gradient terms:
    Sx = (
        -0.25 * d_zeta2_dx
        - 0.5 * d_zetaD_dx
        - 0.5 * d_zetaDtilde_dy
    )

    return Sx
