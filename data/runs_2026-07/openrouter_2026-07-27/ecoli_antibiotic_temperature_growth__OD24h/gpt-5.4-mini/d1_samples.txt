import numpy as np

def predict(X):
    T = np.asarray(X)[:, 0].astype(float)

    # A standard closed-form temperature-response shape: Gaussian peak near optimal growth temperature.
    # Since no fitted coefficients are provided, use a biologically plausible generic form.
    A = 1.0      # peak OD scale
    Topt = 37.0  # typical E. coli optimum
    sigma = 7.0  # broadness of temperature response

    return A * np.exp(-0.5 * ((T - Topt) / sigma) ** 2)

    # a peaked (approximately Gaussian) dependence on temperature with an optimum near 37°C.
    # Since only temperature is provided, use a generic bell-shaped empirical form for OD after 24 h.
    ODmax = 1.0
    Topt = 37.0
    sigma = 7.0
    y = ODmax * np.exp(-0.5 * ((T - Topt) / sigma) ** 2)
