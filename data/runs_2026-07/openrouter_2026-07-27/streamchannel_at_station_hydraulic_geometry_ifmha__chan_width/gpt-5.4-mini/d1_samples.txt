import numpy as np

def predict(X):
    Q = np.asarray(X)[:, 0]
    # Common hydraulic geometry form: w = a * Q^b
    # Representative alluvial-stream scaling (general-purpose):
    a = 6.0
    b = 0.5
    return a * np.power(Q, b)
