import numpy as np

def predict(X):
    D = np.asarray(X)[:, 0]
    # Canonical pantropical tree height-diameter relationship (Chave et al.-style)
    # H = 1.3 + a * (1 - exp(-b * D))^c
    a = 60.0
    b = 0.06
    c = 1.0
    return 1.3 + a * (1.0 - np.exp(-b * D)) ** c
