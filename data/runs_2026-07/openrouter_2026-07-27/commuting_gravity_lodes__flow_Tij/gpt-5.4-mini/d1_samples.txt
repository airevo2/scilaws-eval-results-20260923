import numpy as np

def predict(X):
    """
    Commonly used gravity-style commuting model:
        log10(T_ij + 1) = a + b1*log10(m_i) + b2*log10(m_j) + b3*log10(d_ij) + b4*log10(s_ij + 1)
    using a simple canonical parameterization with positive attraction from masses,
    negative distance friction, and positive intervening-opportunity effect.
    """
    X = np.asarray(X, dtype=float)

    m_i = X[:, 0]
    m_j = X[:, 1]
    d_ij = X[:, 2]
    s_ij = X[:, 3]

    # Canonical closed-form gravity / intervening-opportunities hybrid
    # (chosen as a standard, commonly cited functional form)
    a = -1.0
    b_mi = 0.5
    b_mj = 0.5
    b_d = -1.0
    b_s = 0.25

    y = (
        a
        + b_mi * np.log10(m_i)
        + b_mj * np.log10(m_j)
        + b_d * np.log10(d_ij)
        + b_s * np.log10(s_ij + 1.0)
    )
    return y
