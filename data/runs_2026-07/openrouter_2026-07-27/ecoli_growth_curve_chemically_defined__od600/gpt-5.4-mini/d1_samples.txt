import numpy as np

def predict(X):
    t = X[:, 0].astype(float)

    # Canonical closed-form growth curve: logistic model
    # OD(t) = K / (1 + ((K - OD0) / OD0) * exp(-r * t))
    #
    # Since no calibration data were provided, use a generic E. coli-like
    # parameterization that yields a plausible sigmoidal OD trajectory.
    OD0 = 0.03   # initial OD600
    K = 1.2      # carrying capacity / plateau OD600
    r = 0.9      # specific growth rate (1/h)

    return K / (1.0 + ((K - OD0) / OD0) * np.exp(-r * t))
