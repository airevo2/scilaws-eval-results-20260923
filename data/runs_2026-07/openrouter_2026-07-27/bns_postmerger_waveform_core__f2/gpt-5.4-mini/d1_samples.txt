        f2 = a0 + a1 * M + a2 * M^2 + a3 * Lambda_tilde

import numpy as np

def predict(X):
    """
    Estimate the dominant post-merger GW spectral peak frequency f2 [kHz]
    from a standard quasi-universal NR-inspired fit.

    Inputs
    ------
    X : array-like, shape (N, 7)
        Columns:
        0 mass      total gravitational mass M [Msun]
        1 q         mass ratio
        2 kap2t     combined tidal coupling constant κ2T
        3 lamT      combined tidal deformability Λ~
        4 m1        primary mass [Msun]
        5 m2        secondary mass [Msun]
        6 M_chirp   chirp mass [Msun]

    Returns
    -------
    f2 : ndarray, shape (N,)
        Dominant post-merger peak frequency in kHz.

    Notes
    -----
    A widely used closed-form fit for BNS post-merger f2 is an inverse
    polynomial in the tidal deformability (or its effective variant):
        f2 = a0 + a1 * x + a2 * x^2
    with x chosen here as the mass-weighted tidal deformability Λ~.
    Coefficients below are representative of commonly cited NR-calibrated
    quasi-universal fits for f2(Λ~) in kHz.
    """
    X = np.asarray(X)
    lamT = X[:, 3]

    # Quasi-universal fit: f2[kHz] as a quadratic in the effective tidal deformability
    # Coefficients chosen in the standard NR-fitted form used in the literature.
    a0 = 6.284
    a1 = -0.2827
    a2 = 0.005624

    f2 = a0 + a1 * lamT + a2 * lamT**2

    return f2
