import numpy as np

def predict(X):
    h = np.asarray(X)[:, 0]  # matric suction head [m], positive
    # van Genuchten (1980) soil water retention curve
    theta_r = 0.05
    theta_s = 0.45
    alpha = 1.0  # 1/m
    n = 1.6
    m = 1.0 - 1.0 / n

    Se = 1.0 / np.power(1.0 + np.power(alpha * h, n), m)
    theta = theta_r + (theta_s - theta_r) * Se
    return theta

    # Standard van Genuchten water retention curve:
    # theta(h) = theta_r + (theta_s - theta_r) / [1 + (alpha * h)^n]^m
    # with the common constraint m = 1 - 1/n
    theta_r = 0.0
    theta_s = 1.0
    alpha = 1.0
    n = 1.0
    m = 1.0 - 1.0 / n if n != 0 else 0.0

    theta = theta_r + (theta_s - theta_r) / (1.0 + (alpha * h) ** n) ** m
    return theta

import numpy as np

def predict(X):
    h = np.asarray(X[:, 0], dtype=float)

    # van Genuchten (1980) soil water retention curve
    # theta(h) = theta_r + (theta_s - theta_r) / [1 + (alpha * h)^n]^m
    # with m commonly taken as 1 - 1/n
    theta_r = 0.0
    theta_s = 1.0
    alpha = 1.0
    n = 1.6
    m = 1.0 - 1.0 / n

    theta = theta_r + (theta_s - theta_r) / np.power(1.0 + np.power(alpha * h, n), m)
    return theta
