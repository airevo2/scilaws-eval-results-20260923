import numpy as np

def predict(X):
    """
    Net CO2 assimilation rate A_n as a function of incident PPFD using the
    standard non-rectangular hyperbola light-response model.

    X[:, 0] = Qin (incident PPFD)
    """
    Q = np.asarray(X)[:, 0]

    # Canonical light-response parameters (general standard form).
    # These are placeholders / generic physiological defaults, since no fit data are provided.
    Amax = 12.0   # light-saturated net assimilation (umol CO2 m-2 s-1)
    phi = 0.05    # apparent quantum yield (umol CO2 per umol photons)
    Rd = 1.0      # dark respiration (umol CO2 m-2 s-1, positive magnitude)
    theta = 0.7   # curvature parameter (0<theta<=1)

    # Non-rectangular hyperbola:
    # A = [phi*Q + Amax - sqrt((phi*Q + Amax)^2 - 4*theta*phi*Q*Amax)] / (2*theta) - Rd
    term = phi * Q + Amax
    disc = np.maximum(term * term - 4.0 * theta * phi * Q * Amax, 0.0)
    A = (term - np.sqrt(disc)) / (2.0 * theta) - Rd

    return A
