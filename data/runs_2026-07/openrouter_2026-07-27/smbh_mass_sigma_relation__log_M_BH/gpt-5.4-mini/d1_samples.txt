import numpy as np

def predict(X):
    """
    Predict log10(M_BH/M_sun) from the canonical M_BH-sigma relation.

    Uses the commonly cited form for dynamically measured supermassive
    black holes:
        log10(M_BH/M_sun) = alpha + beta * log10(sigma / 200 km/s)

    A widely used calibration is from Kormendy & Ho (2013) for classical
    bulges and ellipticals:
        alpha = 8.49
        beta  = 4.38
    """
    sigma = np.asarray(X)[:, 0]
    return 8.49 + 4.38 * np.log10(sigma / 200.0)

import numpy as np

def predict(X):
    """
    Predict log10(M_BH / M_sun) using the standard M_BH-sigma relation.
    
    Parameters
    ----------
    X : np.ndarray
        2-D array with columns:
        X[:, 0] = sigma0 [km/s]
        X[:, 1] = log_M_sph
        X[:, 2] = B_T
        X[:, 3] = Pseudobulge
    """
    sigma = np.asarray(X)[:, 0]
    # Commonly used canonical form for dynamically measured BH masses:
    # log10(M_BH/M_sun) = 8.32 + 5.64 * log10(sigma / 200 km/s)
    return 8.32 + 5.64 * np.log10(sigma / 200.0)
