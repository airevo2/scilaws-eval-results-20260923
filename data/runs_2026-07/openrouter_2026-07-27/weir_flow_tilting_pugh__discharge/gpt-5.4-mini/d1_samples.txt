import numpy as np

def predict(X):
    h = X[:, 0]
    p = X[:, 1]
    b = X[:, 2]

    # Standard sharp-crested rectangular weir discharge with end-contraction correction
    # Francis-type form:
    # Q = (2/3) * C_d * b * sqrt(2g) * h^(3/2)
    # Commonly written in laboratory/engineering practice as:
    # Q = 1.84 * b * (h + 0.0015)**1.5   (SI units, for a sharp-crested rectangular weir)
    # Here, using the canonical closed-form with a typical discharge coefficient.
    g = 9.81
    Cd = 0.611  # standard sharp-crested weir coefficient in SI

    Q = (2.0 / 3.0) * Cd * b * np.sqrt(2.0 * g) * np.power(h, 1.5)
    return Q
