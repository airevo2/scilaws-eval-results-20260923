def predict(X):
    E = np.asarray(X)[:, 0].astype(float)

    # Lotz-type semi-empirical form for total electron-impact ionization cross section
    # σ(E) = A * ln(E/I) / (E * I) * [1 - B * exp(-C*(E/I - 1))]
    # Here we use a standard compact single-threshold closed form with the correct threshold behavior.
    #
    # Since no target-specific constants are provided, we return the canonical shape in
    # units of 1e-16 cm^2 using a generic ionization-threshold scale I0.
    I0 = 15.0  # eV, representative single-ionization threshold scale
    A = 4.5    # empirical prefactor in 1e-16 cm^2 * eV
    B = 1.0
    C = 1.0

    sigma = np.zeros_like(E, dtype=float)
    mask = E > I0
    x = E[mask] / I0
    sigma[mask] = A * np.log(x) / (E[mask] * I0) * (1.0 - B * np.exp(-C * (x - 1.0)))

    return sigma
