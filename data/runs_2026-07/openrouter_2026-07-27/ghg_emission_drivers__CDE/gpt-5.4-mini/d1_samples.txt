    DP  = X[:, 3]
    UR  = X[:, 4]

    b2 = 0.85

    b4 = 0.10
    b5 = 0.30
    b6 = -0.02

        b0

import numpy as np

def predict(X):
    # Kaya identity / IPAT-style standard closed-form approximation:
    # CO2 emissions = population × affluence × emissions intensity
    # Here we use GDP per capita as the affluence term and a simple
    # multiplicative form for national fossil CO2 emissions.
    gdp_pc = X[:, 0]
    pop = X[:, 1]
    
    # Emissions intensity proxy calibrated generically as proportional to economic activity
    # (no data provided, so use the standard structural form only).
    #
    # Since the quantity is national annual emissions in kg CO2, and GDP per capita is in
    # constant 2015 USD, this returns emissions in proportional units consistent with the
    # canonical closed-form relationship.
    return pop * gdp_pc
