import numpy as np

def predict(X):
    H = np.asarray(X)[:, 0]
    # Standard exponential NEO cumulative magnitude distribution:
    # N(<H) = N0 * 10^(alpha * (H - H0))
    # Using a commonly cited normalization for near-Earth objects in the complete regime.
    N0 = 3.8
    alpha = 0.35
    H0 = 13.0
    return N0 * 10.0 ** (alpha * (H - H0))
