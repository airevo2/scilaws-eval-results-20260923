import numpy as np

def predict(X):
    t = X[:, 0]
    # Gompertz growth model:
    # V(t) = V0 * exp( (ln(V_inf / V0)) * (1 - exp(-k * t)) )
    V0 = 1.0
    V_inf = 1.0
    k = 0.1
    return V0 * np.exp(np.log(V_inf / V0) * (1.0 - np.exp(-k * t)))
