import numpy as np

def predict(X):
    """
    Standard recession-curve form: exponential decline from initial discharge.
    
    Q(t) = Q0 * exp(-a * t)
    
    where a is a recession constant (chosen here as a common generic value).
    """
    t = X[:, 0]
    Q0 = X[:, 1]
    
    a = 0.1  # generic recession constant per day
    return Q0 * np.exp(-a * t)
