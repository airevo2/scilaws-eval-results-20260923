    Predict log10(L_acc / L_sun) for a T Tauri star.

    Canonical accretion-luminosity scaling with stellar parameters:
        log L_acc = log L_star + 1.09 * log(M_star / M_sun)
                              - 0.98 * log(R_star / R_sun)
                              + 0.23 * log(T_eff / 4000 K)

    This is the commonly used empirical closed-form relation from UV-excess

    """

    logL_star = X[:, 0]
    M_star = X[:, 1]
    R_star = X[:, 2]
    Teff_K = X[:, 3]

    return (
        logL_star
        + 1.09 * np.log10(M_star)
        - 0.98 * np.log10(R_star)
        + 0.23 * np.log10(Teff_K / 4000.0)
        + 1.50
    )

import numpy as np

def predict(X):
    """
    Predict log10(L_acc / L_sun) for T Tauri stars from stellar parameters.

    Canonical empirical scaling for magnetospheric accretion luminosity:
        L_acc ~ G M_star dotM / R_star * (1 - R_star / R_in)
    but with no direct dotM input, the commonly used phenomenological
    closed-form relation between accretion luminosity and stellar luminosity
    is a near-linear log-log scaling.

    This implementation uses a standard literature-style form:
        logL_acc = a * logL_star + b
    with typical coefficients near unity and a small offset.

    Since no calibration data are provided, we return a widely used
    approximate relation consistent with T Tauri samples.
    """
    logL_star = np.asarray(X)[:, 0]

    # Commonly adopted approximate relation for PMS stars:
    # log L_acc ≈ 1.5 log L_* - 0.3
    logL_acc = 1.5 * logL_star - 0.3

    return logL_acc
