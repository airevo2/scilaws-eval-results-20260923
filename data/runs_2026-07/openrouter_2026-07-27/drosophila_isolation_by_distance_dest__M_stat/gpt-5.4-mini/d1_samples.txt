import numpy as np

def predict(X):
    # Rousset's linearized differentiation statistic is commonly modeled as increasing
    # approximately linearly with geographic distance under isolation by distance.
    # A standard simple closed-form form is:
    #   M = a + b * ln(d)
    # where d is distance in km. Since no fitted coefficients are provided, we return
    # the canonical unparameterized form with unit slope and zero intercept.
    log_distance_km = X[:, 0]
    M = log_distance_km
    return M
