    #   S(p) ≈ exp(a + b*ln p + c*t + region FE)
    # Here we use the most standard closed-form approximation for log top share:
    #   ln S(p) = alpha + beta * ln(p)
    # with optional weak time/region adjustments omitted as generic placeholders.
    #
    # Since no data are provided, this is a standard theoretical/log-linear form,
    # not a fitted empirical WID-specific model.

    log_p = X[:, 0].astype(float)
    year = X[:, 2].astype(float)
    region = X[:, 3]

    # Baseline Pareto-log share elasticity:
    # In top-share applications, beta is typically positive and around 0.7-1.0
    # depending on income definition and upper-tail concentration.
    alpha = -0.35
    beta = 0.85

    # Mild secular trend placeholder centered around 2000
    gamma = 0.002
    year_centered = year - 2000.0

    # Region fixed effects (generic placeholders for WID macro-regions)
    region_fe = {
        "Europe": 0.00,
        "Asia": -0.03,
        "Africa": -0.05,
        "Americas": 0.02,
        "Oceania": -0.01,
    }

    # Support both string-like and numeric region codes by falling back to 0.0
    reg_adj = np.array([region_fe.get(str(r), 0.0) for r in region], dtype=float)

    return alpha + beta * log_p + gamma * year_centered + reg_adj

def predict(X):
    # Standard Pareto-style approximation for top income shares:
    # ln S(p) = a(t, r) + b(t, r) * ln(p)
    # For a simple closed-form baseline, use a linear form in ln(p), year, and region.
    # Region is encoded as integers; if your coding differs, adjust the mapping below.
    x0 = X[:, 0]  # log_p_above
    x2 = X[:, 2]  # year
    x3 = X[:, 3]  # region code

    # Center year for numerical stability
    t = x2 - 2000.0

    # Region effects: Europe, Asia, Africa, Americas, Oceania
    # Assumes region codes are 0..4 in that order.
    region_effect = np.zeros_like(t, dtype=float)
    region_effect = np.where(x3 == 0, 0.000, region_effect)  # Europe
    region_effect = np.where(x3 == 1, 0.020, region_effect)  # Asia
    region_effect = np.where(x3 == 2, 0.040, region_effect)  # Africa
    region_effect = np.where(x3 == 3, 0.015, region_effect)  # Americas
    region_effect = np.where(x3 == 4, 0.010, region_effect)  # Oceania

    # Canonical semilog specification:
    # top share rises as p falls, and varies smoothly over time/region.
    return (
        -0.85
        + 0.92 * x0
        + 0.010 * t
        + region_effect
    )

import numpy as np

def predict(X):
    """
    Predict log top income share ln S(p) using a standard Pareto-style closed form.

    Canonical form:
        S(p) = A(t, r) * p^alpha(t, r)
    so
        ln S(p) = ln A(t, r) + alpha(t, r) * ln(p)

    Since no data are provided, this uses a standard flexible empirical specification:
        ln S(p) = beta0 + beta1 * ln(p) + beta2 * year + region fixed effects
                  + beta3 * ln(p) * year

    Notes:
    - top_pct is accepted as an equivalent scale for p; if log_p_above is unavailable,
      one could use ln(top_pct / 100). Here we rely on the provided log_p_above.
    - region is treated as a categorical code and mapped to fixed effects by sorted unique values.
    """
    X = np.asarray(X)

    log_p = X[:, 0].astype(float)
    top_pct = X[:, 1].astype(float)
    year = X[:, 2].astype(float)
    region = X[:, 3]

    # Use log_p_above directly; if it contains missing/infinite values, fall back to top_pct
    log_p_fallback = np.log(np.clip(top_pct / 100.0, 1e-300, 1.0))
    log_p = np.where(np.isfinite(log_p), log_p, log_p_fallback)

    # Center year for numerical stability and to represent slow secular shifts
    t = year - 2000.0

    # Region fixed effects: deterministic mapping from sorted unique codes
    uniq_regions = np.unique(region)
    uniq_regions = np.sort(uniq_regions.astype(str))

    # Plausible baseline fixed effects by broad WID region category
    # (unknown numeric coding is mapped to these broad effects in sorted order)
    region_fe_values = np.array([
        0.00,  # first region category
        -0.08, # second
        -0.12, # third
        0.06,  # fourth
        0.02   # fifth
    ], dtype=float)

    region_str = region.astype(str)
    region_index = np.zeros(region_str.shape[0], dtype=int)
    for i, r in enumerate(uniq_regions):
        region_index[region_str == r] = i
    region_index = np.clip(region_index, 0, len(region_fe_values) - 1)
    region_fe = region_fe_values[region_index]

    # Standard top-income-share shape: decreasing in p, slightly time-varying slope
    beta0 = -0.75
    beta1 = 0.62
    beta2 = 0.010
    beta3 = 0.0025

    # A mild convexity correction sometimes used in empirical fits
    beta4 = -0.015
    log_p2 = log_p * log_p

    y = beta0 + beta1 * log_p + beta2 * t + beta3 * log_p * t + beta4 * log_p2 + region_fe
    return y

def predict(X):

    #   S(p) ≈ p^(1 - 1/α)
    # so
    #   ln S(p) = (1 - 1/α) * ln p
    # A common empirical default for top-income shares is α ≈ 1.5,
    # giving ln S(p) ≈ (1/3) * ln p.
    log_p = X[:, 0]
    alpha = 1.5
    return (1.0 - 1.0 / alpha) * log_p

import numpy as np

def predict(X):
    """
    Predict log top income share ln S(p) using a standard Pareto-style form.

    Inputs:
      X[:, 0] = log_p_above = ln(p)
      X[:, 1] = top_pct = 100*p
      X[:, 2] = year
      X[:, 3] = region code (ignored here)

    Canonical closed-form relationship:
      ln S(p) = a + b * ln(p)

    For a Pareto upper tail, the top share scales approximately as a power
    of the cutoff p, so a linear function in ln(p) is the standard form.
    """
    p = np.exp(X[:, 0])
    # Standard Pareto-like approximation: S(p) ≈ p^(1 - 1/α)
    # Using a common stylized exponent for top-income shares.
    a = 0.0
    b = 0.75
    return a + b * np.log(p)
