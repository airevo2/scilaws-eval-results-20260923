def predict(X):

    T_in = X[:, 1]
    T_amb = X[:, 2]

    # Reversed-order Hottel-Whillier-Bliss / ISO 9806-style quadratic efficiency curve
    # eta = eta0 - a1 * ((T_in - T_amb)/G) - a2 * ((T_in - T_amb)^2 / G)
    # Coefficients are representative of a high-performance solar-thermal collector field.
    eta0 = 0.75
    a1 = 1.70
    a2 = 0.010

    deltaT = T_in - T_amb
    eta = eta0 - a1 * (deltaT / G) - a2 * (deltaT * deltaT / G)

    return eta

import numpy as np

def predict(X):
    G = X[:, 0]
    T_in = X[:, 1]
    T_amb = X[:, 2]
    
    # Common flat-plate/collector-array efficiency form:
    # eta = eta0 - a1 * ((T_in - T_amb)/G) - a2 * ((T_in - T_amb)^2 / G)
    # Using a simple canonical reduced-temperature quadratic form.
    dT = T_in - T_amb
    x = dT / G
    
    eta0 = 0.62
    a1 = 3.20
    a2 = 0.010
    
    eta = eta0 - a1 * x - a2 * x * x
    return eta
