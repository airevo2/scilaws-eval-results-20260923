import numpy as np

def predict(X):
    # S_terceira: Arthropod species richness per 1-ha pitfall-trap site on Terceira Island, Azores
    d = X[:, 0]   # anthropogenic habitat-modification index
    h = X[:, 1]   # altitude
    p = X[:, 2]   # annual precipitation
    t = X[:, 3]   # annual temperature
    sl = X[:, 4]  # slope

    # Standard closed-form ecological richness model: multiplicative power-law
    # (species richness is commonly modeled as an exponential / power function of environmental gradients)
    return np.exp(
        3.401
        - 0.0123 * d
        - 0.00042 * h
        + 0.00019 * p
        + 0.156 * t
        + 0.018 * sl
    )
