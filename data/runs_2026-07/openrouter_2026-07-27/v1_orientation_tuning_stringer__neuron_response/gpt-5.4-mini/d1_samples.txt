import numpy as np

def predict(X):
    """
    Orientation tuning curve: circular Gaussian / von Mises form.
    X[:, 0] = orientation in degrees, folded to [0, 180)
    Returns mean response R(theta).
    """
    theta = np.deg2rad(X[:, 0])
    
    # Canonical 180-deg periodic tuning centered at pref_deg
    pref_deg = 90.0
    pref = np.deg2rad(pref_deg)
    
    # Free parameters as generic constants
    R0 = 0.0      # baseline
    A = 1.0       # amplitude
    kappa = 2.0   # concentration / tuning sharpness
    
    # 180-degree periodic von Mises / circular Gaussian on orientation
    return R0 + A * np.exp(kappa * np.cos(2.0 * (theta - pref)))
