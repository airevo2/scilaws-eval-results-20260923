import numpy as np

def predict(X):
    N = X[:, 0].astype(float)
    # Self-thinning / density-dependent size scaling:
    # mean mass per plant declines approximately as a power law with density.
    # w = a * N^b
    a = 1.0
    b = -0.5
    return a * np.power(N, b)

import numpy as np

def predict(X):
