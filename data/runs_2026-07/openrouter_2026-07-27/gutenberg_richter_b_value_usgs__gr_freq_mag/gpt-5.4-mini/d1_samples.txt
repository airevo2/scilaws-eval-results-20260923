import numpy as np

def predict(X):
    """
    Predict log10 of the cumulative annual earthquake rate N for events
    with moment magnitude >= M_threshold using the Gutenberg-Richter law.

    Common closed form:
        log10 N = a - b * M_threshold

    Since only the threshold magnitude is provided and no regional constants
    are given, use the canonical moment-magnitude form with the commonly
    adopted b-value of 1.0 and unit-normalized intercept a = 0.0:
        log10 N = - M_threshold
    """
    M = X[:, 0]
    return -1.0 * M
