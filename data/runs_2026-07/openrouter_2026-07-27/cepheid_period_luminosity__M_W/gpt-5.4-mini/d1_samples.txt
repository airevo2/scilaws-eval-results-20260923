import numpy as np

def predict(X):
    logP = X[:, 0]
    feh = X[:, 1]
    
    # Canonical linear PLZ form for an absolute Wesenheit magnitude:
    # M_W = a + b*(log P) + c*[Fe/H]
    # For Gaia Wesenheit relations, a commonly used reddening-free form is:
    # W = G - 1.90*(G_BP - G_RP)
    # with an absolute calibration of the form below.
    a = -1.0
    b = -2.7
    c = 0.15
    
    return a + b * logP + c * feh
