import numpy as np

def predict(X):
    m = X[:, 0]
    S = X[:, 1]
    
    # Wing loading
    W_S = m / S  # kg/m^2
    
    # Empirical allometric relationship for mean equivalent airspeed (m/s)
    # Ue ≈ 16.5 * (m / S) ^ 0.5
    Ue = 16.5 * np.sqrt(W_S)
    
    return Ue

import numpy as np

def predict(X):
    m = X[:, 0]
    S = X[:, 1]
    
    # Standard scaling relationship for equivalent airspeed in flapping birds:
    # Ue ≈ k * sqrt(m / S)
    # with k chosen to give a typical sea-level equivalent airspeed in m/s.

    return k * np.sqrt(m / S)
