import numpy as np

def predict(X):
    K = X[:, 0]
    L = X[:, 1]
    h = X[:, 2]
    
    # Canonical Cobb-Douglas per-worker form with human capital:
    # Y/L = A * (K/L)^alpha * h^(1-alpha)
    # Using the standard textbook calibration alpha = 1/3.
    alpha = 1.0 / 3.0
    
    return np.log(K / L) * alpha + np.log(h) * (1.0 - alpha)
