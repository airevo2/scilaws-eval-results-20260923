import numpy as np

def predict(X):
    P = np.asarray(X)[:, 0]
    
    # Langmuir isotherm (most common closed-form for equilibrium adsorption)
    q_max = 1.0   # mmol/g
    b = 1.0       # bar^-1
    
    q = (q_max * b * P) / (1.0 + b * P)
    return q
