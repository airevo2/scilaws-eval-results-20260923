def predict(X):
    X = np.asarray(X)
    GDP = X[:, 0]
    TP = X[:, 1]
    AP = X[:, 2]
    DP = X[:, 3]
    UR = X[:, 4]
    AT = X[:, 5]

    # Standard reduced-form CH4 emissions relationship (IPAT-style, multiplicative),
    # written in a smooth closed form and returning emissions in kg CH4.
    # Log-linear elasticities are not fixed from data here, so use the canonical
    # structural form: scale with population, affluence, urbanization, density,
    # working-age share, and temperature.
    #
    # Since no fitted coefficients are provided, this is a generic closed-form
    # proxy commonly used in environmental-economics applications.
    #
    # Units: kg CH4 / year
    E = (
        TP
        * np.power(GDP, 0.55)
        * np.power(AP / 100.0, 0.30)
        * np.power(DP, 0.10)
        * np.power(UR / 100.0, 0.20)
        * np.exp(0.08 * AT)
        * 1e-3
    )
    return E

def predict(X):
    # Common log-linear (Kaya-style / STIRPAT-like) closed-form specification
    # for national methane emissions:
    # ln(ME) = a + b1 ln(GDP) + b2 ln(PP) + b3 ln(AP) + b4 ln(DP) + b5 ln(UR) + b6 AT
    # Since no fitted coefficients are provided, this returns a standard
    # structurally consistent form with unit elasticities as a canonical baseline.
    GDP = X[:, 0]
    TP  = X[:, 1]
    AP  = X[:, 2]
    DP  = X[:, 3]
    UR  = X[:, 4]
    AT  = X[:, 5]

    eps = 1e-12
    ln_me = (
        0.0
        + 1.0 * np.log(np.maximum(GDP, eps))
        + 1.0 * np.log(np.maximum(TP, eps))
        + 1.0 * np.log(np.maximum(AP, eps))
        + 1.0 * np.log(np.maximum(DP, eps))
        + 1.0 * np.log(np.maximum(UR, eps))
        + 1.0 * AT
    )

    return np.exp(ln_me)

import numpy as np

def predict(X):
    # Standard reduced-form environmental emissions specification:
    # log(CH4) = β0 + β1*log(GDP) + β2*log(P) + β3*AP + β4*DP + β5*UR + β6*AT
    # Here implemented as a generic multiplicative-log form with unit elasticity placeholders.
    gdp = X[:, 0]
    pop = X[:, 1]
    ap = X[:, 2]
    dp = X[:, 3]
    ur = X[:, 4]
    at = X[:, 5]

    # Avoid issues with nonpositive values in logs
    eps = 1e-12
    y_log = (
        np.log(eps)
        + 1.0 * np.log(np.maximum(gdp, eps))
        + 1.0 * np.log(np.maximum(pop, eps))
        + 0.0 * ap
        + 0.0 * dp
        + 0.0 * ur
        + 0.0 * at
    )
    return np.exp(y_log)

import numpy as np

def predict(X):
    # Common closed-form choice for national CH4 emissions:
    # multiplicative log-linear (Cobb-Douglas-style) relationship
    GDP = X[:, 0]
    TP  = X[:, 1]
    AP  = X[:, 2]
    DP  = X[:, 3]
    UR  = X[:, 4]
    AT  = X[:, 5]

    # Using all predictors in a simple reduced-form elasticity specification.
    # Coefficients are generic stylized elasticities, not data-fitted here.
    ln_Y = (
        1.20
        + 0.35 * np.log(GDP)
        + 0.88 * np.log(TP)
        + 0.22 * np.log(AP)
        + 0.10 * np.log(DP)
        + 0.18 * np.log(UR)
        + 0.04 * AT
    )

    return np.exp(ln_Y)
