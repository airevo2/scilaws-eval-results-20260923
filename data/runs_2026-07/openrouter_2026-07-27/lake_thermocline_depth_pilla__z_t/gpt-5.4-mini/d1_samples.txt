import numpy as np

def predict(X):
    """
    Predict summertime thermocline depth (z_t, m) from lake area and max depth.

    Canonical empirical form:
        z_t = a + b * ln(A_s) + c * ln(Z_max)

    where A_s is lake surface area (km^2) and Z_max is maximum depth (m).
    Coefficients below are representative literature-style values for a
    simple closed-form thermocline-depth scaling and are not dataset-fitted.
    """
    A_s = np.asarray(X[:, 0], dtype=float)
    Z_max = np.asarray(X[:, 1], dtype=float)

    # Guard against invalid values for logarithms
    A_s = np.maximum(A_s, 1e-12)
    Z_max = np.maximum(Z_max, 1e-12)

    a = -0.45
    b = 1.20
    c = 0.35

    z_t = a + b * np.log(A_s) + c * np.log(Z_max)

    # Thermocline depth cannot be negative and should not exceed maximum depth
    z_t = np.clip(z_t, 0.0, Z_max)
    return z_t
