    #
    # Since only distance and sex are provided, a commonly used closed-form

    #   v̄ = a * d^b * exp(c * s_M)
    # with b < 0 so velocity decreases with distance.
    #
    # Coefficients chosen to reflect the well-known power-law decline of

import numpy as np

def predict(X):
    d = X[:, 0].astype(float)
    s_M = X[:, 1].astype(float)

    # Average race velocity from distance and world-record finishing time.
    # Common closed-form representation: v̄ = d / t_wr.
    # For prediction without record time data, a simple sex-specific
    # allometric approximation is used as a canonical power-law form:
    # v̄ = a * d^b * exp(c * s_M)
    a = 8.0
    b = -0.05
    c = 0.04

    return a * np.power(d, b) * np.exp(c * s_M)
