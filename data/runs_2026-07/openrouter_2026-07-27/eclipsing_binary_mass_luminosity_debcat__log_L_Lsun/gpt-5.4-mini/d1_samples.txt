import numpy as np

def predict(X):
    """
    Predict log10(L/Lsun) from log10(M/Msun) using the standard mass-luminosity
    relation as a piecewise power law.
    """
    x = np.asarray(X)
    logM = x[:, 0]
    M = 10.0 ** logM

    # Piecewise mass-luminosity relation (approximate, commonly cited)
    L = np.empty_like(M, dtype=float)

    # M < 0.43 Msun: L ~ 0.23 M^2.3
    m1 = M < 0.43
    L[m1] = 0.23 * M[m1] ** 2.3

    # 0.43 <= M < 2 Msun: L ~ M^4
    m2 = (M >= 0.43) & (M < 2.0)
    L[m2] = M[m2] ** 4.0

    # 2 <= M < 20 Msun: L ~ 1.5 M^3.5
    m3 = (M >= 2.0) & (M < 20.0)
    L[m3] = 1.5 * M[m3] ** 3.5

    # M >= 20 Msun: L ~ 32000 M
    m4 = M >= 20.0
    L[m4] = 32000.0 * M[m4]

    return np.log10(L)

    Predict bolometric luminosity in log10(L/Lsun) from stellar mass and metallicity.

    mask3 = (M >= 2.0) & (M < 20.0)
    mask4 = M >= 20.0

    L[mask1] = 0.23 * M[mask1] ** 2.3
