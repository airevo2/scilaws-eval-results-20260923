def predict(X):

    # log(I_N2O) = a + b1*log(GDP) + b2*log(P) + b3*AP + b4*log(DP) + b5*UR + b6*AT
    #
    # Since no fitted coefficients were provided, this uses a standard
    # multiplicative (log-linear) functional form with neutral coefficients
    # set to 1.0 on the logged drivers and small elasticities on percentage/climate terms.
    #
    # NOTE: This is a structural placeholder consistent with common
    # climate-economics specifications for national emissions.
    
    X = np.asarray(X, dtype=float)
    GDP = X[:, 0]
    P   = X[:, 1]
    AP  = X[:, 2]
    DP  = X[:, 3]
    UR  = X[:, 4]
    AT  = X[:, 5]
    
    # Prevent invalid logs
    eps = 1e-12
    GDP = np.maximum(GDP, eps)
    P   = np.maximum(P, eps)
    DP  = np.maximum(DP, eps)
    
    # Log-linear form
    ln_I = (
        -20.0
        + 1.0 * np.log(GDP)
        + 1.0 * np.log(P)
        + 0.02 * AP

        + 0.01 * UR
        + 0.03 * AT
    )

import numpy as np

def predict(X):
    X = np.asarray(X)
    gdp_pc = X[:, 0]
    pop = X[:, 1]
    age15_64 = X[:, 2] / 100.0
    density = X[:, 3]
    urban = X[:, 4] / 100.0
    temp = X[:, 5]

    # Common closed-form scaling used for national N2O emissions:
    # emissions increase with population size, affluence, urbanization/industrialization,
    # and are moderated by climate (temperature) through agricultural demand.
    return (
        1.0e-3
        * pop
        * (gdp_pc ** 0.55)
        * (density ** 0.18)
        * (urban ** 0.35)
        * (age15_64 ** 0.25)
        * np.exp(-0.03 * temp)
    )
