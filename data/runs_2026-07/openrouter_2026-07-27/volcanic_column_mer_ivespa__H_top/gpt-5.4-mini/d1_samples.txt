import numpy as np

def predict(X):
    MER = np.asarray(X)[:, 0].astype(float)
    # Classical empirical plume-height scaling (Sparks et al.-type power law):
    # H_top ≈ 2.00 * MER^0.259, with H in km above vent and MER in kg/s.
    return 2.00 * np.power(MER, 0.259)
