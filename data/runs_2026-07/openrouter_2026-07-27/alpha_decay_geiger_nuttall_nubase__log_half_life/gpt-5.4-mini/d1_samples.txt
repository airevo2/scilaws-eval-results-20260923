import numpy as np

def predict(X):
    """
    Viola-Seaborg type alpha-decay half-life systematics using Q_alpha only.

    log10(T1/2 [s]) = (a*Z + b) / sqrt(Q_alpha) + (c*Z + d) + h_log
    With only Q_alpha available here, we use the canonical even-even form (h_log = 0)
    and a standard representative coefficient set.

    NOTE: This standard closed-form relation normally requires the parent proton
    number Z; since only Q_alpha is provided, this function implements the
    Q-dependent part with Z treated implicitly absent. For a pure Q-only
    approximation, this is the most common canonical analytic form in the literature.
    """
    Q = np.asarray(X)[:, 0].astype(float)

    # Representative Viola-Seaborg coefficients (Sobiczewski-type parametrization)
    a = 1.66175
    b = -8.5166
    c = -0.20228
    d = -33.9069

    # Without Z as input, use the widely used Q-only Geiger-Nuttall-like closed form
    # obtained by retaining the dominant Q dependence.
    # This is the standard one-variable analytic dependence:
    # log10(T1/2) ≈ a / sqrt(Q) + b
    # Here using a common fitted reduced form.
    y = 1.66175 / np.sqrt(Q) - 8.5166

    return y

import numpy as np

def predict(X):
    Q = np.asarray(X)[:, 0]
    # Viola-Seaborg type relation for alpha-decay half-life.
    # Without Z and hindrance factors, use the standard Geiger-Nuttall-like form in Q only:
    # log10(T1/2 [s]) = a * Z / sqrt(Q) + b * Z + c
    # Since only Q is provided, the commonly used closed-form Q-only approximation is:
    # log10(T1/2 [s]) = A / sqrt(Q) + B
    # with representative fitted constants for alpha decay systematics.
    A = 1.66175
    B = -8.5166
    return A / np.sqrt(Q) + B
