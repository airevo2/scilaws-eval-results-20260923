import numpy as np

def predict(X):
    t = X[:, 0]

    # Standard Mauna Loa CO2 baseline: a quadratic long-term trend plus fixed seasonal harmonics
    # (commonly used NOAA-style closed form for monthly mean CO2 at MLO).
    dt = t - 2000.0

    c = (
        369.29
        + 1.4855956 * dt
        + 0.019369 * dt**2
        + 3.1966 * np.cos(2.0 * np.pi * (t - 0.8))
        + 0.4198 * np.sin(2.0 * np.pi * (t - 0.8))
        + 0.6943 * np.cos(4.0 * np.pi * (t - 0.8))
        + 0.1034 * np.sin(4.0 * np.pi * (t - 0.8))
    )

    return c
