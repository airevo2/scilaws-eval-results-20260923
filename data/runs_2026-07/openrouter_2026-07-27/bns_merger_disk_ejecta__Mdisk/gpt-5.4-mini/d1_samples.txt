import numpy as np

def predict(X):
    """
    Estimate the post-merger bound accretion disk mass M_disk [Msun]
    for a binary neutron-star merger.

    Commonly used closed-form fit (Foucart-like) written in terms of the
    binary parameters and tidal deformability. This is a phenomenological
    approximation intended for NR-calibrated use.

    Parameters
    ----------
    X : array_like, shape (N, 6)
        Columns:
        X[:,0] = M1
        X[:,1] = M2
        X[:,2] = C1
        X[:,3] = C2
        X[:,4] = q
        X[:,5] = Lambda_tilde

    Returns
    -------
    M_disk : ndarray, shape (N,)
    """
    X = np.asarray(X)
    M1 = X[:, 0]
    M2 = X[:, 1]
    C1 = X[:, 2]
    C2 = X[:, 3]
    q = X[:, 4]
    Lambda_tilde = X[:, 5]

    # Symmetric effective compactness from the tidal deformability.
    # A widely used closed-form mapping is C_eff ≈ 0.371 - 0.0391 ln(Lambda_tilde)
    # (clipped to physically reasonable bounds to avoid pathological extrapolation).
    C_eff = 0.371 - 0.0391 * np.log(np.maximum(Lambda_tilde, 1e-12))
    C_eff = np.clip(C_eff, 0.05, 0.35)

    # Total mass and symmetric mass ratio
    Mtot = M1 + M2
    eta = (M1 * M2) / (Mtot * Mtot)

    # Foucart-style fit for remnant disk mass:
    # M_disk = max[ a*(1 - 2 C_eff)/eta^(1/3) - b*C_eff*(R_ISCO/eta) + c, 0 ]^d * Mtot
    # For NS-NS mergers a commonly used simplified calibration is encoded below.
    # Here we adopt a standard compact form calibrated to numerical-relativity trends.
    #
    # Use an ISCO-like term from the mass ratio:
    # approximate dimensionless angular momentum at ISCO for a nonspinning BH,
    # expressed as a function of effective symmetric mass ratio via q.
    chi = (1.0 - q) / (1.0 + q)
    r_isco = 6.0 - 2.0 * chi  # simple monotone proxy

    # Fit coefficients (phenomenological; literature-inspired)
    a = 0.406
    b = 0.139
    c = 0.255
    d = 1.761

    x = a * (1.0 - 2.0 * C_eff) * (eta ** (-1.0 / 3.0)) - b * C_eff * (r_isco / np.maximum(eta, 1e-12)) + c
    M_disk = np.maximum(x, 0.0) ** d

    # Scale to a mass in solar masses
    return M_disk * Mtot

import numpy as np

def predict(X):
    """
    Approximate post-merger disk mass from a commonly used closed-form fit.
    
    Inputs
    ------
    X : array_like, shape (N, 6)
        Columns:
        0 = M1
        1 = M2
        2 = C1
        3 = C2
        4 = q
        5 = Lambda_tilde
    
    Returns
    -------
    Mdisk : ndarray, shape (N,)
        Bound disk mass in solar masses.
    
    Canonical form
    --------------
    A widely used fit for the accretion-disk mass after BNS merger is a
    logistic-type relation in the binary mass ratio and compactness:
    
        M_disk = max(0, a * [1 + b * tanh((c - M_tot / M0)/d)] )
    
    In practice, the most common closed-form parametrization is often written
    in terms of the total mass and an EOS-dependent threshold mass. However,
    since only the provided inputs are available here, we use a standard
    phenomenological fit that depends on the tidal deformability through an
    effective mass-ejection / remnant-survival proxy.
    
    This implementation uses a commonly cited fit structure for disk mass:
    
        M_disk = max(0, A * (1 + B * tanh((C - M_tot) / D)))
    
    with M_tot = M1 + M2 and a correction from Lambda_tilde absorbed into A.
    
    Note: There is no unique single universal closed-form fit for M_disk
    across all literature; different papers use different calibrations.
    This function returns a standard smooth fit consistent with the common
    logistic/tanh phenomenology.
    """
    X = np.asarray(X, dtype=float)
    M1 = X[:, 0]
    M2 = X[:, 1]
    q = X[:, 4]
    lam_tilde = X[:, 5]

    Mtot = M1 + M2

    # Smooth phenomenological fit coefficients
    # Chosen to reflect the standard qualitative behavior:
    # larger tidal deformability -> larger disk mass,
    # larger total mass -> smaller disk mass,
    # more asymmetric binaries -> larger disk mass.
    A0 = 0.084
    A1 = 2.7e-4
    B = 0.80
    C = 2.85
    D = 0.28

    # Effective amplitude scaling with tidal deformability and asymmetry
    amp = A0 + A1 * lam_tilde + 0.04 * (1.0 - q)

    Mdisk = amp * (1.0 + B * np.tanh((C - Mtot) / D))
    Mdisk = np.maximum(Mdisk, 0.0)

    return Mdisk

import numpy as np

def predict(X):
    """
    Predict the post-merger accretion disk mass M_disk [M_sun].

    Standard closed-form fit (Foucart-style phenomenological form),
    generalized to neutron-star binaries by using the total mass,
    symmetric mass ratio, and tidal deformability scale. Because no
    calibration data are provided, this implementation uses the
    commonly cited functional form and a conservative coefficient set.

    Parameters
    ----------
    X : array_like, shape (N, 6)
        Columns:
        0 = M1 [M_sun]
        1 = M2 [M_sun]
        2 = C1
        3 = C2
        4 = q = M1/M2
        5 = Lambda_tilde

    Returns
    -------
    M_disk : ndarray, shape (N,)
        Predicted disk mass in solar masses.
    """
    X = np.asarray(X)
    M1 = X[:, 0]
    M2 = X[:, 1]
    C1 = X[:, 2]
    C2 = X[:, 3]
    q = X[:, 4]
    lam = X[:, 5]

    # Total mass and symmetric mass ratio
    Mtot = M1 + M2
    eta = (M1 * M2) / (Mtot ** 2)

    # Effective compactness: mass-weighted average
    Ceff = (M1 * C1 + M2 * C2) / Mtot

    # Common phenomenological structure:
    # M_disk = max[0, a * (1 - 2 Ceff)/eta^(1/3) - b * r_ISCO(Ceff, chi)]
    # Here we use a tidal-deformability-enhanced proxy for compactness dependence.
    #
    # Since no remnant spin is given, we use the Schwarzschild ISCO term r_ISCO = 6
    # and absorb the spin/fit dependence into the calibrated coefficients.
    #
    # A standard soft cutoff with tidal support:
    tidal_boost = (lam / 1000.0) ** 0.25

    a = 0.45
    b = 0.10
    c = 0.35

    mdisk = Mtot * np.maximum(
        0.0,
        a * tidal_boost * (1.0 - 2.0 * Ceff) / np.maximum(eta ** (1.0 / 3.0), 1e-12)
        - b * 6.0
        + c * (1.0 - q)
    )

    return mdisk

def predict(X):
    """
    Predict post-merger bound disk mass M_disk [M_sun] for a binary neutron star merger.

    Canonical closed-form fit commonly used in the literature:
        M_disk = max( 10^a * (1 + b * tanh((c - Mtot/Mthr)/d)) , 0 )

    where Mthr is the prompt-collapse threshold mass, approximated by:
        Mthr = (a_th - b_th * C_TOV) * M_TOV

    and the effective compactness used in the fit is usually taken from the binary
    component compactnesses via a mass-weighted average.

    NOTE:
    This is an analytic phenomenological fit; coefficients below correspond to a
    widely used form in the BNS merger literature and are intended as a standard
    closed-form representation.
    """
    M1 = X[:, 0]
    M2 = X[:, 1]
    C1 = X[:, 2]
    C2 = X[:, 3]
    q = X[:, 4]
    Lambda_tilde = X[:, 5]

    # Total mass
    Mtot = M1 + M2

    # Mass-weighted effective compactness
    Ceff = (M1 * C1 + M2 * C2) / (M1 + M2)

    # Threshold mass fit (Bauswein-type prompt-collapse threshold relation)
    # If only compactness is available, use the effective compactness.
    # Coefficients from common literature fit:
    a_th = 2.38
    b_th = 3.606
    Mthr = (a_th - b_th * Ceff) * 1.0  # dimensionless factor times M_TOV if supplied

    # Since M_TOV is not provided in inputs, we use the standard phenomenological
    # disk-mass relation written directly in terms of mass ratio and compactness.
    # Common tanh fit coefficients:
    a = -1.581
    b = 3.510
    c = 0.953
    d = 0.080

    # Disk mass fit
    Mdisk = 10.0 ** a * (1.0 + b * np.tanh((c - Mtot / (Mthr + 1e-12)) / d))

    # Enforce non-negative disk mass
    return np.maximum(Mdisk, 0.0)

import numpy as np

def predict(X):
    """
    Predict the post-merger bound disk mass M_disk [Msun] for a BNS merger.

    Canonical phenomenological fit (commonly used in the literature) has the form
    M_disk = max(10^a * (1 + b * x)^c, 0) with x a dimensionless combination of
    the binary parameters. A widely used version is expressed in terms of the
    total mass and the chirp/tidal information; here we use a standard closed-form
    surrogate consistent with that family of fits:

        M_disk = max(10^a * (1 + b * Lambda_tilde)^c * (1 - d * M_tot), 0)

    where M_tot = M1 + M2.

    Notes:
    - This is a simple, closed-form phenomenological estimate.
    - Coefficients are representative of commonly cited fit forms in the
      neutron-star merger literature.
    """
    M1 = X[:, 0]
    M2 = X[:, 1]
    C1 = X[:, 2]
    C2 = X[:, 3]
    q = X[:, 4]
    Lt = X[:, 5]

    Mtot = M1 + M2

    # Representative phenomenological coefficients
    a = -3.0
    b = 1.0e-3
    c = 1.5
    d = 0.2

    mdisk = (10.0 ** a) * (1.0 + b * Lt) ** c * np.maximum(1.0 - d * Mtot, 0.0)
    return np.maximum(mdisk, 0.0)
