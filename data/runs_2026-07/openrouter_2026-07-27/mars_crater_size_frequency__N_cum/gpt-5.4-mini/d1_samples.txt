import numpy as np

def predict(X):
    """
    Cumulative areal density of Mars craters N(>=D) [km^-2]
    as a function of lower diameter threshold D (km).

    Canonical approximation: power law in log-log space
        N(>=D) = A * D^(-b)

    with commonly used Mars crater-population exponents near b ~ 2.
    """
    D = np.asarray(X)[:, 0]
    A = 5.0e-3
    b = 2.0
    return A * np.power(D, -b)
