import numpy as np

def predict(X):
    # Canonical gravity equation in log-linear form:
    # ln X_ij = β0 + β1 ln Y_i + β2 ln Y_j + β3 ln d_ij
    #           + β4 contig + β5 comlang_off + β6 col_dep_ever + β7 fta_wto
    #
    # Standard textbook/theoretical coefficient values are not universal, so this
    # returns the usual structural/log-linear form with unit coefficients on the
    # listed regressors except distance, which enters negatively in the canonical way.
    X = np.asarray(X)

    log_gdp_o = X[:, 0]
    log_gdp_d = X[:, 1]
    log_dist = X[:, 2]
    contig = X[:, 3]
    comlang_off = X[:, 4]
    col_dep_ever = X[:, 5]
    fta_wto = X[:, 6]

    # Intercept absorbed into a constant; set to 0 in the generic closed form.
    return (
        0.0
        + 1.0 * log_gdp_o
        + 1.0 * log_gdp_d
        - 1.0 * log_dist
        + 1.0 * contig
        + 1.0 * comlang_off
        + 1.0 * col_dep_ever
        + 1.0 * fta_wto
    )
