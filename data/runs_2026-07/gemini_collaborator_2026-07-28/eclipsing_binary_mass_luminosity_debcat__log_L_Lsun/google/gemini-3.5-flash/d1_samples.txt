import numpy as np


def predict(X):
    # Extract log10(M/M_sun)
    log_M = X[:, 0]

    # Define standard transition mass boundaries in log10 space
    m1 = np.log10(0.43)
    m2 = np.log10(2.0)
    m3 = np.log10(20.0)

    # Calculate continuous offsets based on the solar normalization (log_L = 0 when log_M = 0)
    # The standard scaling relations are:
    # L ~ M^2.3  for M < 0.43
    # L ~ M^4.0  for 0.43 <= M < 2.0
    # L ~ M^3.5  for 2.0 <= M < 20.0
    # L ~ M^1.0  for M >= 20.0
    c1 = 1.7 * m1
    c3 = 0.5 * m2
    c4 = 2.5 * m3 + c3

    # Define conditions for piecewise mass-luminosity relation
    condlist = [
        log_M < m1,
        (log_M >= m1) & (log_M < m2),
        (log_M >= m2) & (log_M < m3),
        log_M >= m3,
    ]

    funclist = [
        lambda m: 2.3 * m + c1,
        lambda m: 4.0 * m,
        lambda m: 3.5 * m + c3,
        lambda m: 1.0 * m + c4,
    ]

    return np.piecewise(log_M, condlist, funclist)

import numpy as np


def predict(X):
    # X[:, 0] is log_M_Msun (log10(M/Msun))
    log_m = X[:, 0]
    m = 10**log_m

    # Standard piecewise mass-luminosity relation
    condlist = [
        m < 0.43,
        (m >= 0.43) & (m < 2.0),
        (m >= 2.0) & (m < 20.0),
        m >= 20.0,
    ]

    funclist = [
        lambda x: 2.3 * x + np.log10(0.23),
        lambda x: 4.0 * x,
        lambda x: 3.5 * x + np.log10(1.5),
        lambda x: 1.0 * x + np.log10(3200.0),
    ]

    return np.piecewise(log_m, condlist, funclist)
