import numpy as np


def predict(X):
    """Predicts the aboveground biomass (AGB) in kg of individual tropical trees

    using the pantropical allometric equation from Chave et al. (2014).

    Inputs:
    - X[:, 0]: wood_density (rho) [g/cm^3]
    - X[:, 1]: diameter_cm (D) [cm]
    - X[:, 2]: height_m (H) [m]

    Output:
    - 1-D numpy array of AGB in kg
    """
    rho = X[:, 0]
    D = X[:, 1]
    H = X[:, 2]

    # Chave et al. (2014) universal pantropical model:
    # AGB = 0.0673 * (rho * D^2 * H)^0.976
    agb_kg = 0.0673 * (rho * (D**2) * H) ** 0.976

    return agb_kg
