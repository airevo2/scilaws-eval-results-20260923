import numpy as np


def predict(X):
    # Extract orientation in degrees
    theta = X[:, 0]

    # Standard tuning curve parameters for a typical V1 neuron
    R0 = 0.1  # Baseline response (arbitrary units)
    A = 2.0  # Peak response amplitude above baseline (arbitrary units)
    theta_pref = 90.0  # Preferred orientation in degrees
    sigma = 20.0  # Tuning width (standard deviation) in degrees

    # Compute angular difference accounting for 180-degree periodicity
    delta_theta = ((theta - theta_pref + 90) % 180) - 90

    # Gaussian tuning function
    R = R0 + A * np.exp(-(delta_theta**2) / (2 * sigma**2))

    return R

def predict(X):
    # X[:, 0] = orientation_deg (theta) in [0, 180)
    theta = X[:, 0]

    # Canonical parameters for a representative V1 orientation-selective neuron
    R_0 = 0.1  # Baseline response (spontaneous activity) [au]
    A = 1.5  # Peak response amplitude above baseline [au]
    theta_pref = 90.0  # Preferred orientation [deg]
    kappa = 1.5  # Concentration parameter (determines tuning bandwidth)

    # Convert angles to radians
    theta_rad = np.radians(theta)
    theta_pref_rad = np.radians(theta_pref)

    # Von Mises tuning function formulation for 180-degree periodic orientation data
    # Multiplying the angular difference by 2 accounts for the pi-periodicity of gratings
    R = R_0 + A * np.exp(kappa * (np.cos(2 * (theta_rad - theta_pref_rad)) - 1))

    return R

def predict(X):

    theta = X[:, 0]

    theta_pref = 90.0     # Preferred orientation in degrees

    R_baseline = 0.1      # Baseline response (arbitrary units)
    
    # Convert angles to radians
    theta_rad = np.radians(theta)
    theta_pref_rad = np.radians(theta_pref)
    
    # Von Mises tuning curve with 180-degree periodicity (using 2 * theta)
    response = R_baseline + A * np.exp(kappa * (np.cos(2 * (theta_rad - theta_pref_rad)) - 1))
    
    return response
