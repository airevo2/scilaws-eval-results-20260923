import numpy as np


def predict(X):
    """Predicts the natural log of Gross Metropolitan Product (log_GMP)

    based on the natural log of population (log_pop) using the canonical
    urban scaling relationship (agglomeration economies).

    Inputs:
    - X: 2-D numpy array where X[:, 0] is log_pop (ln N) in persons.

    Outputs:
    - 1-D numpy array of predicted log_GMP (ln Y) in millions of chained 2017 USD.
    """
    log_pop = X[:, 0]

    # In urban economics, economic output scales superlinearly with population:
    # Y = Y_0 * N^beta  =>  ln(Y) = ln(Y_0) + beta * ln(N)
    # For GDP/GMP, the scaling exponent beta is canonically estimated to be ~1.15.
    # The intercept is calibrated for GMP in millions of 2017 USD.
    beta = 1.15
    alpha = -4.89

    log_GMP = alpha + beta * log_pop
    return log_GMP
