import numpy as np


def predict(X):
    # Extract inputs
    Z = X[:, 0]
    A = X[:, 2]
    asym = X[:, 4]
    coul_proxy = X[:, 5]
    pair = X[:, 6]

    # Bethe-Weizsäcker semi-empirical mass formula coefficients (in MeV)
    a_v = 15.75  # Volume term coefficient
    a_s = 17.8  # Surface term coefficient
    a_c = 0.711  # Coulomb term coefficient
    a_a = 23.7  # Asymmetry term coefficient
    a_p = 11.18  # Pairing term coefficient

    # Volume term per nucleon
    vol_term = a_v

    # Surface term per nucleon (a_s * A^(2/3) / A = a_s * A^(-1/3))
    surf_term = a_s / (A ** (1.0 / 3.0))

    # Coulomb term per nucleon (a_c * coul_proxy / A)
    # coul_proxy is Z*(Z-1)/A^(1/3), so dividing by A yields Z*(Z-1)/A^(4/3)
    coul_term = a_c * coul_proxy / A

    # Asymmetry term per nucleon (a_a * asym^2)
    # asym is (N-Z)/A
    asym_term = a_a * (asym**2)

    # Pairing term per nucleon (a_p * pair * A^(-1/2) / A = a_p * pair * A^(-1.5))
    pair_term = a_p * pair / (A**1.5)

    # Binding energy per nucleon
    be_per_A = vol_term - surf_term - coul_term - asym_term + pair_term

    return be_per_A

import numpy as np


def predict(X):
    # Extract features from input array
    Z = X[:, 0]
    A = X[:, 2]
    asym = X[:, 4]  # (N - Z) / A
    coul_proxy = X[:, 5]  # Z(Z-1) / A**(1/3)
    pair = X[:, 6]  # +1 for even-even, 0 for odd-mass, -1 for odd-odd

    # Semi-empirical mass formula coefficients (in MeV)
    # Standard values from Krane (Introductory Nuclear Physics)
    a_v = 15.56  # Volume term
    a_s = 17.23  # Surface term
    a_c = 0.697  # Coulomb term
    a_a = 23.285  # Asymmetry term
    a_p = 12.0  # Pairing term

    # Safe division array for A
    A_safe = np.maximum(A, 1.0)

    # Calculate individual terms per nucleon (B/A)
    term_volume = a_v
    term_surface = a_s / (A_safe ** (1.0 / 3.0))
    term_coulomb = a_c * coul_proxy / A_safe
    term_asymmetry = a_a * (asym**2)
    term_pairing = pair * a_p / (A_safe**1.5)

    # Total binding energy per nucleon
    be_per_A = (
        term_volume - term_surface - term_coulomb - term_asymmetry + term_pairing
    )

    # Binding energy per nucleon is 0 for A <= 1 (single nucleons)
    be_per_A = np.where(A <= 1, 0.0, be_per_A)

    return be_per_A

    A = X[:, 2]
    asym = X[:, 4]
    coul_proxy = X[:, 5]

    a_s = 17.80  # Surface term coefficient
    a_c = 0.711  # Coulomb term coefficient
    a_a = 23.70  # Asymmetry term coefficient
    a_p = 11.18  # Pairing term coefficient

    A_safe = np.where(A > 0, A, 1.0)

    # Component terms of the binding energy per nucleon (B/A)
    vol_term = a_v
    surf_term = a_s / (A_safe ** (1.0 / 3.0))
    coul_term = a_c * coul_proxy / A_safe

    pair_term = a_p * pair * (A_safe ** (-1.5))

    # Total binding energy per nucleon
    be_per_A = vol_term - surf_term - coul_term - asym_term + pair_term

    # For single nucleons (A <= 1), the binding energy per nucleon is 0
    be_per_A = np.where(A <= 1, 0.0, be_per_A)

    return be_per_A
