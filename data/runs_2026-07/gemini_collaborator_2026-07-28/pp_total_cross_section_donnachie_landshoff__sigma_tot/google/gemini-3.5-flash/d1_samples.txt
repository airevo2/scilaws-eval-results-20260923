def predict(X):
    # X[:, 0] = s (Mandelstam s in GeV^2)
    s = X[:, 0]

    # Standard PDG (COMPETE collaboration) fit parameters for pp collisions
    H = 0.272  # in mb
    s0 = 16.0  # in GeV^2
    beta = 34.41  # in mb
    Y1 = 13.07  # in mb
    Y2 = 7.39  # in mb
    eta1 = 0.4473
    eta2 = 0.5486

    # Total cross-section formula:
    # sigma_tot = H * ln^2(s / s0) + beta + Y1 * s^(-eta1) - Y2 * s^(-eta2)
    term_froissart = H * (np.log(s / s0)) ** 2
    term_reggeon1 = Y1 * (s**-eta1)
    term_reggeon2 = -Y2 * (s**-eta2)

    sigma_tot = term_froissart + beta + term_reggeon1 + term_reggeon2
    return sigma_tot

import numpy as np


def predict(X):
    # Extract Mandelstam s from the first column of X
    s = X[:, 0]

    # Standard COMPASS / PDG (Particle Data Group) fit parameters for pp total cross-section
    # sigma_tot = Z + B * ln^2(s/s0) + Y1 * (s1/s)^eta1 - Y2 * (s1/s)^eta2
    Z = 35.45  # Constant term in mb
    B = 0.272  # Froissart-bound-associated coefficient in mb
    Y1 = 42.53  # Reggeon contribution coefficient 1 in mb
    Y2 = 33.34  # Reggeon contribution coefficient 2 in mb
    eta1 = 0.447  # Regge exponent 1
    eta2 = 0.548  # Regge exponent 2

    s0 = 15.972  # Scale parameter (2*m_p + M)^2 in GeV^2, with M = 2.120 GeV
    s1 = 1.0  # Scale parameter in GeV^2

    # Calculate the components of the total cross-section
    term_log = B * np.log(s / s0) ** 2
    term_regge1 = Y1 * (s1 / s) ** eta1
    term_regge2 = Y2 * (s1 / s) ** eta2

    # Total hadronic cross-section for pp collisions
    sigma_tot = Z + term_log + term_regge1 - term_regge2

    return sigma_tot

import numpy as np


def predict(X):
    # Extract Mandelstam s from the first column of X
    s = X[:, 0]

    # PDG (Particle Data Group) standard fit parameters for pp total cross-section
    H = 0.2720  # mb
    B = 34.41  # mb
    s_0 = 33.9  # GeV²
    s_1 = 1.0  # GeV²
    Y_1 = 13.07  # mb
    Y_2 = 7.39  # mb
    eta_1 = 0.4473
    eta_2 = 0.5486

    # Calculate the components of the total cross-section
    term_froissart = H * (np.log(s / s_0) ** 2)
    term_reggeon1 = Y_1 * (s_1 / s) ** eta_1
    term_reggeon2 = Y_2 * (s_1 / s) ** eta_2

    # For pp collisions, the Y_2 reggeon term is subtracted
    sigma_tot = term_froissart + B + term_reggeon1 - term_reggeon2

    return sigma_tot
