import numpy as np


def predict(X):
    """Calculates the refractive index of ethylene glycol as a function of

    vacuum wavelength (in micrometres) and temperature (in °C).

    Parameters:
    X : ndarray of shape (N, 2)
        X[:, 0] : wavelength in micrometres (lambda_um)
        X[:, 1] : temperature in °C (temperature_C)

    Returns:
    n : ndarray of shape (N,)
        Refractive index of ethylene glycol.
    """
    lam = X[:, 0]
    T = X[:, 1]

    # Cauchy coefficients for ethylene glycol at reference temperature T0 = 20 °C
    A = 1.423
    B = 0.0025  # um^2
    C = 0.00018  # um^4

    # Thermo-optic coefficient (dn/dT) for ethylene glycol
    dn_dT = -0.00026  # per degree Celsius

    # Refractive index calculation using Cauchy dispersion with temperature correction
    n = A + (B / lam**2) + (C / lam**4) + dn_dT * (T - 20.0)

    return n

import numpy as np


def predict(X):
    """Calculates the refractive index of ethylene glycol as a function of

    wavelength and temperature using a temperature-dependent Cauchy equation.

    Parameters:
    X : ndarray of shape (N, 2)
        X[:, 0] = vacuum wavelength (lambda) in micrometers
        X[:, 1] = temperature (T) in degrees Celsius

    Returns:
    n : ndarray of shape (N,)
        Refractive index of ethylene glycol.
    """
    lam = X[:, 0]
    T = X[:, 1]

    # Cauchy coefficients at reference temperature T0 = 20 °C
    A = 1.4194
    B = 0.00407
    C = 0.00010

    # Thermo-optic coefficient dn/dT (in 1/°C)
    dn_dT = -0.00026

    # Calculate refractive index
    n = A + B / (lam**2) + C / (lam**4) + dn_dT * (T - 20.0)

    return n
