import numpy as np

def predict(X):
    # X[:, 0] = R_m (Hydraulic depth/radius in meters)
    # X[:, 1] = SLOPE (Channel bed slope, dimensionless)
    
    R = X[:, 0]
    S = X[:, 1]
    
    # Manning's roughness coefficient (n) typically averages around 0.035 
    # for natural channels monitored by the USGS.
    n = 0.035
    
    # Manning's equation in SI units: V = (1/n) * R^(2/3) * S^(1/2)
    # Using np.maximum to avoid issues with any potential non-positive values
    R_clipped = np.maximum(R, 0.0)
    S_clipped = np.maximum(S, 0.0)
    
    V = (1.0 / n) * np.power(R_clipped, 2.0/3.0) * np.sqrt(S_clipped)
    
    return V

    # X[:, 1] = SLOPE (Channel bed/friction slope, dimensionless)

    # Manning's equation for mean velocity in metric units:

    return V
