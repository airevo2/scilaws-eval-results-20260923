def predict(X):
    # X[:, 0] = A_s (Lake surface area in km^2)
    # X[:, 1] = Z_max (Lake maximum depth in m)
    A_s = X[:, 0]
    Z_max = X[:, 1]

    # Empirical power-law relationship based on Shuter et al. (1983) and Hanna (1990)
    # relating thermocline depth to lake surface area (fetch proxy)
    z_t_est = 3.3 * (A_s**0.225)

    # Apply physical constraint: thermocline depth cannot exceed maximum depth.
    # We cap it at 90% of Z_max to account for the bottom boundary layer.
    z_t = np.minimum(z_t_est, 0.9 * Z_max)

    # Round to the nearest 0.5 m to match the uniform interpolation grid of the source dataset.
    z_t_grid = np.round(z_t * 2.0) / 2.0

    # Ensure thermocline depth is at least 0.5 m
    return np.maximum(z_t_grid, 0.5)
