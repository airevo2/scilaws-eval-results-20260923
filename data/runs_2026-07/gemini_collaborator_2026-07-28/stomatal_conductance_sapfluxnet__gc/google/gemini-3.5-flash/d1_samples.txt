import numpy as np


def predict(X):
    # Extract VPD (D) in kPa from the first column of X
    D = X[:, 0]

    # Reference canopy conductance at D = 1 kPa (cm3 cm-2 h-1 kPa-1)
    Gc_ref = 50.0

    # Sensitivity parameter (m), typically ~0.5 to 0.6 of Gc_ref
    m = 25.0

    # Calculate Gc using the canonical log-linear relationship
    # We clip D to be at least 0.3 kPa as specified by the input range
    D_clipped = np.maximum(D, 0.3)
    Gc = Gc_ref - m * np.log(D_clipped)

    # Ensure Gc remains non-negative
    Gc = np.maximum(Gc, 0.0)

    return Gc

import numpy as np


def predict(X):
    # X[:, 0] is VPD (D) in kPa
    D = X[:, 0]

    # Reference canopy conductance at VPD = 1 kPa (Gc_ref)
    # Typical value for sapwood-area-normalized canopy conductance is around 20.0 cm3 cm-2 h-1 kPa-1
    Gc_ref = 20.0

    # Stomatal sensitivity to VPD (m)
    # Oren et al. (1999) showed that m is theoretically and empirically ~0.6 * Gc_ref
    m = 0.6 * Gc_ref

    # Log-linear decline of canopy conductance with VPD
    Gc = Gc_ref - m * np.log(D)

    # Apply a biological minimum conductance boundary (e.g., 5% of Gc_ref)
    # to prevent negative or unrealistically low conductance at very high VPD.
    Gc = np.maximum(Gc, 0.05 * Gc_ref)

    return Gc
