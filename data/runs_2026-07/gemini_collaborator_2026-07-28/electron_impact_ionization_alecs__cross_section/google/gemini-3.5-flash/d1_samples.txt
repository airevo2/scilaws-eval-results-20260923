import numpy as np

def predict(X):
    """
    Calculates the total electron-impact single-ionization cross-section 
    for atomic hydrogen (as a representative canonical target) in units of 10^-16 cm^2
    using the Binary-Encounter-Bethe (BEB) model.
    
    Parameters:
    X : ndarray of shape (N, 1)
        X[:, 0] = electron_energy_eV (E_e) [eV]
        
    Returns:
    sigma : ndarray of shape (N,)
        Cross-section in units of 10^-16 cm^2
    """
    Ee = X[:, 0]
    
    # Constants for Hydrogen 1s orbital
    B = 13.605693  # Binding energy in eV
    U = 13.605693  # Average kinetic energy of the orbital electron in eV
    
    # S = 4 * pi * a0^2 * N * (R/B)^2 in units of 10^-16 cm^2
    # For H: N=1, R=13.605693 eV, B=13.605693 eV, 4*pi*a0^2 ≈ 3.5182
    S = 3.5182
    
    t = Ee / B
    u = U / B
    
    # Avoid division by zero or log of negative numbers for energies below threshold
    t_safe = np.maximum(t, 1e-9)
    
    # Binary-Encounter-Bethe (BEB) formula terms
    term1 = 0.5 * np.log(t_safe) * (1.0 - 1.0 / (t_safe**2))
    term2 = 1.0 - 1.0 / t_safe
    term3 = np.log(t_safe) / (t_safe + 1.0)
    
    sigma = (S / (t + u + 1.0)) * (term1 + term2 - term3)
    
    # Set cross-section to 0 for energies below the threshold (t < 1.0)
    sigma = np.where(t > 1.0, sigma, 0.0)
    
    return sigma

import numpy as np


def predict(X):
    """Calculates the total electron-impact single-ionization cross-section of

    atomic Hydrogen using the Binary-Encounter-Bethe (BEB) model.

    Parameters:
    X : ndarray of shape (N, 1)
        X[:, 0] contains the incident electron energy E_e in eV.

    Returns:
    ndarray of shape (N,)
        The ionization cross-section in units of 10^-16 cm^2.
    """
    Ee = X[:, 0]

    # Physical parameters for Hydrogen atom (ground state)
    B = 13.6057  # Binding energy (ionization potential) in eV
    U = 13.6057  # Average kinetic energy of the orbital electron in eV
    N = 1  # Orbital occupation number

    # S = 4 * pi * a_0^2 * N * (R / B)^2
    # where R = 13.6057 eV, a_0 = 0.529177e-8 cm.
    # 4 * pi * a_0^2 approx 3.5182 * 10^-16 cm^2.
    S = 3.5182 * N * (13.6057 / B) ** 2  # prefactor in units of 10^-16 cm^2

    sigma = np.zeros_like(Ee)
    mask = Ee > B

    t = Ee[mask] / B
    u = U / B

    term1 = (np.log(t) / 2.0) * (1.0 - 1.0 / t**2)
    term2 = 1.0 - 1.0 / t
    term3 = np.log(t) / (t + u)

    sigma[mask] = (S / (t + u + 1.0)) * (term1 + term2 - term3)

    return sigma

import numpy as np

def predict(X):
    # Electron energy in eV
    Ee = X[:, 0]
    
    # Target: Hydrogen atom (H), the standard benchmark for single-ionization.
    # Using the Binary-Encounter-Bethe (BEB) model by Kim and Rudd (1994).
    B = 13.605693  # Ionization energy of H (1s orbital) in eV
    U = 13.605693  # Average kinetic energy of the orbital electron in eV
    N = 1          # Number of electrons in the shell
    
    # S = 4 * pi * a0^2 * N * (Rydberg / B)^2
    # pi * a0^2 approx 0.87973556 x 10^-16 cm^2
    # For H, B = Rydberg, so S = 4 * 0.87973556 * 1 = 3.51894224
    S = 3.51894224
    
    # Normalized energies
    t = Ee / B
    u = U / B
    
    # Safe thresholding to avoid division by zero or negative logs
    t_safe = np.maximum(t, 1.0 + 1e-9)
    
    # BEB formula terms
    term1 = (np.log(t_safe) / 2.0) * (1.0 - 1.0 / (t_safe**2))
    term2 = 1.0 - 1.0 / t_safe
    term3 = -np.log(t_safe) / (t_safe + 1.0)
    
    # Cross section in units of 10^-16 cm^2
    sigma = (S / (t_safe + u + 1.0)) * (term1 + term2 + term3)
    
    # Set cross section to 0 below the ionization threshold
    sigma = np.where(Ee >= B, sigma, 0.0)
    
    return sigma

import numpy as np


def predict(X):
    """Calculates the total electron-impact single-ionization cross-section

    using the canonical Lotz empirical formula for a hydrogen-like atom.

    Inputs:
    - X: 2-D numpy array where X[:, 0] is the electron kinetic energy (E_e) in
    eV.

    Returns:
    - 1-D numpy array of the cross-section in units of 10^-16 cm^2.
    """
    E = X[:, 0]

    # Ionization energy of Hydrogen (I = 13.605 eV)
    I = 13.605693

    # Lotz parameter a = 4.0 * 10^-14 cm^2 eV^2
    # In units of 10^-16 cm^2 eV^2, a = 400.0
    a = 400.0

    # Initialize cross-section array to 0
    sigma = np.zeros_like(E)

    # The cross-section is non-zero only above the ionization threshold (E > I)
    mask = E > I
    if np.any(mask):
        E_m = E[mask]
        sigma[mask] = a * np.log(E_m / I) / (E_m * I)

    return sigma
