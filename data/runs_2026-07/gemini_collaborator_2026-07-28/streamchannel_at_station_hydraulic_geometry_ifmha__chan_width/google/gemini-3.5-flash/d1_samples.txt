import numpy as np

def predict(X):
    # X[:, 0] = chan_discharge (Q) [cfs]
    Q = X[:, 0]
    
    # Clip to avoid negative or zero values in power calculation
    Q_safe = np.clip(Q, a_min=1e-5, a_max=None)
    
    # Classical downstream hydraulic geometry (DHG) relationship (Leopold and Maddock, 1953)
    # w = a * Q^b, where typically a ≈ 4.5 and b ≈ 0.5 for channel width in feet
    w = 4.5 * np.pow(Q_safe, 0.5)
    
    return w
