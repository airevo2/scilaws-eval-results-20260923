import numpy as np

def predict(X):
    # Rousset's linearized genetic differentiation statistic M = F_ST / (1 - F_ST)
    # under the standard 2-dimensional isolation-by-distance (IBD) model.
    # The theoretical expectation is: M = a + b * ln(distance)
    # where b = 1 / (4 * pi * D * sigma^2) (D: population density, sigma^2: dispersal variance)
    # We use representative empirical parameters for a typical species.
    
    ln_d = X[:, 0]  # Natural logarithm of distance in km
    
    # Representative coefficients
    intercept = 0.010
    slope = 0.005
    
    M_stat = intercept + slope * ln_d
    return M_stat

import numpy as np

def predict(X):
    """
    Predicts Rousset's linearized differentiation statistic, M = F_ST / (1 - F_ST),
    using the standard 2-dimensional isolation-by-distance (IBD) relationship:
    
        M = a + b * ln(d)
        
    where d is the geographic distance, and the slope b is theoretically 
    related to the population density (D) and mean squared dispersal distance (sigma^2) 
    by b = 1 / (4 * pi * D * sigma^2).
    
    Inputs:
    - X: 2-D numpy array where:
        X[:, 0] = log_distance_km (ln(d))
        X[:, 1] = distance_km (d)
        
    Returns:
    - 1-D numpy array of M values.
    """
    # Extract the natural logarithm of the great-circle distance
    log_distance = X[:, 0]
    
    # Representative empirical coefficients for 2D isolation by distance
    intercept = 0.010
    slope = 0.005
    
    # Calculate Rousset's linearized statistic
    M = intercept + slope * log_distance
    
    return M
