import numpy as np


def predict(X):
    # Extract CMB-frame redshift z_hd
    z = X[:, 0]

    # Standard flat LCDM cosmological parameters (Pantheon/Planck-like)
    c = 299792.458  # Speed of light in km/s
    H0 = 70.0  # Hubble constant in km/s/Mpc
    Omega_m = 0.3
    Omega_L = 0.7

    # Vectorized numerical integration of the comoving distance integrand:
    # 1 / E(z) = 1 / sqrt(Omega_m * (1+z)^3 + Omega_L)
    # Using 1000 steps for trapezoidal integration per redshift
    steps = 1000
    t = np.linspace(0.0, 1.0, steps)
    grid = z[:, np.newaxis] * t[np.newaxis, :]  # Shape: (len(z), steps)

    integrand = 1.0 / np.sqrt(Omega_m * (1.0 + grid) ** 3 + Omega_L)
    integral = np.trapz(integrand, grid, axis=1)

    # Luminosity distance d_L in Mpc
    dL = (1.0 + z) * (c / H0) * integral

    # Avoid log10(0) for any edge cases near z=0
    dL = np.maximum(dL, 1e-10)

    # Distance modulus: mu = 5 * log10(dL_pc / 10) = 5 * log10(dL_Mpc) + 25
    mu = 5.0 * np.log10(dL) + 25.0

    return mu

import numpy as np


def predict(X):
    # Extract Hubble-diagram CMB-frame redshift
    z = X[:, 0]

    # Cosmological parameters (Standard Flat Lambda-CDM)
    H0 = 70.0  # Hubble constant in km/s/Mpc
    Omega_m = 0.3  # Matter density parameter
    Omega_L = 1.0 - Omega_m  # Dark energy density parameter
    c = 299792.458  # Speed of light in km/s

    # Numerical integration using Trapezoidal rule over N steps
    N = 1000
    t = np.linspace(0.0, 1.0, N)
    # u shape: (len(z), N)
    u = z[:, np.newaxis] * t[np.newaxis, :]

    # Integrand 1 / E(u) where E(u) = sqrt(Omega_m * (1+u)^3 + Omega_L)
    integrand = 1.0 / np.sqrt(Omega_m * (1.0 + u) ** 3 + Omega_L)

    # Perform the integration: integral of 1/E(u) du from 0 to z
    # Since u = z * t, du = z * dt
    dt = 1.0 / (N - 1)
    integral = (
        z
        * dt
        * (
            np.sum(integrand, axis=1)
            - 0.5 * (integrand[:, 0] + integrand[:, -1])
        )
    )

    # Luminosity distance d_L in Mpc
    d_L = (c / H0) * (1.0 + z) * integral

    # Distance modulus mu = 5 * log10(d_L / 10 pc) = 5 * log10(d_L_Mpc) + 25
    mu = 5.0 * np.log10(d_L) + 25.0

    return mu

    integrand = 1.0 / Ez

    # dx step size for each z
    dx = z / (N - 1)

    integral = (
        np.sum(integrand[:, 1:-1], axis=1)
        + 0.5 * (integrand[:, 0] + integrand[:, -1])
    ) * dx

    # Luminosity distance d_L in Mpc
    dL = (c / H0) * (1.0 + z) * integral

    # Avoid division/log errors for extremely small distances
    dL = np.maximum(dL, 1e-10)

    # Distance modulus: mu = 5 * log10(dL / 10 pc) = 5 * log10(dL_Mpc) + 25
    mu = 5.0 * np.log10(dL) + 25.0

    return mu

import numpy as np


def predict(X):
    # Extract Hubble-diagram CMB-frame redshift
    z = X[:, 0]

    # Cosmological parameters (Flat LCDM, Planck-like)
    H0 = 70.0  # km/s/Mpc
    Om = 0.3  # Matter density parameter
    Ode = 0.7  # Dark energy density parameter (1 - Om)
    c = 299792.458  # Speed of light in km/s

    mu = np.zeros_like(z)

    # Compute distance modulus for each redshift
    for i, zi in enumerate(z):
        if zi <= 0:
            mu[i] = np.nan
            continue

        # Numerical integration of 1/E(z) using trapezoidal rule
        num_steps = 200
        x = np.linspace(0, zi, num_steps)
        integrand = 1.0 / np.sqrt(Om * (1 + x) ** 3 + Ode)

        dx = zi / (num_steps - 1)
        integral = dx * (np.sum(integrand) - 0.5 * (integrand[0] + integrand[-1]))

        # Luminosity distance in Mpc
        dL = (c / H0) * (1 + zi) * integral

        # Distance modulus
        mu[i] = 5.0 * np.log10(dL) + 25.0

    return mu

import numpy as np


def predict(X):
    # Extract Hubble-diagram redshift
    z = X[:, 0]

    # Cosmological parameters (Standard flat Lambda-CDM)
    H0 = 70.0  # Hubble constant in km/s/Mpc
    Om = 0.30  # Matter density parameter
    Ol = 0.70  # Dark energy density parameter (1 - Om)
    c = 299792.458  # Speed of light in km/s

    # Vectorized numerical integration using the midpoint rule
    # to compute the comoving distance integral: \int_0^z dx / E(x)
    N = 1000
    dz = z / N
    steps = np.arange(N)[:, None] + 0.5
    z_grid = steps * dz[None, :]

    E_z = np.sqrt(Om * (1.0 + z_grid) ** 3 + Ol)
    integral = np.sum(1.0 / E_z, axis=0) * dz

    # Luminosity distance in Mpc
    dL = (1.0 + z) * (c / H0) * integral

    # Distance modulus mu = 5 * log10(dL / 10 pc) = 5 * log10(dL_Mpc) + 25
    mu = 5.0 * np.log10(dL) + 25.0

    return mu
