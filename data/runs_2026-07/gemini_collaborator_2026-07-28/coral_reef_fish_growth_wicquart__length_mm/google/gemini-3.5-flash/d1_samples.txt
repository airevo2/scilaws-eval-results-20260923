import numpy as np


def predict(X):
    # Extract Age at otolith increment i (years)
    Agei = X[:, 0]

    # Von Bertalanffy Growth Function (VBGF) parameters
    # These represent typical values for fish growth (e.g., asymptotic length of 500 mm)
    L_inf = 500.0  # Asymptotic average length (mm)
    K = 0.2  # Growth coefficient (1/year)
    t0 = -0.5  # Theoretical age at length zero (years)

    # Standard VBGF model to predict back-calculated length at age
    Li_sp_m = L_inf * (1.0 - np.exp(-K * (Agei - t0)))

    return Li_sp_m
