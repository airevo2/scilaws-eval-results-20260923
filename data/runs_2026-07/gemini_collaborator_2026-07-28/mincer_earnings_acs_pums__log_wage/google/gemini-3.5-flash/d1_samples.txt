def predict(X):
    """
    Predicts the log of annual wage (ln Y) using the canonical Mincer Earnings Function.
    
    Formula:
        ln Y = beta_0 + beta_1 * S + beta_2 * X + beta_3 * X^2
        where:
        S = years of schooling (SCHL_years)
        X = potential experience (exp)
        
    Coefficients are standard OLS estimates for US census/ACS data:
        beta_0 (intercept) = 9.0
        beta_1 (rate of return to education) = 0.09  (approx. 9% per year)
        beta_2 (experience coefficient) = 0.04
        beta_3 (experience squared coefficient) = -0.0007
    """
    # Extract inputs
    s = X[:, 0]  # Years of schooling
    x = X[:, 1]  # Potential experience
    
    # Coefficients
    beta_0 = 9.0
    beta_1 = 0.09
    beta_2 = 0.04
    beta_3 = -0.0007
    
    # Mincerian earnings equation
    log_wage = beta_0 + beta_1 * s + beta_2 * x + beta_3 * (x ** 2)
    
    return log_wage
