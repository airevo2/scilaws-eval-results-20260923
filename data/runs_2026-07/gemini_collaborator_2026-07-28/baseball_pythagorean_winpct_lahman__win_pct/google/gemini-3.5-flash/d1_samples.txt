import numpy as np


def predict(X):
    """Predicts the season win percentage (W%) using the Pythagorean expectation.

    Inputs:
    - X: 2-D numpy array where:
        X[:, 0] = R (Runs Scored)
        X[:, 1] = RA (Runs Allowed)

    Returns:
    - 1-D numpy array of predicted win percentages.
    """
    R = X[:, 0]
    RA = X[:, 1]

    # The standard empirically derived exponent for baseball is 1.83
    exponent = 1.83

    R_exp = np.power(R, exponent)
    RA_exp = np.power(RA, exponent)

    win_pct = R_exp / (R_exp + RA_exp)

    return win_pct
