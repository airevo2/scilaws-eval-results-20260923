import numpy as np


def predict(X):
    # Extract principal quantum number n and orbital angular momentum l
    n = X[:, 0]
    l = X[:, 1]

    # Rydberg constant for infinite mass in cm^-1
    R_y = 109737.31568

    # Empirical quantum defect delta(l) which decreases as l increases.
    # This represents a typical average behavior for neutral Rydberg atoms.
    delta = 1.5 / (l + 0.5) ** 2

    # Avoid division by zero or unphysical values if n is extremely small
    effective_n = np.maximum(n - delta, 0.5)

    # Rydberg formula for binding energy
    E_bind = R_y / (effective_n**2)

    return E_bind

import numpy as np


def predict(X):
    # Extract principal quantum number n and orbital angular momentum l
    n = X[:, 0]
    l = X[:, 1]

    # Rydberg constant for infinite mass in cm^-1
    R_inf = 109737.315685

    # Representative quantum defects (delta) for a typical neutral atom.
    # Quantum defects decrease rapidly as the orbital angular momentum l increases
    # because the centrifugal barrier prevents the electron from penetrating the core.
    delta_table = np.array([1.35, 0.85, 0.15, 0.05, 0.01, 0.0])

    # Map l to the quantum defect, capping at the maximum indexed l
    l_idx = np.clip(l.astype(int), 0, len(delta_table) - 1)
    delta = delta_table[l_idx]

    # Calculate effective principal quantum number (n* = n - delta)
    # Ensure n_eff is positive to avoid division by zero
    n_eff = np.maximum(n - delta, 0.5)

    # Rydberg formula for binding energy
    E_bind = R_inf / (n_eff**2)

    return E_bind
