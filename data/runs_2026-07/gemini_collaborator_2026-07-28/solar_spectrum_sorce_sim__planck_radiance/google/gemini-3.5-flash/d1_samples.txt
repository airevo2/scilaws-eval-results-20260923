import numpy as np


def predict(X):
    # X[:, 0] is wavelength in nm
    wavelength_nm = X[:, 0]

    # Physical constants
    h = 6.62607015e-34  # Planck constant (J s)
    c = 299792458.0  # Speed of light (m/s)
    k_B = 1.380649e-23  # Boltzmann constant (J/K)
    T = 5778.0  # Canonical solar effective temperature (K)

    # Convert wavelength from nm to m
    wavelength_m = wavelength_nm * 1e-9

    # First and second radiation constants
    c1 = 2.0 * h * c**2
    c2 = (h * c) / k_B

    # Calculate Planck spectral radiance in W m^-2 m^-1 sr^-1
    exponent = c2 / (wavelength_m * T)
    # Clip exponent to avoid numerical overflow in np.exp
    exponent = np.clip(exponent, None, 700.0)
    B_lambda = c1 / (wavelength_m**5 * (np.exp(exponent) - 1.0))

    # Convert to W m^-2 nm^-1 sr^-1 (multiply by 1e-9)
    L = B_lambda * 1e-9

    return L
