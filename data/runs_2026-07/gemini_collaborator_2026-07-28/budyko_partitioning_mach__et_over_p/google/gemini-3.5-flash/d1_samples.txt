import numpy as np


def predict(X):
    """Calculates the long-term mean annual ratio of actual evapotranspiration

    to precipitation (E/P) using Budyko's classic semi-empirical relationship.

    Parameters:
    X : 2-D numpy array where:
        X[:, 0] = pet_over_p (phi), the climate aridity index (Ep/P)

    Returns:
    1-D numpy array of E/P
    """
    phi = X[:, 0]

    # Prevent division by zero for extremely dry/wet or boundary cases
    phi_safe = np.maximum(phi, 1e-12)

    # Budyko (1974) canonical formula: E/P = [ phi * tanh(1/phi) * (1 - exp(-phi)) ]^0.5
    et_over_p = np.sqrt(
        phi_safe * np.tanh(1.0 / phi_safe) * (1.0 - np.exp(-phi_safe))
    )

    # For any actual 0 values in phi, E/P should be 0
    et_over_p = np.where(phi == 0, 0.0, et_over_p)

    return et_over_p

    # Set E/P to 0 where phi is 0 or negative
    ep = np.where(phi > 0, ep, 0.0)

    return ep
