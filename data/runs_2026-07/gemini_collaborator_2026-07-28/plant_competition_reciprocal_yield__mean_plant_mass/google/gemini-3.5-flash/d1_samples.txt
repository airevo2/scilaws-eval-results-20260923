import numpy as np


def predict(X):
    # Extract density (N) in plants/pot
    N = X[:, 0]

    # Shinozaki-Kira reciprocal yield-density model:
    # w = 1 / (a + b * N)
    # where:
    # - 1/a is the asymptotic maximum individual plant biomass at zero competition (N -> 0)
    # - 1/b is the asymptotic maximum total yield per pot (carrying capacity) as N -> infinity
    # Typical values chosen for herbaceous plants in a pot experiment:
    a = 0.25  # Corresponds to a maximum individual dry mass of ~4.0 g
    b = 0.05  # Corresponds to a maximum total pot yield of ~20.0 g

    w = 1.0 / (a + b * N)
    return w

def predict(X):
    """Predicts the mean oven-dry mass per surviving plant (w) in grams

    using the reciprocal equation of plant competition (Shinozaki & Kira, 1956).

    Parameters:
    X : 2-D numpy array
        X[:, 0] = density (N) [plants/pot]

    Returns:
    w : 1-D numpy array
        Mean oven-dry mass per surviving plant [g/plant]
    """
    # Extract density (N)
    N = X[:, 0]

    # Kira's reciprocal equation parameters:
    # w = 1 / (a + b * N)
    # where:
    # - 1/a is the theoretical maximum mass of an isolated plant (N -> 0)
    # - 1/b is the theoretical maximum asymptotic yield per pot (N -> infinity)
    a = 0.10
    b = 0.02

    w = 1.0 / (a + b * N)

    return w

import numpy as np


def predict(X):
    # X[:, 0] is the density of surviving plants at harvest (N) [plants/pot]
    N = X[:, 0]

    # Reciprocal equation of plant competition (Shinozaki & Kira, 1956),
    # which describes the relationship between mean plant mass (w) and density (N):
    # w = 1 / (a * N + b)
    # where 1/a represents the asymptotic maximum yield per pot (constant final yield)
    # and 1/b represents the potential biomass of an isolated individual plant.
    # Representative parameters fitted for typical herbaceous species in pots:
    a = 0.125
    b = 0.375

    w = 1.0 / (a * N + b)
    return w
