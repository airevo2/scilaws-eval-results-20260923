import numpy as np


def predict(X):
    """Predicts the absorbance (A) of a potassium dichromate solution

    using the Beer-Lambert law at a standard calibration wavelength of 350 nm.

    Parameters:
    X : ndarray of shape (N, 2)
        X[:, 0] : concentration (c) in g/L
        X[:, 1] : path_length (l) in cm

    Returns:
    A : ndarray of shape (N,)
        Absorbance (dimensionless)
    """
    c = X[:, 0]
    l = X[:, 1]

    # Specific absorptivity (epsilon) of potassium dichromate in acidic solution
    # (0.005 M H2SO4) at 350 nm is approximately 10.7 L/(g*cm).
    epsilon = 10.7  # L / (g * cm)

    # Beer-Lambert Law: A = epsilon * c * l
    A = epsilon * c * l

    return A
