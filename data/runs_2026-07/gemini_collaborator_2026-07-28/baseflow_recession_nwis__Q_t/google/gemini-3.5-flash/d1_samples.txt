import numpy as np


def predict(X):
    """
    Predicts the daily-mean streamflow discharge Q(t) during a recession event
    using the classic Maillet (1905) exponential recession model.

    Inputs:
    - X: 2-D numpy array where:
        - X[:, 0] = t (elapsed time in days since the start of recession)
        - X[:, 1] = Q_0 (observed discharge at the start of the recession, t=0)

    Returns:
    - 1-D numpy array of predicted streamflow discharge Q_t [ft3/s]
    """
    t = X[:, 0]
    Q_0 = X[:, 1]

    # k is the daily recession depletion coefficient (typical range: 0.01 to 0.2 day^-1)
    # A standard median value of 0.1 day^-1 is used here.
    k = 0.1

    Q_t = Q_0 * np.exp(-k * t)

    return Q_t

    using the classical Maillet exponential recession model.

    Inputs:
    - X: 2-D numpy array where:

    Returns:
    - Q_t: 1-D numpy array of predicted streamflow discharge [ft3/s]
    """

    Q_0 = X[:, 1]

    # k is the daily recession constant (dimension day^-1).
    # A value of 0.1 is highly representative of typical basin recession rates.

    return Q_t
