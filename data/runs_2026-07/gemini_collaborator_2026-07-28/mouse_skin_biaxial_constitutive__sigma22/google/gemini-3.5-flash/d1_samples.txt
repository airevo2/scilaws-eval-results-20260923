def predict(X):
    # Inputs: Stretch ratios in lateral (11) and cranial-caudal (22) directions
    lambda11 = X[:, 0]
    lambda22 = X[:, 1]

    # Compute Green-Lagrange strains
    E11 = 0.5 * (lambda11**2 - 1.0)
    E22 = 0.5 * (lambda22**2 - 1.0)

    # Canonical Fung-type orthotropic model parameters for mouse skin (in MPa)
    c = 0.05  # Scaling parameter
    a1 = 1.8  # Stiffness parameter in lateral direction
    a2 = 2.2  # Stiffness parameter in cranial-caudal direction
    a3 = 0.4  # Shear/coupling interaction parameter

    # Exponent Q in the Fung strain energy density function
    Q = a1 * E11**2 + a2 * E22**2 + 2.0 * a3 * E11 * E22

    # Compute second principal Cauchy stress sigma22
    sigma22 = (lambda22**2) * c * np.exp(Q) * (a2 * E22 + a3 * E11)

    return sigma22
