import numpy as np

def predict(X):
    # X[:, 0] is the adult body mass in grams
    body_mass_g = X[:, 0]
    
    # Convert body mass to kilograms
    body_mass_kg = body_mass_g / 1000.0
    
    # Kleiber's Law: BMR (Watts) = a * (M_kg ** 0.75)
    # Standard coefficient 'a' for mammals is ~3.39 Watts/kg^0.75 (equivalent to 70 kcal/day/kg^0.75)
    bmr_W = 3.39 * (body_mass_kg ** 0.75)
    
    # Return the base-10 logarithm of BMR in Watts
    return np.log10(bmr_W)

import numpy as np

def predict(X):
    # X[:, 0] is body mass in grams
    body_mass_g = X[:, 0]
    
    # Convert body mass to kilograms
    body_mass_kg = body_mass_g / 1000.0
    
    # Kleiber's Law: BMR (Watts) = 3.39 * (M_kg ** 0.75)
    # This is derived from the standard 70 kcal/day scaling constant:
    # 70 kcal/day * (4184 J / kcal) / (86400 s / day) ≈ 3.39 Watts
    bmr_W = 3.39 * np.power(body_mass_kg, 0.75)
    
    return np.log10(bmr_W)
