import numpy as np


def predict(X):
    # X[:, 0] = gdppc (Y) [USD_1963]
    Y = X[:, 0]

    # Preston (1975) estimated the 1960s relationship using a logistic function:
    # e0 = 80 / (1 + exp(a) * Y**(-b))
    # where a = 2.654 and b = 0.575
    # Mathematically equivalent to: 80 / (1 + exp(2.654 - 0.575 * ln(Y)))
    e0 = 80.0 / (1.0 + np.exp(2.654 - 0.575 * np.log(Y)))

    return e0

import numpy as np

def predict(X):
    """
    Predicts the sex-averaged life expectancy at birth (e0) from GDP per capita
    expressed in constant 1963 US dollars, using the standard logistic Preston Curve
    relationship estimated for the 1960s (Preston, 1975).
    
    Inputs:
    - X: 2-D numpy array where X[:, 0] is gdppc (Y) in USD_1963.
    
    Returns:
    - 1-D numpy array of predicted e0 in years.
    """
    Y = X[:, 0]
    # Standard Preston (1975) logistic formulation for the 1960s
    e0 = 80.0 / (1.0 + np.exp(0.6125 - 0.00127 * Y))
    return e0
