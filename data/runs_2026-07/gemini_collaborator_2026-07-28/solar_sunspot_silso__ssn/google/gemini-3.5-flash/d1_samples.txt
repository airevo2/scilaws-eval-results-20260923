    Predicts the monthly mean total International Sunspot Number (ISN v2.0 scale)
    using a canonical multi-harmonic model representing the primary solar cycles:
    the ~11.06-year Schwabe cycle and the ~88-year Gleissberg cycle.
    
    Parameters:
    X : np.ndarray

        - X[:, 1] = year (integer calendar year)
        - X[:, 2] = month (calendar month, 1 to 12)
        
    Returns:
    np.ndarray

    """

    t = X[:, 0]

    # Mean sunspot number in v2.0 is higher than v1.0 (by a factor of ~1.4)
    mean_ssn = 82.0
    
    # Schwabe Cycle (primary ~11-year solar cycle)
    amp_schwabe = 68.0
    period_schwabe = 11.06
    phase_schwabe = 2014.3  # Approximate peak of Solar Cycle 24
    
    # Gleissberg Cycle (~88-year modulation envelope)
    amp_gleissberg = 18.0
    period_gleissberg = 88.0
    phase_gleissberg = 1980.0
    
    # Calculate the cyclic components
    theta_schwabe = 2 * np.pi * (t - phase_schwabe) / period_schwabe
    theta_gleissberg = 2 * np.pi * (t - phase_gleissberg) / period_gleissberg
    
    schwabe = amp_schwabe * np.cos(theta_schwabe)
    gleissberg = amp_gleissberg * np.cos(theta_gleissberg)
    
    # Combine components and enforce non-negativity
    r_i = np.maximum(0.0, mean_ssn + schwabe + gleissberg)
    
    return r_i

        [1996.3, 180.0, 5.6], # Cycle 23
        [2008.9, 116.0, 5.4], # Cycle 24
        [2019.8, 160.0, 4.7]  # Cycle 25
    ])

    contrib = np.where(mask, contrib, 0.0)
    
    # Sum overlapping active cycles to model smooth transitions at minima
    ssn = np.sum(contrib, axis=1)
    
    return ssn

    schwabe_fundamental = 48.5 * np.cos(omega_s * (t - 2000.3))
    schwabe_harmonic = 15.2 * np.cos(2 * omega_s * (t - 2000.3) - 0.8)

    # Gleissberg cycle amplitude modulation
    gleissberg_modulation = 1.0 + 0.25 * np.cos(omega_g * (t - 1980.0))

    # Combine into the International Sunspot Number (ISN v2.0 scale)
    ssn = 55.0 + (schwabe_fundamental + schwabe_harmonic) * gleissberg_modulation

    # Sunspot numbers must be non-negative

import numpy as np


def predict(X):
    """Predicts the Monthly mean total International Sunspot Number (ISN v2.0 scale)

    using Hathaway's canonical solar cycle formula for an average 11-year cycle.

    Parameters:
    - X: 2-D numpy array where:
        - X[:, 0] = t_year (decimal year)
        - X[:, 1] = year (integer year)
        - X[:, 2] = month (1 to 12)

    Returns:
    - 1-D numpy array of predicted sunspot numbers (R_I)
    """
    t = X[:, 0]

    # Parameters for the average solar cycle (Schwabe cycle)
    t_start = 1755.2  # Start of Solar Cycle 1 (reference epoch)
    period = 11.04  # Average cycle duration in years

    # Compute phase time (years since start of the current cycle)
    tau = np.mod(t - t_start, period)

    # Hathaway's simplified solar cycle shape parameters (scaled for ISN v2.0)
    a = 3.5
    b = 4.0
    c = 0.7

    # Hathaway's formula: R(t) = a * tau^3 / [exp((tau/b)^2) - c]
    # Written as exp(2 * tau / b) for numerical stability and standard scaling
    ssn = a * (tau**3) / (np.exp(2 * tau / b) - c)

    return ssn
