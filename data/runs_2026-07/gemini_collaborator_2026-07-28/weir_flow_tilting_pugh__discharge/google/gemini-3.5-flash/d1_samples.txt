import numpy as np

def predict(X):
    """
    Predicts the total volumetric discharge (Q) over a suppressed rectangular 
    sharp-crested weir using the Kindsvater-Carter formula (ISO 1438 standard).
    
    Inputs:
    - X: 2-D numpy array where:
        - X[:, 0] = h [m], upstream head over the weir crest
        - X[:, 1] = p [m], height of the weir crest above the channel bed
        - X[:, 2] = b [m], length of the weir crest (channel width)
        
    Returns:
    - Q: 1-D numpy array of volumetric discharge [m³ s⁻¹]
    """
    h = X[:, 0]
    p = X[:, 1]
    b = X[:, 2]
    
    g = 9.80665  # Acceleration due to gravity [m s⁻²]
    
    # Kindsvater-Carter discharge coefficient for a suppressed weir
    C_e = 0.602 + 0.075 * (h / p)
    
    # Effective head (accounting for fluid properties/surface tension)
    h_e = h + 0.001
    
    # Kindsvater-Carter formula
    Q = (2.0 / 3.0) * np.sqrt(2 * g) * C_e * b * (h_e ** 1.5)
    
    return Q

import numpy as np


def predict(X):
    # Inputs
    h = X[:, 0]  # Upstream head over weir [m]
    p = X[:, 1]  # Weir height [m]
    b = X[:, 2]  # Weir crest length / channel width [m]

    # Constants
    g = 9.81  # Acceleration due to gravity [m s^-2]

    # Kindsvater-Carter formula for a full-width (suppressed) sharp-crested weir
    # Discharge coefficient Cd as a function of h/p
    C_d = 0.602 + 0.075 * (h / p)

    # Effective head (incorporating surface tension and viscosity corrections)
    h_e = h + 0.0012

    # Volumetric discharge Q [m^3 s^-1]
    Q = (2.0 / 3.0) * C_d * np.sqrt(2.0 * g) * b * (h_e**1.5)

    return Q
