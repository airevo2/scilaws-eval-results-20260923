import numpy as np


def predict(X):
    """Predicts the indentation force (F) using the Hertzian contact model

    for a spherical probe on a flat elastic half-space.

    Parameters:
    X : 2-D numpy array
        X[:, 0] = indentation depth (delta) [m]
        X[:, 1] = probe radius (R) [m]

    Returns:
    F : 1-D numpy array
        Indentation force [N]
    """
    delta = X[:, 0]
    R = X[:, 1]

    # Effective Young's modulus (E*) in Pa.
    # A typical value for soft biological samples (e.g., cells or embryos) is ~1 kPa.
    E_star = 1000.0

    # Hertz model: F = (4/3) * E* * R^(1/2) * delta^(3/2)
    F = (4.0 / 3.0) * E_star * np.sqrt(R) * (np.maximum(delta, 0) ** 1.5)

    return F
