import numpy as np


def predict(X):
    """Predicts the fluorometric enzyme activity rate (v) in soil using the

    canonical Michaelis-Menten kinetic model.

    Inputs:
    - X: 2-D numpy array where X[:, 0] is the substrate concentration (S) [uM].

    Outputs:
    - v: 1-D numpy array of enzyme activity rate [nmol g^-1 h^-1].
    """
    # Extract substrate concentration (S)
    S = X[:, 0]

    # Standard representative Michaelis-Menten parameters for soil enzymes
    # (e.g., BG, AP, LAP) under typical assay conditions:
    # Vmax: Maximum enzyme activity rate (typical range: 100 - 1000 nmol g^-1 h^-1)
    # Km: Michaelis constant / half-saturation constant (typical range: 30 - 150 uM)
    V_max = 350.0  # nmol g^-1 h^-1
    K_m = 100.0  # uM

    # Michaelis-Menten equation
    v = (V_max * S) / (K_m + S)

    return v
