import numpy as np


def predict(X):
    """Predicts the absolute Gaia Wesenheit magnitude (M_W) for classical Cepheids

    using the Period-Luminosity-Metallicity (PLZ) relation from Gaia DR3 (Ripepi et al. 2023).

    Inputs:
    - X: 2-D numpy array where:
        - X[:, 0] = log_P: Base-10 logarithm of the fundamental-mode period in days.
        - X[:, 1] = feh: Metallicity [Fe/H] in dex.

    Returns:
    - 1-D numpy array of M_W.
    """
    log_P = X[:, 0]
    feh = X[:, 1]

    # Standard PLZ relation for Fundamental-mode Classical Cepheids in the Gaia Wesenheit band:
    # M_W = alpha * (log_P - 1.0) + beta * [Fe/H] + gamma
    alpha = -3.268
    beta = -0.11
    gamma = -5.92

    M_W = alpha * (log_P - 1.0) + beta * feh + gamma
    return M_W
