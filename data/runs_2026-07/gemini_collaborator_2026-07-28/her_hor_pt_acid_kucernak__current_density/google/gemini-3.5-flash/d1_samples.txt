import numpy as np


def predict(X):
    # Inputs
    eta = X[:, 0]  # Overpotential [V]
    T = X[:, 1]  # Temperature [K]
    P_H2 = X[:, 2]  # H2 partial pressure [bar]
    c_acid = X[:, 3]  # Acid concentration [mol dm-3]

    # Physical constants
    F = 96485.33  # Faraday constant [C mol-1]
    R = 8.31446  # Gas constant [J mol-1 K-1]

    # Kinetic parameters for Pt HOR/HER
    T_ref = 298.15  # Reference temperature [K]
    P_ref = 1.0  # Reference pressure [bar]
    c_ref = 0.5  # Reference acid concentration [mol dm-3]

    # Exchange current density at reference conditions [A cm-2]
    # For highly active Pt in acid (floating electrode), j0_ref is typically ~ 20-40 mA cm-2
    j0_ref = 0.030

    # Activation energy [J mol-1]
    E_a = 24000.0

    # Reaction orders
    m_H2 = 0.5  # Reaction order with respect to H2
    m_H = 0.5  # Reaction order with respect to protons (acid)

    # Calculate exchange current density j0 as a function of T, P_H2, and c_acid
    j0 = (
        j0_ref
        * (P_H2 / P_ref) ** m_H2
        * (c_acid / c_ref) ** m_H
        * np.exp(-(E_a / R) * (1.0 / T - 1.0 / T_ref))
    )

    # Butler-Volmer equation (transfer coefficients alpha_a = alpha_c = 0.5)
    alpha_a = 0.5
    alpha_c = 0.5

    j = j0 * (
        np.exp(alpha_a * F * eta / (R * T))
        - np.exp(-alpha_c * F * eta / (R * T))
    )

    return j

import numpy as np


def predict(X):
    # Constants
    F = 96485.33  # Faraday constant, C/mol
    R = 8.31446  # Gas constant, J/(mol*K)
    T_ref = 298.15  # Reference temperature, K

    # Inputs
    eta = X[:, 0]  # Overpotential, V vs RHE
    T = X[:, 1]  # Temperature, K
    P_H2 = X[:, 2]  # H2 partial pressure, bar
    c_acid = X[:, 3]  # Acid concentration, mol/dm3 (notably 0.5 M or 4.0 M)

    # Kinetic parameters for HOR/HER on Pt (derived from floating electrode studies)
    j0_ref = 0.067  # Reference exchange current density, A/cm2
    E_a = 24000.0  # Activation energy, J/mol
    gamma = 0.5  # Reaction order with respect to H2 pressure
    alpha = 0.5  # Charge transfer coefficient (symmetric barrier)

    # Exchange current density scaling with T and P_H2
    # The effect of acid concentration on Pt HOR/HER kinetics in highly acidic
    # media is typically negligible when evaluated on the RHE scale.
    j0 = (
        j0_ref
        * (P_H2**gamma)
        * np.exp(-(E_a / R) * (1.0 / T - 1.0 / T_ref))
    )

    # Standard Butler-Volmer formulation for HOR/HER current density
    j = j0 * (
        np.exp(alpha * F * eta / (R * T)) - np.exp(-alpha * F * eta / (R * T))
    )

    return j
