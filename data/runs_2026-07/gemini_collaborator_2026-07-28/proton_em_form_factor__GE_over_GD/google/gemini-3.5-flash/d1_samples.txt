import numpy as np


def predict(X):
    # X[:, 0] is the four-momentum transfer squared Q^2 in GeV^2
    Q2 = X[:, 0]

    # Proton mass in GeV
    M_p = 0.938272
    tau = Q2 / (4.0 * M_p**2)

    # Kelly (2004) parametrization for the proton electric form factor G_E
    a1 = -0.24
    b1 = 10.98
    b2 = 12.82
    b3 = 21.97

    G_E = (1.0 + a1 * tau) / (1.0 + b1 * tau + b2 * tau**2 + b3 * tau**3)

    # Standard dipole form factor G_D
    G_D = (1.0 + Q2 / 0.71) ** (-2)

    # Ratio G_E / G_D
    return G_E / G_D

    # Kelly (2004) parameterization coefficients for G_E

    b1 = 10.98

    b3 = 21.97

    # Calculate G_E using the rational function parameterization

    # Return the ratio G_E / G_D
    return G_E / G_D
