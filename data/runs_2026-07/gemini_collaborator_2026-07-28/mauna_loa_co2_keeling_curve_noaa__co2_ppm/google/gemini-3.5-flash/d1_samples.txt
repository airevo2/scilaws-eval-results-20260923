def predict(X):
    """Predicts the monthly mean CO2 mole fraction (in ppm) at Mauna Loa

    Observatory using a standard quadratic trend and two-harmonic seasonal
    model.

    Parameters:
    - X: 2-D numpy array where X[:, 0] is the decimal year (t).

    Returns:
    - 1-D numpy array of predicted CO2 concentrations in ppm.
    """
    t = X[:, 0]

    # Reference epoch (1958.0, near the start of the Mauna Loa record)
    t0 = 1958.0
    dt = t - t0

    # Quadratic trend coefficients (empirically fitted for 1958-2024)
    a0 = 314.57
    a1 = 0.792
    a2 = 0.0129

    # Fourier coefficients for the seasonal cycle (annual and semi-annual harmonics)
    # Peak in May, trough in October/November
    c1, s1 = -2.12, 2.12  # First harmonic (annual)
    c2, s2 = 0.79, -0.36  # Second harmonic (semi-annual)

    # Compute trend
    trend = a0 + a1 * dt + a2 * dt**2

    # Compute seasonal cycle
    omega = 2 * np.pi * t
    seasonal = (
        c1 * np.cos(omega)
        + s1 * np.sin(omega)
        + c2 * np.cos(2 * omega)
        + s2 * np.sin(2 * omega)
    )

    return trend + seasonal
