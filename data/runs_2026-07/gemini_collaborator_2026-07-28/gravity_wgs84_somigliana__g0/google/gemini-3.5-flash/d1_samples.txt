import numpy as np


def predict(X):
    # Convert latitude from degrees to radians
    phi = np.radians(X[:, 0])

    # WGS84 Ellipsoid Constants
    gamma_e = 9.7803253359  # Normal gravity at equator [m/s^2]
    k = 0.00193185265241  # Somigliana's constant
    e2 = 0.00669437999014  # First eccentricity squared

    # Somigliana closed-form equation for normal gravity
    sin2_phi = np.sin(phi) ** 2
    gamma = gamma_e * (1 + k * sin2_phi) / np.sqrt(1 - e2 * sin2_phi)

    return gamma
