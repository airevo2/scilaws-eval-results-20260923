import numpy as np


def predict(X):
    """Estimates the annual national primary energy consumption (I_E) in TJ

    using a calibrated STIRPAT (Stochastic Impacts by Regression on Population,
    Affluence, and Technology) formulation.

    Inputs:
    - X[:, 0] = GDP per capita (constant 2015 USD)
    - X[:, 1] = Total population (persons)
    - X[:, 2] = Working-age fraction (%)
    - X[:, 3] = Population density (persons/km2)
    - X[:, 4] = Urbanization rate (%)
    - X[:, 5] = Mean annual temperature (°C)
    """
    # Extract features
    A = np.maximum(X[:, 0], 1.0)  # GDP per capita
    P = np.maximum(X[:, 1], 1.0)  # Population
    f_active = X[:, 2]  # Active population (%)
    rho = np.maximum(X[:, 3], 0.1)  # Population density
    f_urban = X[:, 4]  # Urbanization (%)
    T = X[:, 5]  # Temperature (°C)

    # STIRPAT model coefficients
    beta_0 = -11.26
    beta_P = 1.02
    beta_A = 0.85
    beta_rho = -0.04
    beta_urban = 0.003
    beta_active = 0.005
    beta_T = -0.005

    # Compute predicted energy consumption in log space, then exponentiate
    log_IE = (
        beta_0
        + beta_P * np.log(P)
        + beta_A * np.log(A)
        + beta_rho * np.log(rho)
        + beta_urban * f_urban
        + beta_active * f_active
        + beta_T * T
    )

    return np.exp(log_IE)

import numpy as np


def predict(X):
    """Predicts the annual national primary energy consumption (I_E) in TJ

    using a STIRPAT-based macroeconomic formulation.

    Parameters:
    X : 2-D numpy array
        X[:, 0] = GDP (A) [constant 2015 USD per capita]
        X[:, 1] = TP (P) [persons]
        X[:, 2] = AP (f_active) [%]
        X[:, 3] = DP (rho) [persons per km2]
        X[:, 4] = UR (f_urban) [%]
        X[:, 5] = AT (T) [degrees C]

    Returns:
    1-D numpy array of predicted energy consumption in TJ.
    """
    A = X[:, 0]
    P = X[:, 1]
    f_active = X[:, 2]
    rho = X[:, 3]
    f_urban = X[:, 4]
    T = X[:, 5]

    # Model coefficients based on empirical global energy-economic relationships
    beta_0 = -13.5
    beta_1 = 1.00  # Population elasticity
    beta_2 = 0.95  # Affluence (GDP per capita) elasticity
    beta_3 = 0.50  # Working-age population share elasticity
    beta_4 = -0.05  # Population density elasticity (efficiency/compactness)
    beta_5 = 0.25  # Urbanization elasticity
    beta_6 = -0.005  # Temperature effect (heating demand proxy)

    # Prevent potential log of zero or negative values
    eps = 1e-5
    ln_P = np.log(np.maximum(P, eps))
    ln_A = np.log(np.maximum(A, eps))
    ln_active = np.log(np.maximum(f_active, eps))
    ln_rho = np.log(np.maximum(rho, eps))
    ln_urban = np.log(np.maximum(f_urban, eps))

    ln_I_E = (
        beta_0
        + beta_1 * ln_P
        + beta_2 * ln_A
        + beta_3 * ln_active
        + beta_4 * ln_rho
        + beta_5 * ln_urban
        + beta_6 * T
    )

    return np.exp(ln_I_E)
