def predict(X):
    # Inputs:
    # X[:, 0] = G (Plane-of-array irradiance, W/m^2)
    # X[:, 1] = T_c (Module temperature, deg C)
    # X[:, 2] = WS (Wind speed, m/s)
    # X[:, 3] = AM (Air mass, dimensionless)
    G = X[:, 0]
    T_c = X[:, 1]
    AM = X[:, 3]

    # Nominal STC Parameters for a typical crystalline silicon module
    P_stc = 250.0  # Rated power at STC [W]
    G_ref = 1000.0  # Reference irradiance [W/m^2]
    T_ref = 25.0  # Reference temperature [deg C]
    gamma = -0.0041  # Temperature coefficient of power [1/deg C]

    # Air Mass modifier (SAPM generic crystalline silicon polynomial)
    a0 = 0.98485
    a1 = 0.03456
    a2 = -0.01522
    a3 = 0.00263
    a4 = -0.000174

    f_AM = a0 + a1 * AM + a2 * AM**2 + a3 * AM**3 + a4 * AM**4
    f_AM = np.clip(f_AM, 0.0, 1.1)

    # Osterwald model for maximum power with temperature and air mass corrections
    P_mp = P_stc * (G / G_ref) * f_AM * (1.0 + gamma * (T_c - T_ref))

    # Power output cannot be negative
    P_mp = np.maximum(0.0, P_mp)

    return P_mp
