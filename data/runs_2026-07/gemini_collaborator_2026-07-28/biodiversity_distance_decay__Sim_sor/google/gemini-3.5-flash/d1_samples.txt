import numpy as np


def predict(X):
    # Extract the mean environmental distance (d)
    d = X[:, 0]

    # Standard exponential distance-decay relationship for ecological similarity:
    # S(d) = S_0 * exp(-decay_rate * d)
    # S_0 is the estimated taxonomic similarity at zero environmental distance.
    # decay_rate determines how fast similarity decreases with environmental distance.
    S_0 = 0.65
    decay_rate = 0.25

    Sim_sor = S_0 * np.exp(-decay_rate * d)

    # Ensure the output is strictly bounded between [0, 1]
    return np.clip(Sim_sor, 0.0, 1.0)
