import numpy as np


def predict(X):
    """Predicts the natural log of the central period death rate (ln m)

    using a Gompertz model with a log-linear time trend.

    Inputs:
    - X: 2-D numpy array where:
        X[:, 0] = age (x) [30 to 95]
        X[:, 1] = calendar year (t) [1950 to 2020]

    Returns:
    - 1-D numpy array of ln m
    """
    age = X[:, 0]
    year = X[:, 1]

    # Standard parameterized Gompertz-time model:
    # ln m(x, t) = A + B * x - C * t
    # where B is the rate of aging (mortality doubling time of ~8 years -> B ≈ 0.085)
    # and C is the annual rate of mortality decline (~1.2% per year -> C ≈ 0.012)
    A = 14.5
    B = 0.085
    C = 0.012

    return A + B * age - C * year

    b = 0.085
    c = -0.012

    log_m = a + b * age + c * (year - 2000)
    return log_m
