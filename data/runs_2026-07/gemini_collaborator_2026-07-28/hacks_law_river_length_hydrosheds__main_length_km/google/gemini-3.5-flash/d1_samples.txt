import numpy as np

def predict(X):
    """
    Predicts the main-stem river length (L) in km using Hack's Law,
    a standard empirical scaling relationship in hydrology relating 
    drainage basin area (A) to the main-stem length.

    Inputs:
    - X[:, 0]: upland_area_skm (A) [km^2]
    - X[:, 2]: segment_length_km (L_s) [km]
    """
    A = X[:, 0]
    L_s = X[:, 2]
    
    # Hack's Law in metric units: L = C * A^h
    # Canonical global values: C ≈ 1.4, h ≈ 0.56
    L = 1.4 * (A ** 0.56)
    
    # Physical constraint: The cumulative main-stem length must be 
    # at least as long as the terminal outlet segment length.
    return np.maximum(L, L_s)

import numpy as np

def predict(X):
    """
    Predicts the main-stem river length (L) in km using Hack's Law,
    which is the standard empirical scaling relationship in hydrology
    relating basin drainage area to main-stem length.
    
    Formula: L = C * A**h
    Where:
    - A is the upstream drainage area (upland_area_skm, X[:, 0])
    - C is a metric scaling constant (typically ~1.4)
    - h is the Hack exponent (typically ~0.568)
    
    Inputs:
    - X: 2-D numpy array where X[:, 0] is upland_area_skm
    """
    A = X[:, 0]
    
    # Hack's Law parameters for metric units (km and km^2)
    C = 1.4
    h = 0.568
    
    L = C * (A ** h)
    return L
