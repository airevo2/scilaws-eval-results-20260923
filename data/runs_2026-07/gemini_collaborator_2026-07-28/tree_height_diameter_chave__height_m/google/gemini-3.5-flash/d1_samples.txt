import numpy as np


def predict(X):
    # Extract DBH (cm) from the first column of the input array
    D = X[:, 0]

    # Using the canonical Feldpausch et al. (2011) pantropical Weibull H-D model:
    # H = a * (1 - exp(-b * D^c))
    # where:
    # a (asymptotic maximum height) = 55.4
    # b = 0.059
    # c = 0.744
    a = 55.4
    b = 0.059
    c = 0.744

    H = a * (1.0 - np.exp(-b * (D**c)))
    return H
