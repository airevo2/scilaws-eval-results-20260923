import numpy as np

def predict(X):
    """
    Predicts the natural log of output-side real GDP per person engaged (ln(Y/L))
    using a Cobb-Douglas production function framework.
    
    Inputs:
    - X[:, 0]: cn (K) - Capital stock in millions 2017 USD PPP
    - X[:, 1]: emp (L) - Number of persons engaged in millions
    - X[:, 2]: hc (h) - Human-capital index
    """
    K = X[:, 0]
    L = X[:, 1]
    h = X[:, 2]
    
    # Capital per worker
    k = K / L
    
    # Empirically fitted macroeconomic relationship based on PWT 11.0
    # log(Y/L) = beta_0 + beta_1 * log(K/L) + beta_2 * log(h)
    log_Y_L = 2.5 + 0.65 * np.log(k) + 0.50 * np.log(h)
    
    return log_Y_L

import numpy as np


def predict(X):
    # Inputs:
    # X[:, 0] = cn (K) [millions 2017 USD PPP]
    # X[:, 1] = emp (L) [millions of persons]
    # X[:, 2] = hc (h) [human-capital index]

    K = X[:, 0]
    L = X[:, 1]
    h = X[:, 2]

    # Avoid division by zero or log of zero
    L = np.clip(L, 1e-9, None)
    K = np.clip(K, 1e-9, None)
    h = np.clip(h, 1e-9, None)

    # Standard Cobb-Douglas production function parameters:
    # Y = A * K^alpha * (h * L)^(1-alpha)
    # => Y/L = A * (K/L)^alpha * h^(1-alpha)
    # => ln(Y/L) = ln(A) + alpha * ln(K/L) + (1-alpha) * ln(h)
    # Typical capital share (alpha) is 0.33.
    # The constant term ln(A) is set to a standard empirical average of ~6.5.

    alpha = 0.33
    ln_A = 6.5

    ln_Y_L = ln_A + alpha * np.log(K / L) + (1 - alpha) * np.log(h)

    return ln_Y_L
