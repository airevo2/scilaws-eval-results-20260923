    - X[:, 0] = GDP (A) [constant 2015 USD per capita]
    - X[:, 1] = TP (P) [persons]
    - X[:, 2] = AP (AP) [%] (Share of working-age population)

    TP = X[:, 1]
    AP = X[:, 2]
    DP = X[:, 3]
    UR = X[:, 4]
    AT = X[:, 5]

    beta_0 = -1.50  # Constant intercept
    beta_1 = 1.00  # Population elasticity
    beta_2 = 0.35  # GDP per capita (Affluence) elasticity
    beta_3 = 0.20  # Active population share elasticity

    beta_5 = 0.15  # Urbanization rate elasticity
    beta_6 = 0.01  # Temperature semi-elasticity (waste/agricultural decay)

        + beta_4 * log_DP
        + beta_5 * log_UR
        + beta_6 * AT

import numpy as np


def predict(X):
    """Predicts the national annual CH4 emissions from all sectors excluding

    land-use change and forestry (ME (I_CH4) [kg CH4]) using a STIRPAT-based
    environmental-economic formulation.

    Inputs:
    - X[:, 0]: GDP per capita (constant 2015 USD)
    - X[:, 1]: Total population (persons)
    - X[:, 2]: Share of working-age population (%)
    - X[:, 3]: Population density (persons per km2)
    - X[:, 4]: Urbanization rate (%)
    - X[:, 5]: Annual mean surface air temperature (degrees C)
    """
    # Extract variables and handle potential non-positive values safely
    gdp = np.clip(X[:, 0], 1e-5, None)
    tp = np.clip(X[:, 1], 1e-5, None)
    ap = X[:, 2]
    dp = np.clip(X[:, 3], 1e-5, None)
    ur = X[:, 4]
    at = X[:, 5]

    # STIRPAT model coefficients calibrated for global methane emissions
    b0 = -1.2  # Constant intercept
    b1 = 1.02  # Population elasticity (near-unitary scaling)
    b2 = 0.32  # Affluence (GDP per capita) elasticity
    b3 = 0.008  # Working-age population share coefficient
    b4 = -0.04  # Population density elasticity (efficiency of concentration)
    b5 = 0.003  # Urbanization rate coefficient
    b6 = 0.008  # Temperature coefficient (agricultural/waste decay scaling)

    # Calculate log emissions
    ln_I = (
        b0
        + b1 * np.log(tp)
        + b2 * np.log(gdp)
        + b3 * ap
        + b4 * np.log(dp)
        + b5 * ur
        + b6 * at
    )

    return np.exp(ln_I)
