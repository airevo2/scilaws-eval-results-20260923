import numpy as np


def predict(X):
    # Extract inputs
    B_r = X[:, 0]  # radial magnetic field component [nT]
    v_sw = X[:, 1]  # solar wind speed [km/s]
    r = X[:, 2]  # heliocentric distance [AU]

    # Physical constants
    omega = 2.7e-6  # Solar angular rotation rate [rad/s]
    AU_in_km = 1.495978707e8  # 1 Astronomical Unit [km]

    # Parker spiral relationship: B_phi = - B_r * (omega * r) / v_sw
    # Convert heliocentric distance from AU to km to match v_sw in km/s
    B_phi = -B_r * (omega * r * AU_in_km) / v_sw

    return B_phi
