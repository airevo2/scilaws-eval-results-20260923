import numpy as np


def predict(X):
    """Predicts Forced Vital Capacity (FVC) in liters using a unisex average

    of the standard Hankinson (NHANES III) reference equations for pediatric/
    adolescent populations (ages 8-19).

    Inputs:
    - X[:, 0] = age (t) [years]
    - X[:, 1] = height (h) [cm]

    Output:
    - 1-D numpy array of FVC [L]
    """
    age = X[:, 0]
    height = X[:, 1]

    # Averaged coefficients for boys (8-19) and girls (8-17) from NHANES III:
    # Boys:  -0.2584 - 0.20415 * age + 0.01013 * age^2 + 0.00018642 * height^2
    # Girls: -0.3160 - 0.13019 * age + 0.00676 * age^2 + 0.00016021 * height^2
    b0 = -0.2872
    b1 = -0.16717
    b2 = 0.008445
    b3 = 0.00017331

    fvc = b0 + b1 * age + b2 * (age**2) + b3 * (height**2)

    # Ensure physical plausibility (FVC cannot be negative)
    return np.maximum(fvc, 0.5)
