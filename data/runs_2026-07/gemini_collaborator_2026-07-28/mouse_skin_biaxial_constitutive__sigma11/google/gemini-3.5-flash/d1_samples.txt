import numpy as np


def predict(X):
    """Predicts the Cauchy stress sigma11 [MPa] in a mouse-skin specimen

    under biaxial loading using the standard orthotropic Fung-elasticity model.

    Parameters:
    - X: 2-D numpy array of shape (N, 2)
         X[:, 0] = lambda11 (stretch in the first/lateral direction)
         X[:, 1] = lambda22 (stretch in the second/cranial-caudal direction)

    Returns:
    - 1-D numpy array of shape (N,) containing the predicted sigma11 [MPa].
    """
    # Extract principal stretches
    l1 = X[:, 0]
    l2 = X[:, 1]

    # Compute Green-Lagrange strains
    E1 = 0.5 * (l1**2 - 1.0)
    E2 = 0.5 * (l2**2 - 1.0)

    # Representative material parameters for mouse skin (Fung model)
    c = 0.045  # Stress scaling parameter [MPa]
    a1 = 9.2  # Anisotropic stiffness parameter in lateral direction
    a2 = 7.1  # Anisotropic stiffness parameter in cranial-caudal direction
    a3 = 3.5  # Interaction parameter

    # Exponent Q in the Fung strain energy function
    Q = a1 * E1**2 + a2 * E2**2 + 2.0 * a3 * E1 * E2

    # 2nd Piola-Kirchhoff stress: S11 = dW/dE1 = c * (a1 * E1 + a3 * E2) * exp(Q)
    # Cauchy stress: sigma11 = l1^2 * S11 (assuming incompressibility)
    sigma11 = (l1**2) * c * (a1 * E1 + a3 * E2) * np.exp(Q)

    return sigma11
