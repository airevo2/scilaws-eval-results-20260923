import numpy as np


def predict(X):
    """Predicts period life expectancy at birth (e0) as a function of calendar year

    using a logistic (sigmoidal) model that represents the historical
    demographic transition from pre-industrial to modern eras.

    Parameters:
    X : ndarray of shape (N, M)
        X[:, 0] contains the calendar year (t) [years]

    Returns:
    ndarray of shape (N,)
        Predicted period life expectancy at birth (e0) [years]
    """
    t = X[:, 0]

    # Parameters representing the canonical demographic transition of e0
    # based on long-run historical data (e.g., Sweden 1751-2024)
    e_min = 32.0  # Baseline pre-industrial life expectancy
    e_max = 85.0  # Modern upper asymptotic life expectancy
    t0 = 1925.0  # Inflection point (midpoint of the rapid transition)
    k = 0.022  # Rate of transition parameter

    # Logistic growth model
    e0 = e_min + (e_max - e_min) / (1.0 + np.exp(-k * (t - t0)))

    return e0
