    # Population size N

    S = np.maximum(S_mean + z, 1e-5 * N)

    # TSIR exponent alpha (typically ~0.97 for measles to account for discretization/clustering)
    alpha = 0.97

    # Deterministic TSIR relationship: I_t = beta_t * I_{t-1}^alpha * (S_{t-1} / N)
    I_pred = beta * (I_prev**alpha) * (S / N)

    # Clean up any potential numerical issues (e.g., negative values or NaNs)
    I_pred = np.nan_to_num(I_pred, nan=0.0, posinf=0.0, neginf=0.0)

import numpy as np


def predict(X):
    # Extract inputs
    cases_prev = X[:, 0]
    biweek = X[:, 1]
    log_pop = X[:, 3]
    z_t = X[:, 4]

    # Calculate population N from log10 population
    N = 10**log_pop

    # Reconstruct Susceptibles: S_t = S_mean + z_t
    # In prevaccination eras, the mean susceptible fraction (S_mean/N) is typically around 3.5% to 5%.
    # We use 4.0% (0.04) as a canonical representation.
    S_mean = 0.04 * N
    S = S_mean + z_t

    # Ensure S remains physically plausible (positive)
    S = np.maximum(S, 0.01 * N)

    # TSIR Model parameters:
    # alpha: transmission exponent (slightly less than 1 to account for spatial clustering/discretization)
    alpha = 0.97

    # Seasonal transmission rate beta_s
    # Modeled via a standard sinusoidal profile peaking in late winter / early spring (around biweek 5)
    beta_mean = 28.0
    beta_amplitude = 0.15
    peak_biweek = 5.0
    theta = 2.0 * np.pi * (biweek - peak_biweek) / 26.0
    beta = beta_mean * (1.0 + beta_amplitude * np.cos(theta))

    # Calculate predicted cases: I_t = beta_s * I_{t-1}^alpha * (S / N)
    cases_prev_alpha = np.power(np.maximum(cases_prev, 0.0), alpha)
    I_t = beta * cases_prev_alpha * (S / N)

    return I_t
