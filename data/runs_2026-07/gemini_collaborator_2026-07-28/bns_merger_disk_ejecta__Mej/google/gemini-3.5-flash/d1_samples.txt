import numpy as np


def predict(X):
    """Predicts the dynamical ejecta mass (M_ej) in solar masses

    for a binary neutron star merger using the standard phenomenological
    fitting formula from Dietrich & Uryu (2017).

    Inputs:
    X[:, 0] = M1 (gravitational mass of primary)
    X[:, 1] = M2 (gravitational mass of secondary)
    X[:, 2] = Mb1 (baryonic mass of primary)
    X[:, 3] = Mb2 (baryonic mass of secondary)
    X[:, 4] = C1 (compactness of primary)
    X[:, 5] = C2 (compactness of secondary)
    X[:, 6] = q (mass ratio, not used in this formulation)
    X[:, 7] = Lambda_tilde (effective tidal deformability, not used)
    """
    M1 = X[:, 0]
    M2 = X[:, 1]
    Mb1 = X[:, 2]
    Mb2 = X[:, 3]
    C1 = X[:, 4]
    C2 = X[:, 5]

    # Standard fitting coefficients from Dietrich et al. (2017)
    a = -0.0719
    b = 0.2116
    c = -2.42
    d = -0.2905
    n = -0.5
    e = 0.0957

    # Compute the components of the fitting formula
    term1 = a * (M2 / M1) ** (1.0 / 3.0) * (1.0 - 2.0 * C1) / C1
    term2 = b * (M1 / M2) ** n * (1.0 - 2.0 * C2) / C2
    term3 = c * (M1 - Mb1)
    term4 = d * (M2 - Mb2)

    # Convert from the fit scale (10^-3 Solar Masses) to Solar Masses
    M_ej = (term1 + term2 + term3 + term4 + e) * 1e-3

    # Enforce physical boundary (ejecta mass must be non-negative)
    return np.maximum(0.0, M_ej)

    M2 = X[:, 1]
    Mb1 = X[:, 2]
    Mb2 = X[:, 3]
    C1 = X[:, 4]
    C2 = X[:, 5]

    # Fitting coefficients from the canonical Dietrich & Uryu (2017) model
    a = -0.0719
    b = 0.2116
    c = -0.242
    d = -2.42

    # Contribution from the primary neutron star (i=1, j=2)
    term1 = (
        a * (M2 / M1) ** (1.0 / 3.0) * (1.0 - 2.0 * C1) / C1
        + b * (M2 / M1) ** d
        + c * (1.0 - M1 / Mb1)
    ) * Mb1

    # Contribution from the secondary neutron star (i=2, j=1)
    term2 = (
        a * (M1 / M2) ** (1.0 / 3.0) * (1.0 - 2.0 * C2) / C2
        + b * (M1 / M2) ** d
        + c * (1.0 - M2 / Mb2)
    ) * Mb2

    # The fitting formula predicts 10^3 * M_ej
    M_ej_103 = term1 + term2

    # Convert to solar masses and enforce a physical lower bound of 0
    M_ej = np.maximum(0.0, 1e-3 * M_ej_103)

    return M_ej

import numpy as np


def predict(X):
    """Predicts the dynamical ejecta mass (M_ej) in solar masses for a binary

    neutron star merger using the Dietrich & Uryu (2017) fitting formula.

    Inputs:
    X[:, 0] = M1 [M_sun] — Gravitational mass of the primary neutron star.
    X[:, 1] = M2 [M_sun] — Gravitational mass of the secondary neutron star.
    X[:, 2] = Mb1 [M_sun] — Baryonic mass of the primary neutron star.
    X[:, 3] = Mb2 [M_sun] — Baryonic mass of the secondary neutron star.
    X[:, 4] = C1 — Compactness of the primary neutron star.
    X[:, 5] = C2 — Compactness of the secondary neutron star.
    X[:, 6] = q — Binary mass ratio (M2/M1).
    X[:, 7] = Lambda_tilde — Effective binary tidal deformability.
    """
    M1 = X[:, 0]
    M2 = X[:, 1]
    Mb1 = X[:, 2]
    Mb2 = X[:, 3]
    C1 = X[:, 4]
    C2 = X[:, 5]

    # Fitting coefficients from Dietrich & Uryu (2017)
    a = -0.0719
    b = 0.2116
    n = -0.2437
    d = -2.42

    # Mass ratios
    q1 = M2 / M1
    q2 = M1 / M2

    # Contributions from the primary star
    term1_a = a * (q1 ** (1.0 / 3.0)) * ((1.0 - 2.0 * C1) / C1) * Mb1
    term1_b = b * (q2**n) * (Mb1 - M1)

    # Contributions from the secondary star
    term2_a = a * (q2 ** (1.0 / 3.0)) * ((1.0 - 2.0 * C2) / C2) * Mb2
    term2_b = b * (q1**n) * (Mb2 - M2)

    # Total ejecta mass (scaled from 10^-3 M_sun)
    M_ej = (term1_a + term1_b + term2_a + term2_b + d) * 1e-3

    # Ensure the ejecta mass is non-negative
    return np.maximum(M_ej, 0.0)
