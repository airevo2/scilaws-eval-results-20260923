import numpy as np

def predict(X):
    """
    Predicts the gravitational mass of the bound, post-merger accretion disk 
    surrounding the BNS merger remnant (M_disk in solar masses) using the 
    standard phenomenological fit from Radice et al. (2018).
    
    Inputs:
    X[:, 0] = M1 [M_sun]
    X[:, 1] = M2 [M_sun]
    X[:, 5] = Lambda_tilde
    """
    M1 = X[:, 0]
    M2 = X[:, 1]
    Lambda_tilde = X[:, 5]
    
    M_tot = M1 + M2
    
    # Radice et al. (2018) fit parameters (arXiv:1804.02689)
    a = 0.084
    b = 0.078
    c = 368.0
    d = 265.0
    
    # Calculate the fraction of total mass that goes into the disk
    disk_fraction = a + b * np.tanh((Lambda_tilde - c) / d)
    
    # Apply the physical lower bound of 10^-3
    disk_fraction = np.maximum(1e-3, disk_fraction)
    
    M_disk = M_tot * disk_fraction
    return M_disk

    log10_Mdisk = np.maximum(-3.0, log10_Mdisk)

    return 10**log10_Mdisk

    disk_fraction = np.maximum(1e-3, disk_fraction)

import numpy as np

def predict(X):
    # Extract required inputs
    M1 = X[:, 0]  # Gravitational mass of the lighter neutron star [M_sun]
    M2 = X[:, 1]  # Gravitational mass of the heavier neutron star [M_sun]
    Lambda_tilde = X[:, 5]  # Binary-averaged dimensionless tidal deformability
    
    # Total gravitational mass of the binary
    M_tot = M1 + M2
    
    # Coefficients from Radice et al. (2018) fit for BNS disk mass
    a = 0.084
    b = 0.078
    c = 368.0
    d = 211.0
    
    # Calculate the accretion disk mass
    M_disk = M_tot * np.maximum(1e-3, a + b * np.tanh((Lambda_tilde - c) / d))
    
    return M_disk
