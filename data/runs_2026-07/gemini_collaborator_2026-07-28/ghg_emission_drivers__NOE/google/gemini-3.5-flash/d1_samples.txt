def predict(X):

    Predicts the national annual N2O emissions (excluding land-use change and forestry)
    in kilograms of N2O per country per year using a STIRPAT-based formulation.

    X[:, 1] = TP (P) [persons]
    X[:, 2] = AP (AP) [%]
    X[:, 3] = DP (DP) [persons per km2]
    X[:, 4] = UR (UR) [%]
    X[:, 5] = AT (AT) [degrees C]
    
    Returns:
    1-D numpy array of predicted N2O emissions in kg.
    """
    # Extract inputs
    A = X[:, 0]
    P = X[:, 1]
    AP = X[:, 2]
    DP = X[:, 3]
    UR = X[:, 4]
    AT = X[:, 5]
    
    # Coefficients based on standard empirical STIRPAT models for N2O emissions
    beta_0 = -2.70   # Intercept
    beta_1 = 1.02    # Population elasticity
    beta_2 = 0.35    # GDP per capita elasticity
    beta_3 = 0.15    # Active population share elasticity
    beta_4 = -0.08   # Population density elasticity
    beta_5 = -0.12   # Urbanization rate elasticity
    beta_6 = 0.005   # Temperature semi-elasticity
    
    # Calculate log of emissions
    log_I_N2O = (beta_0 + 
                 beta_1 * np.log(P) + 
                 beta_2 * np.log(A) + 
                 beta_3 * np.log(AP) + 
                 beta_4 * np.log(DP) + 
                 beta_5 * np.log(UR) + 
                 beta_6 * AT)
    
    # Return emissions in kg N2O
    return np.exp(log_I_N2O)
